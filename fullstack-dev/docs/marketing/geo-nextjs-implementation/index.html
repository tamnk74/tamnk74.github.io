GEO Implementation Guide for Next.js

Overview​

This guide provides a comprehensive, step-by-step implementation of Generative Engine Optimization (GEO) strategies in a Next.js application. You'll learn how to structure your content, implement semantic markup, optimize metadata, and configure your Next.js app to be AI-friendly.

Prerequisites​

  • Next.js 14+ (App Router or Pages Router)
  • Basic understanding of React and Next.js
  • Node.js 18+ installed
  • Familiarity with GEO concepts

Project Setup​

Step 1: Create a New Next.js Project​

npx create-next-app@latest my-geo-optimized-site
cd my-geo-optimized-site

# Choose the following options:
# ✔ Would you like to use TypeScript? Yes
# ✔ Would you like to use ESLint? Yes
# ✔ Would you like to use Tailwind CSS? Yes
# ✔ Would you like to use `src/` directory? Yes
# ✔ Would you like to use App Router? Yes
# ✔ Would you like to customize the default import alias? No

Step 2: Install Required Dependencies​

npm install next-seo schema-dts next-sitemap react-schemaorg
npm install -D @types/react @types/node

Package purposes:

  • next-seo: SEO and metadata management
  • schema-dts: TypeScript definitions for schema.org
  • next-sitemap: Generate sitemap and robots.txt
  • react-schemaorg: React components for schema.org JSON-LD

/** @type {import('next-sitemap').IConfig} */
module.exports = {
  siteUrl: process.env.SITE_URL || 'https://yourdomain.com',
  generateRobotsTxt: false, // We created custom robots.txt
  generateIndexSitemap: true,
  exclude: ['/api/*', '/admin/*', '/private/*', '/_next/*'],
  robotsTxtOptions: {
    policies: [
      {
        userAgent: '*',
        allow: '/',
      },
      {
        userAgent: 'GPTBot',
        allow: '/',
      },
      {
        userAgent: 'Google-Extended',
        allow: '/',
      },
    ],
  },
  transform: async (config, path) => {
    // Customize priority based on page type
    let priority = 0.7;
    let changefreq = 'weekly';

    if (path === '/') {
      priority = 1.0;
      changefreq = 'daily';
    } else if (path.includes('/articles/')) {
      priority = 0.8;
      changefreq = 'weekly';
    } else if (path.includes('/faq')) {
      priority = 0.9;
      changefreq = 'monthly';
    }

    return {
      loc: path,
      changefreq,
      priority,
      lastmod: new Date().toISOString(),
      alternateRefs: config.alternateRefs ?? [],
    };
  },
};

Step 9: Create AI-Friendly Content Components​

Hierarchical Heading Component​

Create src/components/Heading.tsx:

import React from 'react';

interface HeadingProps {
  level: 1 | 2 | 3 | 4 | 5 | 6;
  children: React.ReactNode;
  id?: string;
  className?: string;
}

export default function Heading({ level, children, id, className = '' }: HeadingProps) {
  const Tag = `h${level}` as keyof JSX.IntrinsicElements;
  const baseClasses = 'font-semibold scroll-mt-20';

  const sizeClasses = {
    1: 'text-4xl mb-6',
    2: 'text-3xl mb-5 mt-8',
    3: 'text-2xl mb-4 mt-6',
    4: 'text-xl mb-3 mt-5',
    5: 'text-lg mb-2 mt-4',
    6: 'text-base mb-2 mt-3',
  };

  // Auto-generate ID from content if not provided
  const headingId =
    id ||
    (typeof children === 'string'
      ? children
          .toLowerCase()
          .replace(/\s+/g, '-')
          .replace(/[^\w-]/g, '')
      : undefined);

  return (
    <Tag id={headingId} className={`${baseClasses} ${sizeClasses[level]} ${className}`}>
      {children}
    </Tag>
  );
}

Table of Contents Component​

Create src/components/TableOfContents.tsx:

'use client';

import { useEffect, useState } from 'react';

interface TocItem {
  id: string;
  text: string;
  level: number;
}

export default function TableOfContents() {
  const [toc, setToc] = useState<TocItem[]>([]);
  const [activeId, setActiveId] = useState<string>('');

  useEffect(() => {
    // Extract headings from the document
    const headings = Array.from(document.querySelectorAll('h2, h3, h4')).map((heading) => ({
      id: heading.id,
      text: heading.textContent || '',
      level: parseInt(heading.tagName[1]),
    }));
    setToc(headings);

    // Track active heading
    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {
            setActiveId(entry.target.id);
          }
        });
      },
      { rootMargin: '-80px 0px -80% 0px' },
    );

    headings.forEach(({ id }) => {
      const element = document.getElementById(id);
      if (element) observer.observe(element);
    });

    return () => observer.disconnect();
  }, []);

  return (
    <nav className="sticky top-20 max-h-[calc(100vh-5rem)] overflow-auto">
      <h3 className="font-semibold mb-4">Table of Contents</h3>
      <ul className="space-y-2 text-sm">
        {toc.map((item) => (
          <li
            key={item.id}
            style={{ paddingLeft: `${(item.level - 2) * 1}rem` }}
            className={activeId === item.id ? 'font-semibold text-blue-600' : 'text-gray-600'}
          >
            <a href={`#${item.id}`} className="hover:text-blue-600">
              {item.text}
            </a>
          </li>
        ))}
      </ul>
    </nav>
  );
}

Step 10: Implement Entity Markup​

interface EntityMentionProps {
  type: 'Person' | 'Organization' | 'Product' | 'Place';
  name: string;
  url?: string;
  children: React.ReactNode;
}

export default function EntityMention({ type, name, url, children }: EntityMentionProps) {
  const schema = {
    '@context': 'https://schema.org',
    '@type': type,
    name: name,
    ...(url && { url }),
  };

  return (
    <>
      <span itemScope itemType={`https://schema.org/${type}`} itemProp="name">
        {url ? (
          <a href={url} itemProp="url" className="text-blue-600 hover:underline">
            {children}
          </a>
        ) : (
          children
        )}
      </span>
      <script type="application/ld+json" dangerouslySetInnerHTML={{ __html: JSON.stringify(schema) }} />
    </>
  );
}

Step 14: Implement Reading Time Calculator​

Create src/lib/readingTime.ts:

export function calculateReadingTime(text: string): string {
  const wordsPerMinute = 200;
  const wordCount = text.trim().split(/\s+/).length;
  const minutes = Math.ceil(wordCount / wordsPerMinute);
  return `${minutes} min read`;
}

export function getWordCount(text: string): number {
  return text.trim().split(/\s+/).length;
}

  const aiReferrers = ['chat.openai.com', 'gemini.google.com', 'claude.ai', 'perplexity.ai', 'you.com'];

  const isAIReferrer = aiReferrers.some((ai) => referrer.includes(ai));

  if (isAIReferrer && typeof window !== 'undefined') {
    // Google Analytics 4
    if (window.gtag) {
      window.gtag('event', 'ai_referral', {
        source: referrer,
        page_path: window.location.pathname,
      });
    }

    // Custom analytics
    console.log('AI Referral detected:', referrer);
  }
};

Create src/components/AnalyticsTracker.tsx:

'use client';

import { useEffect } from 'react';
import { trackAIReferral } from '@/lib/analytics';

export default function AnalyticsTracker() {
  useEffect(() => {
    const referrer = document.referrer;
    if (referrer) {
      trackAIReferral(referrer);
    }
  }, []);

  return null;
}

Add to your layout:

  );
}

import fs from 'fs';
import path from 'path';

const robotsTxt = fs.readFileSync(path.join(process.cwd(), 'public', 'robots.txt'), 'utf-8');

const aiCrawlers = ['GPTBot', 'Google-Extended', 'CCBot', 'anthropic-ai', 'Claude-Web'];

aiCrawlers.forEach((crawler) => {
  const regex = new RegExp(`User-agent: ${crawler}[\\s\\S]*?Allow: /`, 'i');
  if (regex.test(robotsTxt)) {
    console.log(`✅ ${crawler} is allowed`);

Troubleshooting​

Common Issues​

  1. Schema validation errors:

    • Ensure all required fields are present
    • Check date formats (ISO 8601)
    • Verify URLs are absolute, not relative

  2. Sitemap not generating:

    npm run postbuild

  3. Metadata not appearing:

    • Check that you're using App Router correctly
    • Ensure metadata is exported at page level
    • Verify no conflicting metadata in layout
