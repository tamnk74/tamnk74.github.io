Building AI Document Wiki Agent with LangChain + OpenAI

Overview​

This guide provides a complete implementation using LangChain framework with OpenAI's GPT models and embeddings. This is the recommended approach for quick start and production-ready applications.

Pros:

  • ✅ Easy to set up with excellent documentation
  • ✅ Battle-tested in production environments
  • ✅ Rich ecosystem of integrations
  • ✅ Active community support
  • ✅ High-quality responses from GPT-4

Cons:

  • ❌ Requires OpenAI API subscription (cost varies by usage)
  • ❌ Data sent to external API (privacy considerations)
  • ❌ Dependent on OpenAI service availability

Step 3: Document Loader Implementation​

Create src/ingestion/documentLoader.js:

import { DirectoryLoader } from 'langchain/document_loaders/fs/directory';
import { TextLoader } from 'langchain/document_loaders/fs/text';
import { PDFLoader } from 'langchain/document_loaders/fs/pdf';
import { JSONLoader } from 'langchain/document_loaders/fs/json';
import { CSVLoader } from 'langchain/document_loaders/fs/csv';
import path from 'path';

export class DocumentLoader {
  constructor(docsPath) {
    this.docsPath = docsPath;
  }

  async loadDocuments() {
    console.log(`Loading documents from ${this.docsPath}...`);

    const loader = new DirectoryLoader(
      this.docsPath,
      {
        '.md': (path) => new TextLoader(path),
        '.mdx': (path) => new TextLoader(path),
        '.txt': (path) => new TextLoader(path),
        '.pdf': (path) => new PDFLoader(path),
        '.json': (path) => new JSONLoader(path),
        '.csv': (path) => new CSVLoader(path),
      },
      true, // Recursive
    );

    const docs = await loader.load();
    console.log(`Loaded ${docs.length} documents`);

    // Add metadata
    return docs.map((doc) => ({
      ...doc,
      metadata: {
        ...doc.metadata,
        loadedAt: new Date().toISOString(),
        relativePath: path.relative(this.docsPath, doc.metadata.source),
      },
    }));
  }

  async loadCodeFiles(codePath, extensions = ['.js', '.ts', '.jsx', '.tsx', '.py']) {
    console.log(`Loading code files from ${codePath}...`);

    const extMapping = {};
    extensions.forEach((ext) => {
      extMapping[ext] = (path) => new TextLoader(path);
    });

    const loader = new DirectoryLoader(codePath, extMapping, true);
    const docs = await loader.load();

    console.log(`Loaded ${docs.length} code files`);

    return docs.map((doc) => ({
      ...doc,
      metadata: {
        ...doc.metadata,
        type: 'code',
        language: path.extname(doc.metadata.source).slice(1),
        relativePath: path.relative(codePath, doc.metadata.source),
      },
    }));
  }
}

Step 4: Document Processing & Chunking​

Create src/ingestion/documentProcessor.js:

import { RecursiveCharacterTextSplitter } from 'langchain/text_splitter';
import { Document } from 'langchain/document';
import path from 'path';

export class DocumentProcessor {
  constructor(options = {}) {
    this.chunkSize = options.chunkSize || 1000;
    this.chunkOverlap = options.chunkOverlap || 200;
    this.splitter = new RecursiveCharacterTextSplitter({
      chunkSize: this.chunkSize,
      chunkOverlap: this.chunkOverlap,
      separators: ['\n\n', '\n', '. ', ' ', ''],
    });
  }

  async processDocuments(documents) {
    console.log(`Processing ${documents.length} documents...`);

    // Split documents into chunks
    const chunks = await this.splitter.splitDocuments(documents);
    console.log(`Created ${chunks.length} chunks`);

    // Enhance metadata
    const enhancedChunks = chunks.map((chunk, index) => {
      return new Document({
        pageContent: chunk.pageContent,
        metadata: {
          ...chunk.metadata,
          chunkId: `chunk_${index}`,
          chunkSize: chunk.pageContent.length,
          title: this.extractTitle(chunk.pageContent, chunk.metadata),
        },
      });
    });

    return enhancedChunks;
  }

  extractTitle(content, metadata) {
    // Try to extract title from markdown headers
    const headerMatch = content.match(/^#\s+(.+)$/m);
    if (headerMatch) {
      return headerMatch[1];
    }

    // Fall back to filename
    if (metadata.source) {
      return path.basename(metadata.source, path.extname(metadata.source));
    }

    return 'Untitled';
  }

  async processWithMetadata(documents, additionalMetadata = {}) {
    const processed = await this.processDocuments(documents);

    return processed.map((doc) => ({
      ...doc,
      metadata: {
        ...doc.metadata,
        ...additionalMetadata,
      },
    }));
  }
}

Step 5: Vector Store with ChromaDB​

Create src/ingestion/vectorStore.js:

import { Chroma } from '@langchain/community/vectorstores/chroma';
import { OpenAIEmbeddings } from '@langchain/openai';
import { ChromaClient } from 'chromadb';

export class VectorStoreManager {
  constructor(config) {
    this.chromaPath = config.chromaPath || './data/chroma_db';
    this.collectionName = config.collectionName || 'project_docs';
    this.embeddings = new OpenAIEmbeddings({
      modelName: config.embeddingModel || 'text-embedding-3-small',
      openAIApiKey: config.apiKey || process.env.OPENAI_API_KEY,
    });
    this.vectorStore = null;
  }

  async initialize() {
    console.log('Initializing vector store...');

    this.vectorStore = await Chroma.fromExistingCollection(this.embeddings, {
      collectionName: this.collectionName,
      url: this.chromaPath,
    }).catch(async () => {
      console.log(`Creating new collection: ${this.collectionName}`);
      return null;
    });

    return this.vectorStore;
  }

  async addDocuments(documents) {
    console.log(`Adding ${documents.length} documents to vector store...`);

    if (!this.vectorStore) {
      this.vectorStore = await Chroma.fromDocuments(documents, this.embeddings, {
        collectionName: this.collectionName,
        url: this.chromaPath,
      });
    } else {
      await this.vectorStore.addDocuments(documents);
    }

    console.log('Documents added successfully');
    return this.vectorStore;
  }

  async similaritySearch(query, k = 4) {
    if (!this.vectorStore) {
      await this.initialize();
    }

    return await this.vectorStore.similaritySearch(query, k);
  }

  async similaritySearchWithScore(query, k = 4) {
    if (!this.vectorStore) {
      await this.initialize();
    }

    return await this.vectorStore.similaritySearchWithScore(query, k);
  }

  async deleteCollection() {
    console.log(`Deleting collection: ${this.collectionName}`);
    const client = new ChromaClient({ path: this.chromaPath });
    await client.deleteCollection({ name: this.collectionName });
  }

  getVectorStore() {
    return this.vectorStore;
  }
}

Step 6: Ingestion Script​

Create src/ingest.js:

import dotenv from 'dotenv';
import { DocumentLoader } from './ingestion/documentLoader.js';
import { DocumentProcessor } from './ingestion/documentProcessor.js';
import { VectorStoreManager } from './ingestion/vectorStore.js';

dotenv.config();

async function ingestDocuments() {
  try {
    console.log('=== Starting Document Ingestion ===\n');

    // 1. Load documents
    const docsPath = process.env.DOCS_PATH || './docs';
    const codePath = process.env.CODE_PATH || './src';

    const loader = new DocumentLoader(docsPath);
    let documents = await loader.loadDocuments();

    // Optionally load code files
    if (process.env.INCLUDE_CODE === 'true') {
      const codeFiles = await loader.loadCodeFiles(codePath);
      documents = [...documents, ...codeFiles];
    }

    console.log(`\nTotal documents loaded: ${documents.length}\n`);

    // 2. Process documents (chunking)
    const processor = new DocumentProcessor({
      chunkSize: parseInt(process.env.CHUNK_SIZE || '1000'),
      chunkOverlap: parseInt(process.env.CHUNK_OVERLAP || '200'),
    });

    const chunks = await processor.processDocuments(documents);
    console.log(`\nTotal chunks created: ${chunks.length}\n`);

    // 3. Store in vector database
    const vectorStore = new VectorStoreManager({
      chromaPath: process.env.CHROMA_PATH,
      collectionName: process.env.CHROMA_COLLECTION_NAME,
      embeddingModel: process.env.EMBEDDING_MODEL,
    });

    // Delete existing collection if reingest flag is set
    if (process.env.REINGEST === 'true') {
      try {
        await vectorStore.deleteCollection();
        console.log('Existing collection deleted\n');
      } catch (error) {
        console.log('No existing collection to delete\n');
      }
    }

    await vectorStore.addDocuments(chunks);

    console.log('\n=== Ingestion Complete ===');
    console.log(`✓ Processed ${documents.length} documents`);
    console.log(`✓ Created ${chunks.length} chunks`);
    console.log(`✓ Stored in collection: ${process.env.CHROMA_COLLECTION_NAME}`);
  } catch (error) {
    console.error('Error during ingestion:', error);
    process.exit(1);
  }
}

ingestDocuments();

Step 7: RAG Chain Implementation​

Create src/retrieval/ragChain.js:

import { ChatOpenAI } from '@langchain/openai';
import { PromptTemplate } from '@langchain/core/prompts';
import { StringOutputParser } from '@langchain/core/output_parsers';
import { RunnableSequence, RunnablePassthrough } from '@langchain/core/runnables';

export class RAGChain {
  constructor(vectorStore, options = {}) {
    this.vectorStore = vectorStore;
    this.llm = new ChatOpenAI({
      modelName: options.modelName || 'gpt-4-turbo-preview',
      temperature: options.temperature || 0.2,
      maxTokens: options.maxTokens || 2000,
    });
    this.k = options.k || 4;

    this.promptTemplate = this.createPromptTemplate();
  }

  createPromptTemplate() {
    const template = `You are an AI assistant that helps answer questions about a software project based on its documentation and codebase.

Use the following context from the project documentation to answer the question. If you cannot find the answer in the context, say so clearly. Do not make up information.

Context:
{context}

Question: {question}

Instructions:
- Provide accurate, detailed answers based on the context
- Include relevant code examples if mentioned in the context
- Reference specific document sections or files when applicable
- If the answer requires multiple parts, structure your response clearly
- If you're not sure, say "I don't have enough information in the documentation to answer this confidently"

Answer:`;

    return PromptTemplate.fromTemplate(template);
  }

  async query(question, options = {}) {
    console.log(`\nQuery: ${question}`);

    // Retrieve relevant documents
    const relevantDocs = await this.vectorStore.similaritySearchWithScore(question, options.k || this.k);

    console.log(`Found ${relevantDocs.length} relevant chunks`);

    // Format context
    const context = relevantDocs
      .map(([doc, score], index) => {
        const source = doc.metadata.relativePath || doc.metadata.source || 'Unknown';
        return `[Document ${index + 1}] (Source: ${source}, Relevance: ${(score * 100).toFixed(1)}%)
${doc.pageContent}
---`;
      })
      .join('\n\n');

    // Build RAG chain
    const chain = RunnableSequence.from([
      {
        context: () => context,
        question: new RunnablePassthrough(),
      },
      this.promptTemplate,
      this.llm,
      new StringOutputParser(),
    ]);

    // Get answer
    const answer = await chain.invoke(question);

    return {
      question,
      answer,
      sources: relevantDocs.map(([doc, score]) => ({
        source: doc.metadata.relativePath || doc.metadata.source,
        relevance: score,
        title: doc.metadata.title,
        snippet: doc.pageContent.substring(0, 200) + '...',
      })),
    };
  }

  async queryWithHistory(question, chatHistory = []) {
    const historyContext = chatHistory.map((msg) => `${msg.role}: ${msg.content}`).join('\n');

    const enhancedQuestion =
      chatHistory.length > 0 ? `Previous conversation:\n${historyContext}\n\nCurrent question: ${question}` : question;

    return await this.query(enhancedQuestion);
  }

  updateSystemPrompt(newTemplate) {
    this.promptTemplate = PromptTemplate.fromTemplate(newTemplate);
  }
}

Create src/retrieval/queryEngine.js:

import { VectorStoreManager } from '../ingestion/vectorStore.js';
import { RAGChain } from './ragChain.js';

export class QueryEngine {
  constructor(config) {
    this.config = config;
    this.vectorStoreManager = null;
    this.ragChain = null;
    this.initialized = false;
  }

  async initialize() {
    if (this.initialized) return;

    console.log('Initializing Query Engine...');

    this.vectorStoreManager = new VectorStoreManager(this.config);
    await this.vectorStoreManager.initialize();

    this.ragChain = new RAGChain(this.vectorStoreManager, {
      modelName: this.config.modelName,
      temperature: this.config.temperature,
      k: this.config.retrievalK,
    });

    this.initialized = true;
    console.log('Query Engine initialized successfully');
  }

  async ask(question, options = {}) {
    if (!this.initialized) {
      await this.initialize();
    }

    return await this.ragChain.query(question, options);
  }

  async askWithHistory(question, chatHistory) {
    if (!this.initialized) {
      await this.initialize();
    }

    return await this.ragChain.queryWithHistory(question, chatHistory);
  }

  async getRelevantDocs(query, k = 4) {
    if (!this.initialized) {
      await this.initialize();
    }

    return await this.vectorStoreManager.similaritySearchWithScore(query, k);
  }
}

Create src/api/server.js:

import express from 'express';
import cors from 'cors';
import dotenv from 'dotenv';
import { QueryEngine } from '../retrieval/queryEngine.js';

dotenv.config();

const app = express();
const port = process.env.PORT || 3000;

app.use(cors({ origin: process.env.CORS_ORIGIN || '*' }));
app.use(express.json());

const queryEngine = new QueryEngine({
  chromaPath: process.env.CHROMA_PATH,
  collectionName: process.env.CHROMA_COLLECTION_NAME,
  embeddingModel: process.env.EMBEDDING_MODEL,
  modelName: process.env.OPENAI_MODEL,
  temperature: 0.2,
  retrievalK: 4,
});

const sessions = new Map();

app.get('/health', (req, res) => {
  res.json({ status: 'ok', timestamp: new Date().toISOString() });
});

app.post('/api/query', async (req, res) => {
  try {
    const { question, sessionId } = req.body;

    if (!question) {
      return res.status(400).json({ error: 'Question is required' });
    }

    console.log(`\n[API] Query from session ${sessionId}: ${question}`);

    let chatHistory = [];
    if (sessionId && sessions.has(sessionId)) {
      chatHistory = sessions.get(sessionId);
    }

    const result = await queryEngine.askWithHistory(question, chatHistory);

    if (sessionId) {
      chatHistory.push({ role: 'user', content: question }, { role: 'assistant', content: result.answer });
      sessions.set(sessionId, chatHistory);
    }

    res.json({
      success: true,
      data: result,
      sessionId: sessionId || null,
    });
  } catch (error) {
    console.error('Query error:', error);
    res.status(500).json({
      success: false,
      error: error.message,
    });
  }
});

app.post('/api/search', async (req, res) => {
  try {
    const { query, k = 4 } = req.body;

    if (!query) {
      return res.status(400).json({ error: 'Query is required' });
    }

    const docs = await queryEngine.getRelevantDocs(query, k);

    res.json({
      success: true,
      data: docs.map(([doc, score]) => ({
        content: doc.pageContent,
        metadata: doc.metadata,
        relevance: score,
      })),
    });
  } catch (error) {
    console.error('Search error:', error);
    res.status(500).json({
      success: false,
      error: error.message,
    });
  }
});

app.post('/api/session/clear', (req, res) => {
  const { sessionId } = req.body;

  if (sessionId && sessions.has(sessionId)) {
    sessions.delete(sessionId);
  }

  res.json({ success: true });
});

app.listen(port, async () => {
  console.log(`\n🚀 AI Document Wiki Agent API running on port ${port}`);
  console.log(`📚 Initializing query engine...`);

  try {
    await queryEngine.initialize();
    console.log(`✓ Query engine ready`);
    console.log(`\nAvailable endpoints:`);
    console.log(`  - POST /api/query - Ask questions about the documentation`);
    console.log(`  - POST /api/search - Search for relevant documents`);
    console.log(`  - POST /api/session/clear - Clear chat history`);
    console.log(`  - GET /health - Health check\n`);
  } catch (error) {
    console.error('Failed to initialize query engine:', error);
    process.exit(1);
  }
});

Step 9: CLI Interface​

Create src/cli.js:

import dotenv from 'dotenv';
import readline from 'readline';
import { QueryEngine } from './retrieval/queryEngine.js';
import chalk from 'chalk';

dotenv.config();

const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout,
});

async function main() {
  console.log(chalk.blue.bold('\n=== AI Document Wiki Agent (LangChain + OpenAI) ===\n'));

  const queryEngine = new QueryEngine({
    chromaPath: process.env.CHROMA_PATH,
    collectionName: process.env.CHROMA_COLLECTION_NAME,
    embeddingModel: process.env.EMBEDDING_MODEL,
    modelName: process.env.OPENAI_MODEL,
    temperature: 0.2,
    retrievalK: 4,
  });

  console.log(chalk.yellow('Initializing...'));
  await queryEngine.initialize();
  console.log(chalk.green('✓ Ready to answer your questions!\n'));

  const chatHistory = [];

  const askQuestion = () => {
    rl.question(chalk.cyan('You: '), async (question) => {
      if (!question.trim()) {
        askQuestion();
        return;
      }

      if (question.toLowerCase() === 'exit' || question.toLowerCase() === 'quit') {
        console.log(chalk.yellow('\nGoodbye!\n'));
        rl.close();
        process.exit(0);
      }

      if (question.toLowerCase() === 'clear') {
        chatHistory.length = 0;
        console.log(chalk.yellow('\n✓ Chat history cleared\n'));
        askQuestion();
        return;
      }

      try {
        console.log(chalk.gray('\nThinking...\n'));

        const result = await queryEngine.askWithHistory(question, chatHistory);

        console.log(chalk.green.bold('Agent: ') + result.answer + '\n');

        if (result.sources && result.sources.length > 0) {
          console.log(chalk.gray('Sources:'));
          result.sources.forEach((source, index) => {
            console.log(
              chalk.gray(`  ${index + 1}. ${source.source} (${(source.relevance * 100).toFixed(1)}% relevant)`),
            );
          });
          console.log();
        }

        chatHistory.push({ role: 'user', content: question }, { role: 'assistant', content: result.answer });
      } catch (error) {
        console.error(chalk.red('Error:'), error.message);
      }

      askQuestion();
    });
  };

  console.log(chalk.gray('Type "exit" to quit, "clear" to clear history\n'));
  askQuestion();
}

main().catch(console.error);

Cost Optimization Tips​

1. Use GPT-3.5 for Development​

# In .env file
OPENAI_MODEL=gpt-3.5-turbo  # Much cheaper than GPT-4

2. Implement Caching​

import { InMemoryCache } from '@langchain/core/caches';

const cache = new InMemoryCache();

const llm = new ChatOpenAI({
  modelName: 'gpt-4-turbo-preview',
  cache: cache, // Reuse responses for identical queries
});

3. Optimize Chunk Retrieval​

# Retrieve fewer chunks to reduce context size
CHUNK_SIZE=800  # Smaller chunks
RETRIEVAL_K=3   # Fewer chunks per query

4. Use Smaller Embedding Model​

EMBEDDING_MODEL=text-embedding-3-small  # vs text-embedding-3-large

Issue: High API Costs​

Solutions:

  • Use GPT-3.5 instead of GPT-4
  • Implement semantic caching
  • Reduce number of retrieved chunks
  • Use smaller embedding models

Issue: Poor Answer Quality​

Solutions:

  • Increase chunk overlap for better context
  • Use GPT-4 instead of GPT-3.5
  • Improve document structure
  • Refine system prompts
  • Increase number of retrieved chunks
