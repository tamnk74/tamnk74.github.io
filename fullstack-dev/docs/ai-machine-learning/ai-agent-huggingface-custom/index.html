Building AI Document Wiki Agent with Custom Hugging Face Solution

Overview​

This guide demonstrates building a fully customized AI documentation agent using Hugging Face models and the Transformers library. This approach gives you complete control over every aspect of the system.

Pros:

  • ✅ Complete control and customization
  • ✅ Can fine-tune models on your specific documentation
  • ✅ Access to thousands of open-source models
  • ✅ Can run completely locally or in cloud
  • ✅ Highly optimizable for specific use cases
  • ✅ No vendor lock-in

Cons:

  • ❌ Requires more development and ML expertise
  • ❌ More complex setup and maintenance
  • ❌ Need to handle model hosting and scaling
  • ❌ Longer development time

Step 3: Custom Embedding Model​

Create src/models/embedder.py:

import torch
from sentence_transformers import SentenceTransformer
from typing import List
import numpy as np

class DocumentEmbedder:
    def __init__(self, model_name: str, device: str = "cuda"):
        """Initialize embedding model"""
        self.device = device if torch.cuda.is_available() else "cpu"
        print(f"Loading embedding model: {model_name} on {self.device}")

        self.model = SentenceTransformer(model_name, device=self.device)
        self.embedding_dim = self.model.get_sentence_embedding_dimension()

        print(f"Model loaded. Embedding dimension: {self.embedding_dim}")

    def embed_documents(self, texts: List[str], batch_size: int = 32) -> np.ndarray:
        """Embed a list of documents"""
        print(f"Embedding {len(texts)} documents...")

        embeddings = self.model.encode(
            texts,
            batch_size=batch_size,
            show_progress_bar=True,
            convert_to_numpy=True,
            normalize_embeddings=True,
        )

        return embeddings

    def embed_query(self, query: str) -> np.ndarray:
        """Embed a single query"""
        embedding = self.model.encode(
            query,
            convert_to_numpy=True,
            normalize_embeddings=True,
        )

        return embedding

    def get_embedding_dim(self) -> int:
        """Get embedding dimension"""
        return self.embedding_dim

Step 4: Custom LLM Wrapper​

Create src/models/llm.py:

import torch
from transformers import AutoTokenizer, AutoModelForSeq2SeqLM, AutoModelForCausalLM, pipeline
from typing import List, Optional

class CustomLLM:
    def __init__(
        self,
        model_name: str,
        device: str = "cuda",
        max_length: int = 512,
        temperature: float = 0.7,
    ):
        """Initialize LLM"""
        self.device = device if torch.cuda.is_available() else "cpu"
        self.max_length = max_length
        self.temperature = temperature

        print(f"Loading LLM: {model_name} on {self.device}")

        # Load tokenizer
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)

        # Determine model type and load accordingly
        if "t5" in model_name.lower() or "flan" in model_name.lower():
            self.model = AutoModelForSeq2SeqLM.from_pretrained(
                model_name,
                torch_dtype=torch.float16 if self.device == "cuda" else torch.float32,
                low_cpu_mem_usage=True,
            )
            self.model_type = "seq2seq"
        else:
            self.model = AutoModelForCausalLM.from_pretrained(
                model_name,
                torch_dtype=torch.float16 if self.device == "cuda" else torch.float32,
                low_cpu_mem_usage=True,
            )
            self.model_type = "causal"

        self.model.to(self.device)
        self.model.eval()

        print(f"Model loaded. Type: {self.model_type}")

    def generate(
        self,
        prompt: str,
        max_new_tokens: int = 256,
        temperature: Optional[float] = None,
    ) -> str:
        """Generate text from prompt"""
        temp = temperature if temperature is not None else self.temperature

        # Tokenize input
        inputs = self.tokenizer(
            prompt,
            return_tensors="pt",
            truncation=True,
            max_length=self.max_length,
        ).to(self.device)

        # Generate
        with torch.no_grad():
            if self.model_type == "seq2seq":
                outputs = self.model.generate(
                    **inputs,
                    max_new_tokens=max_new_tokens,
                    temperature=temp,
                    do_sample=temp > 0,
                    top_p=0.9,
                    num_beams=1,
                )
            else:
                outputs = self.model.generate(
                    **inputs,
                    max_new_tokens=max_new_tokens,
                    temperature=temp,
                    do_sample=temp > 0,
                    top_p=0.9,
                    pad_token_id=self.tokenizer.eos_token_id,
                )

        # Decode
        generated_text = self.tokenizer.decode(outputs[0], skip_special_tokens=True)

        return generated_text

    def __call__(self, prompt: str, **kwargs) -> str:
        """Make the class callable"""
        return self.generate(prompt, **kwargs)

Step 5: Document Loader​

Create src/ingestion/loader.py:

import os
from pathlib import Path
from typing import List, Dict
import json
import pypdf
import markdown
from docx import Document as DocxDocument

class Document:
    def __init__(self, text: str, metadata: Dict):
        self.text = text
        self.metadata = metadata

class DocumentLoader:
    def __init__(self, docs_path: str):
        self.docs_path = Path(docs_path)

    def load_text_file(self, file_path: Path) -> Document:
        """Load a text file"""
        with open(file_path, 'r', encoding='utf-8') as f:
            text = f.read()

        return Document(
            text=text,
            metadata={
                "source": str(file_path),
                "type": "text",
                "extension": file_path.suffix,
            }
        )

    def load_pdf_file(self, file_path: Path) -> Document:
        """Load a PDF file"""
        with open(file_path, 'rb') as f:
            pdf = pypdf.PdfReader(f)
            text = ""
            for page in pdf.pages:
                text += page.extract_text()

        return Document(
            text=text,
            metadata={
                "source": str(file_path),
                "type": "pdf",
                "pages": len(pdf.pages),
            }
        )

    def load_markdown_file(self, file_path: Path) -> Document:
        """Load a markdown file"""
        with open(file_path, 'r', encoding='utf-8') as f:
            text = f.read()

        # Convert markdown to plain text (optional)
        # html = markdown.markdown(text)

        return Document(
            text=text,
            metadata={
                "source": str(file_path),
                "type": "markdown",
            }
        )

    def load_documents(self) -> List[Document]:
        """Load all documents from directory"""
        print(f"Loading documents from {self.docs_path}...")

        documents = []

        for file_path in self.docs_path.rglob("*"):
            if not file_path.is_file():
                continue

            try:
                if file_path.suffix in [".txt", ".text"]:
                    doc = self.load_text_file(file_path)
                    documents.append(doc)
                elif file_path.suffix == ".pdf":
                    doc = self.load_pdf_file(file_path)
                    documents.append(doc)
                elif file_path.suffix in [".md", ".markdown", ".mdx"]:
                    doc = self.load_markdown_file(file_path)
                    documents.append(doc)
            except Exception as e:
                print(f"Error loading {file_path}: {e}")

        print(f"Loaded {len(documents)} documents")

        return documents

class DocumentChunker:
    def __init__(self, chunk_size: int = 512, chunk_overlap: int = 50):
        self.chunk_size = chunk_size
        self.chunk_overlap = chunk_overlap

    def chunk_text(self, text: str) -> List[str]:
        """Split text into overlapping chunks"""
        words = text.split()
        chunks = []

        for i in range(0, len(words), self.chunk_size - self.chunk_overlap):
            chunk = " ".join(words[i:i + self.chunk_size])
            if chunk:
                chunks.append(chunk)

        return chunks

    def chunk_documents(self, documents: List[Document]) -> List[Document]:
        """Chunk all documents"""
        print(f"Chunking {len(documents)} documents...")

        chunked_docs = []

        for doc in documents:
            chunks = self.chunk_text(doc.text)

            for i, chunk in enumerate(chunks):
                chunked_doc = Document(
                    text=chunk,
                    metadata={
                        **doc.metadata,
                        "chunk_index": i,
                        "total_chunks": len(chunks),
                    }
                )
                chunked_docs.append(chunked_doc)

        print(f"Created {len(chunked_docs)} chunks")

        return chunked_docs

Step 6: FAISS Vector Store​

Create src/ingestion/vector_store.py:

import faiss
import numpy as np
import json
from pathlib import Path
from typing import List, Tuple
from ..models.embedder import DocumentEmbedder

class FAISSVectorStore:
    def __init__(
        self,
        embedder: DocumentEmbedder,
        index_path: str,
        metadata_path: str,
    ):
        self.embedder = embedder
        self.index_path = Path(index_path)
        self.metadata_path = Path(metadata_path)
        self.index = None
        self.metadata = []
        self.embedding_dim = embedder.get_embedding_dim()

    def create_index(self, documents: List, batch_size: int = 32):
        """Create FAISS index from documents"""
        print("Creating FAISS index...")

        # Extract texts
        texts = [doc.text for doc in documents]

        # Generate embeddings
        embeddings = self.embedder.embed_documents(texts, batch_size=batch_size)

        # Create FAISS index
        self.index = faiss.IndexFlatIP(self.embedding_dim)  # Inner product (cosine sim)
        self.index.add(embeddings.astype('float32'))

        # Store metadata
        self.metadata = [doc.metadata for doc in documents]

        # Save
        self.save()

        print(f"Index created with {self.index.ntotal} vectors")

    def save(self):
        """Save index and metadata to disk"""
        self.index_path.parent.mkdir(parents=True, exist_ok=True)

        # Save FAISS index
        faiss.write_index(self.index, str(self.index_path))

        # Save metadata
        with open(self.metadata_path, 'w') as f:
            json.dump(self.metadata, f)

        print(f"Index saved to {self.index_path}")

    def load(self):
        """Load index and metadata from disk"""
        if not self.index_path.exists():
            raise FileNotFoundError(f"Index not found at {self.index_path}")

        # Load FAISS index
        self.index = faiss.read_index(str(self.index_path))

        # Load metadata
        with open(self.metadata_path, 'r') as f:
            self.metadata = json.load(f)

        print(f"Index loaded with {self.index.ntotal} vectors")

    def search(self, query: str, k: int = 4) -> List[Tuple[str, Dict, float]]:
        """Search for similar documents"""
        # Embed query
        query_embedding = self.embedder.embed_query(query)
        query_embedding = query_embedding.reshape(1, -1).astype('float32')

        # Search
        scores, indices = self.index.search(query_embedding, k)

        # Prepare results
        results = []
        for score, idx in zip(scores[0], indices[0]):
            if idx < len(self.metadata):
                results.append((
                    f"Document {idx}",  # You'd load actual text here
                    self.metadata[idx],
                    float(score),
                ))

        return results

Step 7: RAG Pipeline​

Create src/retrieval/rag_pipeline.py:

from typing import List, Dict
from ..models.llm import CustomLLM
from ..ingestion.vector_store import FAISSVectorStore

class RAGPipeline:
    def __init__(
        self,
        llm: CustomLLM,
        vector_store: FAISSVectorStore,
        k: int = 4,
    ):
        self.llm = llm
        self.vector_store = vector_store
        self.k = k

    def create_prompt(self, question: str, context: List[str]) -> str:
        """Create RAG prompt"""
        context_str = "\n\n".join([
            f"[Document {i+1}]\n{text}"
            for i, text in enumerate(context)
        ])

        prompt = f"""You are an AI assistant answering questions about a software project based on its documentation.

Context:
{context_str}

Question: {question}

Instructions:
- Answer based only on the context provided
- If you cannot find the answer in the context, say so clearly
- Do not make up information
- Be concise and accurate

Answer:"""

        return prompt

    def query(self, question: str) -> Dict:
        """Query the RAG pipeline"""
        print(f"\nQuery: {question}")

        # Retrieve relevant documents
        results = self.vector_store.search(question, k=self.k)

        # Extract context
        context = [metadata.get("text", "") for _, metadata, _ in results]

        # Create prompt
        prompt = self.create_prompt(question, context)

        # Generate answer
        answer = self.llm.generate(prompt, max_new_tokens=256)

        return {
            "question": question,
            "answer": answer,
            "sources": [
                {
                    "metadata": metadata,
                    "relevance": score,
                }
                for _, metadata, score in results
            ],
        }

Step 8: Ingestion Script​

Create src/ingest.py:

import os
from dotenv import load_dotenv
from ingestion.loader import DocumentLoader, DocumentChunker, Document
from ingestion.vector_store import FAISSVectorStore
from models.embedder import DocumentEmbedder

load_dotenv()

def main():
    print("=== Starting Document Ingestion ===\n")

    # Configuration
    docs_path = os.getenv("DOCS_PATH", "./docs")
    embedding_model = os.getenv("EMBEDDING_MODEL", "sentence-transformers/all-MiniLM-L6-v2")
    device = os.getenv("DEVICE", "cuda")
    chunk_size = int(os.getenv("CHUNK_SIZE", "512"))
    chunk_overlap = int(os.getenv("CHUNK_OVERLAP", "50"))

    # Load documents
    loader = DocumentLoader(docs_path)
    documents = loader.load_documents()

    # Chunk documents
    chunker = DocumentChunker(chunk_size=chunk_size, chunk_overlap=chunk_overlap)
    chunked_docs = chunker.chunk_documents(documents)

    # Initialize embedder
    embedder = DocumentEmbedder(embedding_model, device=device)

    # Create vector store
    vector_store = FAISSVectorStore(
        embedder=embedder,
        index_path=os.getenv("FAISS_INDEX_PATH", "./data/faiss_index"),
        metadata_path=os.getenv("METADATA_PATH", "./data/metadata.json"),
    )

    vector_store.create_index(chunked_docs)

    print("\n=== Ingestion Complete ===")
    print(f"✓ Processed {len(documents)} documents")
    print(f"✓ Created {len(chunked_docs)} chunks")

if __name__ == "__main__":
    main()

Step 9: FastAPI Server​

Create src/api/server.py:

import os
from dotenv import load_dotenv
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
from typing import Optional
import uvicorn

from models.embedder import DocumentEmbedder
from models.llm import CustomLLM
from ingestion.vector_store import FAISSVectorStore
from retrieval.rag_pipeline import RAGPipeline

load_dotenv()

app = FastAPI(title="AI Document Wiki Agent (Custom)")

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

class QueryRequest(BaseModel):
    question: str
    sessionId: Optional[str] = None

# Initialize components
print("Initializing AI Agent...")

embedder = DocumentEmbedder(
    model_name=os.getenv("EMBEDDING_MODEL"),
    device=os.getenv("DEVICE", "cuda"),
)

llm = CustomLLM(
    model_name=os.getenv("LLM_MODEL"),
    device=os.getenv("DEVICE", "cuda"),
    max_length=int(os.getenv("MAX_LENGTH", "512")),
)

vector_store = FAISSVectorStore(
    embedder=embedder,
    index_path=os.getenv("FAISS_INDEX_PATH"),
    metadata_path=os.getenv("METADATA_PATH"),
)
vector_store.load()

rag_pipeline = RAGPipeline(llm=llm, vector_store=vector_store, k=4)

print("✓ Agent ready!")

@app.get("/health")
async def health():
    return {"status": "ok"}

@app.post("/api/query")
async def query(request: QueryRequest):
    try:
        result = rag_pipeline.query(request.question)
        return {"success": True, "data": result}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

if __name__ == "__main__":
    port = int(os.getenv("PORT", 8000))
    uvicorn.run(app, host="0.0.0.0", port=port)

Fine-tune Embedding Model​

from sentence_transformers import SentenceTransformer, InputExample, losses
from torch.utils.data import DataLoader

# Prepare training data
train_examples = [
    InputExample(texts=['query1', 'relevant doc1'], label=1.0),
    InputExample(texts=['query2', 'irrelevant doc'], label=0.0),
]

train_dataloader = DataLoader(train_examples, shuffle=True, batch_size=16)

model = SentenceTransformer('sentence-transformers/all-MiniLM-L6-v2')
train_loss = losses.CosineSimilarityLoss(model)

model.fit(
    train_objectives=[(train_dataloader, train_loss)],
    epochs=3,
    warmup_steps=100,
)

model.save('./data/models/custom-embedder')
