MVP Stack Implementation Guide

Stack: NestJS monorepo → PostgreSQL + Redis → RabbitMQ → Next.js App Router

This guide walks through the concrete implementation of the small-team MVP stack for the stock trading app. Every layer is production-realistic but intentionally minimal — no premature abstractions, no second runtime, no Kafka cluster ops overhead.

3. NestJS API — Core Setup​

apps/api/src/main.ts​

import { NestFactory } from '@nestjs/core';
import { ValidationPipe, VersioningType } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { AppModule } from './app.module';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);

  app.useGlobalPipes(
    new ValidationPipe({ whitelist: true, forbidNonWhitelisted: true, transform: true }),
  );

  app.enableVersioning({ type: VersioningType.URI, defaultVersion: '1' });
  app.enableCors({ origin: process.env.WEB_ORIGIN ?? 'http://localhost:3000' });
  app.enableShutdownHooks();

  const config = app.get(ConfigService);
  await app.listen(config.get('PORT', 3001));
}
bootstrap();

4. Database — TypeORM Entities​

// apps/api/src/orders/order.entity.ts
import {
  Entity, PrimaryGeneratedColumn, Column, CreateDateColumn,
  ManyToOne, JoinColumn, Index,
} from 'typeorm';
import { Account } from '../users/account.entity';

export enum OrderStatus {
  PENDING         = 'pending',
  ACCEPTED        = 'accepted',
  PARTIALLY_FILLED = 'partially_filled',
  FILLED          = 'filled',
  CANCELLED       = 'cancelled',
  REJECTED        = 'rejected',
  UNKNOWN         = 'unknown',
}

export enum OrderSide { BUY = 'buy', SELL = 'sell' }
export enum OrderType { MARKET = 'market', LIMIT = 'limit' }

@Entity('orders')
@Index(['account', 'status'])
@Index(['idempotencyKey'], { unique: true })
export class Order {
  @PrimaryGeneratedColumn('uuid')
  id: string;

  @ManyToOne(() => Account, { nullable: false })
  @JoinColumn({ name: 'account_id' })
  account: Account;

  @Column({ length: 10 })
  symbol: string;

  @Column({ type: 'enum', enum: OrderSide })
  side: OrderSide;

  @Column({ type: 'enum', enum: OrderType })
  orderType: OrderType;

  @Column({ type: 'numeric', precision: 18, scale: 8 })
  qty: string;

  @Column({ type: 'numeric', precision: 18, scale: 8, nullable: true })
  limitPrice: string | null;

  @Column({ type: 'enum', enum: OrderStatus, default: OrderStatus.PENDING })
  status: OrderStatus;

  @Column({ nullable: true, length: 64 })
  brokerOrderId: string | null;

  @Column({ length: 64 })
  idempotencyKey: string;

  @CreateDateColumn()
  placedAt: Date;

  @Column({ nullable: true })
  filledAt: Date | null;

  @Column({ type: 'numeric', precision: 18, scale: 8, nullable: true })
  avgFillPrice: string | null;
}

6. RabbitMQ Messaging Module​

Using @nestjs/microservices with the RMQ transport.

// apps/api/src/shared/messaging/messaging.module.ts
import { Module, Global } from '@nestjs/common';
import { ClientsModule, Transport } from '@nestjs/microservices';
import { ConfigModule, ConfigService } from '@nestjs/config';

export const RABBITMQ_CLIENT = 'RABBITMQ_CLIENT';

@Global()
@Module({
  imports: [
    ClientsModule.registerAsync([
      {
        name: RABBITMQ_CLIENT,
        imports: [ConfigModule],
        inject: [ConfigService],
        useFactory: (config: ConfigService) => ({
          transport: Transport.RMQ,
          options: {
            urls: [config.getOrThrow<string>('RABBITMQ_URL')],
            queue: 'trading_events',
            queueOptions: {
              durable: true,
              arguments: {
                'x-dead-letter-exchange': 'trading_events.dlx',
              },
            },
            noAck: false,          // manual ack — no message loss
            prefetchCount: 10,
          },
        }),
      },
    ]),
  ],
  exports: [ClientsModule],
})
export class MessagingModule {}

Publishing an event​

// apps/api/src/orders/orders.service.ts (excerpt)
import { Inject, Injectable } from '@nestjs/common';
import { ClientProxy } from '@nestjs/microservices';
import { RABBITMQ_CLIENT } from '../shared/messaging/messaging.module';

@Injectable()
export class OrdersService {
  constructor(
    @Inject(RABBITMQ_CLIENT) private readonly rmq: ClientProxy,
  ) {}

  async publishOrderFilled(payload: OrderFilledEvent) {
    // fire-and-forget; RabbitMQ confirms delivery via publisher confirm
    this.rmq.emit('order.filled', payload);
  }
}

Consuming events​

// apps/api/src/portfolio/portfolio.consumer.ts
import { Controller } from '@nestjs/common';
import { EventPattern, Payload, Ctx, RmqContext } from '@nestjs/microservices';
import { PortfolioService } from './portfolio.service';

@Controller()
export class PortfolioConsumer {
  constructor(private readonly portfolioService: PortfolioService) {}

  @EventPattern('order.filled')
  async handleOrderFilled(
    @Payload() event: OrderFilledEvent,
    @Ctx() context: RmqContext,
  ) {
    await this.portfolioService.applyFill(event);
    // Manual ack after successful processing
    context.getChannelRef().ack(context.getMessage());
  }
}

orders.service.ts​

import { Injectable, ConflictException, NotFoundException, Logger } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { DataSource, Repository } from 'typeorm';
import { Order, OrderStatus } from './order.entity';
import { OutboxEntry } from './outbox.entity';
import { CreateOrderDto } from './dto/create-order.dto';
import { BrokerAdapterService } from './broker-adapter.service';

@Injectable()
export class OrdersService {
  private readonly logger = new Logger(OrdersService.name);

  constructor(
    @InjectRepository(Order) private readonly orders: Repository<Order>,
    @InjectRepository(OutboxEntry) private readonly outbox: Repository<OutboxEntry>,
    private readonly brokerAdapter: BrokerAdapterService,
    private readonly dataSource: DataSource,
  ) {}

  async create(accountId: string, dto: CreateOrderDto, idempotencyKey: string) {
    // Idempotency: return existing order if key already seen
    const existing = await this.orders.findOne({ where: { idempotencyKey } });
    if (existing) return existing;

    return this.dataSource.transaction(async (manager) => {
      const order = manager.create(Order, {
        account: { id: accountId },
        symbol: dto.symbol.toUpperCase(),
        side: dto.side,
        orderType: dto.orderType,
        qty: dto.qty,
        limitPrice: dto.limitPrice ?? null,
        status: OrderStatus.PENDING,
        idempotencyKey,
      });
      await manager.save(order);

      // Outbox entry — picked up by OutboxWorkerService and sent to broker
      await manager.save(OutboxEntry, {
        orderId: order.id,
        payload: JSON.stringify(dto),
        processedAt: null,
      });

      return order;
    });
  }

  async markAccepted(orderId: string, brokerOrderId: string) {
    await this.orders.update(orderId, { status: OrderStatus.ACCEPTED, brokerOrderId });
  }

  async markFilled(orderId: string, avgFillPrice: string) {
    await this.orders.update(orderId, {
      status: OrderStatus.FILLED,
      avgFillPrice,
      filledAt: new Date(),
    });
  }

  async cancel(accountId: string, orderId: string) {
    const order = await this.orders.findOne({
      where: { id: orderId, account: { id: accountId } },
    });
    if (!order) throw new NotFoundException('Order not found');
    if (order.status !== OrderStatus.ACCEPTED && order.status !== OrderStatus.PENDING) {
      throw new ConflictException(`Cannot cancel order in status: ${order.status}`);
    }
    await this.brokerAdapter.cancelOrder(order.brokerOrderId!);
    await this.orders.update(orderId, { status: OrderStatus.CANCELLED });
    return { cancelled: true };
  }

  async findAll(accountId: string) {
    return this.orders.find({
      where: { account: { id: accountId } },
      order: { placedAt: 'DESC' },
      take: 100,
    });
  }
}

8. Market Data — Quote Caching in Redis​

// apps/api/src/market-data/market-data.service.ts
import { Injectable, Inject, Logger, OnModuleInit } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { Redis } from 'ioredis';
import WebSocket from 'ws';
import { REDIS_CLIENT } from '../shared/redis/redis.module';

const QUOTE_TTL_SECONDS = 2;

@Injectable()
export class MarketDataService implements OnModuleInit {
  private readonly logger = new Logger(MarketDataService.name);
  private providerWs: WebSocket | null = null;

  constructor(
    @Inject(REDIS_CLIENT) private readonly redis: Redis,
    private readonly config: ConfigService,
  ) {}

  onModuleInit() {
    this.connectProvider();
  }

  private connectProvider() {
    const url = this.config.getOrThrow<string>('MARKET_DATA_WS_URL');
    this.providerWs = new WebSocket(url);

    this.providerWs.on('open', () => {
      this.logger.log('Connected to market data provider');
    });

    this.providerWs.on('message', async (raw) => {
      try {
        const quote = JSON.parse(raw.toString()) as ProviderQuote;
        await this.cacheAndPublish(quote);
      } catch (err) {
        this.logger.error(`Failed to process quote: ${(err as Error).message}`);
      }
    });

    this.providerWs.on('close', () => {
      this.logger.warn('Provider WS closed — reconnecting in 2s');
      setTimeout(() => this.connectProvider(), 2_000);
    });
  }

  private async cacheAndPublish(quote: ProviderQuote) {
    const key = `quote:${quote.symbol}`;
    const payload = JSON.stringify({
      type: 'quote',
      symbol: quote.symbol,
      price: quote.price,
      bid: quote.bid,
      ask: quote.ask,
      volume: quote.volume,
      ts: Date.now(),
    });

    // Cache latest quote with TTL
    await this.redis.set(key, payload, 'EX', QUOTE_TTL_SECONDS);

    // Fan-out to WebSocket gateway subscribers via Redis Pub/Sub
    await this.redis.publish(key, payload);
  }

  async getQuote(symbol: string): Promise<string | null> {
    return this.redis.get(`quote:${symbol.toUpperCase()}`);
  }
}

const API_BASE = process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001/v1';

async function apiFetch<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(`${API_BASE}${path}`, {
    ...init,
    headers: {
      'Content-Type': 'application/json',
      ...init?.headers,
    },
    credentials: 'include',
  });

  if (!res.ok) {
    const body = await res.json().catch(() => ({}));
    throw new Error(body?.message ?? `API error ${res.status}`);
  }

  return res.json() as Promise<T>;
}

Live quotes hook — apps/web/lib/ws.ts​

'use client';

import { useEffect, useRef, useState, useCallback } from 'react';

interface Quote {
  symbol: string;
  price: number;
  bid: number;
  ask: number;
  ts: number;
}

const WS_URL = process.env.NEXT_PUBLIC_WS_URL ?? 'ws://localhost:3001/ws';
const RECONNECT_DELAY_MS = 2_000;
const MAX_RECONNECT_DELAY_MS = 30_000;

export function useQuotes(symbols: string[]) {
  const [quotes, setQuotes] = useState<Record<string, Quote>>({});
  const ws = useRef<WebSocket | null>(null);
  const delay = useRef(RECONNECT_DELAY_MS);
  const mounted = useRef(true);

  const connect = useCallback(() => {
    if (!mounted.current) return;

    const token = localStorage.getItem('access_token');
    const socket = new WebSocket(`${WS_URL}?token=${token}`);
    ws.current = socket;

    socket.onopen = () => {
      delay.current = RECONNECT_DELAY_MS;
      if (symbols.length > 0) {
        socket.send(JSON.stringify({ type: 'subscribe', symbols }));
      }
    };

    socket.onmessage = (ev) => {
      const frame = JSON.parse(ev.data as string);
      if (frame.type === 'quote') {
        setQuotes((prev) => ({ ...prev, [frame.symbol]: frame as Quote }));
      }
      if (frame.type === 'reconnect') {
        socket.close();
      }
    };

    socket.onclose = () => {
      if (!mounted.current) return;
      setTimeout(connect, delay.current);
      delay.current = Math.min(delay.current * 2, MAX_RECONNECT_DELAY_MS);
    };
  }, [symbols]);

  useEffect(() => {
    mounted.current = true;
    connect();
    return () => {
      mounted.current = false;
      ws.current?.close();
    };
  }, [connect]);

  const subscribe = useCallback((newSymbols: string[]) => {
    ws.current?.send(JSON.stringify({ type: 'subscribe', symbols: newSymbols }));
  }, []);

  const unsubscribe = useCallback((removeSymbols: string[]) => {
    ws.current?.send(JSON.stringify({ type: 'unsubscribe', symbols: removeSymbols }));
  }, []);

  return { quotes, subscribe, unsubscribe };
}

import { useQuotes } from '@/lib/ws';

interface Props { initialItems: WatchlistItem[] }

export function WatchlistClient({ initialItems }: Props) {

          );
        })}
      </tbody>
    </table>
  );
}

Upgrade Path​

This MVP stack is designed to evolve without rewrites:

WhenUpgrade
Quote fan-out latency > 100 msExtract WebSocket Gateway to a dedicated service (see WebSocket Gateway)
Order p99 > 500 msProfile and extract hot path to Go service
RabbitMQ throughput saturated or replay neededMigrate events to Kafka — consumer interface unchanged if using event pattern
synchronize: true causes issuesAdd TypeORM migrations: pnpm typeorm migration:generate
Multi-region neededIntroduce Cloud SQL read replica + Redis cluster + Kafka
