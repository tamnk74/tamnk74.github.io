Google Analytics Setup for Next.js

A complete guide to implementing Google Analytics 4 (GA4) in Next.js applications, covering both App Router and Pages Router architectures with TypeScript support.

Prerequisites​

Required​

  • Next.js 12.0.0 or higher
  • Node.js 16+
  • Google Analytics 4 property with Measurement ID (format: G-XXXXXXXXXX)
  • Basic understanding of React and Next.js routing

Recommended​

  • TypeScript for type safety
  • Environment variable management
  • Basic understanding of Google Analytics concepts

App Router Setup (Next.js 13+)​

Method 1: Using @next/third-parties (Recommended)​

This is the official Next.js solution with optimal loading strategy.

Installation​

npm install @next/third-parties@latest

Implementation​

1. Create Analytics Component

// app/components/analytics.tsx
'use client';

import { GoogleAnalytics } from '@next/third-parties/google';

  const measurementId = process.env.NEXT_PUBLIC_GA_MEASUREMENT_ID;

Method 3: Manual Script Injection (Maximum Control)​

For advanced use cases requiring custom configuration.

Implementation​

1. Create Analytics Provider

// app/components/analytics-provider.tsx
'use client';

import Script from 'next/script';
import { usePathname, useSearchParams } from 'next/navigation';
import { useEffect } from 'react';

declare global {
  interface Window {
    gtag: (command: string, ...args: any[]) => void;
    dataLayer: any[];
  }
}

export function AnalyticsProvider() {
  const pathname = usePathname();
  const searchParams = useSearchParams();
  const measurementId = process.env.NEXT_PUBLIC_GA_MEASUREMENT_ID;

  useEffect(() => {
    if (!measurementId || !window.gtag) return;

    const url = pathname + searchParams.toString();

    window.gtag('config', measurementId, {
      page_path: url,
    });
  }, [pathname, searchParams, measurementId]);

  if (!measurementId) return null;

  return (
    <>
      <Script strategy="afterInteractive" src={`https://www.googletagmanager.com/gtag/js?id=${measurementId}`} />
      <Script
        id="google-analytics"
        strategy="afterInteractive"
        dangerouslySetInnerHTML={{
          __html: `
            window.dataLayer = window.dataLayer || [];
            function gtag(){dataLayer.push(arguments);}
            gtag('js', new Date());
            gtag('config', '${measurementId}', {
              page_path: window.location.pathname,
              send_page_view: false
            });
          `,
        }}
      />
    </>
  );
}

3. Create Custom Hook for Events

// app/hooks/use-analytics.ts
'use client';

import { useCallback } from 'react';

interface EventParams {
  [key: string]: any;
}

export function useAnalytics() {
  const trackEvent = useCallback((eventName: string, params?: EventParams) => {
    if (typeof window !== 'undefined' && window.gtag) {
      window.gtag('event', eventName, params);
    }
  }, []);

  const trackPageView = useCallback((url: string) => {
    if (typeof window !== 'undefined' && window.gtag) {
      window.gtag('config', process.env.NEXT_PUBLIC_GA_MEASUREMENT_ID!, {
        page_path: url,
      });
    }
  }, []);

  const setUserProperties = useCallback((properties: EventParams) => {
    if (typeof window !== 'undefined' && window.gtag) {
      window.gtag('set', 'user_properties', properties);
    }
  }, []);

  return {
    trackEvent,
    trackPageView,
    setUserProperties,
  };
}

Pages Router Setup (Next.js 12 and below)​

Using Custom _app.tsx​

1. Install Dependencies

npm install react-ga4

2. Configure _app.tsx

// pages/_app.tsx
import type { AppProps } from 'next/app';
import { useRouter } from 'next/router';
import { useEffect } from 'react';
import ReactGA from 'react-ga4';

const GA_MEASUREMENT_ID = process.env.NEXT_PUBLIC_GA_MEASUREMENT_ID;

function MyApp({ Component, pageProps }: AppProps) {
  const router = useRouter();

  useEffect(() => {
    if (!GA_MEASUREMENT_ID) return;

    // Initialize GA
    ReactGA.initialize(GA_MEASUREMENT_ID, {
      gaOptions: {
        siteSpeedSampleRate: 100,
      },
    });

    // Track initial page load
    ReactGA.send({
      hitType: 'pageview',
      page: router.pathname,
    });
  }, []);

  useEffect(() => {
    const handleRouteChange = (url: string) => {
      ReactGA.send({
        hitType: 'pageview',
        page: url,
      });
    };

    router.events.on('routeChangeComplete', handleRouteChange);

    return () => {
      router.events.off('routeChangeComplete', handleRouteChange);
    };
  }, [router.events]);

  return <Component {...pageProps} />;
}

export default MyApp;

Event Tracking​

Standard E-commerce Events​

// lib/analytics/ecommerce.ts
export const ecommerceEvents = {
  // View Item
  viewItem: (item: { id: string; name: string; category: string; price: number }) => {
    window.gtag?.('event', 'view_item', {
      currency: 'USD',
      value: item.price,
      items: [
        {
          item_id: item.id,
          item_name: item.name,
          item_category: item.category,
          price: item.price,
        },
      ],
    });
  },

  // Add to Cart
  addToCart: (item: { id: string; name: string; price: number; quantity: number }) => {
    window.gtag?.('event', 'add_to_cart', {
      currency: 'USD',
      value: item.price * item.quantity,
      items: [
        {
          item_id: item.id,
          item_name: item.name,
          price: item.price,
          quantity: item.quantity,
        },
      ],
    });
  },

  // Remove from Cart
  removeFromCart: (item: { id: string; name: string; price: number; quantity: number }) => {
    window.gtag?.('event', 'remove_from_cart', {
      currency: 'USD',
      value: item.price * item.quantity,
      items: [
        {
          item_id: item.id,
          item_name: item.name,
          price: item.price,
          quantity: item.quantity,
        },
      ],
    });
  },

  // Begin Checkout
  beginCheckout: (items: any[], value: number) => {
    window.gtag?.('event', 'begin_checkout', {
      currency: 'USD',
      value,
      items,
    });
  },

  // Purchase
  purchase: (transaction: {
    transactionId: string;
    value: number;
    tax: number;
    shipping: number;
    items: any[];
    coupon?: string;
  }) => {
    window.gtag?.('event', 'purchase', {
      transaction_id: transaction.transactionId,
      value: transaction.value,
      tax: transaction.tax,
      shipping: transaction.shipping,
      currency: 'USD',
      coupon: transaction.coupon,
      items: transaction.items,
    });
  },
};

Privacy & Consent Management​

GDPR Compliant Consent Mode v2​

// app/components/consent-manager.tsx
'use client';

import Script from 'next/script';
import { useState, useEffect } from 'react';

export function ConsentManager() {
  const [consentGiven, setConsentGiven] = useState(false);
  const measurementId = process.env.NEXT_PUBLIC_GA_MEASUREMENT_ID;

  useEffect(() => {
    // Check for existing consent
    const consent = localStorage.getItem('cookie-consent');
    if (consent) {
      setConsentGiven(JSON.parse(consent));
    }
  }, []);

  const handleAcceptConsent = () => {
    setConsentGiven(true);
    localStorage.setItem('cookie-consent', 'true');

    // Update consent
    if (window.gtag) {
      window.gtag('consent', 'update', {
        ad_storage: 'granted',
        ad_user_data: 'granted',
        ad_personalization: 'granted',
        analytics_storage: 'granted',
      });
    }
  };

  const handleRejectConsent = () => {
    setConsentGiven(false);
    localStorage.setItem('cookie-consent', 'false');
  };

  return (
    <>
      {/* Default Consent State (Before user choice) */}
      <Script
        id="google-consent-default"
        strategy="beforeInteractive"
        dangerouslySetInnerHTML={{
          __html: `
            window.dataLayer = window.dataLayer || [];
            function gtag(){dataLayer.push(arguments);}

            gtag('consent', 'default', {
              'ad_storage': 'denied',
              'ad_user_data': 'denied',
              'ad_personalization': 'denied',
              'analytics_storage': 'denied',
              'functionality_storage': 'denied',
              'personalization_storage': 'denied',
              'security_storage': 'granted',
              'wait_for_update': 500
            });
          `,
        }}
      />

      {/* Analytics Script */}
      <Script strategy="afterInteractive" src={`https://www.googletagmanager.com/gtag/js?id=${measurementId}`} />
      <Script
        id="google-analytics"
        strategy="afterInteractive"
        dangerouslySetInnerHTML={{
          __html: `
            window.dataLayer = window.dataLayer || [];
            function gtag(){dataLayer.push(arguments);}
            gtag('js', new Date());
            gtag('config', '${measurementId}');
          `,
        }}
      />

      {/* Consent Banner */}
      {!consentGiven && (
        <div className="fixed bottom-0 left-0 right-0 bg-gray-900 text-white p-4 z-50">
          <div className="container mx-auto flex items-center justify-between">
            <p>We use cookies to improve your experience. By using our site, you agree to our cookie policy.</p>
            <div className="flex gap-2">
              <button onClick={handleAcceptConsent} className="bg-blue-600 px-4 py-2 rounded">
                Accept
              </button>
              <button onClick={handleRejectConsent} className="bg-gray-600 px-4 py-2 rounded">
                Reject
              </button>
            </div>
          </div>
        </div>
      )}
    </>
  );
}

Performance Optimization​

Lazy Loading Analytics​

// app/components/lazy-analytics.tsx
'use client';

import { useEffect, useState } from 'react';
import dynamic from 'next/dynamic';

const Analytics = dynamic(() => import('./analytics'), {
  ssr: false,
});

export function LazyAnalytics() {
  const [shouldLoad, setShouldLoad] = useState(false);

  useEffect(() => {
    // Load after user interaction or timeout
    const timer = setTimeout(() => {
      setShouldLoad(true);
    }, 3000);

    const handleInteraction = () => {
      setShouldLoad(true);
      window.removeEventListener('scroll', handleInteraction);
      window.removeEventListener('click', handleInteraction);
    };

    window.addEventListener('scroll', handleInteraction, { once: true });
    window.addEventListener('click', handleInteraction, { once: true });

    return () => {
      clearTimeout(timer);
      window.removeEventListener('scroll', handleInteraction);
      window.removeEventListener('click', handleInteraction);
    };
  }, []);

  if (!shouldLoad) return null;

  return <Analytics />;
}

2. Centralized Analytics Module​

// lib/analytics/index.ts
import { ecommerceEvents } from './ecommerce';
import { customEvents } from './custom-events';
import { config } from './config';

class Analytics {
  private initialized = false;

  init() {
    if (this.initialized || !config.measurementId) return;

    // Initialization logic
    this.initialized = true;
    console.log('Analytics initialized');
  }

  trackPageView(url: string) {
    if (!this.initialized) return;

    window.gtag?.('config', config.measurementId!, {
      page_path: url,
    });
  }

  trackEvent(eventName: string, params?: Record<string, any>) {
    if (!this.initialized) return;

    window.gtag?.('event', eventName, params);
  }

  // Expose sub-modules
  ecommerce = ecommerceEvents;
  custom = customEvents;
}

export const analytics = new Analytics();

3. Error Boundaries​

// app/components/analytics-error-boundary.tsx
'use client';

import { Component, ReactNode } from 'react';

interface Props {
  children: ReactNode;
}

interface State {
  hasError: boolean;
}

export class AnalyticsErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props);
    this.state = { hasError: false };
  }

  static getDerivedStateFromError() {
    return { hasError: true };
  }

  componentDidCatch(error: Error, errorInfo: any) {
    console.error('Analytics Error:', error, errorInfo);

    // Track the error
    window.gtag?.('event', 'exception', {
      description: error.message,
      fatal: false,
    });
  }

  render() {
    if (this.state.hasError) {
      return this.props.children; // Fail gracefully
    }

    return this.props.children;
  }
}

4. Data Layer Management​

// lib/analytics/data-layer.ts
export const dataLayer = {
  push: (data: Record<string, any>) => {
    if (typeof window !== 'undefined') {
      window.dataLayer = window.dataLayer || [];
      window.dataLayer.push(data);
    }
  },

  getUserProperties: () => {
    if (typeof window === 'undefined') return {};

    return {
      user_id: localStorage.getItem('userId'),
      user_type: localStorage.getItem('userType'),
      is_logged_in: !!localStorage.getItem('authToken'),
    };
  },

  setUserProperties: (properties: Record<string, any>) => {
    window.gtag?.('set', 'user_properties', properties);
  },
};

5. Server-Side Tracking (API Routes)​

// app/api/analytics/route.ts
import { NextRequest, NextResponse } from 'next/server';

export async function POST(request: NextRequest) {
  const { event, params, clientId } = await request.json();

  // Send to GA4 Measurement Protocol
  const measurementId = process.env.NEXT_PUBLIC_GA_MEASUREMENT_ID;
  const apiSecret = process.env.GA_MEASUREMENT_API_SECRET;

  const response = await fetch(
    `https://www.google-analytics.com/mp/collect?measurement_id=${measurementId}&api_secret=${apiSecret}`,
    {
      method: 'POST',
      body: JSON.stringify({
        client_id: clientId,
        events: [
          {
            name: event,
            params,
          },
        ],
      }),
    },
  );

  return NextResponse.json({ success: response.ok });
}
