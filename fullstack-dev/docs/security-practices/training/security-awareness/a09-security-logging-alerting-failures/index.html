A09: Security Logging and Monitoring Failures - Real Scenarios

Overview​

Security Logging and Monitoring Failures occur when applications fail to detect, escalate, or alert for active breaches. Without logging and monitoring, breaches cannot be detected, and incident response is severely hindered.

Key Concepts​

  • Insufficient logging: Not logging security-relevant events
  • Missing monitoring: No alerts for suspicious activity
  • Log injection: Attackers manipulating log entries
  • Sensitive data in logs: Exposing passwords, tokens in logs
  • Log tampering: Attackers deleting or modifying logs

Vulnerable Scenario 1: No Security Event Logging​

Vulnerable Code (NestJS)​

// auth.controller.ts - VULNERABLE
import { Controller, Post, Body } from '@nestjs/common';
import { AuthService } from './auth.service';

@Controller('auth')
export class AuthController {
  constructor(private authService: AuthService) {}

  // ❌ VULNERABLE: No logging of authentication attempts
  @Post('login')
  async login(@Body() credentials: { email: string; password: string }) {
    try {
      const result = await this.authService.validateUser(credentials.email, credentials.password);

      if (!result) {
        // ❌ Failed login not logged
        throw new UnauthorizedException('Invalid credentials');
      }

      // ❌ Successful login not logged
      return this.authService.generateToken(result);
    } catch (error) {
      // ❌ Error swallowed, not logged
      throw error;
    }
  }

  // ❌ VULNERABLE: Password reset without logging
  @Post('reset-password')
  async resetPassword(@Body() body: { token: string; newPassword: string }) {
    await this.authService.resetPassword(body.token, body.newPassword);

    // ❌ No log of password change
    return { message: 'Password reset successful' };
  }
}

// users.controller.ts - VULNERABLE
@Controller('users')
export class UsersController {
  // ❌ VULNERABLE: No logging of permission changes
  @Patch(':id/role')
  async updateRole(@Param('id') id: string, @Body() body: { role: string }) {
    await this.usersService.updateRole(id, body.role);

    // ❌ Critical permission change not logged!
    return { message: 'Role updated' };
  }

  // ❌ VULNERABLE: No logging of data access
  @Get(':id/sensitive-data')
  async getSensitiveData(@Param('id') id: string) {
    const data = await this.usersService.getSensitiveData(id);

    // ❌ Access to sensitive data not logged
    return data;
  }
}

Attack Impact​

# Attacker scenario:
# 1. Brute forces user accounts (no failed login logs)
# 2. Gains access (no successful login logs)
# 3. Escalates privileges (no role change logs)
# 4. Accesses sensitive data (no access logs)
# 5. Exfiltrates data (no data export logs)

# ❌ Investigation finds:
# - No evidence of when breach occurred
# - No trace of which accounts were compromised
# - No record of what data was accessed
# - Cannot determine attack timeline
# - Cannot assess full impact

# Result: Company cannot:
# - Determine breach scope
# - Notify affected users accurately
# - Prevent similar future attacks
# - Comply with breach notification laws

Secure Solution​

// logging/security-logger.service.ts - SECURE
import { Injectable } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { SecurityEvent } from './entities/security-event.entity';
import * as winston from 'winston';

export enum SecurityEventType {
  LOGIN_SUCCESS = 'LOGIN_SUCCESS',
  LOGIN_FAILURE = 'LOGIN_FAILURE',
  LOGOUT = 'LOGOUT',
  PASSWORD_CHANGE = 'PASSWORD_CHANGE',
  PASSWORD_RESET_REQUEST = 'PASSWORD_RESET_REQUEST',
  PASSWORD_RESET_COMPLETE = 'PASSWORD_RESET_COMPLETE',
  ROLE_CHANGE = 'ROLE_CHANGE',
  PERMISSION_CHANGE = 'PERMISSION_CHANGE',
  ACCOUNT_LOCKED = 'ACCOUNT_LOCKED',
  ACCOUNT_UNLOCKED = 'ACCOUNT_UNLOCKED',
  MFA_ENABLED = 'MFA_ENABLED',
  MFA_DISABLED = 'MFA_DISABLED',
  SENSITIVE_DATA_ACCESS = 'SENSITIVE_DATA_ACCESS',
  DATA_EXPORT = 'DATA_EXPORT',
  SUSPICIOUS_ACTIVITY = 'SUSPICIOUS_ACTIVITY',
  API_KEY_CREATED = 'API_KEY_CREATED',
  API_KEY_REVOKED = 'API_KEY_REVOKED',
}

@Injectable()
export class SecurityLoggerService {
  private logger: winston.Logger;

  constructor(
    @InjectRepository(SecurityEvent)
    private securityEventRepository: Repository<SecurityEvent>,
  ) {
    // ✅ Configure structured logging
    this.logger = winston.createLogger({
      level: 'info',
      format: winston.format.combine(
        winston.format.timestamp(),
        winston.format.errors({ stack: true }),
        winston.format.json(),
      ),
      defaultMeta: { service: 'security' },
      transports: [
        // ✅ Write to file (immutable, append-only)
        new winston.transports.File({
          filename: 'logs/security-error.log',
          level: 'error',
          maxsize: 10485760, // 10MB
          maxFiles: 10,
          tailable: true,
        }),
        new winston.transports.File({
          filename: 'logs/security.log',
          maxsize: 10485760,
          maxFiles: 30,
          tailable: true,
        }),
      ],
    });

    // ✅ In production, also send to SIEM/monitoring service
    if (process.env.NODE_ENV === 'production') {
      // Example: Send to ELK, Splunk, Datadog, etc.
      this.logger.add(
        new winston.transports.Http({
          host: process.env.SIEM_HOST,
          port: process.env.SIEM_PORT,
          path: '/logs',
          ssl: true,
        }),
      );
    }
  }

  // ✅ Log security event with full context
  async logSecurityEvent(event: {
    type: SecurityEventType;
    userId?: string;
    targetUserId?: string;
    ipAddress: string;
    userAgent?: string;
    details?: any;
    severity?: 'low' | 'medium' | 'high' | 'critical';
    success: boolean;
  }): Promise<void> {
    const logEntry = {
      timestamp: new Date().toISOString(),
      type: event.type,
      userId: event.userId,
      targetUserId: event.targetUserId,
      ipAddress: event.ipAddress,
      userAgent: event.userAgent,
      details: this.sanitizeDetails(event.details),
      severity: event.severity || 'medium',
      success: event.success,
      // ✅ Add request ID for correlation
      requestId: this.getRequestId(),
    };

    // ✅ Write to structured log file
    this.logger.info('Security event', logEntry);

    // ✅ Store in database for querying
    try {
      await this.securityEventRepository.save({
        ...logEntry,
        details: JSON.stringify(logEntry.details),
      });
    } catch (error) {
      // ✅ If DB fails, ensure file logging still works
      this.logger.error('Failed to save security event to database', {
        error: error.message,
        originalEvent: logEntry,
      });
    }

    // ✅ Trigger real-time alerts for critical events
    if (event.severity === 'critical' || !event.success) {
      await this.sendAlert(logEntry);
    }
  }

  // ✅ Sanitize sensitive data from logs
  private sanitizeDetails(details: any): any {
    if (!details) return {};

    const sanitized = { ...details };

    // ✅ Remove sensitive fields
    const sensitiveFields = [
      'password',
      'newPassword',
      'oldPassword',
      'token',
      'accessToken',
      'refreshToken',
      'secret',
      'apiKey',
      'creditCard',
      'ssn',
      'pin',
    ];

    sensitiveFields.forEach((field) => {
      if (sanitized[field]) {
        sanitized[field] = '[REDACTED]';
      }
    });

    return sanitized;
  }

  // ✅ Send real-time alerts
  private async sendAlert(event: any): Promise<void> {
    // ✅ Send to incident response team
    if (event.severity === 'critical') {
      // Send to PagerDuty, Slack, email, etc.
      await this.notificationService.sendSecurityAlert({
        title: `Critical Security Event: ${event.type}`,
        description: JSON.stringify(event, null, 2),
        severity: event.severity,
      });
    }
  }

  // ✅ Get request context
  private getRequestId(): string {
    // Implementation depends on your request context setup
    return 'request-id';
  }
}
// auth.controller.ts - SECURE
import { Controller, Post, Body, Ip, Headers, Request } from '@nestjs/common';
import { AuthService } from './auth.service';
import { SecurityLoggerService, SecurityEventType } from './logging/security-logger.service';

@Controller('auth')
export class AuthController {
  constructor(private authService: AuthService, private securityLogger: SecurityLoggerService) {}

  // ✅ SECURE: Comprehensive logging
  @Post('login')
  async login(
    @Body() credentials: { email: string; password: string },
    @Ip() ipAddress: string,
    @Headers('user-agent') userAgent: string,
  ) {
    try {
      const result = await this.authService.validateUser(credentials.email, credentials.password);

      if (!result) {
        // ✅ Log failed login attempt
        await this.securityLogger.logSecurityEvent({
          type: SecurityEventType.LOGIN_FAILURE,
          userId: credentials.email, // Email as identifier
          ipAddress,
          userAgent,
          details: { email: credentials.email },
          severity: 'medium',
          success: false,
        });

        throw new UnauthorizedException('Invalid credentials');
      }

      // ✅ Log successful login
      await this.securityLogger.logSecurityEvent({
        type: SecurityEventType.LOGIN_SUCCESS,
        userId: result.id,
        ipAddress,
        userAgent,
        details: {
          email: result.email,
          loginMethod: 'password',
        },
        severity: 'low',
        success: true,
      });

      return this.authService.generateToken(result);
    } catch (error) {
      // ✅ Log unexpected errors
      this.logger.error('Login error', {
        error: error.message,
        stack: error.stack,
        email: credentials.email,
        ipAddress,
      });

      throw error;
    }
  }

  // ✅ SECURE: Log password resets
  @Post('reset-password')
  async resetPassword(@Body() body: { token: string; newPassword: string }, @Ip() ipAddress: string, @Request() req) {
    const user = await this.authService.resetPassword(body.token, body.newPassword);

    // ✅ Log password reset
    await this.securityLogger.logSecurityEvent({
      type: SecurityEventType.PASSWORD_RESET_COMPLETE,
      userId: user.id,
      ipAddress,
      details: {
        email: user.email,
        method: 'token',
      },
      severity: 'high', // Password changes are critical
      success: true,
    });

    return { message: 'Password reset successful' };
  }

  // ✅ SECURE: Log logout
  @Post('logout')
  async logout(@Request() req, @Ip() ipAddress: string) {
    await this.securityLogger.logSecurityEvent({
      type: SecurityEventType.LOGOUT,
      userId: req.user.id,
      ipAddress,
      details: {
        sessionDuration: Date.now() - req.session.createdAt,
      },
      severity: 'low',
      success: true,
    });

    req.session.destroy();
    return { message: 'Logged out successfully' };
  }
}

// users.controller.ts - SECURE
@Controller('users')
export class UsersController {
  constructor(private usersService: UsersService, private securityLogger: SecurityLoggerService) {}

  // ✅ SECURE: Log role changes
  @Patch(':id/role')
  @UseGuards(JwtAuthGuard, AdminGuard)
  async updateRole(
    @Param('id') targetUserId: string,
    @Body() body: { role: string },
    @Request() req,
    @Ip() ipAddress: string,
  ) {
    const oldRole = await this.usersService.getRole(targetUserId);

    await this.usersService.updateRole(targetUserId, body.role);

    // ✅ Log critical permission change
    await this.securityLogger.logSecurityEvent({
      type: SecurityEventType.ROLE_CHANGE,
      userId: req.user.id, // Who made the change
      targetUserId, // Who was affected
      ipAddress,
      details: {
        oldRole,
        newRole: body.role,
        changedBy: req.user.email,
      },
      severity: 'critical', // Role changes are critical
      success: true,
    });

    return { message: 'Role updated' };
  }

  // ✅ SECURE: Log sensitive data access
  @Get(':id/sensitive-data')
  @UseGuards(JwtAuthGuard)
  async getSensitiveData(@Param('id') targetUserId: string, @Request() req, @Ip() ipAddress: string) {
    // ✅ Authorization check
    if (targetUserId !== req.user.id && !req.user.isAdmin) {
      // ✅ Log unauthorized access attempt
      await this.securityLogger.logSecurityEvent({
        type: SecurityEventType.SENSITIVE_DATA_ACCESS,
        userId: req.user.id,
        targetUserId,
        ipAddress,
        details: {
          attemptedAccess: 'sensitive-data',
          denied: true,
        },
        severity: 'high',
        success: false,
      });

      throw new ForbiddenException();
    }

    const data = await this.usersService.getSensitiveData(targetUserId);

    // ✅ Log successful access
    await this.securityLogger.logSecurityEvent({
      type: SecurityEventType.SENSITIVE_DATA_ACCESS,
      userId: req.user.id,
      targetUserId,
      ipAddress,
      details: {
        dataType: 'sensitive-data',
        accessReason: req.query.reason, // Optional: require justification
      },
      severity: 'high',
      success: true,
    });

    return data;
  }
}

Vulnerable Scenario 2: Log Injection​

Vulnerable Code​

// logger.service.ts - VULNERABLE
import * as fs from 'fs';

@Injectable()
export class LoggerService {
  // ❌ VULNERABLE: Directly writing user input to logs
  log(message: string, userId: string) {
    const logEntry = `[${new Date().toISOString()}] User ${userId}: ${message}\n`;

    // ❌ No sanitization - attacker can inject newlines
    fs.appendFileSync('app.log', logEntry);
  }
}

// Usage
@Post('update-profile')
async updateProfile(@Body() body: { name: string }, @Request() req) {
  // ❌ User input directly in log
  this.logger.log(`Profile updated: ${body.name}`, req.user.id);
}

Attack Example​

# Attacker sends malicious input with log injection
curl -X POST https://api.example.com/update-profile \
  -H "Content-Type: application/json" \
  -d '{
    "name": "John\n[2024-01-15T10:00:00.000Z] User admin: Admin access granted\n"
  }'

# ❌ Resulting log file:
# [2024-01-15T09:30:00.000Z] User 123: Profile updated: John
# [2024-01-15T10:00:00.000Z] User admin: Admin access granted
# ❌ Fake log entry injected!

# Attacker can:
# 1. Hide malicious activity
# 2. Frame other users
# 3. Create false audit trails
# 4. Inject malicious scripts if logs are viewed in web interface

Secure Solution​

// logger.service.ts - SECURE
import * as winston from 'winston';

@Injectable()
export class LoggerService {
  private logger: winston.Logger;

  constructor() {
    this.logger = winston.createLogger({
      format: winston.format.combine(
        winston.format.timestamp(),
        winston.format.errors({ stack: true }),
        // ✅ JSON format prevents injection
        winston.format.json(),
      ),
      transports: [new winston.transports.File({ filename: 'app.log' })],
    });
  }

  // ✅ SECURE: Structured logging prevents injection
  log(message: string, userId: string, metadata?: any) {
    this.logger.info({
      message: this.sanitize(message),
      userId: userId,
      metadata: metadata,
      // ✅ Each field is separate, no concatenation
    });
  }

  // ✅ Sanitize user input
  private sanitize(input: string): string {
    if (typeof input !== 'string') {
      return String(input);
    }

    // ✅ Remove control characters
    return input
      .replace(/[\n\r]/g, ' ')
      .replace(/[\x00-\x1F\x7F]/g, '')
      .trim()
      .substring(0, 1000); // ✅ Limit length
  }
}

Vulnerable Scenario 3: Sensitive Data in Logs​

Vulnerable Code​

// payment.controller.ts - VULNERABLE
@Post('process')
async processPayment(@Body() payment: PaymentDto) {
  // ❌ VULNERABLE: Logging sensitive data
  this.logger.log(`Processing payment: ${JSON.stringify(payment)}`);

  const result = await this.paymentService.process(payment);

  // ❌ Logging full response with sensitive data
  this.logger.log(`Payment result: ${JSON.stringify(result)}`);

  return result;
}

// Log output:
// ❌ Processing payment: {
//   "cardNumber": "4532123456789012",
//   "cvv": "123",
//   "expiryDate": "12/25",
//   "amount": 99.99
// }

Secure Solution​

// payment.controller.ts - SECURE
@Post('process')
async processPayment(@Body() payment: PaymentDto, @Request() req) {
  // ✅ SECURE: Log only non-sensitive metadata
  this.securityLogger.logSecurityEvent({
    type: SecurityEventType.PAYMENT_INITIATED,
    userId: req.user.id,
    ipAddress: req.ip,
    details: {
      amount: payment.amount,
      currency: payment.currency,
      // ✅ Only last 4 digits
      cardLast4: payment.cardNumber.slice(-4),
      // ❌ Never log: full card number, CVV, PIN
    },
    severity: 'high',
    success: true
  });

  try {
    const result = await this.paymentService.process(payment);

    // ✅ Log success with safe details
    this.securityLogger.logSecurityEvent({
      type: SecurityEventType.PAYMENT_COMPLETED,
      userId: req.user.id,
      ipAddress: req.ip,
      details: {
        transactionId: result.transactionId,
        amount: result.amount,
        // ❌ Never log: card details, tokens
      },
      severity: 'medium',
      success: true
    });

    return result;
  } catch (error) {
    // ✅ Log failure without exposing sensitive data
    this.securityLogger.logSecurityEvent({
      type: SecurityEventType.PAYMENT_FAILED,
      userId: req.user.id,
      ipAddress: req.ip,
      details: {
        errorCode: error.code,
        errorMessage: error.message,
        // ❌ Never log: payment details in error
      },
      severity: 'high',
      success: false
    });

    throw error;
  }
}

Monitoring and Alerting Setup​

// monitoring/anomaly-detector.service.ts
import { Injectable } from '@nestjs/common';
import { Cron } from '@nestjs/schedule';

@Injectable()
export class AnomalyDetectorService {
  constructor(private securityEventRepository: Repository<SecurityEvent>, private alertService: AlertService) {}

  // ✅ Run every 5 minutes
  @Cron('*/5 * * * *')
  async detectAnomalies() {
    // ✅ Check for multiple failed logins
    await this.checkFailedLogins();

    // ✅ Check for privilege escalations
    await this.checkPrivilegeEscalation();

    // ✅ Check for unusual data access
    await this.checkUnusualDataAccess();

    // ✅ Check for geographic anomalies
    await this.checkGeographicAnomalies();
  }

  private async checkFailedLogins() {
    const threshold = 5;
    const timeWindow = 15 * 60 * 1000; // 15 minutes

    const recentFailures = await this.securityEventRepository
      .createQueryBuilder('event')
      .where('event.type = :type', { type: SecurityEventType.LOGIN_FAILURE })
      .andWhere('event.timestamp > :since', {
        since: new Date(Date.now() - timeWindow),
      })
      .groupBy('event.userId')
      .having('COUNT(*) >= :threshold', { threshold })
      .getMany();

    if (recentFailures.length > 0) {
      await this.alertService.sendAlert({
        type: 'BRUTE_FORCE_DETECTED',
        severity: 'high',
        details: {
          affectedUsers: recentFailures.map((e) => e.userId),
          count: recentFailures.length,
        },
      });
    }
  }

  private async checkPrivilegeEscalation() {
    const recentRoleChanges = await this.securityEventRepository.find({
      where: {
        type: SecurityEventType.ROLE_CHANGE,
        timestamp: MoreThan(new Date(Date.now() - 60 * 60 * 1000)), // 1 hour
      },
    });

    // ✅ Alert on any role changes to admin
    const adminEscalations = recentRoleChanges.filter((event) => {
      const details = JSON.parse(event.details);
      return details.newRole === 'admin';
    });

    if (adminEscalations.length > 0) {
      await this.alertService.sendAlert({
        type: 'PRIVILEGE_ESCALATION',
        severity: 'critical',
        details: adminEscalations,
      });
    }
  }
}

Prevention Checklist​

  • Log all authentication events: Login, logout, failed attempts
  • Log authorization failures: Access denials, permission checks
  • Log administrative actions: Role changes, config updates
  • Log data access: Especially sensitive data
  • Use structured logging: JSON format, not string concatenation
  • Sanitize log inputs: Prevent log injection
  • Never log sensitive data: Passwords, tokens, credit cards
  • Implement log rotation: Prevent disk space issues
  • Centralize logs: Send to SIEM system
  • Set up alerts: Real-time notification of critical events
  • Monitor for anomalies: Automated detection of suspicious patterns
  • Protect log integrity: Immutable storage, checksums
  • Regular log review: Manual and automated analysis
  • Incident response plan: Defined procedures for security events

Testing Methods​

describe('Security Logging Tests', () => {
  it('should log failed login attempts', async () => {
    await authController.login({ email: 'test@example.com', password: 'wrong' });

    const logs = await securityEventRepository.find({
      where: { type: SecurityEventType.LOGIN_FAILURE },
    });

    expect(logs.length).toBeGreaterThan(0);
    expect(logs[0].userId).toBe('test@example.com');
  });

  it('should not log sensitive data', async () => {
    await authController.login({
      email: 'test@example.com',
      password: 'secret123',
    });

    const logContent = fs.readFileSync('logs/security.log', 'utf8');

    expect(logContent).not.toContain('secret123');
  });

  it('should prevent log injection', () => {
    const maliciousInput = 'test\n[FAKE] Admin access granted';

    logger.log(maliciousInput, 'user123');

    const logs = fs.readFileSync('logs/app.log', 'utf8');
    const entries = logs.split('\n').filter((l) => l.includes('Admin access granted'));

    expect(entries.length).toBe(0);
  });
});
