A02: Security Misconfiguration - Real Scenarios

Overview​

Security Misconfiguration occurs when security settings are defined, implemented, or maintained improperly. This includes missing security patches, unnecessary features enabled, default accounts/passwords, overly informative error messages, and misconfigured security headers.

Key Concepts​

  • Default configurations: Unchanged default credentials or settings
  • Unnecessary features: Enabled services, pages, accounts not needed
  • Missing patches: Outdated software with known vulnerabilities
  • Verbose errors: Exposing stack traces and system info
  • Misconfigured permissions: Overly permissive file/database access

// middleware.ts - SECURE: Remove debug endpoints in production
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

export function middleware(request: NextRequest) {
  // ✅ Block debug/admin endpoints in production
  if (process.env.NODE_ENV === 'production') {
    const debugPaths = ['/api/debug', '/api/admin/debug', '/.env', '/config'];

    if (debugPaths.some((path) => request.nextUrl.pathname.startsWith(path))) {
      return NextResponse.json({ error: 'Not found' }, { status: 404 });
    }
  }

  return NextResponse.next();
}

Secure Solution​

// app.module.ts - SECURE
import { Module } from '@nestjs/common';
import { ConfigModule, ConfigService } from '@nestjs/config';
import { TypeOrmModule } from '@nestjs/typeorm';

@Module({
  imports: [
    // ✅ Load configuration from environment
    ConfigModule.forRoot({
      isGlobal: true,
      envFilePath: `.env.${process.env.NODE_ENV}`,
      validate: validateConfig, // ✅ Validate required vars
    }),

    // ✅ Conditional admin module (only in development)
    ...(process.env.NODE_ENV === 'development'
      ? [
          AdminModule.createAdmin({
            auth: {
              authenticate: async (email, password) => {
                // ✅ Use environment variables
                if (email === process.env.ADMIN_EMAIL && password === process.env.ADMIN_PASSWORD) {
                  return { email };
                }
                return null;
              },
              cookiePassword: process.env.ADMIN_COOKIE_SECRET,
            },
            // ✅ Restrict accessible resources
            resources: [
              {
                resource: User,
                options: {
                  properties: {
                    password: { isVisible: false }, // ✅ Hide sensitive fields
                    ssn: { isVisible: false },
                  },
                },
              },
            ],
          }),
        ]
      : []),
  ],
})
export class AppModule {}

// ✅ Configuration validation
function validateConfig(config: Record<string, any>) {
  const required = ['DATABASE_URL', 'JWT_SECRET', 'NODE_ENV'];

  for (const key of required) {
    if (!config[key]) {
      throw new Error(`Missing required environment variable: ${key}`);
    }
  }

  // ✅ Validate JWT_SECRET strength
  if (config.JWT_SECRET.length < 32) {
    throw new Error('JWT_SECRET must be at least 32 characters');
  }

  return config;
}
// main.ts - SECURE
import { NestFactory } from '@nestjs/core';
import { AppModule } from './app.module';
import { ConfigService } from '@nestjs/config';
import helmet from 'helmet';
import { DocumentBuilder, SwaggerModule } from '@nestjs/swagger';

async function bootstrap() {
  const app = await NestFactory.create(AppModule, {
    // ✅ Minimal logging in production
    logger: process.env.NODE_ENV === 'production' ? ['error', 'warn'] : ['log', 'error', 'warn', 'debug'],
  });

  const configService = app.get(ConfigService);

  // ✅ Security headers with helmet
  app.use(
    helmet({
      contentSecurityPolicy: {
        directives: {
          defaultSrc: ["'self'"],
          styleSrc: ["'self'", "'unsafe-inline'"],
          scriptSrc: ["'self'"],
          imgSrc: ["'self'", 'data:', 'https:'],
        },
      },
      hsts: {
        maxAge: 31536000,
        includeSubDomains: true,
        preload: true,
      },
    }),
  );

  // ✅ Strict CORS configuration
  const allowedOrigins = configService.get('ALLOWED_ORIGINS')?.split(',') || [];

  app.enableCors({
    origin: (origin, callback) => {
      // ✅ Whitelist specific origins
      if (!origin || allowedOrigins.includes(origin)) {
        callback(null, true);
      } else {
        callback(new Error('Not allowed by CORS'));
      }
    },
    credentials: true,
    methods: ['GET', 'POST', 'PUT', 'DELETE', 'PATCH'],
    allowedHeaders: ['Content-Type', 'Authorization'],
  });

  // ✅ Conditional Swagger (development only)
  if (process.env.NODE_ENV === 'development') {
    const config = new DocumentBuilder()
      .setTitle('API Documentation')
      .setDescription('Development API documentation')
      .setVersion('1.0')
      .addBearerAuth()
      .build();

    const document = SwaggerModule.createDocument(app, config);
    SwaggerModule.setup('api-docs', app, document);
  }

  // ✅ Global error handler
  app.useGlobalFilters(new AllExceptionsFilter());

  const port = configService.get('PORT', 3000);
  await app.listen(port);

  // ✅ Minimal startup info
  if (process.env.NODE_ENV === 'development') {
    console.log(`Application running in development mode on port ${port}`);
  }
}
bootstrap();
// all-exceptions.filter.ts - SECURE: Don't expose sensitive errors
import { ExceptionFilter, Catch, ArgumentsHost, HttpException, HttpStatus } from '@nestjs/common';
import { Request, Response } from 'express';

@Catch()
export class AllExceptionsFilter implements ExceptionFilter {
  catch(exception: unknown, host: ArgumentsHost) {
    const ctx = host.switchToHttp();
    const response = ctx.getResponse<Response>();
    const request = ctx.getRequest<Request>();

    let status = HttpStatus.INTERNAL_SERVER_ERROR;
    let message = 'Internal server error';

    if (exception instanceof HttpException) {
      status = exception.getStatus();
      const exceptionResponse = exception.getResponse();
      message = typeof exceptionResponse === 'string' ? exceptionResponse : (exceptionResponse as any).message;
    }

    // ✅ Log full error internally
    console.error('Exception occurred:', {
      timestamp: new Date().toISOString(),
      path: request.url,
      method: request.method,
      error: exception,
      stack: exception instanceof Error ? exception.stack : undefined,
    });

    // ✅ Return minimal error info to client
    response.status(status).json({
      statusCode: status,
      message:
        process.env.NODE_ENV === 'production'
          ? 'An error occurred' // ✅ Generic message in production
          : message, // ✅ Detailed message in development
      timestamp: new Date().toISOString(),
      path: request.url,
      // ❌ Never include: stack trace, internal paths, database errors
    });
  }
}

Vulnerable Configuration (AWS S3)​

export class AwsService {
  private s3: AWS.S3;

  constructor() {
    // ❌ VULNERABLE: Credentials in code
    this.s3 = new AWS.S3({
      accessKeyId: 'AKIAIOSFODNN7EXAMPLE',
      secretAccessKey: 'wJalrXUtnFEMI/K7MDENG/bPxRfiCYEXAMPLEKEY',
      region: 'us-east-1',

  async uploadFile(file: Express.Multer.File) {
    const params = {
      Bucket: 'my-public-bucket',

Secure Solution​

// aws.service.ts - SECURE
import { Injectable } from '@nestjs/common';
import { S3 } from '@aws-sdk/client-s3';
import { getSignedUrl } from '@aws-sdk/s3-request-presigner';
import { ConfigService } from '@nestjs/config';
import * as crypto from 'crypto';
import * as path from 'path';

@Injectable()
export class AwsService {
  private s3: S3;
  private bucketName: string;

  constructor(private configService: ConfigService) {
    // ✅ SECURE: Use IAM roles or environment variables
    this.s3 = new S3({
      region: this.configService.get('AWS_REGION'),
      // ✅ Credentials automatically loaded from:
      // 1. Environment variables (AWS_ACCESS_KEY_ID, AWS_SECRET_ACCESS_KEY)
      // 2. IAM role (recommended for EC2/ECS/Lambda)
      // 3. AWS credentials file
    });

    this.bucketName = this.configService.get('AWS_S3_BUCKET');
  }

  async uploadFile(file: Express.Multer.File, userId: string, isPublic: boolean = false) {
    // ✅ Generate secure filename
    const fileExtension = path.extname(file.originalname);
    const randomName = crypto.randomBytes(16).toString('hex');
    const key = `uploads/${userId}/${randomName}${fileExtension}`;

    // ✅ Validate file type
    const allowedTypes = ['image/jpeg', 'image/png', 'application/pdf'];
    if (!allowedTypes.includes(file.mimetype)) {
      throw new BadRequestException('Invalid file type');
    }

    // ✅ Validate file size (10MB max)
    if (file.size > 10 * 1024 * 1024) {
      throw new BadRequestException('File too large');
    }

    const params = {
      Bucket: this.bucketName,
      Key: key,
      Body: file.buffer,
      ContentType: file.mimetype,
      // ✅ SECURE: Private by default
      ACL: isPublic ? 'public-read' : 'private',
      // ✅ Server-side encryption
      ServerSideEncryption: 'AES256',
      // ✅ Add metadata
      Metadata: {
        userId: userId,
        uploadedAt: new Date().toISOString(),
      },
    };

    await this.s3.putObject(params);

    return {
      key,
      url: isPublic ? `https://${this.bucketName}.s3.amazonaws.com/${key}` : await this.getPresignedUrl(key), // ✅ Temporary signed URL
    };
  }

  // ✅ Generate temporary signed URL for private files
  async getPresignedUrl(key: string, expiresIn: number = 3600): Promise<string> {
    const command = new GetObjectCommand({
      Bucket: this.bucketName,
      Key: key,
    });

    return await getSignedUrl(this.s3, command, { expiresIn });
  }

  async deleteFile(key: string, userId: string) {
    // ✅ Verify ownership before deletion
    if (!key.startsWith(`uploads/${userId}/`)) {
      throw new ForbiddenException('Cannot delete this file');
    }

    await this.s3.deleteObject({
      Bucket: this.bucketName,
      Key: key,
    });
  }
}

Prevention Checklist​

  • Remove debug features in production
  • Change default credentials immediately
  • Disable unnecessary services and endpoints
  • Keep software updated with security patches
  • Implement security headers (HSTS, CSP, etc.)
  • Restrict CORS to specific origins
  • Hide error details from external users
  • Review cloud permissions (S3, IAM roles)
  • Use environment variables for secrets
  • Implement proper logging without sensitive data
  • Secure admin panels with strong authentication
  • Regular security audits of configuration

Testing Methods​

describe('Security Configuration Tests', () => {
  it('should not expose environment variables in production', () => {
    process.env.NODE_ENV = 'production';

    const config = getConfig();
    expect(config.debug).toBe(false);
    expect(config.verbose).toBe(false);
  });

  it('should have security headers', async () => {
    const response = await request(app).get('/');

    expect(response.headers['x-frame-options']).toBeDefined();
    expect(response.headers['x-content-type-options']).toBe('nosniff');
    expect(response.headers['strict-transport-security']).toBeDefined();
  });

  it('should not expose Swagger in production', async () => {
    process.env.NODE_ENV = 'production';

    const response = await request(app).get('/api-docs');
    expect(response.status).toBe(404);
  });

  it('should reject default admin credentials', async () => {
    const response = await request(app).post('/admin/login').send({ email: 'admin', password: 'admin' });

    expect(response.status).toBe(401);
  });
});
