A06: Insecure Design - Real Scenarios

Overview​

Insecure Design represents flaws in the design and architecture phase before implementation. These are different from insecure implementation - even perfect code cannot fix flawed design. This requires threat modeling, secure design patterns, and reference architectures.

Key Concepts​

  • Missing security controls: No rate limiting, no account lockout
  • Flawed business logic: Price manipulation, quantity bypass
  • Insufficient isolation: Shared resources without proper boundaries
  • Trust boundary violations: Trusting user input or external systems
  • Missing abuse cases: Not considering malicious user behavior

Vulnerable Scenario 1: Missing Rate Limiting on Critical Functions​

Vulnerable Code (Next.js API)​

// pages/api/auth/forgot-password.ts - VULNERABLE
import type { NextApiRequest, NextApiResponse } from 'next';
import { sendPasswordResetEmail } from '@/lib/email';
import { db } from '@/lib/db';

// ❌ VULNERABLE: No rate limiting on password reset
export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  if (req.method !== 'POST') {
    return res.status(405).json({ error: 'Method not allowed' });
  }

  const { email } = req.body;

  // ❌ No rate limiting - attacker can spam requests
  const user = await db.user.findUnique({ where: { email } });

  if (user) {
    const resetToken = generateResetToken();

    await db.user.update({
      where: { id: user.id },
      data: { resetToken, resetTokenExpiry: new Date(Date.now() + 3600000) },
    });

    // ❌ Attacker can flood victim's inbox
    await sendPasswordResetEmail(email, resetToken);
  }

  // ❌ Timing attack: response differs if email exists
  return res.status(200).json({
    message: user ? 'Email sent' : 'Email not found',
  });
}

Secure Solution​

// pages/api/auth/forgot-password.ts - SECURE
import type { NextApiRequest, NextApiResponse } from 'next';
import { sendPasswordResetEmail } from '@/lib/email';
import { db } from '@/lib/db';
import rateLimit from 'express-rate-limit';
import RedisStore from 'rate-limit-redis';
import { createClient } from 'redis';

// ✅ SECURE: Rate limiter with Redis
const redisClient = createClient({
  url: process.env.REDIS_URL,
});

const limiter = rateLimit({
  store: new RedisStore({
    client: redisClient,
    prefix: 'rl:forgot-password:',
  }),
  // ✅ Strict limits on sensitive endpoint
  windowMs: 15 * 60 * 1000, // 15 minutes
  max: 3, // Max 3 requests per 15 minutes per IP
  message: 'Too many password reset attempts, please try again later',
  standardHeaders: true,
  legacyHeaders: false,

  // ✅ Key by IP + email (more granular)
  keyGenerator: (req) => {
    const ip = req.headers['x-forwarded-for'] || req.socket.remoteAddress;
    const email = req.body.email || '';
    return `${ip}:${email}`;
  },

  // ✅ Skip successful requests from counting
  skipSuccessfulRequests: false,
  skipFailedRequests: false,
});

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  if (req.method !== 'POST') {
    return res.status(405).json({ error: 'Method not allowed' });
  }

  // ✅ Apply rate limiting
  await new Promise((resolve, reject) => {
    limiter(req as any, res as any, (result: any) => {
      if (result instanceof Error) {
        return reject(result);
      }
      resolve(result);
    });
  });

  const { email } = req.body;

  // ✅ Input validation
  if (!email || !isValidEmail(email)) {
    return res.status(400).json({ error: 'Invalid email' });
  }

  // ✅ Always do database lookup to maintain constant timing
  const user = await db.user.findUnique({ where: { email } });

  if (user) {
    // ✅ Check if recent reset was already requested
    const recentReset = await db.passwordReset.findFirst({
      where: {
        userId: user.id,
        createdAt: { gte: new Date(Date.now() - 5 * 60 * 1000) }, // 5 min
      },
    });

    if (recentReset) {
      // ✅ Don't send another email too soon
      return res.status(200).json({
        message: 'If that email exists, a reset link has been sent',
      });
    }

    const resetToken = generateSecureToken();

    // ✅ Store in separate table with expiry
    await db.passwordReset.create({
      data: {
        userId: user.id,
        token: await hashToken(resetToken),
        expiresAt: new Date(Date.now() + 3600000), // 1 hour
        ipAddress: req.headers['x-forwarded-for'] || req.socket.remoteAddress,
      },
    });

    // ✅ Send email (with error handling)
    try {
      await sendPasswordResetEmail(email, resetToken);
    } catch (error) {
      console.error('Failed to send reset email:', error);
      // ✅ Don't expose email sending failure to user
    }
  } else {
    // ✅ Simulate email sending delay to prevent timing attacks
    await new Promise((resolve) => setTimeout(resolve, Math.random() * 200 + 100));
  }

  // ✅ Generic response prevents user enumeration
  return res.status(200).json({
    message: 'If that email exists, a reset link has been sent',
  });
}

// ✅ Secure token generation
function generateSecureToken(): string {
  return crypto.randomBytes(32).toString('hex');
}

// ✅ Hash tokens before storage
async function hashToken(token: string): Promise<string> {
  return crypto.createHash('sha256').update(token).digest('hex');
}

import type { NextApiRequest, NextApiResponse } from 'next';
import { db } from '@/lib/db';

// ❌ VULNERABLE: Trusts price from client

        userId: req.user.id,

}

Secure Solution​

// pages/api/checkout.ts - SECURE
import type { NextApiRequest, NextApiResponse } from 'next';
import { db } from '@/lib/db';
import { processPayment } from '@/lib/payment';
import Decimal from 'decimal.js'; // ✅ Use decimal library for money

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  const { items } = req.body;

  // ✅ Input validation
  if (!Array.isArray(items) || items.length === 0) {
    return res.status(400).json({ error: 'Invalid items' });
  }

  if (items.length > 100) {
    return res.status(400).json({ error: 'Too many items' });
  }

  // ✅ SERVER-SIDE: Fetch current prices from database
  const productIds = items.map((item) => item.id);
  const products = await db.product.findMany({
    where: { id: { in: productIds } },
    select: { id: true, price: true, stock: true, isActive: true },
  });

  // ✅ Create price lookup map
  const priceMap = new Map(products.map((p) => [p.id, { price: p.price, stock: p.stock, isActive: p.isActive }]));

  // ✅ Validate and calculate total server-side
  let calculatedTotal = new Decimal(0);
  const validatedItems = [];

  for (const item of items) {
    const product = priceMap.get(item.id);

    // ✅ Validate product exists and is active
    if (!product || !product.isActive) {
      return res.status(400).json({
        error: `Product ${item.id} is not available`,
      });
    }

    // ✅ Validate quantity
    if (!Number.isInteger(item.quantity) || item.quantity < 1) {
      return res.status(400).json({ error: 'Invalid quantity' });
    }

    if (item.quantity > 100) {
      return res.status(400).json({ error: 'Quantity exceeds limit' });
    }

    // ✅ Check stock availability
    if (item.quantity > product.stock) {
      return res.status(400).json({
        error: `Insufficient stock for product ${item.id}`,
      });
    }

    // ✅ Calculate using server-side price (not client price!)
    const itemTotal = new Decimal(product.price).times(item.quantity);
    calculatedTotal = calculatedTotal.plus(itemTotal);

    validatedItems.push({
      productId: item.id,
      quantity: item.quantity,
      unitPrice: product.price,
      totalPrice: itemTotal.toFixed(2),
    });
  }

  // ✅ Apply discounts/coupons server-side
  const discount = await calculateDiscount(req.user.id, validatedItems);
  calculatedTotal = calculatedTotal.minus(discount);

  // ✅ Ensure non-negative
  if (calculatedTotal.lessThan(0)) {
    calculatedTotal = new Decimal(0);
  }

  const finalAmount = calculatedTotal.toFixed(2);

  // ✅ Start database transaction
  const result = await db.$transaction(async (tx) => {
    // ✅ Lock products to prevent race conditions
    const lockedProducts = await tx.product.findMany({
      where: { id: { in: productIds } },
      select: { id: true, stock: true },
    });

    // ✅ Recheck stock within transaction
    for (const item of validatedItems) {
      const product = lockedProducts.find((p) => p.id === item.productId);
      if (!product || product.stock < item.quantity) {
        throw new Error(`Insufficient stock for product ${item.productId}`);
      }
    }

    // ✅ Process payment with calculated amount
    const paymentResult = await processPayment({
      amount: parseFloat(finalAmount),
      currency: 'USD',
      userId: req.user.id,
      idempotencyKey: req.body.idempotencyKey, // ✅ Prevent double-charging
    });

    if (!paymentResult.success) {
      throw new Error('Payment failed');
    }

    // ✅ Deduct stock
    for (const item of validatedItems) {
      await tx.product.update({
        where: { id: item.productId },
        data: { stock: { decrement: item.quantity } },
      });
    }

    // ✅ Create order with validated data
    const order = await tx.order.create({
      data: {
        userId: req.user.id,
        items: {
          create: validatedItems,
        },
        subtotal: calculatedTotal.plus(discount).toFixed(2),
        discount: discount.toFixed(2),
        total: finalAmount,
        status: 'completed',
        paymentId: paymentResult.id,
      },
    });

    return order;
  });

  return res.status(200).json({ order: result });
}

Vulnerable Scenario 3: Insufficient Account Lockout​

Vulnerable Code (Authentication)​

// auth.service.ts - VULNERABLE
import { Injectable } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { User } from './entities/user.entity';
import * as bcrypt from 'bcrypt';

@Injectable()
export class AuthService {
  constructor(
    @InjectRepository(User)
    private userRepository: Repository<User>,
  ) {}

  // ❌ VULNERABLE: No account lockout
  async login(email: string, password: string) {
    const user = await this.userRepository.findOne({ where: { email } });

    if (!user) {
      throw new UnauthorizedException('Invalid credentials');
    }

    const isValid = await bcrypt.compare(password, user.password);

    if (!isValid) {
      // ❌ No tracking of failed attempts
      throw new UnauthorizedException('Invalid credentials');
    }

    return this.generateToken(user);
  }
}

Secure Solution​

// auth.service.ts - SECURE
import { Injectable, UnauthorizedException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { User } from './entities/user.entity';
import { LoginAttempt } from './entities/login-attempt.entity';
import * as bcrypt from 'bcrypt';
import { Cache } from 'cache-manager';
import { Inject } from '@nestjs/common';
import { CACHE_MANAGER } from '@nestjs/cache-manager';

@Injectable()
export class AuthService {
  // ✅ Account lockout configuration
  private readonly MAX_ATTEMPTS = 5;
  private readonly LOCKOUT_DURATION_MINUTES = 15;
  private readonly ATTEMPT_WINDOW_MINUTES = 15;

  constructor(
    @InjectRepository(User)
    private userRepository: Repository<User>,
    @InjectRepository(LoginAttempt)
    private loginAttemptRepository: Repository<LoginAttempt>,
    @Inject(CACHE_MANAGER) private cacheManager: Cache,
  ) {}

  async login(email: string, password: string, ipAddress: string) {
    // ✅ Check if account is locked
    const isLocked = await this.isAccountLocked(email);
    if (isLocked) {
      const lockoutRemaining = await this.getLockoutTimeRemaining(email);
      throw new UnauthorizedException(`Account is locked. Try again in ${lockoutRemaining} minutes.`);
    }

    // ✅ Check failed attempts from this IP
    const ipAttempts = await this.getRecentAttempts(ipAddress);
    if (ipAttempts >= this.MAX_ATTEMPTS) {
      throw new UnauthorizedException('Too many failed attempts from this IP address');
    }

    const user = await this.userRepository.findOne({
      where: { email },
      select: ['id', 'email', 'password', 'isActive', 'failedLoginAttempts', 'lockedUntil'],
    });

    if (!user || !user.isActive) {
      // ✅ Record failed attempt even if user doesn't exist
      await this.recordFailedAttempt(email, ipAddress);

      // ✅ Add delay to slow down brute force
      await new Promise((resolve) => setTimeout(resolve, 1000));

      throw new UnauthorizedException('Invalid credentials');
    }

    const isValid = await bcrypt.compare(password, user.password);

    if (!isValid) {
      // ✅ Increment failed attempts
      await this.handleFailedLogin(user, ipAddress);

      const remainingAttempts = this.MAX_ATTEMPTS - (user.failedLoginAttempts + 1);

      if (remainingAttempts > 0) {
        throw new UnauthorizedException(`Invalid credentials. ${remainingAttempts} attempts remaining.`);
      } else {
        throw new UnauthorizedException(
          `Account locked due to too many failed attempts. Try again in ${this.LOCKOUT_DURATION_MINUTES} minutes.`,
        );
      }
    }

    // ✅ Successful login - reset failed attempts
    await this.handleSuccessfulLogin(user, ipAddress);

    delete user.password;
    return this.generateToken(user);
  }

  // ✅ Track failed login attempts
  private async handleFailedLogin(user: User, ipAddress: string) {
    const attempts = user.failedLoginAttempts + 1;

    if (attempts >= this.MAX_ATTEMPTS) {
      // ✅ Lock the account
      const lockedUntil = new Date(Date.now() + this.LOCKOUT_DURATION_MINUTES * 60 * 1000);

      await this.userRepository.update(user.id, {
        failedLoginAttempts: attempts,
        lockedUntil: lockedUntil,
      });

      // ✅ Send alert email
      await this.sendAccountLockedEmail(user.email, lockedUntil);

      // ✅ Log security event
      await this.logSecurityEvent({
        type: 'ACCOUNT_LOCKED',
        userId: user.id,
        ipAddress,
        reason: `${this.MAX_ATTEMPTS} failed login attempts`,
      });
    } else {
      await this.userRepository.update(user.id, {
        failedLoginAttempts: attempts,
      });
    }

    // ✅ Record attempt in separate table
    await this.recordFailedAttempt(user.email, ipAddress);
  }

  private async handleSuccessfulLogin(user: User, ipAddress: string) {
    // ✅ Reset failed attempts counter
    await this.userRepository.update(user.id, {
      failedLoginAttempts: 0,
      lockedUntil: null,
      lastLoginAt: new Date(),
      lastLoginIp: ipAddress,
    });

    // ✅ Clear IP-based attempts from cache
    await this.cacheManager.del(`login_attempts:${ipAddress}`);

    // ✅ Log successful login
    await this.logSecurityEvent({
      type: 'LOGIN_SUCCESS',
      userId: user.id,
      ipAddress,
    });
  }

  private async isAccountLocked(email: string): Promise<boolean> {
    const user = await this.userRepository.findOne({
      where: { email },
      select: ['lockedUntil'],
    });

    if (!user || !user.lockedUntil) {
      return false;
    }

    if (new Date() < user.lockedUntil) {
      return true;
    }

    // ✅ Lockout period expired, unlock account
    await this.userRepository.update(user.id, {
      lockedUntil: null,
      failedLoginAttempts: 0,
    });

    return false;
  }

  private async getLockoutTimeRemaining(email: string): Promise<number> {
    const user = await this.userRepository.findOne({
      where: { email },
      select: ['lockedUntil'],
    });

    if (!user?.lockedUntil) {
      return 0;
    }

    const remaining = user.lockedUntil.getTime() - Date.now();
    return Math.ceil(remaining / 60000); // Convert to minutes
  }

  // ✅ Track attempts by IP address (prevents distributed attacks)
  private async getRecentAttempts(ipAddress: string): Promise<number> {
    const cacheKey = `login_attempts:${ipAddress}`;
    const attempts = await this.cacheManager.get<number>(cacheKey);
    return attempts || 0;
  }

  private async recordFailedAttempt(email: string, ipAddress: string) {
    // ✅ Persist to database
    await this.loginAttemptRepository.create({
      email,
      ipAddress,
      success: false,
      timestamp: new Date(),
    });

    // ✅ Increment cache counter
    const cacheKey = `login_attempts:${ipAddress}`;
    const currentAttempts = await this.getRecentAttempts(ipAddress);
    await this.cacheManager.set(
      cacheKey,
      currentAttempts + 1,
      this.ATTEMPT_WINDOW_MINUTES * 60 * 1000, // TTL in milliseconds
    );
  }
}

Prevention Checklist​

  • Threat modeling: Identify threats in design phase
  • Rate limiting: Apply to all sensitive endpoints
  • Account lockout: Implement progressive delays
  • Server-side validation: Never trust client calculations
  • Secure defaults: Fail securely, deny by default
  • Business logic review: Consider abuse cases
  • Transaction integrity: Use database transactions
  • Idempotency: Prevent duplicate operations
  • Resource limits: Limit quantities, file sizes, etc.
  • Audit logging: Log all security-relevant events
