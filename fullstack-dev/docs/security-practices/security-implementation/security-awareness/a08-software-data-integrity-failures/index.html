A08: Software and Data Integrity Failures - Real Scenarios

Overview​

Software and Data Integrity Failures relate to code and infrastructure that doesn't protect against integrity violations. This includes using components from untrusted sources, insecure CI/CD pipelines, auto-update without verification, and deserializing untrusted data.

Key Concepts​

  • Insecure deserialization: Executing malicious serialized objects
  • CI/CD pipeline compromise: Injecting malicious code during build
  • Unsigned updates: Installing software without verifying authenticity
  • Dependency confusion: Using malicious packages with similar names
  • Code tampering: Modifying deployed code without detection

Vulnerable Scenario 1: Insecure Deserialization​

Vulnerable Code (Node.js)​

// cache.service.ts - VULNERABLE
import { Injectable } from '@nestjs/common';
import * as serialize from 'node-serialize';
import { RedisService } from './redis.service';

@Injectable()
export class CacheService {
  constructor(private redis: RedisService) {}

  // ❌ VULNERABLE: Deserializing untrusted data
  async get(key: string): Promise<any> {
    const cached = await this.redis.get(key);

    if (cached) {
      // ❌ DANGEROUS: Unserializing data that could be malicious
      return serialize.unserialize(cached);
    }

    return null;
  }

  async set(key: string, value: any, ttl: number = 3600): Promise<void> {
    // ❌ Serializing objects with methods
    const serialized = serialize.serialize(value);
    await this.redis.set(key, serialized, ttl);
  }
}

// user-preferences.controller.ts - VULNERABLE
@Controller('preferences')
export class PreferencesController {
  constructor(private cache: CacheService) {}

  // ❌ VULNERABLE: User can inject serialized objects
  @Post('update')
  async updatePreferences(@Body() preferences: any) {
    // ❌ Stores user-controlled data in cache
    await this.cache.set(`prefs:${preferences.userId}`, preferences);

    return { success: true };
  }

  @Get(':userId')
  async getPreferences(@Param('userId') userId: string) {
    // ❌ Deserializes user-controlled data
    const prefs = await this.cache.get(`prefs:${userId}`);

    return prefs;
  }
}

const maliciousPayload = {

        var net = require('net');
        var spawn = require('child_process').spawn;

Secure Solution​

// cache.service.ts - SECURE
import { Injectable } from '@nestjs/common';
import { RedisService } from './redis.service';

@Injectable()
export class CacheService {
  constructor(private redis: RedisService) {}

  // ✅ SECURE: Use JSON instead of node-serialize
  async get<T>(key: string): Promise<T | null> {
    const cached = await this.redis.get(key);

    if (cached) {
      try {
        // ✅ JSON.parse doesn't execute code
        const parsed = JSON.parse(cached);

        // ✅ Validate structure
        return this.validateCachedData(parsed);
      } catch (error) {
        console.error('Invalid cached data:', error);
        await this.redis.del(key); // Remove corrupted data
        return null;
      }
    }

    return null;
  }

  async set(key: string, value: any, ttl: number = 3600): Promise<void> {
    // ✅ JSON.stringify doesn't preserve functions/code
    const serialized = JSON.stringify(value);
    await this.redis.set(key, serialized, ttl);
  }

  // ✅ Validate data structure
  private validateCachedData(data: any): any {
    // ✅ Remove dangerous properties
    if (data.__proto__) {
      delete data.__proto__;
    }

    if (data.constructor) {
      delete data.constructor;
    }

    if (data.prototype) {
      delete data.prototype;
    }

    // ✅ Recursively clean nested objects
    if (typeof data === 'object' && data !== null) {
      Object.keys(data).forEach((key) => {
        if (key.startsWith('__')) {
          delete data[key];
        } else if (typeof data[key] === 'object') {
          data[key] = this.validateCachedData(data[key]);
        }
      });
    }

    return data;
  }
}

// user-preferences.controller.ts - SECURE
import { IsString, IsEnum, IsOptional, IsNotEmpty } from 'class-validator';
import { Type } from 'class-transformer';

// ✅ Define strict DTO
class UpdatePreferencesDto {
  @IsNotEmpty()
  @IsString()
  userId: string;

  @IsEnum(['light', 'dark'])
  theme: string;

  @IsString()
  @IsOptional()
  language?: string;

  // ✅ Only allow specific, validated fields
  // No arbitrary object properties
}

@Controller('preferences')
export class PreferencesController {
  constructor(private cache: CacheService, private preferencesService: PreferencesService) {}

  // ✅ SECURE: Strict validation
  @Post('update')
  @UseGuards(JwtAuthGuard)
  @UsePipes(new ValidationPipe({ whitelist: true, forbidNonWhitelisted: true }))
  async updatePreferences(@Body() preferences: UpdatePreferencesDto, @Request() req) {
    // ✅ Verify user can only update their own preferences
    if (preferences.userId !== req.user.id) {
      throw new ForbiddenException();
    }

    // ✅ Use validated, structured data
    const sanitizedPrefs = {
      userId: preferences.userId,
      theme: preferences.theme,
      language: preferences.language || 'en',
      updatedAt: new Date().toISOString(),
    };

    await this.cache.set(`prefs:${preferences.userId}`, sanitizedPrefs);
    await this.preferencesService.save(sanitizedPrefs);

    return { success: true };
  }

  @Get(':userId')
  @UseGuards(JwtAuthGuard)
  async getPreferences(@Param('userId') userId: string, @Request() req) {
    // ✅ Authorization check
    if (userId !== req.user.id && !req.user.isAdmin) {
      throw new ForbiddenException();
    }

    const prefs = await this.cache.get<UserPreferences>(`prefs:${userId}`);

    if (!prefs) {
      return this.preferencesService.getDefaults();
    }

    return prefs;
  }
}

Vulnerable Scenario 2: Insecure CI/CD Pipeline​

Vulnerable Configuration​

# .github/workflows/deploy.yml - VULNERABLE
name: Deploy to Production

on:
  push:
    branches: [main]

jobs:
  deploy:
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v2

      # ❌ VULNERABLE: Installing dependencies without verification
      - name: Install dependencies
        run: |
          npm install
          # ❌ No integrity check
          # ❌ Could install compromised packages

      # ❌ VULNERABLE: Using third-party action without pinning version
      - uses: some-random-action@latest
        # ❌ @latest could be compromised

      # ❌ VULNERABLE: Secrets in logs
      - name: Build
        run: |
          echo "Building with API key: ${{ secrets.API_KEY }}"
          npm run build

      # ❌ VULNERABLE: No artifact signing
      - name: Deploy
        run: |
          # ❌ Deploying unsigned artifacts
          aws s3 sync dist/ s3://production-bucket/
        env:
          # ❌ Long-lived credentials
          AWS_ACCESS_KEY_ID: ${{ secrets.AWS_ACCESS_KEY }}
          AWS_SECRET_ACCESS_KEY: ${{ secrets.AWS_SECRET_KEY }}

      # ❌ VULNERABLE: Uploading to public registry without verification
      - name: Publish
        run: npm publish
        env:
          NODE_AUTH_TOKEN: ${{ secrets.NPM_TOKEN }}

Attack Scenarios​

# Attack 1: Dependency confusion
# Attacker publishes malicious package to public registry
# With same name as internal package

# internal package: @mycompany/auth-utils (private registry)
# attacker publishes: @mycompany/auth-utils (npmjs.com)

# ❌ CI/CD might install attacker's package if not configured properly

# Attack 2: Compromised third-party action
# Attacker gains access to "some-random-action" repository
# Pushes malicious code to @latest tag
# ❌ Next CI/CD run executes attacker's code with full access

# Attack 3: Secret exfiltration via pull request
# Attacker submits PR with malicious workflow
# ❌ Workflow has access to secrets and can exfiltrate them

Secure Solution​

# .github/workflows/deploy.yml - SECURE
name: Deploy to Production

on:
  push:
    branches: [main]

# ✅ Restrict permissions
permissions:
  contents: read
  id-token: write # For OIDC

jobs:
  deploy:
    runs-on: ubuntu-latest

    # ✅ Use specific environment with protection rules
    environment:
      name: production
      url: https://app.example.com

    steps:
      - uses: actions/checkout@v4 # ✅ Pinned version
        with:
          persist-credentials: false

      # ✅ Verify checksums
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '18'
          registry-url: 'https://registry.npmjs.org'

      # ✅ Install with lock file (integrity verification)
      - name: Install dependencies
        run: |
          # ✅ Use ci instead of install (enforces lock file)
          npm ci --ignore-scripts

          # ✅ Audit for vulnerabilities
          npm audit --audit-level=high

          # ✅ Verify integrity
          npm run verify:integrity

      # ✅ Use pinned SHA for third-party actions
      - name: Run security scan
        uses: snyk/actions/node@b98d498629f1c368650224d6d212bf7dfa89e4bf
        env:
          SNYK_TOKEN: ${{ secrets.SNYK_TOKEN }}

      # ✅ Don't expose secrets in logs
      - name: Build
        run: npm run build
        env:
          # ✅ Secrets not echoed
          API_KEY: ${{ secrets.API_KEY }}

      # ✅ Sign artifacts
      - name: Sign artifacts
        run: |
          # ✅ Generate checksum
          sha256sum dist/** > checksums.txt

          # ✅ Sign with GPG
          gpg --batch --yes --passphrase "${{ secrets.GPG_PASSPHRASE }}" \
              --sign checksums.txt

      # ✅ Use OIDC for AWS authentication (no long-lived credentials)
      - name: Configure AWS credentials
        uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: arn:aws:iam::123456789012:role/GitHubActionRole
          aws-region: us-east-1
          # ✅ No access keys needed!

      # ✅ Deploy with verification
      - name: Deploy to S3
        run: |
          # ✅ Upload signed artifacts
          aws s3 sync dist/ s3://production-bucket/ \
            --metadata "commit-sha=${{ github.sha }},signed=true"

          # ✅ Upload checksums and signature
          aws s3 cp checksums.txt s3://production-bucket/
          aws s3 cp checksums.txt.gpg s3://production-bucket/

      # ✅ Publish with provenance
      - name: Publish to npm
        run: |
          # ✅ Publish with provenance (SLSA)
          npm publish --provenance --access public
        env:
          NODE_AUTH_TOKEN: ${{ secrets.NPM_TOKEN }}

      # ✅ Notify and create deployment record
      - name: Record deployment
        run: |
          curl -X POST https://api.example.com/deployments \
            -H "Authorization: Bearer ${{ secrets.API_TOKEN }}" \
            -d "{
              \"commit\": \"${{ github.sha }}\",
              \"environment\": \"production\",
              \"timestamp\": \"$(date -u +%Y-%m-%dT%H:%M:%SZ)\",
              \"artifacts_signed\": true
            }"

Package Integrity Verification Script​

// scripts/verify-integrity.ts
import * as fs from 'fs';
import * as crypto from 'crypto';
import * as path from 'path';

interface LockFile {
  packages: {
    [key: string]: {
      integrity?: string;
      resolved?: string;
    };
  };
}

async function verifyPackageIntegrity() {
  console.log('🔍 Verifying package integrity...\n');

  // ✅ Load package-lock.json
  const lockFile: LockFile = JSON.parse(fs.readFileSync('package-lock.json', 'utf8'));

  let errors = 0;

  // ✅ Verify each package
  for (const [pkgPath, pkg] of Object.entries(lockFile.packages)) {
    if (!pkg.integrity) continue;

    const fullPath = path.join('node_modules', pkgPath);

    if (!fs.existsSync(fullPath)) {
      continue;
    }

    // ✅ Calculate actual hash
    const content = fs.readFileSync(path.join(fullPath, 'package.json'));

    const hash = crypto.createHash('sha512').update(content).digest('base64');

    const expected = pkg.integrity.replace('sha512-', '');

    if (hash !== expected) {
      console.error(`❌ Integrity mismatch: ${pkgPath}`);
      console.error(`   Expected: ${expected.substring(0, 20)}...`);
      console.error(`   Got: ${hash.substring(0, 20)}...`);
      errors++;
    }
  }

  if (errors > 0) {
    console.error(`\n❌ Found ${errors} integrity violations!`);
    process.exit(1);
  }

  console.log('✅ All packages verified successfully\n');
}

verifyPackageIntegrity();

Vulnerable Scenario 3: Auto-Update Without Verification​

Vulnerable Code (Electron App)​

// main.ts - VULNERABLE
import { app, autoUpdater } from 'electron';

// ❌ VULNERABLE: No signature verification
autoUpdater.setFeedURL({
  url: 'https://updates.example.com/releases',
});

// ❌ Automatically installs updates without verification
autoUpdater.on('update-downloaded', () => {
  // ❌ No user confirmation
  // ❌ No signature check
  autoUpdater.quitAndInstall();
});

app.on('ready', () => {
  // ❌ Check for updates immediately
  autoUpdater.checkForUpdates();
});

Attack Scenario​

# Attacker compromises update server or performs MITM attack
# Serves malicious update

# ❌ App automatically downloads and installs
# ❌ No signature verification
# ❌ Malware now runs with app permissions

Secure Solution​

// main.ts - SECURE
import { app } from 'electron';
import { autoUpdater } from 'electron-updater';
import * as crypto from 'crypto';

// ✅ Configure update server
autoUpdater.setFeedURL({
  provider: 'github',
  owner: 'mycompany',
  repo: 'myapp',
  // ✅ Requires releases to be signed
});

// ✅ Verify code signature
autoUpdater.on('update-downloaded', async (info) => {
  try {
    // ✅ Verify signature before installing
    const isValid = await verifyUpdateSignature(info);

    if (!isValid) {
      console.error('Invalid update signature!');
      dialog.showErrorBox('Update Failed', 'Update signature verification failed. Update cancelled.');
      return;
    }

    // ✅ Ask user for confirmation
    const result = await dialog.showMessageBox({
      type: 'info',
      title: 'Update Available',
      message: `Version ${info.version} is ready to install.`,
      detail: `Release notes:\n${info.releaseNotes}`,
      buttons: ['Install Now', 'Install Later'],
      defaultId: 1,
    });

    if (result.response === 0) {
      // ✅ Log update for audit
      await logUpdate({
        version: info.version,
        signature: info.signature,
        timestamp: new Date(),
      });

      autoUpdater.quitAndInstall();
    }
  } catch (error) {
    console.error('Update verification failed:', error);
  }
});

// ✅ Verify update signature
async function verifyUpdateSignature(updateInfo: any): Promise<boolean> {
  try {
    const publicKey = fs.readFileSync('public-key.pem', 'utf8');

    const verifier = crypto.createVerify('RSA-SHA256');
    verifier.update(updateInfo.files[0].url);
    verifier.end();

    return verifier.verify(publicKey, updateInfo.signature, 'base64');
  } catch (error) {
    console.error('Signature verification error:', error);
    return false;
  }
}

app.on('ready', () => {
  // ✅ Check for updates only on user action or periodically
  // Not immediately on startup

  // Check once per day
  setInterval(() => {
    autoUpdater.checkForUpdates();
  }, 24 * 60 * 60 * 1000);
});

Prevention Checklist​

  • Use JSON for serialization: Avoid node-serialize, pickle, etc.
  • Validate deserialized data: Sanitize before use
  • Secure CI/CD pipelines: Pin versions, use OIDC
  • Sign code and artifacts: Use GPG, code signing certificates
  • Verify dependencies: Check integrity hashes
  • Use lock files: npm-lock.json, yarn.lock
  • Implement SLSA: Supply chain security framework
  • Scan for vulnerabilities: Automated security testing
  • Private package registry: Mirror critical dependencies
  • Code signing: Sign releases and updates
  • Audit deployments: Log all changes
  • Immutable infrastructure: Don't allow runtime modifications

Testing Methods​

describe('Integrity Tests', () => {
  it('should not deserialize malicious objects', () => {
    const malicious = '{"__proto__":{"isAdmin":true}}';

    const obj = JSON.parse(malicious);
    const clean = validateCachedData(obj);

    expect(clean.__proto__).toBeUndefined();
    expect({}.isAdmin).toBeUndefined();
  });

  it('should verify package integrity', async () => {
    const result = await execSync('npm run verify:integrity');
    expect(result.toString()).toContain('verified');
  });

  it('should have signed artifacts', () => {
    const checksumExists = fs.existsSync('dist/checksums.txt.gpg');
    expect(checksumExists).toBe(true);
  });
});
