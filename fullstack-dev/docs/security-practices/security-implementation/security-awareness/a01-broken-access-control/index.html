A01:2025 - Broken Access Control (includes SSRF) - Real Scenarios

Overview​

Broken Access Control occurs when users can act outside their intended permissions. This remains the #1 vulnerability in OWASP Top 10 2025, with 3.73% of applications tested having one or more related weaknesses.

NEW in 2025: Server-Side Request Forgery (SSRF) has been rolled into this category as it represents unauthorized access to network resources.

Key Concepts​

  • Vertical Privilege Escalation: Regular user accessing admin functions
  • Horizontal Privilege Escalation: User accessing another user's data
  • Insecure Direct Object References (IDOR): Manipulating parameters to access unauthorized resources
  • Server-Side Request Forgery (SSRF): Application fetching unvalidated URLs, accessing internal resources

Vulnerable Scenario 1: IDOR in Order Management​

Vulnerable Code (NestJS)​

// orders.controller.ts
import { Controller, Get, Param } from '@nestjs/common';
import { OrdersService } from './orders.service';

@Controller('orders')
export class OrdersController {
  constructor(private ordersService: OrdersService) {}

  // ❌ VULNERABLE: No authorization check
  @Get(':id')
  async getOrder(@Param('id') id: string) {
    return this.ordersService.findOne(id);
  }

  // ❌ VULNERABLE: User can access any order by changing the ID
  @Get(':id/invoice')
  async downloadInvoice(@Param('id') orderId: string) {
    const order = await this.ordersService.findOne(orderId);
    return this.generateInvoice(order);
  }
}

Attack Example​

# Attacker is user ID 123
# They place an order with ID 5001
curl -H "Authorization: Bearer <token>" \
  https://api.example.com/orders/5001
# Returns their order

# Attacker tries incrementing the order ID
curl -H "Authorization: Bearer <token>" \
  https://api.example.com/orders/5002
# ❌ VULNERABLE: Returns another user's order!

# Attacker downloads someone else's invoice
curl -H "Authorization: Bearer <token>" \
  https://api.example.com/orders/5002/invoice
# ❌ VULNERABLE: Gets another user's sensitive financial data

Secure Solution​

// orders.controller.ts
import { Controller, Get, Param, UseGuards, Request, ForbiddenException } from '@nestjs/common';
import { JwtAuthGuard } from '../auth/jwt-auth.guard';
import { OrdersService } from './orders.service';

@Controller('orders')
@UseGuards(JwtAuthGuard)
export class OrdersController {
  constructor(private ordersService: OrdersService) {}

  // ✅ SECURE: Check ownership before returning data
  @Get(':id')
  async getOrder(@Param('id') id: string, @Request() req) {
    const order = await this.ordersService.findOne(id);

    if (!order) {
      throw new NotFoundException('Order not found');
    }

    // Verify the order belongs to the authenticated user
    if (order.userId !== req.user.id && !req.user.isAdmin) {
      throw new ForbiddenException('Access denied');
    }

    return order;
  }

  @Get(':id/invoice')
  async downloadInvoice(@Param('id') orderId: string, @Request() req) {
    const order = await this.ordersService.findOne(orderId);

    if (!order) {
      throw new NotFoundException('Order not found');
    }

    // Authorization check
    if (order.userId !== req.user.id && !req.user.isAdmin) {
      throw new ForbiddenException('Access denied');
    }

    return this.generateInvoice(order);
  }
}

Vulnerable Scenario 2: Admin Function Without Authorization​

Vulnerable Code (Next.js API Route)​

// pages/api/admin/users/[id].ts
import type { NextApiRequest, NextApiResponse } from 'next';
import { db } from '@/lib/db';

// ❌ VULNERABLE: No authentication or authorization check
export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  const { id } = req.query;

  if (req.method === 'DELETE') {
    // ❌ VULNERABLE: Any user can delete any account!
    await db.user.delete({ where: { id: Number(id) } });
    return res.status(200).json({ message: 'User deleted' });
  }

  if (req.method === 'PATCH') {
    // ❌ VULNERABLE: Any user can make any account an admin!
    const { isAdmin, email } = req.body;
    await db.user.update({
      where: { id: Number(id) },
      data: { isAdmin, email },
    });
    return res.status(200).json({ message: 'User updated' });
  }
}

Attack Example​

// Attacker's malicious script
const makeMyAccountAdmin = async () => {
  // Attacker knows their user ID is 42
  const response = await fetch('/api/admin/users/42', {
    method: 'PATCH',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ isAdmin: true }),
  });

  if (response.ok) {
    console.log('I am now an admin!');
    // ❌ Attacker now has full system access
  }
};

// Delete other users
const deleteCompetitor = async () => {
  await fetch('/api/admin/users/10', {
    method: 'DELETE',
  });
  // ❌ Deleted another user's account
};

Secure Solution​

// pages/api/admin/users/[id].ts
import type { NextApiRequest, NextApiResponse } from 'next';
import { getServerSession } from 'next-auth/next';
import { authOptions } from '../auth/[...nextauth]';
import { db } from '@/lib/db';

// ✅ SECURE: Middleware to check admin role
async function requireAdmin(req: NextApiRequest, res: NextApiResponse) {
  const session = await getServerSession(req, res, authOptions);

  if (!session) {
    return { authorized: false, error: 'Not authenticated' };
  }

  if (!session.user.isAdmin) {
    return { authorized: false, error: 'Admin access required' };
  }

  return { authorized: true, user: session.user };
}

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  // ✅ Check authorization first
  const auth = await requireAdmin(req, res);
  if (!auth.authorized) {
    return res.status(403).json({ error: auth.error });
  }

  const { id } = req.query;

  if (req.method === 'DELETE') {
    // ✅ Additional check: prevent deleting yourself
    if (Number(id) === auth.user.id) {
      return res.status(400).json({ error: 'Cannot delete your own account' });
    }

    await db.user.delete({ where: { id: Number(id) } });

    // ✅ Log the action for audit trail
    await db.auditLog.create({
      data: {
        action: 'USER_DELETED',
        performedBy: auth.user.id,
        targetUserId: Number(id),
        timestamp: new Date(),
      },
    });

    return res.status(200).json({ message: 'User deleted' });
  }

  if (req.method === 'PATCH') {
    const { isAdmin, email } = req.body;

    // ✅ Additional validation
    if (Number(id) === auth.user.id && isAdmin === false) {
      return res.status(400).json({
        error: 'Cannot remove your own admin privileges',
      });
    }

    await db.user.update({
      where: { id: Number(id) },
      data: { isAdmin, email },
    });

    // ✅ Audit log
    await db.auditLog.create({
      data: {
        action: 'USER_UPDATED',
        performedBy: auth.user.id,
        targetUserId: Number(id),
        changes: { isAdmin, email },
        timestamp: new Date(),
      },
    });

    return res.status(200).json({ message: 'User updated' });
  }

  return res.status(405).json({ error: 'Method not allowed' });
}

Vulnerable Scenario 3: Client-Side Access Control​

Vulnerable Code (React Component)​

// components/AdminPanel.tsx
import { useSession } from 'next-auth/react';

// ❌ VULNERABLE: Client-side only check
export default function AdminPanel() {
  const { data: session } = useSession();

  // ❌ Attacker can bypass this by modifying JavaScript
  if (!session?.user?.isAdmin) {
    return <div>Access Denied</div>;
  }

  const deleteAllUsers = async () => {
    // ❌ VULNERABLE: API doesn't check authorization
    await fetch('/api/admin/delete-all-users', { method: 'POST' });
  };

  return (
    <div>
      <h1>Admin Panel</h1>
      <button onClick={deleteAllUsers}>Delete All Users</button>
    </div>
  );
}

Attack Example​

// Attacker opens browser console
// Bypasses UI check by calling API directly
fetch('/api/admin/delete-all-users', {
  method: 'POST',
  credentials: 'include',
})
  .then((r) => r.json())
  .then((data) => console.log('All users deleted!'));
// ❌ VULNERABLE: API executes without checking permissions

Secure Solution​

// components/AdminPanel.tsx
import { useSession } from 'next-auth/react';

// ✅ Client-side check is for UX only
export default function AdminPanel() {
  const { data: session } = useSession();

  // UI-level check (can be bypassed, but that's OK)
  if (!session?.user?.isAdmin) {
    return <div>Access Denied</div>;
  }

  const deleteAllUsers = async () => {
    // ✅ Server will enforce authorization
    const response = await fetch('/api/admin/delete-all-users', {
      method: 'POST',
    });

    if (response.status === 403) {
      alert('Access denied - admin privileges required');
    }
  };

  return (
    <div>
      <h1>Admin Panel</h1>
      <button onClick={deleteAllUsers}>Delete All Users</button>
    </div>
  );
}
// pages/api/admin/delete-all-users.ts
import { getServerSession } from 'next-auth/next';
import { authOptions } from '../auth/[...nextauth]';

export default async function handler(req, res) {
  // ✅ SECURE: Server-side authorization is mandatory
  const session = await getServerSession(req, res, authOptions);

  if (!session?.user?.isAdmin) {
    return res.status(403).json({ error: 'Admin access required' });
  }

  // Additional safeguards for dangerous operations
  if (process.env.NODE_ENV === 'production') {
    return res.status(403).json({
      error: 'This operation is disabled in production',
    });
  }

  // Require additional confirmation token
  const { confirmationToken } = req.body;
  if (confirmationToken !== session.user.confirmationToken) {
    return res.status(400).json({ error: 'Invalid confirmation token' });
  }

  // Perform the operation
  await db.user.deleteMany({ where: { isAdmin: false } });

  return res.status(200).json({ message: 'Operation completed' });
}

Prevention Checklist​

  • Deny by default: Require explicit access grants
  • Server-side enforcement: Never rely on client-side checks alone
  • Object ownership validation: Verify resource belongs to user
  • Role-based access control: Implement consistent authorization model
  • Audit logging: Log all access control decisions
  • Automated testing: Test authorization for each endpoint
  • Rate limiting: Prevent enumeration attacks
  • Disable directory listing: Don't expose file structures

Testing Methods​

Manual Testing​

# Test IDOR vulnerability
# 1. Login as User A, note your resource IDs
# 2. Try accessing User B's resources by changing IDs
curl -H "Authorization: Bearer <user-a-token>" \
  https://api.example.com/orders/USER_B_ORDER_ID

# Test vertical escalation
# 1. Login as regular user
# 2. Try accessing admin endpoints
curl -H "Authorization: Bearer <regular-user-token>" \
  https://api.example.com/admin/users

Automated Testing (Jest)​

// orders.controller.spec.ts
describe('OrdersController - Access Control', () => {
  it('should prevent users from accessing other users orders', async () => {
    const user1Token = await getAuthToken('user1@example.com');
    const user2Token = await getAuthToken('user2@example.com');

    // User 1 creates an order
    const order = await request(app.getHttpServer())
      .post('/orders')
      .set('Authorization', `Bearer ${user1Token}`)
      .send({ productId: 1, quantity: 1 })
      .expect(201);

    const orderId = order.body.id;

    // User 2 tries to access User 1's order
    await request(app.getHttpServer())
      .get(`/orders/${orderId}`)
      .set('Authorization', `Bearer ${user2Token}`)
      .expect(403); // ✅ Should be forbidden
  });

  it('should prevent regular users from accessing admin functions', async () => {
    const userToken = await getAuthToken('user@example.com');

    await request(app.getHttpServer()).delete('/admin/users/1').set('Authorization', `Bearer ${userToken}`).expect(403); // ✅ Should be forbidden
  });
});

Vulnerable Scenario 4: Server-Side Request Forgery (SSRF)​

NEW in 2025: SSRF is now part of Broken Access Control as it involves unauthorized access to network resources.

Vulnerable Code (NestJS)​

// content.controller.ts - VULNERABLE
import { Controller, Get, Query } from '@nestjs/common';
import fetch from 'node-fetch';

@Controller('content')
export class ContentController {
  // ❌ VULNERABLE: Fetches any URL provided by user
  @Get('fetch')
  async fetchContent(@Query('url') url: string) {
    // ❌ No validation on URL
    const response = await fetch(url);
    const content = await response.text();

    return { content };
  }

  // ❌ VULNERABLE: Image proxy without validation
  @Get('image-proxy')
  async proxyImage(@Query('imageUrl') imageUrl: string) {
    const response = await fetch(imageUrl);
    const buffer = await response.buffer();

    return buffer;
  }
}

Attack Examples​

Attack 1: Accessing AWS Metadata​

# Attacker exploits SSRF to read AWS credentials
curl "https://api.example.com/content/fetch?url=http://169.254.169.254/latest/meta-data/iam/security-credentials/"

# Response: ["MyAppRole"]

curl "https://api.example.com/content/fetch?url=http://169.254.169.254/latest/meta-data/iam/security-credentials/MyAppRole"

# ❌ Returns AWS credentials!
{
  "AccessKeyId": "ASIA...",
  "SecretAccessKey": "wJalr...",
  "Token": "IQoJb3..."
}

Attack 2: Internal Network Scanning​

# Scan internal services
curl "https://api.example.com/content/fetch?url=http://internal-redis:6379"
curl "https://api.example.com/content/fetch?url=http://internal-db:5432"
curl "https://api.example.com/content/fetch?url=http://admin-panel.internal:8080"

# ❌ Attacker can discover and access internal services

Secure Solution​

// content.controller.ts - SECURE
import { Controller, Get, Query, BadRequestException } from '@nestjs/common';
import fetch from 'node-fetch';
import { URL } from 'url';
import * as dns from 'dns/promises';

@Controller('content')
export class ContentController {
  // ✅ Whitelist of allowed domains
  private readonly ALLOWED_DOMAINS = ['api.example.com', 'cdn.example.com', 'trusted-partner.com'];

  // ✅ Blocked IP ranges
  private readonly BLOCKED_IP_RANGES = [
    { start: '10.0.0.0', end: '10.255.255.255' },
    { start: '172.16.0.0', end: '172.31.255.255' },
    { start: '192.168.0.0', end: '192.168.255.255' },
    { start: '127.0.0.0', end: '127.255.255.255' },
    { start: '169.254.169.254', end: '169.254.169.254' }, // AWS metadata
  ];

  @Get('fetch')
  async fetchContent(@Query('url') urlString: string) {
    // ✅ Validate URL
    await this.validateUrl(urlString);

    // ✅ Fetch with timeout and size limits
    const controller = new AbortController();
    const timeout = setTimeout(() => controller.abort(), 5000);

    try {
      const response = await fetch(urlString, {
        signal: controller.signal,
        headers: { 'User-Agent': 'MyApp/1.0' },
        redirect: 'manual',
        size: 1048576, // Max 1MB
      });

      clearTimeout(timeout);

      if (!response.ok) {
        throw new BadRequestException('Failed to fetch content');
      }

      const content = await response.text();
      return { content: content.substring(0, 10000) };
    } catch (error) {
      throw new BadRequestException('Invalid or forbidden URL');
    }
  }

  // ✅ Comprehensive URL validation
  private async validateUrl(urlString: string): Promise<void> {
    let url: URL;

    try {
      url = new URL(urlString);
    } catch {
      throw new BadRequestException('Invalid URL format');
    }

    // ✅ Only allow HTTP/HTTPS
    if (!['http:', 'https:'].includes(url.protocol)) {
      throw new BadRequestException('Only HTTP and HTTPS are allowed');
    }

    // ✅ Check domain whitelist
    const hostname = url.hostname.toLowerCase();
    const isAllowed = this.ALLOWED_DOMAINS.some((domain) => hostname === domain || hostname.endsWith(`.${domain}`));

    if (!isAllowed) {
      throw new BadRequestException('Domain not in whitelist');
    }

    // ✅ Resolve hostname and check IP
    try {
      const addresses = await dns.resolve4(hostname);

      for (const ip of addresses) {
        if (this.isBlockedIp(ip)) {
          throw new BadRequestException('Access to this IP is forbidden');
        }
      }
    } catch {
      throw new BadRequestException('Failed to resolve hostname');
    }
  }

  private isBlockedIp(ip: string): boolean {
    const ipNum = this.ipToNumber(ip);

    for (const range of this.BLOCKED_IP_RANGES) {
      const startNum = this.ipToNumber(range.start);
      const endNum = this.ipToNumber(range.end);

      if (ipNum >= startNum && ipNum <= endNum) {
        return true;
      }
    }

    return false;
  }

  private ipToNumber(ip: string): number {
    return ip.split('.').reduce((acc, octet) => (acc << 8) + parseInt(octet), 0) >>> 0;
  }
}
