SANS CWE Top 25 Training

Overview​

This document provides a reference implementation for a secure coding training track aligned to the SANS CWE Top 25, including modular training content, assessments, and an optional workshop format.

Implementation Reference​

CWE Training Module Creator​

// src/training/cwe-training.service.ts
import { Injectable } from '@nestjs/common';

@Injectable()
export class CWETrainingService {
  createCWEModules(): TrainingModule[] {
    return [
      {
        id: 'cwe-79-xss',
        title: 'CWE-79: Cross-Site Scripting (XSS)',
        category: 'CWE',
        level: 'BEGINNER',
        duration: 75,
        prerequisites: [],
        objectives: [
          'Understand XSS vulnerability types',
          'Implement output encoding and input validation',
          'Configure Content Security Policy (CSP)',
          'Use React security best practices',
        ],
        content: [
          {
            type: 'VIDEO',
            title: 'Understanding XSS Attacks',
            url: '/training/videos/xss-explained.mp4',
            duration: 20,
            description: 'Types of XSS attacks and their impact',
          },
          {
            type: 'LAB',
            title: 'XSS Prevention in Next.js',
            url: '/training/labs/nextjs-xss-prevention',
            duration: 40,
            description: 'Hands-on XSS prevention techniques',
          },
          {
            type: 'DOCUMENT',
            title: 'CSP Implementation Guide',
            url: '/training/docs/csp-guide.pdf',
            duration: 15,
            description: 'Implementing Content Security Policy',
          },
        ],
        assessment: {
          type: 'QUIZ',
          passingScore: 80,
          timeLimit: 30,
          questions: [
            {
              id: 'cwe79-001',
              type: 'MULTIPLE_CHOICE',
              question: 'Which React feature helps prevent XSS by default?',
              options: ['dangerouslySetInnerHTML', 'JSX auto-escaping', 'useEffect hook', 'useState hook'],
              correctAnswer: 'JSX auto-escaping',
              explanation: 'React JSX automatically escapes values, preventing XSS attacks.',
            },
          ],
        },
        certification: false,
      },
      {
        id: 'cwe-89-sql-injection',
        title: 'CWE-89: SQL Injection',
        category: 'CWE',
        level: 'BEGINNER',
        duration: 90,
        prerequisites: [],
        objectives: [
          'Understand SQL injection mechanisms',
          'Use parameterized queries and ORMs',
          'Implement input validation',
          'Apply least privilege principles',
        ],
        content: [
          {
            type: 'VIDEO',
            title: 'SQL Injection Deep Dive',
            url: '/training/videos/sql-injection-deep-dive.mp4',
            duration: 25,
            description: 'Detailed explanation of SQL injection attacks',
          },
          {
            type: 'LAB',
            title: 'TypeORM Security Lab',
            url: '/training/labs/typeorm-security',
            duration: 50,
            description: 'Secure database access with TypeORM',
          },
        ],
        assessment: {
          type: 'PRACTICAL',
          passingScore: 85,
          timeLimit: 45,
          questions: [
            {
              id: 'cwe89-001',
              type: 'CODE_REVIEW',
              question: 'Fix the SQL injection vulnerability in this TypeORM repository method.',
              correctAnswer: 'Use parameterized queries with createQueryBuilder',
              explanation: 'Replace string concatenation with parameterized queries.',
            },
          ],
        },
        certification: true,
      },
      {
        id: 'cwe-20-input-validation',
        title: 'CWE-20: Improper Input Validation',
        category: 'CWE',
        level: 'INTERMEDIATE',
        duration: 100,
        prerequisites: ['cwe-79-xss', 'cwe-89-sql-injection'],
        objectives: [
          'Implement comprehensive input validation',
          'Use validation decorators in NestJS',
          'Configure sanitization pipelines',
          'Handle validation errors securely',
        ],
        content: [
          {
            type: 'VIDEO',
            title: 'Input Validation Strategies',
            url: '/training/videos/input-validation.mp4',
            duration: 30,
            description: 'Comprehensive input validation strategies',
          },
          {
            type: 'LAB',
            title: 'NestJS Validation Lab',
            url: '/training/labs/nestjs-validation',
            duration: 60,
            description: 'Implement robust validation in NestJS',
          },
        ],
        assessment: {
          type: 'PROJECT',
          passingScore: 80,
          timeLimit: 90,
          questions: [
            {
              id: 'cwe20-001',
              type: 'SCENARIO',
              question: 'Design a validation strategy for a user registration API.',
              correctAnswer: 'Multi-layer validation with DTOs, custom validators, and sanitization',
              explanation: 'Implement validation at multiple layers for defense in depth.',
            },
          ],
        },
        certification: true,
      },
    ];
  }

  createSecureCodingWorkshop(): any {
    return {
      id: 'secure-coding-workshop',
      title: 'Secure Coding Workshop: CWE Top 25',
      duration: 480, // 8 hours
      format: 'WORKSHOP',
      agenda: [
        {
          time: '09:00-09:30',
          topic: 'Workshop Introduction and CWE Overview',
          type: 'PRESENTATION',
        },
        {
          time: '09:30-11:00',
          topic: 'Injection Flaws (CWE-79, CWE-89)',
          type: 'HANDS_ON_LAB',
        },
        {
          time: '11:15-12:30',
          topic: 'Access Control Issues (CWE-22, CWE-352)',
          type: 'HANDS_ON_LAB',
        },
        {
          time: '13:30-15:00',
          topic: 'Cryptographic Failures (CWE-327, CWE-328)',
          type: 'HANDS_ON_LAB',
        },
        {
          time: '15:15-16:30',
          topic: 'Security Misconfiguration (CWE-798, CWE-862)',
          type: 'HANDS_ON_LAB',
        },
        {
          time: '16:30-17:00',
          topic: 'Wrap-up and Action Planning',
          type: 'DISCUSSION',
        },
      ],
      labs: [
        {
          id: 'lab-cwe-22',
          title: 'Path Traversal Prevention',
          vulnerability: 'CWE-22: Improper Limitation of a Pathname to a Restricted Directory',
          exercise: 'Fix file upload functionality to prevent directory traversal',
        },
        {
          id: 'lab-cwe-352',
          title: 'CSRF Protection Implementation',
          vulnerability: 'CWE-352: Cross-Site Request Forgery',
          exercise: 'Implement CSRF tokens in a Next.js application',
        },
      ],
    };
  }
}

Minimum Deliverables​

  • CWE module catalog aligned to common secure coding failure modes
  • Assessments that include code review and scenario-based questions
  • Optional workshop plan for concentrated skill uplift
