A07: Identification and Authentication Failures - Real Scenarios

Overview​

Authentication and session management failures allow attackers to compromise passwords, keys, or session tokens, or exploit implementation flaws to assume other users' identities temporarily or permanently.

Key Concepts​

  • Weak password policies: Allowing simple passwords
  • Credential stuffing: Using breached credentials from other sites
  • Session fixation: Hijacking or fixing session identifiers
  • Broken authentication: Flawed login/logout mechanisms
  • Missing multi-factor authentication: No second factor verification

Vulnerable Scenario 1: Weak Password Policy and No Account Protection​

Vulnerable Code (NestJS)​

// auth.service.ts - VULNERABLE
import { Injectable } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { User } from './entities/user.entity';
import * as bcrypt from 'bcrypt';

@Injectable()
export class AuthService {
  constructor(
    @InjectRepository(User)
    private userRepository: Repository<User>,
  ) {}

  // ❌ VULNERABLE: Weak password requirements
  async register(email: string, password: string) {
    // ❌ No password strength validation
    if (password.length < 6) {
      throw new BadRequestException('Password too short');
    }

    // ❌ Allows common passwords
    const hashedPassword = await bcrypt.hash(password, 10);

    const user = this.userRepository.create({
      email,
      password: hashedPassword,
    });

    await this.userRepository.save(user);
    return { message: 'User created' };
  }

  // ❌ VULNERABLE: No rate limiting or lockout
  async login(email: string, password: string) {
    const user = await this.userRepository.findOne({
      where: { email },
      select: ['id', 'email', 'password'],
    });

    if (!user) {
      throw new UnauthorizedException('Invalid credentials');
    }

    const isValid = await bcrypt.compare(password, user.password);

    if (!isValid) {
      // ❌ No failed attempt tracking
      throw new UnauthorizedException('Invalid credentials');
    }

    // ❌ Weak session token
    const token = Buffer.from(`${user.id}:${Date.now()}`).toString('base64');

    return { token };
  }
}

Secure Solution​

// auth.service.ts - SECURE
import { Injectable } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { User } from './entities/user.entity';
import { JwtService } from '@nestjs/jwt';
import * as bcrypt from 'bcrypt';
import * as crypto from 'crypto';
import { Cache } from 'cache-manager';

@Injectable()
export class AuthService {
  // ✅ Password policy constants
  private readonly MIN_PASSWORD_LENGTH = 12;
  private readonly COMMON_PASSWORDS = new Set([
    '123456',
    'password',
    '12345678',
    'qwerty',
    '123456789',
    'password123',
    'admin',
    'letmein',
    'welcome',
    'monkey',
    // ... load from file
  ]);

  constructor(
    @InjectRepository(User)
    private userRepository: Repository<User>,
    private jwtService: JwtService,
    @Inject(CACHE_MANAGER) private cacheManager: Cache,
  ) {}

  // ✅ SECURE: Strong password validation
  async register(email: string, password: string) {
    // ✅ Comprehensive password validation
    this.validatePassword(password);

    // ✅ Check if email already exists
    const existing = await this.userRepository.findOne({ where: { email } });
    if (existing) {
      throw new BadRequestException('Email already registered');
    }

    // ✅ Check against known breached passwords (using HaveIBeenPwned API)
    const isBreached = await this.checkPasswordBreach(password);
    if (isBreached) {
      throw new BadRequestException(
        'This password has been exposed in a data breach. Please choose a different password.',
      );
    }

    // ✅ Strong hashing with salt
    const hashedPassword = await bcrypt.hash(password, 12);

    const user = this.userRepository.create({
      email: email.toLowerCase().trim(),
      password: hashedPassword,
      emailVerified: false,
      createdAt: new Date(),
    });

    await this.userRepository.save(user);

    // ✅ Send verification email
    await this.sendVerificationEmail(user);

    return {
      message: 'Registration successful. Please check your email to verify your account.',
    };
  }

  // ✅ Validate password strength
  private validatePassword(password: string): void {
    const errors: string[] = [];

    // Length check
    if (password.length < this.MIN_PASSWORD_LENGTH) {
      errors.push(`Password must be at least ${this.MIN_PASSWORD_LENGTH} characters`);
    }

    // Complexity checks
    if (!/[A-Z]/.test(password)) {
      errors.push('Password must contain at least one uppercase letter');
    }

    if (!/[a-z]/.test(password)) {
      errors.push('Password must contain at least one lowercase letter');
    }

    if (!/[0-9]/.test(password)) {
      errors.push('Password must contain at least one number');
    }

    if (!/[!@#$%^&*(),.?":{}|<>]/.test(password)) {
      errors.push('Password must contain at least one special character');
    }

    // Check for common passwords
    if (this.COMMON_PASSWORDS.has(password.toLowerCase())) {
      errors.push('This password is too common. Please choose a stronger password.');
    }

    // Check for sequential characters
    if (/(.)\1{2,}/.test(password)) {
      errors.push('Password cannot contain repeated characters');
    }

    // Check for keyboard patterns
    const patterns = ['qwerty', 'asdfgh', '12345', 'abcdef'];
    if (patterns.some((pattern) => password.toLowerCase().includes(pattern))) {
      errors.push('Password contains common patterns');
    }

    if (errors.length > 0) {
      throw new BadRequestException(errors.join('. '));
    }
  }

  // ✅ Check against HaveIBeenPwned database
  private async checkPasswordBreach(password: string): Promise<boolean> {
    try {
      // ✅ Use k-anonymity model - only send first 5 chars of hash
      const hash = crypto.createHash('sha1').update(password).digest('hex').toUpperCase();
      const prefix = hash.substring(0, 5);
      const suffix = hash.substring(5);

      const response = await fetch(`https://api.pwnedpasswords.com/range/${prefix}`);

      const text = await response.text();
      const hashes = text.split('\n');

      // Check if our hash suffix appears in results
      return hashes.some((line) => line.startsWith(suffix));
    } catch (error) {
      // ✅ If API is down, log but don't block registration
      console.error('Failed to check password breach:', error);
      return false;
    }
  }

  // ✅ SECURE: Login with protection mechanisms
  async login(email: string, password: string, ipAddress: string) {
    // ✅ Normalize email
    const normalizedEmail = email.toLowerCase().trim();

    // ✅ Check rate limiting by IP
    await this.checkRateLimit(ipAddress);

    // ✅ Check if account is locked
    const lockKey = `account_lock:${normalizedEmail}`;
    const isLocked = await this.cacheManager.get(lockKey);

    if (isLocked) {
      const ttl = await this.cacheManager.store.ttl(lockKey);
      throw new UnauthorizedException(`Account is locked. Try again in ${Math.ceil(ttl / 60)} minutes.`);
    }

    // ✅ Find user
    const user = await this.userRepository.findOne({
      where: { email: normalizedEmail },
      select: [
        'id',
        'email',
        'password',
        'emailVerified',
        'twoFactorEnabled',
        'twoFactorSecret',
        'failedLoginAttempts',
      ],
    });

    if (!user) {
      // ✅ Constant-time response to prevent user enumeration
      await bcrypt.hash(password, 12);
      throw new UnauthorizedException('Invalid credentials');
    }

    // ✅ Check email verification
    if (!user.emailVerified) {
      throw new UnauthorizedException('Please verify your email before logging in');
    }

    // ✅ Verify password
    const isValid = await bcrypt.compare(password, user.password);

    if (!isValid) {
      // ✅ Track failed attempts
      await this.handleFailedLogin(user, ipAddress);
      throw new UnauthorizedException('Invalid credentials');
    }

    // ✅ Reset failed attempts
    await this.userRepository.update(user.id, {
      failedLoginAttempts: 0,
      lastLoginAt: new Date(),
      lastLoginIp: ipAddress,
    });

    // ✅ Check if 2FA is enabled
    if (user.twoFactorEnabled) {
      // Return temporary token for 2FA step
      const tempToken = this.generateTempToken(user.id);
      return {
        requiresTwoFactor: true,
        tempToken,
      };
    }

    // ✅ Generate secure JWT token
    const accessToken = this.generateAccessToken(user);
    const refreshToken = await this.generateRefreshToken(user);

    // ✅ Log successful login
    await this.auditLog.create({
      userId: user.id,
      action: 'LOGIN_SUCCESS',
      ipAddress,
      timestamp: new Date(),
    });

    return {
      accessToken,
      refreshToken,
      user: {
        id: user.id,
        email: user.email,
      },
    };
  }

  // ✅ Rate limiting
  private async checkRateLimit(ipAddress: string): Promise<void> {
    const key = `login_attempts:${ipAddress}`;
    const attempts = (await this.cacheManager.get<number>(key)) || 0;

    if (attempts >= 5) {
      throw new TooManyRequestsException('Too many login attempts. Please try again later.');
    }

    await this.cacheManager.set(key, attempts + 1, 900000); // 15 minutes
  }

  // ✅ Handle failed login
  private async handleFailedLogin(user: User, ipAddress: string): Promise<void> {
    const attempts = (user.failedLoginAttempts || 0) + 1;

    await this.userRepository.update(user.id, {
      failedLoginAttempts: attempts,
    });

    // ✅ Lock account after 5 failed attempts
    if (attempts >= 5) {
      const lockKey = `account_lock:${user.email}`;
      await this.cacheManager.set(lockKey, true, 900000); // Lock for 15 minutes

      // ✅ Send security alert
      await this.sendSecurityAlert(user, ipAddress);

      // ✅ Audit log
      await this.auditLog.create({
        userId: user.id,
        action: 'ACCOUNT_LOCKED',
        ipAddress,
        reason: 'Too many failed login attempts',
        timestamp: new Date(),
      });
    }
  }

  // ✅ Generate secure JWT
  private generateAccessToken(user: User): string {
    return this.jwtService.sign(
      {
        sub: user.id,
        email: user.email,
        type: 'access',
        jti: crypto.randomUUID(), // ✅ Unique token ID
      },
      {
        expiresIn: '15m', // ✅ Short-lived access token
        issuer: 'myapp.com',
        audience: 'myapp.com',
      },
    );
  }

  // ✅ Generate refresh token
  private async generateRefreshToken(user: User): Promise<string> {
    const token = crypto.randomBytes(32).toString('hex');
    const hashedToken = crypto.createHash('sha256').update(token).digest('hex');

    // ✅ Store refresh token in database
    await this.refreshTokenRepository.create({
      userId: user.id,
      token: hashedToken,
      expiresAt: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000), // 7 days
      createdAt: new Date(),
    });

    return token;
  }
}

Vulnerable Scenario 2: Session Fixation and Hijacking​

Vulnerable Code (Express.js)​

// app.ts - VULNERABLE
import express from 'express';
import session from 'express-session';

const app = express();

// ❌ VULNERABLE: Insecure session configuration
app.use(
  session({
    secret: 'keyboard cat', // ❌ Weak secret
    resave: true,
    saveUninitialized: true,
    cookie: {
      secure: false, // ❌ Allows HTTP
      httpOnly: false, // ❌ Accessible via JavaScript
      maxAge: 30 * 24 * 60 * 60 * 1000, // ❌ 30 days - too long
      sameSite: 'none', // ❌ No CSRF protection
    },
    // ❌ No session regeneration on login
  }),
);

// ❌ VULNERABLE: Session fixation
app.post('/login', async (req, res) => {
  const { email, password } = req.body;

  const user = await authenticateUser(email, password);

  if (user) {
    // ❌ Reuses existing session ID - vulnerable to fixation!
    req.session.userId = user.id;
    req.session.email = user.email;

    res.json({ success: true });
  } else {
    res.status(401).json({ error: 'Invalid credentials' });
  }
});

// ❌ VULNERABLE: No session validation
app.get('/api/profile', (req, res) => {
  // ❌ Only checks if session exists
  if (req.session.userId) {
    const user = getUser(req.session.userId);
    res.json(user);
  } else {
    res.status(401).json({ error: 'Not authenticated' });
  }
});

Attack Example: Session Fixation​

# Step 1: Attacker gets a session ID
curl -c cookies.txt https://app.example.com/

# cookies.txt now contains:
# connect.sid=s%3A1234567890abcdef

# Step 2: Attacker tricks victim to use this session
# Sends phishing link:
https://app.example.com/login?sessionid=s%3A1234567890abcdef

# Step 3: Victim logs in with the fixed session
# ❌ Session ID is NOT regenerated on login

# Step 4: Attacker uses the same session ID
curl -b "connect.sid=s%3A1234567890abcdef" https://app.example.com/api/profile

# ❌ Attacker is now logged in as the victim!

Secure Solution​

// app.ts - SECURE
import express from 'express';
import session from 'express-session';
import RedisStore from 'connect-redis';
import { createClient } from 'redis';
import crypto from 'crypto';

const app = express();

// ✅ SECURE: Redis for session storage
const redisClient = createClient({
  url: process.env.REDIS_URL,
  password: process.env.REDIS_PASSWORD,
});
redisClient.connect();

// ✅ SECURE: Strong session configuration
app.use(
  session({
    store: new RedisStore({ client: redisClient }),

    // ✅ Strong secret from environment
    secret: process.env.SESSION_SECRET, // Must be 32+ random characters

    resave: false,
    saveUninitialized: false,

    // ✅ Generate custom session ID
    genid: () => crypto.randomBytes(32).toString('hex'),

    name: 'sessionId', // ✅ Don't use default name

    cookie: {
      secure: process.env.NODE_ENV === 'production', // ✅ HTTPS only in production
      httpOnly: true, // ✅ Not accessible via JavaScript
      maxAge: 1 * 60 * 60 * 1000, // ✅ 1 hour - short-lived
      sameSite: 'strict', // ✅ CSRF protection
      domain: process.env.COOKIE_DOMAIN,
    },

    // ✅ Rolling expiration
    rolling: true,
  }),
);

// ✅ SECURE: Session regeneration on login
app.post('/login', async (req, res) => {
  const { email, password } = req.body;

  const user = await authenticateUser(email, password);

  if (user) {
    // ✅ Destroy old session
    const oldSessionId = req.session.id;

    // ✅ Regenerate session ID after login
    req.session.regenerate(async (err) => {
      if (err) {
        return res.status(500).json({ error: 'Session error' });
      }

      // ✅ Set session data
      req.session.userId = user.id;
      req.session.email = user.email;
      req.session.createdAt = Date.now();
      req.session.ipAddress = req.ip;
      req.session.userAgent = req.headers['user-agent'];

      // ✅ Log session creation
      await logSecurityEvent({
        type: 'SESSION_CREATED',
        userId: user.id,
        sessionId: req.session.id,
        oldSessionId,
        ipAddress: req.ip,
      });

      req.session.save((err) => {
        if (err) {
          return res.status(500).json({ error: 'Session save error' });
        }

        res.json({
          success: true,
          user: {
            id: user.id,
            email: user.email,
          },
        });
      });
    });
  } else {
    res.status(401).json({ error: 'Invalid credentials' });
  }
});

// ✅ SECURE: Session validation middleware
app.use(async (req, res, next) => {
  if (req.session.userId) {
    // ✅ Validate session hasn't been tampered with
    const isValid = await validateSession(req.session);

    if (!isValid) {
      req.session.destroy(() => {
        res.status(401).json({ error: 'Invalid session' });
      });
      return;
    }

    // ✅ Check for session hijacking indicators
    if (req.session.ipAddress !== req.ip) {
      await logSecurityEvent({
        type: 'SESSION_IP_MISMATCH',
        userId: req.session.userId,
        sessionId: req.session.id,
        originalIp: req.session.ipAddress,
        currentIp: req.ip,
      });

      // ✅ Require re-authentication
      req.session.destroy(() => {
        res.status(401).json({
          error: 'Session invalid. Please log in again.',
        });
      });
      return;
    }

    // ✅ Check session age
    const sessionAge = Date.now() - req.session.createdAt;
    const maxAge = 24 * 60 * 60 * 1000; // 24 hours

    if (sessionAge > maxAge) {
      req.session.destroy(() => {
        res.status(401).json({
          error: 'Session expired. Please log in again.',
        });
      });
      return;
    }
  }

  next();
});

// ✅ SECURE: Logout with session destruction
app.post('/logout', (req, res) => {
  const userId = req.session.userId;
  const sessionId = req.session.id;

  // ✅ Destroy session
  req.session.destroy(async (err) => {
    if (err) {
      return res.status(500).json({ error: 'Logout failed' });
    }

    // ✅ Clear cookie
    res.clearCookie('sessionId');

    // ✅ Log logout
    await logSecurityEvent({
      type: 'LOGOUT',
      userId,
      sessionId,
    });

    res.json({ success: true });
  });
});

// ✅ Session validation helper
async function validateSession(session: any): Promise<boolean> {
  if (!session.userId || !session.createdAt) {
    return false;
  }

  // ✅ Check if session exists in Redis
  const exists = await redisClient.exists(`sess:${session.id}`);
  if (!exists) {
    return false;
  }

  // ✅ Verify user still exists and is active
  const user = await getUserById(session.userId);
  if (!user || !user.isActive) {
    return false;
  }

  return true;
}

Vulnerable Scenario 3: Missing Multi-Factor Authentication​

Vulnerable Code​

// auth.controller.ts - VULNERABLE
@Controller('auth')
export class AuthController {
  // ❌ VULNERABLE: No 2FA option
  @Post('login')
  async login(@Body() credentials: LoginDto) {
    const user = await this.authService.validateUser(credentials.email, credentials.password);

    if (!user) {
      throw new UnauthorizedException();
    }

    // ❌ Returns access immediately - no 2FA check
    return {
      accessToken: this.authService.generateToken(user),
    };
  }
}

Secure Solution with 2FA​

// auth.controller.ts - SECURE
import * as speakeasy from 'speakeasy';
import * as QRCode from 'qrcode';

@Controller('auth')
export class AuthController {
  constructor(private authService: AuthService) {}

  // ✅ Step 1: Username/password
  @Post('login')
  async login(@Body() credentials: LoginDto, @Ip() ipAddress: string) {
    const result = await this.authService.login(credentials.email, credentials.password, ipAddress);

    // ✅ Check if 2FA is required
    if (result.requiresTwoFactor) {
      return {
        requiresTwoFactor: true,
        tempToken: result.tempToken,
      };
    }

    return result;
  }

  // ✅ Step 2: Verify 2FA code
  @Post('verify-2fa')
  @UseGuards(TempTokenGuard)
  async verify2FA(@Body() body: Verify2FADto, @Request() req) {
    const { code } = body;
    const userId = req.user.id;

    const user = await this.userRepository.findOne({
      where: { id: userId },
      select: ['id', 'email', 'twoFactorSecret'],
    });

    // ✅ Verify TOTP code
    const isValid = speakeasy.totp.verify({
      secret: user.twoFactorSecret,
      encoding: 'base32',
      token: code,
      window: 2, // Allow 1 minute clock skew
    });

    if (!isValid) {
      throw new UnauthorizedException('Invalid 2FA code');
    }

    // ✅ Generate final access token
    const accessToken = this.authService.generateAccessToken(user);
    const refreshToken = await this.authService.generateRefreshToken(user);

    return {
      accessToken,
      refreshToken,
      user: {
        id: user.id,
        email: user.email,
      },
    };
  }

  // ✅ Setup 2FA
  @Post('setup-2fa')
  @UseGuards(JwtAuthGuard)
  async setup2FA(@Request() req) {
    const user = req.user;

    // ✅ Generate secret
    const secret = speakeasy.generateSecret({
      name: `MyApp (${user.email})`,
      issuer: 'MyApp',
    });

    // ✅ Generate QR code
    const qrCode = await QRCode.toDataURL(secret.otpauth_url);

    // ✅ Store secret (encrypted)
    await this.userRepository.update(user.id, {
      twoFactorSecret: secret.base32,
      twoFactorEnabled: false, // Not enabled until verified
    });

    return {
      secret: secret.base32,
      qrCode,
    };
  }

  // ✅ Enable 2FA after verification
  @Post('enable-2fa')
  @UseGuards(JwtAuthGuard)
  async enable2FA(@Body() body: { code: string }, @Request() req) {
    const user = await this.userRepository.findOne({
      where: { id: req.user.id },
      select: ['id', 'twoFactorSecret'],
    });

    // ✅ Verify code before enabling
    const isValid = speakeasy.totp.verify({
      secret: user.twoFactorSecret,
      encoding: 'base32',
      token: body.code,
      window: 2,
    });

    if (!isValid) {
      throw new BadRequestException('Invalid code');
    }

    // ✅ Enable 2FA
    await this.userRepository.update(user.id, {
      twoFactorEnabled: true,
    });

    // ✅ Generate backup codes
    const backupCodes = this.generateBackupCodes();
    await this.saveBackupCodes(user.id, backupCodes);

    return {
      success: true,
      backupCodes, // ✅ Show once, user must save
    };
  }

  private generateBackupCodes(count: number = 10): string[] {
    const codes: string[] = [];
    for (let i = 0; i < count; i++) {
      codes.push(crypto.randomBytes(4).toString('hex').toUpperCase());
    }
    return codes;
  }
}

Prevention Checklist​

  • Strong password policy: Minimum 12 characters, complexity
  • Breach detection: Check against HaveIBeenPwned
  • Multi-factor authentication: Implement TOTP or SMS
  • Account lockout: Lock after failed attempts
  • Rate limiting: Prevent brute force attacks
  • Session management: Regenerate on login, use secure cookies
  • Secure tokens: Use cryptographically secure JWTs
  • Password reset: Secure token-based process
  • Email verification: Verify email addresses
  • Security alerts: Notify users of suspicious activity
  • Audit logging: Track authentication events

Testing Methods​

describe('Authentication Security', () => {
  it('should reject weak passwords', async () => {
    await expect(authService.register('test@example.com', 'weak')).rejects.toThrow();
  });

  it('should lock account after failed attempts', async () => {
    for (let i = 0; i < 5; i++) {
      try {
        await authService.login('test@example.com', 'wrong', '127.0.0.1');
      } catch {}
    }

    await expect(authService.login('test@example.com', 'correct', '127.0.0.1')).rejects.toThrow('locked');
  });

  it('should regenerate session on login', async () => {
    const oldSessionId = session.id;

    await authController.login(credentials);

    expect(session.id).not.toBe(oldSessionId);
  });

  it('should require 2FA when enabled', async () => {
    await userRepository.update(userId, { twoFactorEnabled: true });

    const result = await authController.login(credentials);

    expect(result.requiresTwoFactor).toBe(true);
    expect(result.accessToken).toBeUndefined();
  });
});

Real-World Impact Examples​

  2. Yahoo (2013-2014): 3 billion accounts compromised due to weak authentication
  3. British Airways (2018): Session hijacking via Magecart attack
  4. Twitter (2020): Account takeover via social engineering and weak MFA
