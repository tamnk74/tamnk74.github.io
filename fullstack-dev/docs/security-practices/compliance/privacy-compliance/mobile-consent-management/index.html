Mobile Consent Management Platform Solutions

Overview​

Mobile consent management requires specialized approaches due to native app constraints, different privacy frameworks (iOS App Tracking Transparency, Android Privacy Sandbox), and app store requirements. This guide covers implementing compliant consent management for iOS, Android, and cross-platform mobile applications.

Why Mobile Consent Management Differs​

Key Differences from Web​

  • ❌ No Cookie Banner - Mobile apps don't use cookies
  • ✅ App Tracking Transparency (ATT) - iOS requires explicit tracking consent
  • ✅ Advertising ID Control - User control over IDFA/AAID
  • ✅ App Store Compliance - Must meet Apple/Google requirements
  • ✅ Native UI/UX - Custom native consent dialogs
  • ✅ Offline Storage - Consent stored locally and synced
  • ✅ SDK Management - Control when SDKs initialize

// ConsentManager.swift
import AppTrackingTransparency
import AdSupport

class ConsentManager: ObservableObject {
    @Published var trackingStatus: ATTrackingManager.AuthorizationStatus = .notDetermined
    @Published var consentPreferences = ConsentPreferences()

    static let shared = ConsentManager()

    struct ConsentPreferences: Codable {
        var essential: Bool = true
        var analytics: Bool = false
        var advertising: Bool = false
        var personalization: Bool = false
        var timestamp: Date = Date()
        var version: String = "1.0"
    }

    private init() {
        loadConsent()
        checkTrackingStatus()
    }

    // MARK: - ATT Management

    func requestTrackingAuthorization(completion: @escaping (Bool) -> Void) {
        // Must be called after app becomes active
        ATTrackingManager.requestTrackingAuthorization { status in
            DispatchQueue.main.async {
                self.trackingStatus = status
                let granted = status == .authorized

                // Update consent based on ATT decision
                self.consentPreferences.advertising = granted
                self.consentPreferences.analytics = granted

                self.saveConsent()
                self.syncWithBackend()

                completion(granted)
            }
        }
    }

    func checkTrackingStatus() {
        trackingStatus = ATTrackingManager.trackingAuthorizationStatus
    }

    var idfa: String {
        guard trackingStatus == .authorized else {
            return "00000000-0000-0000-0000-000000000000"
        }
        return ASIdentifierManager.shared().advertisingIdentifier.uuidString
    }

    // MARK: - Consent Management

    func updateConsent(_ preferences: ConsentPreferences) {
        consentPreferences = preferences
        saveConsent()
        syncWithBackend()

        // Reinitialize SDKs based on consent
        applyConsent()
    }

    func saveConsent() {
        if let encoded = try? JSONEncoder().encode(consentPreferences) {
            UserDefaults.standard.set(encoded, forKey: "consent_preferences")
        }
    }

    func loadConsent() {
        if let data = UserDefaults.standard.data(forKey: "consent_preferences"),
           let decoded = try? JSONDecoder().decode(ConsentPreferences.self, from: data) {
            consentPreferences = decoded
        }
    }

    func withdrawConsent() {
        consentPreferences = ConsentPreferences()
        saveConsent()
        syncWithBackend()
        applyConsent()
    }

    // MARK: - SDK Control

    func applyConsent() {
        // Initialize or disable SDKs based on consent
        if consentPreferences.analytics {
            initializeAnalytics()
        } else {
            disableAnalytics()
        }

        if consentPreferences.advertising {
            initializeAdvertising()
        } else {
            disableAdvertising()
        }

        NotificationCenter.default.post(
            name: .consentChanged,
            object: consentPreferences
        )
    }

    private func initializeAnalytics() {
        // Example: Firebase Analytics
        // Analytics.setAnalyticsCollectionEnabled(true)
        print("✅ Analytics enabled")
    }

    private func disableAnalytics() {
        // Analytics.setAnalyticsCollectionEnabled(false)
        print("❌ Analytics disabled")
    }

    private func initializeAdvertising() {
        // Example: AdMob, Facebook Ads
        print("✅ Advertising enabled")
    }

    private func disableAdvertising() {
        print("❌ Advertising disabled")
    }

    // MARK: - Backend Sync

    func syncWithBackend() {
        let endpoint = URL(string: "https://api.example.com/v1/consent")!
        var request = URLRequest(url: endpoint)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")

        let payload: [String: Any] = [
            "userId": getUserId(),
            "deviceId": getDeviceId(),
            "preferences": [
                "essential": consentPreferences.essential,
                "analytics": consentPreferences.analytics,
                "advertising": consentPreferences.advertising,
                "personalization": consentPreferences.personalization
            ],
            "attStatus": trackingStatus.rawValue,
            "idfa": idfa,
            "timestamp": ISO8601DateFormatter().string(from: consentPreferences.timestamp),
            "platform": "ios",
            "appVersion": Bundle.main.infoDictionary?["CFBundleShortVersionString"] as? String ?? ""
        ]

        request.httpBody = try? JSONSerialization.data(withJSONObject: payload)

        URLSession.shared.dataTask(with: request) { data, response, error in
            if let error = error {
                print("❌ Consent sync failed: \(error.localizedDescription)")
                return
            }
            print("✅ Consent synced to backend")
        }.resume()
    }

    private func getUserId() -> String {
        // Get from auth system
        return UserDefaults.standard.string(forKey: "userId") ?? ""
    }

    private func getDeviceId() -> String {
        // Use app instance ID or vendor ID
        return UIDevice.current.identifierForVendor?.uuidString ?? ""
    }
}

// ConsentManager.kt
package com.example.app.consent

import android.content.Context
import android.content.SharedPreferences
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import com.google.android.gms.ads.identifier.AdvertisingIdClient
import com.google.gson.Gson
import kotlinx.coroutines.*
import java.util.*

class ConsentManager private constructor(private val context: Context) {

    private val prefs: SharedPreferences = context.getSharedPreferences(
        "consent_prefs",
        Context.MODE_PRIVATE
    )

    private val gson = Gson()
    private val scope = CoroutineScope(Dispatchers.IO + SupervisorJob())

    private val _consentPreferences = MutableLiveData<ConsentPreferences>()
    val consentPreferences: LiveData<ConsentPreferences> = _consentPreferences

    data class ConsentPreferences(
        val essential: Boolean = true,
        val analytics: Boolean = false,
        val advertising: Boolean = false,
        val personalization: Boolean = false,
        val timestamp: Long = System.currentTimeMillis(),
        val version: String = "1.0"
    )

    init {
        loadConsent()
    }

    companion object {
        @Volatile
        private var INSTANCE: ConsentManager? = null

        fun getInstance(context: Context): ConsentManager {
            return INSTANCE ?: synchronized(this) {
                INSTANCE ?: ConsentManager(context.applicationContext).also {
                    INSTANCE = it
                }
            }
        }
    }

    // MARK: - Consent Management

    fun updateConsent(preferences: ConsentPreferences) {
        _consentPreferences.postValue(preferences)
        saveConsent(preferences)
        syncWithBackend(preferences)
        applyConsent(preferences)
    }

    private fun saveConsent(preferences: ConsentPreferences) {
        val json = gson.toJson(preferences)
        prefs.edit().putString(CONSENT_KEY, json).apply()
    }

    private fun loadConsent() {
        val json = prefs.getString(CONSENT_KEY, null)
        val preferences = if (json != null) {
            gson.fromJson(json, ConsentPreferences::class.java)
        } else {
            ConsentPreferences()
        }
        _consentPreferences.postValue(preferences)
        applyConsent(preferences)
    }

    fun withdrawConsent() {
        val defaultPreferences = ConsentPreferences()
        updateConsent(defaultPreferences)
    }

    fun hasConsent(): Boolean {
        return prefs.contains(CONSENT_KEY)
    }

    // MARK: - Advertising ID

    suspend fun getAdvertisingId(): String = withContext(Dispatchers.IO) {
        try {
            val adInfo = AdvertisingIdClient.getAdvertisingIdInfo(context)
            if (adInfo.isLimitAdTrackingEnabled) {
                return@withContext "00000000-0000-0000-0000-000000000000"
            }
            return@withContext adInfo.id ?: "00000000-0000-0000-0000-000000000000"
        } catch (e: Exception) {
            return@withContext "00000000-0000-0000-0000-000000000000"
        }
    }

    // MARK: - SDK Control

    private fun applyConsent(preferences: ConsentPreferences) {
        if (preferences.analytics) {
            initializeAnalytics()
        } else {
            disableAnalytics()
        }

        if (preferences.advertising) {
            initializeAdvertising()
        } else {
            disableAdvertising()
        }

        // Broadcast consent change
        context.sendBroadcast(
            android.content.Intent(ACTION_CONSENT_CHANGED).apply {
                putExtra(EXTRA_CONSENT, gson.toJson(preferences))
            }
        )
    }

    private fun initializeAnalytics() {
        // Example: Firebase Analytics
        // FirebaseAnalytics.getInstance(context).setAnalyticsCollectionEnabled(true)
        android.util.Log.d(TAG, "✅ Analytics enabled")
    }

    private fun disableAnalytics() {
        // FirebaseAnalytics.getInstance(context).setAnalyticsCollectionEnabled(false)
        android.util.Log.d(TAG, "❌ Analytics disabled")
    }

    private fun initializeAdvertising() {
        // Example: AdMob, Facebook Ads
        android.util.Log.d(TAG, "✅ Advertising enabled")
    }

    private fun disableAdvertising() {
        android.util.Log.d(TAG, "❌ Advertising disabled")
    }

    // MARK: - Backend Sync

    private fun syncWithBackend(preferences: ConsentPreferences) {
        scope.launch {
            try {
                val advertisingId = getAdvertisingId()
                val payload = mapOf(
                    "userId" to getUserId(),
                    "deviceId" to getDeviceId(),
                    "preferences" to mapOf(
                        "essential" to preferences.essential,
                        "analytics" to preferences.analytics,
                        "advertising" to preferences.advertising,
                        "personalization" to preferences.personalization
                    ),
                    "advertisingId" to advertisingId,
                    "timestamp" to preferences.timestamp,
                    "platform" to "android",
                    "appVersion" to getAppVersion()
                )

                // Make API call
                // val response = apiClient.syncConsent(payload)
                android.util.Log.d(TAG, "✅ Consent synced to backend")
            } catch (e: Exception) {
                android.util.Log.e(TAG, "❌ Consent sync failed", e)
            }
        }
    }

    private fun getUserId(): String {
        return prefs.getString("userId", "") ?: ""
    }

    private fun getDeviceId(): String {
        return android.provider.Settings.Secure.getString(
            context.contentResolver,
            android.provider.Settings.Secure.ANDROID_ID
        )
    }

    private fun getAppVersion(): String {
        return try {
            val packageInfo = context.packageManager.getPackageInfo(context.packageName, 0)
            packageInfo.versionName
        } catch (e: Exception) {
            "unknown"
        }
    }

    companion object {
        private const val TAG = "ConsentManager"
        private const val CONSENT_KEY = "consent_preferences"
        const val ACTION_CONSENT_CHANGED = "com.example.app.CONSENT_CHANGED"
        const val EXTRA_CONSENT = "consent"
    }
}

// ConsentManager.ts
import AsyncStorage from '@react-native-async-storage/async-storage';
import { Platform } from 'react-native';
import { requestTrackingPermission, getTrackingStatus } from 'react-native-tracking-transparency';
import { getAdvertisingId } from '@react-native-community/google-ads-consent';

export interface ConsentPreferences {
  essential: boolean;
  analytics: boolean;
  advertising: boolean;
  personalization: boolean;
  timestamp: number;
  version: string;
}

export type TrackingStatus = 'authorized' | 'denied' | 'restricted' | 'not-determined';

class ConsentManager {
  private static instance: ConsentManager;
  private preferences: ConsentPreferences | null = null;
  private listeners: Array<(prefs: ConsentPreferences) => void> = [];

  private constructor() {
    this.loadConsent();
  }

  static getInstance(): ConsentManager {
    if (!ConsentManager.instance) {
      ConsentManager.instance = new ConsentManager();
    }
    return ConsentManager.instance;
  }

  // MARK: - Consent Management

  async getConsent(): Promise<ConsentPreferences> {
    if (!this.preferences) {
      await this.loadConsent();
    }
    return this.preferences || this.getDefaultPreferences();
  }

  async updateConsent(preferences: Partial<ConsentPreferences>): Promise<void> {
    const fullPreferences: ConsentPreferences = {
      ...this.getDefaultPreferences(),
      ...this.preferences,
      ...preferences,
      timestamp: Date.now(),
    };

    this.preferences = fullPreferences;
    await this.saveConsent(fullPreferences);
    await this.syncWithBackend(fullPreferences);
    this.applyConsent(fullPreferences);
    this.notifyListeners(fullPreferences);
  }

  async withdrawConsent(): Promise<void> {
    const defaultPreferences = this.getDefaultPreferences();
    await this.updateConsent(defaultPreferences);
  }

  async hasConsent(): Promise<boolean> {
    try {
      const stored = await AsyncStorage.getItem('consent_preferences');
      return stored !== null;
    } catch {
      return false;
    }
  }

  subscribe(listener: (prefs: ConsentPreferences) => void): () => void {
    this.listeners.push(listener);
    return () => {
      this.listeners = this.listeners.filter((l) => l !== listener);
    };
  }

  // MARK: - ATT (iOS) / Advertising ID (Android)

  async requestTrackingPermission(): Promise<TrackingStatus> {
    if (Platform.OS === 'ios') {
      const status = await requestTrackingPermission();
      const granted = status === 'authorized';

      // Update consent based on ATT decision
      await this.updateConsent({
        advertising: granted,
        analytics: granted,
      });

      return status;
    } else {
      // Android doesn't have ATT, handle differently
      return 'authorized';
    }
  }

  async getTrackingPermissionStatus(): Promise<TrackingStatus> {
    if (Platform.OS === 'ios') {
      return await getTrackingStatus();
    }
    return 'authorized';
  }

  async getAdvertisingId(): Promise<string> {
    try {
      const preferences = await this.getConsent();

      if (!preferences.advertising) {
        return '00000000-0000-0000-0000-000000000000';
      }

      if (Platform.OS === 'ios') {
        const status = await this.getTrackingPermissionStatus();
        if (status !== 'authorized') {
          return '00000000-0000-0000-0000-000000000000';
        }
      }

      const adId = await getAdvertisingId();
      return adId || '00000000-0000-0000-0000-000000000000';
    } catch {
      return '00000000-0000-0000-0000-000000000000';
    }
  }

  // MARK: - Storage

  private async saveConsent(preferences: ConsentPreferences): Promise<void> {
    try {
      await AsyncStorage.setItem('consent_preferences', JSON.stringify(preferences));
    } catch (error) {
      console.error('Failed to save consent:', error);
    }
  }

  private async loadConsent(): Promise<void> {
    try {
      const stored = await AsyncStorage.getItem('consent_preferences');
      if (stored) {
        this.preferences = JSON.parse(stored);
      }
    } catch (error) {
      console.error('Failed to load consent:', error);
    }
  }

  // MARK: - SDK Control

  private applyConsent(preferences: ConsentPreferences): void {
    if (preferences.analytics) {
      this.initializeAnalytics();
    } else {
      this.disableAnalytics();
    }

    if (preferences.advertising) {
      this.initializeAdvertising();
    } else {
      this.disableAdvertising();
    }
  }

  private initializeAnalytics(): void {
    // Example: Firebase Analytics
    // analytics().setAnalyticsCollectionEnabled(true);
    console.log('✅ Analytics enabled');
  }

  private disableAnalytics(): void {
    // analytics().setAnalyticsCollectionEnabled(false);
    console.log('❌ Analytics disabled');
  }

  private initializeAdvertising(): void {
    // Example: AdMob
    console.log('✅ Advertising enabled');
  }

  private disableAdvertising(): void {
    console.log('❌ Advertising disabled');
  }

  // MARK: - Backend Sync

  private async syncWithBackend(preferences: ConsentPreferences): Promise<void> {
    try {
      const advertisingId = await this.getAdvertisingId();
      const trackingStatus = await this.getTrackingPermissionStatus();

      const response = await fetch('https://api.example.com/v1/consent', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          userId: await this.getUserId(),
          deviceId: await this.getDeviceId(),
          preferences: {
            essential: preferences.essential,
            analytics: preferences.analytics,
            advertising: preferences.advertising,
            personalization: preferences.personalization,
          },
          advertisingId,
          trackingStatus,
          timestamp: preferences.timestamp,
          platform: Platform.OS,
          appVersion: this.getAppVersion(),
        }),
      });

      if (!response.ok) {
        throw new Error('Failed to sync consent');
      }

      console.log('✅ Consent synced to backend');
    } catch (error) {
      console.error('❌ Consent sync failed:', error);
    }
  }

  // MARK: - Helpers

  private getDefaultPreferences(): ConsentPreferences {
    return {
      essential: true,
      analytics: false,
      advertising: false,
      personalization: false,
      timestamp: Date.now(),
      version: '1.0',
    };
  }

  private notifyListeners(preferences: ConsentPreferences): void {
    this.listeners.forEach((listener) => listener(preferences));
  }

  private async getUserId(): Promise<string> {
    try {
      return (await AsyncStorage.getItem('userId')) || '';
    } catch {
      return '';
    }
  }

class ConsentPreferences {
  final bool essential;
  final bool analytics;
  final bool advertising;
  final bool personalization;
  final DateTime timestamp;
  final String version;

  ConsentPreferences({
    this.essential = true,
    this.analytics = false,
    this.advertising = false,
    this.personalization = false,
    DateTime? timestamp,
    this.version = '1.0',
  }) : timestamp = timestamp ?? DateTime.now();

  Map<String, dynamic> toJson() => {
        'essential': essential,
        'analytics': analytics,
        'advertising': advertising,
        'personalization': personalization,
        'timestamp': timestamp.toIso8601String(),
        'version': version,
      };

  factory ConsentPreferences.fromJson(Map<String, dynamic> json) {
    return ConsentPreferences(
      essential: json['essential'] ?? true,
      analytics: json['analytics'] ?? false,
      advertising: json['advertising'] ?? false,
      personalization: json['personalization'] ?? false,
      timestamp: json['timestamp'] != null
          ? DateTime.parse(json['timestamp'])
          : DateTime.now(),
      version: json['version'] ?? '1.0',
    );
  }

  ConsentPreferences copyWith({
    bool? essential,
    bool? analytics,
    bool? advertising,
    bool? personalization,
    DateTime? timestamp,
    String? version,
  }) {
    return ConsentPreferences(
      essential: essential ?? this.essential,
      analytics: analytics ?? this.analytics,
      advertising: advertising ?? this.advertising,
      personalization: personalization ?? this.personalization,
      timestamp: timestamp ?? this.timestamp,
      version: version ?? this.version,
    );
  }
}

class ConsentManager extends ChangeNotifier {
  static final ConsentManager _instance = ConsentManager._internal();
  factory ConsentManager() => _instance;

  ConsentManager._internal() {
    _init();
  }

  ConsentPreferences _preferences = ConsentPreferences();
  ConsentPreferences get preferences => _preferences;

  TrackingStatus _trackingStatus = TrackingStatus.notDetermined;
  TrackingStatus get trackingStatus => _trackingStatus;

  String _advertisingId = '00000000-0000-0000-0000-000000000000';
  String get advertisingId => _advertisingId;

  bool _initialized = false;
  bool get initialized => _initialized;

  static const String _consentKey = 'consent_preferences';

  // MARK: - Initialization

  Future<void> _init() async {
    await _loadConsent();
    await _checkTrackingStatus();
    _initialized = true;
    notifyListeners();
  }

  // MARK: - ATT (iOS) / Advertising ID

  Future<void> requestTrackingPermission() async {
    if (!Platform.isIOS) {
      debugPrint('⚠️ ATT only available on iOS');
      return;
    }

    try {
      final status = await AppTrackingTransparency.requestTrackingAuthorization();
      _trackingStatus = status;

      final granted = status == TrackingStatus.authorized;

      // Update consent based on ATT decision
      await updateConsent(
        _preferences.copyWith(
          advertising: granted,
          analytics: granted,
        ),
      );

      debugPrint('✅ ATT Status: $status');
    } catch (e) {
      debugPrint('❌ ATT request failed: $e');
    }
  }

  Future<void> _checkTrackingStatus() async {
    if (Platform.isIOS) {
      try {
        _trackingStatus = await AppTrackingTransparency.trackingAuthorizationStatus;
      } catch (e) {
        debugPrint('❌ Failed to check ATT status: $e');
      }
    } else {
      _trackingStatus = TrackingStatus.authorized;
    }
  }

  Future<String> getAdvertisingId() async {
    try {
      if (!_preferences.advertising) {
        return '00000000-0000-0000-0000-000000000000';
      }

      if (Platform.isIOS && _trackingStatus != TrackingStatus.authorized) {
        return '00000000-0000-0000-0000-000000000000';
      }

      final adId = await AdvertisingId.id(true);
      _advertisingId = adId ?? '00000000-0000-0000-0000-000000000000';
      return _advertisingId;
    } catch (e) {
      debugPrint('❌ Failed to get advertising ID: $e');
      return '00000000-0000-0000-0000-000000000000';
    }
  }

  // MARK: - Consent Management

  Future<void> updateConsent(ConsentPreferences preferences) async {
    _preferences = preferences.copyWith(timestamp: DateTime.now());
    await _saveConsent();
    await _syncWithBackend();
    await _applyConsent();
    notifyListeners();
  }

  Future<void> withdrawConsent() async {
    await updateConsent(ConsentPreferences());
  }

  Future<bool> hasConsent() async {
    final prefs = await SharedPreferences.getInstance();
    return prefs.containsKey(_consentKey);
  }

  // MARK: - Storage

  Future<void> _saveConsent() async {
    try {
      final prefs = await SharedPreferences.getInstance();
      final json = jsonEncode(_preferences.toJson());
      await prefs.setString(_consentKey, json);
      debugPrint('✅ Consent saved');
    } catch (e) {
      debugPrint('❌ Failed to save consent: $e');
    }
  }

  Future<void> _loadConsent() async {
    try {
      final prefs = await SharedPreferences.getInstance();
      final json = prefs.getString(_consentKey);

      if (json != null) {
        final data = jsonDecode(json) as Map<String, dynamic>;
        _preferences = ConsentPreferences.fromJson(data);
        await _applyConsent();
        debugPrint('✅ Consent loaded');
      }
    } catch (e) {
      debugPrint('❌ Failed to load consent: $e');
    }
  }

  // MARK: - SDK Control

  Future<void> _applyConsent() async {
    if (_preferences.analytics) {
      _initializeAnalytics();
    } else {
      _disableAnalytics();
    }

    if (_preferences.advertising) {
      _initializeAdvertising();
    } else {
      _disableAdvertising();
    }
  }

  void _initializeAnalytics() {
    // Example: Firebase Analytics
    // FirebaseAnalytics.instance.setAnalyticsCollectionEnabled(true);
    debugPrint('✅ Analytics enabled');
  }

  void _disableAnalytics() {
    // FirebaseAnalytics.instance.setAnalyticsCollectionEnabled(false);
    debugPrint('❌ Analytics disabled');
  }

  void _initializeAdvertising() {
    // Example: Google Mobile Ads
    debugPrint('✅ Advertising enabled');
  }

  void _disableAdvertising() {
    debugPrint('❌ Advertising disabled');
  }

  // MARK: - Backend Sync

  Future<void> _syncWithBackend() async {
    try {
      final advertisingId = await getAdvertisingId();
      final deviceInfo = await _getDeviceInfo();
      final packageInfo = await PackageInfo.fromPlatform();

      final payload = {
        'userId': await _getUserId(),
        'deviceId': deviceInfo['deviceId'],
        'preferences': {
          'essential': _preferences.essential,
          'analytics': _preferences.analytics,
          'advertising': _preferences.advertising,
          'personalization': _preferences.personalization,
        },
        'advertisingId': advertisingId,
        'attStatus': _trackingStatus.toString(),
        'timestamp': _preferences.timestamp.toIso8601String(),
        'platform': Platform.operatingSystem,
        'appVersion': packageInfo.version,
        'osVersion': deviceInfo['osVersion'],
      };

      final response = await http.post(
        Uri.parse('https://api.example.com/v1/consent'),
        headers: {'Content-Type': 'application/json'},
        body: jsonEncode(payload),
      );

      if (response.statusCode == 200) {
        debugPrint('✅ Consent synced to backend');
      } else {
        debugPrint('❌ Consent sync failed: ${response.statusCode}');
      }
    } catch (e) {
      debugPrint('❌ Consent sync error: $e');
    }
  }

  // MARK: - Helpers

  Future<String> _getUserId() async {
    final prefs = await SharedPreferences.getInstance();
    return prefs.getString('userId') ?? '';
  }

  Future<Map<String, String>> _getDeviceInfo() async {
    final deviceInfo = DeviceInfoPlugin();

    if (Platform.isIOS) {
      final iosInfo = await deviceInfo.iosInfo;
      return {
        'deviceId': iosInfo.identifierForVendor ?? '',
        'osVersion': iosInfo.systemVersion,
      };
    } else if (Platform.isAndroid) {
      final androidInfo = await deviceInfo.androidInfo;
      return {
        'deviceId': androidInfo.id,
        'osVersion': androidInfo.version.release,
      };
    }

    return {'deviceId': '', 'osVersion': ''};
  }

  // MARK: - Testing

  Future<void> resetConsent() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.remove(_consentKey);
    _preferences = ConsentPreferences();
    _trackingStatus = TrackingStatus.notDetermined;
    _advertisingId = '00000000-0000-0000-0000-000000000000';
    notifyListeners();
    debugPrint('✅ Consent reset');
  }

            ),
          ],
        ),
      ),
    );
  }

  Widget _buildActions(BuildContext context, ConsentManager manager) {
    return Column(
      children: [
        // Accept All Button
        SizedBox(
          width: double.infinity,
          child: ElevatedButton(
            onPressed: () async {
              await manager.updateConsent(
                manager.preferences.copyWith(
                  analytics: true,
                  advertising: true,
                  personalization: true,
                ),
              );

              if (Theme.of(context).platform == TargetPlatform.iOS &&
                  manager.trackingStatus == TrackingStatus.notDetermined) {
                await manager.requestTrackingPermission();
              }

              if (context.mounted) {
                Navigator.of(context).pop();
              }
            },
            style: ElevatedButton.styleFrom(
              padding: const EdgeInsets.symmetric(vertical: 16),
            ),
            child: const Text('Accept All'),
          ),
        ),
        const SizedBox(height: 12),

        // Save Preferences Button
        SizedBox(
          width: double.infinity,
          child: OutlinedButton(
            onPressed: () async {
              await manager.updateConsent(
                manager.preferences.copyWith(
                  analytics: _analytics,
                  advertising: _advertising,
                  personalization: _personalization,
                ),
              );

              if (context.mounted) {
                Navigator.of(context).pop();
              }
            },
            style: OutlinedButton.styleFrom(
              padding: const EdgeInsets.symmetric(vertical: 16),
            ),
            child: const Text('Save Preferences'),
          ),
        ),
        const SizedBox(height: 12),

        // Privacy Policy Link
        TextButton(
          onPressed: () {
            // Open privacy policy
          },
          child: const Text('Privacy Policy'),
        ),
      ],
    );
  }
}

        useMaterial3: true,

      home: const HomePage(),
    );

class HomePage extends StatefulWidget {
  const HomePage({Key? key}) : super(key: key);

  @override
  State<HomePage> createState() => _HomePageState();
}

class _HomePageState extends State<HomePage> {
  @override
  void initState() {
    super.initState();
    _checkConsent();
  }

  Future<void> _checkConsent() async {
    final manager = context.read<ConsentManager>();

    // Wait for initialization
    while (!manager.initialized) {
      await Future.delayed(const Duration(milliseconds: 100));
    }

    // Show consent dialog on first launch
    final hasConsent = await manager.hasConsent();
    if (!hasConsent && mounted) {
      // Delay for better UX
      await Future.delayed(const Duration(milliseconds: 500));

      if (mounted) {
        await Navigator.of(context).push(
          MaterialPageRoute(
            builder: (context) => const ConsentScreen(),
            fullscreenDialog: true,
          ),
        );
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    final manager = context.watch<ConsentManager>();

    return Scaffold(
      appBar: AppBar(
        title: const Text('Home'),
        actions: [
          IconButton(
            icon: const Icon(Icons.settings),
            onPressed: () {
              Navigator.of(context).push(
                MaterialPageRoute(
                  builder: (context) => const ConsentScreen(),
                ),
              );
            },
          ),
        ],
      ),
      body: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            const Text(
              'Consent Status',
              style: TextStyle(fontSize: 24, fontWeight: FontWeight.bold),
            ),
            const SizedBox(height: 20),
            _buildConsentStatus('Analytics', manager.preferences.analytics),
            _buildConsentStatus('Advertising', manager.preferences.advertising),
            _buildConsentStatus('Personalization', manager.preferences.personalization),
            const SizedBox(height: 30),
            ElevatedButton(
              onPressed: () {
                Navigator.of(context).push(
                  MaterialPageRoute(
                    builder: (context) => const ConsentScreen(),
                  ),
                );
              },
              child: const Text('Manage Preferences'),
            ),
          ],
        ),
      ),
    );
  }

  Widget _buildConsentStatus(String label, bool enabled) {
    return Padding(
      padding: const EdgeInsets.symmetric(vertical: 8),
      child: Row(
        mainAxisAlignment: MainAxisAlignment.center,
        children: [
          Icon(
            enabled ? Icons.check_circle : Icons.cancel,
            color: enabled ? Colors.green : Colors.red,
          ),
          const SizedBox(width: 8),
          Text('$label: ${enabled ? "Enabled" : "Disabled"}'),
        ],
      ),
    );
  }
}

Best Practices for Mobile Consent​

1. Timing & UX​

// Show consent on app launch (first time only)
useEffect(() => {
  const checkConsent = async () => {
    const hasConsent = await ConsentManager.hasConsent();
    if (!hasConsent) {
      // Delay slightly for better UX
      setTimeout(() => {
        setShowConsentModal(true);
      }, 1000);
    }
  };

  checkConsent();
}, []);

2. Graceful Degradation​

// Provide value even without consent
func loadContent() {
    let consent = consentManager.consentPreferences

    if consent.personalization {
        loadPersonalizedContent()
    } else {
        loadGenericContent() // Still provide value!
    }
}

3. Settings Access​

// Make consent settings easily accessible
fun navigateToConsentSettings() {
    val intent = Intent(this, ConsentSettingsActivity::class.java)
    startActivity(intent)
}

4. Offline Support​

// Queue consent updates when offline
private async syncWithBackend(preferences: ConsentPreferences): Promise<void> {
  if (!isOnline()) {
    await queueForSync(preferences);
    return;
  }

  // Sync immediately when online
  await this.performSync(preferences);
}

5. Testing​

// Track consent rates
const trackConsentMetrics = (preferences: ConsentPreferences) => {

Summary​

Mobile consent management requires:

  1. Platform-Specific Implementation - Native iOS (ATT), Android (Advertising ID)
  2. User-Friendly UI - Clear, native dialogs
  3. Offline Support - Local storage with backend sync
  4. SDK Control - Initialize/disable based on consent
  5. App Store Compliance - Privacy manifests and data safety
  6. Cross-Platform Support - React Native, Flutter implementations
  7. Regular Audits - Ensure ongoing compliance

Choose your approach:

  • Enterprise: OneTrust or Usercentrics
  • Startups: Build custom (examples provided)
  • Hybrid Apps: Cookiebot or iubenda

Always test consent flows thoroughly and monitor consent rates to optimize your approach!
