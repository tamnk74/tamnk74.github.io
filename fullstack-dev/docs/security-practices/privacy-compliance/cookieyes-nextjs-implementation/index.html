CookieYes Integration with Next.js

Overview​

This guide provides a step-by-step implementation for integrating CookieYes cookie consent management with Next.js applications, supporting both App Router (Next.js 13+) and Pages Router architectures.

Why CookieYes + Next.js?​

  • ✅ Simple Integration - Quick setup in Next.js projects
  • ✅ Affordable - Free tier for small projects, affordable paid plans
  • ✅ SSR Compatible - Works with server-side rendering
  • ✅ Performance - Minimal impact on page load
  • ✅ Multi-language - 20+ languages out of the box
  • ✅ Google Consent Mode v2 - Full support

2.1 Create CookieYes Script Component​

// app/components/CookieYes.tsx
'use client';

import Script from 'next/script';
import { useEffect } from 'react';

interface CookieYesProps {
  siteId: string;
  onConsentUpdate?: (consent: ConsentPreferences) => void;
}

interface ConsentPreferences {
  essential: boolean;
  functional: boolean;
  analytics: boolean;
  performance: boolean;
  advertisement: boolean;
}

export default function CookieYes({ siteId, onConsentUpdate }: CookieYesProps) {
  useEffect(() => {
    // Listen for consent updates
    const handleConsentUpdate = (event: CustomEvent) => {
      const { accepted, rejected } = event.detail;

      const consent: ConsentPreferences = {
        essential: true, // Always true
        functional: accepted.includes('functional'),
        analytics: accepted.includes('analytics'),
        performance: accepted.includes('performance'),
        advertisement: accepted.includes('advertisement'),
      };

      console.log('CookieYes consent updated:', consent);

      // Call parent callback
      if (onConsentUpdate) {
        onConsentUpdate(consent);
      }

      // Sync with your backend
      syncConsentToBackend(consent);
    };

    // Type assertion for custom event
    document.addEventListener('cookieyes_consent_update', handleConsentUpdate as EventListener);

    return () => {
      document.removeEventListener('cookieyes_consent_update', handleConsentUpdate as EventListener);
    };
  }, [onConsentUpdate]);

  return (
    <Script id="cookieyes" src={`https://cdn-cookieyes.com/client_data/${siteId}.js`} strategy="afterInteractive" />
  );
}

// Helper function to sync consent to your backend
async function syncConsentToBackend(consent: ConsentPreferences) {
  try {
    await fetch('/api/consent', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        preferences: consent,
        timestamp: new Date().toISOString(),
        sessionId: getSessionId(),
      }),
    });
  } catch (error) {
    console.error('Failed to sync consent to backend:', error);
  }
}

function getSessionId(): string {
  if (typeof window === 'undefined') return '';

  let sessionId = sessionStorage.getItem('session_id');
  if (!sessionId) {
    sessionId = `${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
    sessionStorage.setItem('session_id', sessionId);
  }
  return sessionId;
}

Step 5: Custom React Hook for Consent​

// hooks/useCookieConsent.ts
'use client';

import { useState, useEffect } from 'react';

interface ConsentState {
  essential: boolean;
  functional: boolean;
  analytics: boolean;
  performance: boolean;
  advertisement: boolean;
}

export function useCookieConsent() {
  const [consent, setConsent] = useState<ConsentState>({
    essential: true,
    functional: false,
    analytics: false,
    performance: false,
    advertisement: false,
  });

  const [isLoaded, setIsLoaded] = useState(false);

  useEffect(() => {
    const updateConsent = (event?: CustomEvent) => {
      if (typeof window !== 'undefined' && (window as any).cookieyes) {
        const cy = (window as any).cookieyes;
        const accepted = event ? event.detail.accepted : cy.consent?.accepted || [];

        setConsent({
          essential: true,
          functional: accepted.includes('functional'),
          analytics: accepted.includes('analytics'),
          performance: accepted.includes('performance'),
          advertisement: accepted.includes('advertisement'),
        });
        setIsLoaded(true);
      }
    };

    // Check if already loaded
    updateConsent();

    // Listen for updates
    document.addEventListener('cookieyes_consent_update', updateConsent as EventListener);

    return () => {
      document.removeEventListener('cookieyes_consent_update', updateConsent as EventListener);
    };
  }, []);

  const updatePreferences = () => {
    if (typeof window !== 'undefined' && (window as any).cookieyes) {
      (window as any).cookieyes.showSettings();
    }
  };

  const acceptAll = () => {
    if (typeof window !== 'undefined' && (window as any).cookieyes) {
      (window as any).cookieyes.acceptAll();
    }
  };

  const rejectAll = () => {
    if (typeof window !== 'undefined' && (window as any).cookieyes) {
      (window as any).cookieyes.rejectAll();
    }
  };

  return {
    consent,
    isLoaded,
    updatePreferences,
    acceptAll,
    rejectAll,
  };
}

Step 6: Server-Side Consent Check​

// middleware.ts
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

export function middleware(request: NextRequest) {
  const consentCookie = request.cookies.get('cookieyes-consent');

  if (consentCookie) {
    try {
      const consent = JSON.parse(consentCookie.value);

      // Add consent info to headers for server components
      const requestHeaders = new Headers(request.headers);
      requestHeaders.set('x-consent-analytics', consent.analytics ? 'true' : 'false');
      requestHeaders.set('x-consent-marketing', consent.advertisement ? 'true' : 'false');

      return NextResponse.next({
        request: {
          headers: requestHeaders,
        },
      });
    } catch (error) {
      console.error('Error parsing consent cookie:', error);
    }
  }

  return NextResponse.next();
}

// app/components/ProductTracking.tsx
'use client';

import { useCookieConsent } from '@/hooks/useCookieConsent';
import { useEffect } from 'react';

interface Product {
  id: string;
  name: string;
  price: number;
}

export default function ProductTracking({ product }: { product: Product }) {
  const { consent } = useCookieConsent();

  useEffect(() => {
    if (consent.analytics && typeof window !== 'undefined') {
      // Track product view
      trackProductView(product);
    }
  }, [consent.analytics, product]);

  const handleAddToCart = () => {
    // Add to cart logic...

    if (consent.analytics) {
      trackAddToCart(product);
    }
  };

  const handlePurchase = () => {
    // Purchase logic...

    if (consent.analytics) {
      trackPurchase(product);
    }

    if (consent.advertisement) {
      trackConversion(product);
    }
  };

  return (

  );
}

function trackProductView(product: Product) {
  if ((window as any).gtag) {
    (window as any).gtag('event', 'view_item', {
      items: [
        {
          id: product.id,
          name: product.name,
          price: product.price,
        },
      ],
    });
  }
}

function trackAddToCart(product: Product) {
  if ((window as any).gtag) {
    (window as any).gtag('event', 'add_to_cart', {
      items: [

2.2 Update Consent Based on CookieYes​

// app/components/CookieYesGTM.tsx
'use client';

import Script from 'next/script';
import { useEffect } from 'react';

interface CookieYesGTMProps {
  siteId: string;
}

export default function CookieYesGTM({ siteId }: CookieYesGTMProps) {
  useEffect(() => {
    const updateGTMConsent = (event?: CustomEvent) => {
      if (typeof window === 'undefined' || !(window as any).gtag) return;

      const gtag = (window as any).gtag;
      let consent = {
        analytics: false,
        advertisement: false,
        functional: false,
      };

      if (event) {
        // From consent update event
        const { accepted } = event.detail;
        consent = {
          analytics: accepted.includes('analytics'),
          advertisement: accepted.includes('advertisement'),
          functional: accepted.includes('functional'),
        };
      } else if ((window as any).cookieyes) {
        // From existing consent
        const cy = (window as any).cookieyes.consent;
        consent = {
          analytics: cy?.analytics || false,
          advertisement: cy?.advertisement || false,
          functional: cy?.functional || false,
        };
      }

      // Update Google Consent Mode
      gtag('consent', 'update', {
        ad_storage: consent.advertisement ? 'granted' : 'denied',
        ad_user_data: consent.advertisement ? 'granted' : 'denied',
        ad_personalization: consent.advertisement ? 'granted' : 'denied',
        analytics_storage: consent.analytics ? 'granted' : 'denied',
        functionality_storage: consent.functional ? 'granted' : 'denied',
        personalization_storage: consent.functional ? 'granted' : 'denied',
      });

      // Push event to dataLayer for GTM triggers
      (window as any).dataLayer = (window as any).dataLayer || [];
      (window as any).dataLayer.push({
        event: 'consent_updated',
        consent_analytics: consent.analytics,
        consent_advertisement: consent.advertisement,
        consent_functional: consent.functional,
      });

      console.log('✅ GTM Consent updated:', consent);
    };

    // Listen for consent updates
    document.addEventListener('cookieyes_consent_update', updateGTMConsent as EventListener);

    // Check initial consent after CookieYes loads
    const checkInterval = setInterval(() => {
      if ((window as any).cookieyes) {
        updateGTMConsent();
        clearInterval(checkInterval);
      }
    }, 100);

    // Cleanup
    return () => {
      document.removeEventListener('cookieyes_consent_update', updateGTMConsent as EventListener);
      clearInterval(checkInterval);
    };
  }, []);

  return (
    <Script id="cookieyes" src={`https://cdn-cookieyes.com/client_data/${siteId}.js`} strategy="afterInteractive" />
  );
}

5.1 Custom Events with Consent Check​

// utils/gtm.ts
export const GTM = {
  // Push event to dataLayer
  push: (event: Record<string, any>) => {
    if (typeof window !== 'undefined' && (window as any).dataLayer) {
      (window as any).dataLayer.push(event);
    }
  },

  // Track page view
  pageView: (url: string) => {
    GTM.push({
      event: 'page_view',
      page_location: url,
    });
  },

  // Track custom event (checks analytics consent)
  trackEvent: (eventName: string, parameters?: Record<string, any>) => {
    const consent = GTM.getConsent();

    if (!consent.analytics) {
      console.warn('Analytics consent not granted, event not tracked:', eventName);
      return;
    }

    GTM.push({
      event: eventName,
      ...parameters,
    });
  },

  // Get current consent state
  getConsent: () => {
    const dataLayer = (window as any).dataLayer || [];
    const consentData = dataLayer.find((item: any) => item.event === 'consent_updated');

    return {
      analytics: consentData?.consent_analytics || false,
      advertisement: consentData?.consent_advertisement || false,
      functional: consentData?.consent_functional || false,
    };
  },
};
