🔧 Installation & Setup​

Docker Compose Setup​

# docker-compose.yml
version: '3.7'

services:
  elasticsearch:
    image: docker.elastic.co/elasticsearch/elasticsearch:8.11.0
    container_name: elasticsearch
    environment:
      - discovery.type=single-node
      - "ES_JAVA_OPTS=-Xms512m -Xmx512m"
      - xpack.security.enabled=false
    ports:
      - "9200:9200"
    volumes:
      - es_data:/usr/share/elasticsearch/data

  logstash:
    image: docker.elastic.co/logstash/logstash:8.11.0
    container_name: logstash
    volumes:
      - ./logstash/pipeline:/usr/share/logstash/pipeline
      - ./logstash/config/logstash.yml:/usr/share/logstash/config/logstash.yml
    ports:
      - "5044:5044"
      - "9600:9600"
    depends_on:
      - elasticsearch

  kibana:
    image: docker.elastic.co/kibana/kibana:8.11.0
    container_name: kibana
    ports:
      - "5601:5601"
    environment:
      - ELASTICSEARCH_HOSTS=http://elasticsearch:9200
    depends_on:
      - elasticsearch

volumes:
  es_data:

Start ELK Stack​

# Start services
docker-compose up -d

# Check status
docker-compose ps

# View logs
docker-compose logs -f

# Access Kibana
open http://localhost:5601

Index Management​

# Create index
curl -X PUT "localhost:9200/security-logs" -H 'Content-Type: application/json' -d'
{
  "settings": {
    "number_of_shards": 1,
    "number_of_replicas": 0
  },
  "mappings": {
    "properties": {
      "timestamp": {"type": "date"},
      "source_ip": {"type": "ip"},
      "destination_ip": {"type": "ip"},
      "event_type": {"type": "keyword"},
      "severity": {"type": "keyword"},
      "message": {"type": "text"}
    }
  }
}'

# View index
curl "localhost:9200/security-logs?pretty"

# Delete index
curl -X DELETE "localhost:9200/security-logs"

Search Queries​

// Simple search
GET /security-logs/_search
{
  "query": {
    "match": {
      "event_type": "failed_login"
    }
  }
}

// Range query
GET /security-logs/_search
{
  "query": {
    "range": {
      "timestamp": {
        "gte": "now-1h",
        "lte": "now"
      }
    }
  }
}

// Boolean query
GET /security-logs/_search
{
  "query": {
    "bool": {
      "must": [
        {"match": {"event_type": "failed_login"}},
        {"range": {"timestamp": {"gte": "now-24h"}}}
      ],
      "filter": [
        {"term": {"severity": "high"}}
      ]
    }
  }
}

// Aggregations
GET /security-logs/_search
{
  "size": 0,
  "aggs": {
    "failed_logins_by_ip": {
      "terms": {
        "field": "source_ip",
        "size": 10
      }
    }
  }
}

🔄 Logstash Pipelines​

SSH Auth Log Pipeline​

# logstash/pipeline/ssh-auth.conf
input {
  file {
    path => "/var/log/auth.log"
    start_position => "beginning"
    sincedb_path => "/dev/null"
  }
}

filter {
  grok {
    match => {
      "message" => "%{SYSLOGTIMESTAMP:timestamp} %{HOSTNAME:hostname} sshd\[%{NUMBER:pid}\]: %{WORD:auth_method} %{WORD:status} for %{USERNAME:username} from %{IP:source_ip} port %{NUMBER:port}"
    }
  }

  if [status] == "Failed" {
    mutate {
      add_field => { "event_type" => "failed_login" }
      add_field => { "severity" => "high" }
    }
  }

  date {
    match => ["timestamp", "MMM dd HH:mm:ss"]
    target => "@timestamp"
  }

  geoip {
    source => "source_ip"
    target => "geoip"
  }
}

output {
  elasticsearch {
    hosts => ["elasticsearch:9200"]
    index => "ssh-auth-%{+YYYY.MM.dd}"
  }
  
  stdout {
    codec => rubydebug
  }
}

Apache Access Log Pipeline​

# logstash/pipeline/apache.conf
input {
  file {
    path => "/var/log/apache2/access.log"
    start_position => "beginning"
  }
}

filter {
  grok {
    match => {
      "message" => '%{IPORHOST:client_ip} %{USER:ident} %{USER:auth} \[%{HTTPDATE:timestamp}\] "%{WORD:method} %{DATA:request} HTTP/%{NUMBER:http_version}" %{NUMBER:response_code} %{NUMBER:bytes}'
    }
  }

  # Detect suspicious activity
  if [response_code] == "404" {
    mutate {
      add_field => { "alert" => "potential_scan" }
    }
  }

  if [response_code] =~ /^5/ {
    mutate {
      add_field => { "alert" => "server_error" }
    }
  }

  # SQL injection detection
  if [request] =~ /(union|select|insert|update|delete|drop|create|alter)/i {
    mutate {
      add_field => { "alert" => "sql_injection_attempt" }
      add_field => { "severity" => "critical" }
    }
  }

  date {
    match => ["timestamp", "dd/MMM/yyyy:HH:mm:ss Z"]
    target => "@timestamp"
  }
}

output {
  elasticsearch {
    hosts => ["elasticsearch:9200"]
    index => "apache-%{+YYYY.MM.dd}"
  }
}

Firewall Log Pipeline​

# logstash/pipeline/firewall.conf
input {
  syslog {
    port => 5514
    type => "firewall"
  }
}

filter {
  if [type] == "firewall" {
    grok {
      match => {
        "message" => "%{WORD:action} %{WORD:protocol} %{IP:src_ip}:%{NUMBER:src_port} -> %{IP:dst_ip}:%{NUMBER:dst_port}"
      }
    }

    if [action] == "DENY" or [action] == "DROP" {
      mutate {
        add_field => { "event_type" => "blocked_connection" }
      }
    }

    # Detect port scanning
    if [dst_port] in ["21", "22", "23", "3389", "445"] {
      mutate {
        add_field => { "alert" => "common_port_scan" }
      }
    }
  }
}

output {
  elasticsearch {
    hosts => ["elasticsearch:9200"]
    index => "firewall-%{+YYYY.MM.dd}"
  }
}

🚨 Alerting Rules​

Watcher for Failed Logins​

PUT _watcher/watch/failed_login_alert
{
  "trigger": {
    "schedule": {
      "interval": "5m"
    }
  },
  "input": {
    "search": {
      "request": {
        "indices": ["ssh-auth-*"],
        "body": {
          "query": {
            "bool": {
              "filter": [
                {
                  "match": {
                    "event_type": "failed_login"
                  }
                },
                {
                  "range": {
                    "@timestamp": {
                      "gte": "now-5m"
                    }
                  }
                }
              ]
            }
          },
          "aggs": {
            "by_ip": {
              "terms": {
                "field": "source_ip.keyword",
                "size": 10
              }
            }
          }
        }
      }
    }
  },
  "condition": {
    "compare": {
      "ctx.payload.aggregations.by_ip.buckets.0.doc_count": {
        "gt": 5
      }
    }
  },
  "actions": {
    "send_email": {
      "email": {
        "to": "security@example.com",
        "subject": "Multiple Failed Login Attempts Detected",
        "body": "IP {{ctx.payload.aggregations.by_ip.buckets.0.key}} had {{ctx.payload.aggregations.by_ip.buckets.0.doc_count}} failed login attempts in the last 5 minutes."
      }
    }
  }
}

🔍 Security Use Cases​

1. Brute Force Detection​

GET /ssh-auth-*/_search
{
  "size": 0,
  "query": {
    "bool": {
      "filter": [
        {"match": {"event_type": "failed_login"}},
        {"range": {"@timestamp": {"gte": "now-1h"}}}
      ]
    }
  },
  "aggs": {
    "brute_force_ips": {
      "terms": {
        "field": "source_ip.keyword",
        "min_doc_count": 10,
        "size": 50
      }
    }
  }
}

2. Unusual Traffic Patterns​

GET /firewall-*/_search
{
  "size": 0,
  "query": {
    "range": {
      "@timestamp": {
        "gte": "now-24h"
      }
    }
  },
  "aggs": {
    "traffic_by_hour": {
      "date_histogram": {
        "field": "@timestamp",
        "interval": "1h"
      },
      "aggs": {
        "total_bytes": {
          "sum": {
            "field": "bytes"
          }
        }
      }
    }
  }
}

3. Web Attack Detection​

GET /apache-*/_search
{
  "query": {
    "bool": {
      "should": [
        {"match": {"alert": "sql_injection_attempt"}},
        {"match": {"request": "union select"}},
        {"match": {"request": "../../../"}},
        {"match": {"request": "<script>"}}
      ],
      "minimum_should_match": 1
    }
  }
}
