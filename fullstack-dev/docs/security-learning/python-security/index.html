Security Tool Example​

#!/usr/bin/env python3
"""
Simple vulnerability scanner
"""

import socket
import sys
from datetime import datetime

def banner_grab(host, port):
    """Grab banner from a service"""
    try:
        sock = socket.socket()
        sock.settimeout(2)
        sock.connect((host, port))
        banner = sock.recv(1024).decode().strip()
        sock.close()
        return banner
    except:
        return None

def scan_ports(host, ports):
    """Scan multiple ports"""
    print(f"Scanning {host}")
    print(f"Started at {datetime.now()}")
    print("-" * 50)
    
    results = {}
    for port in ports:
        if scan_port(host, port):
            banner = banner_grab(host, port)
            results[port] = {
                'status': 'open',
                'banner': banner
            }
            print(f"Port {port}: OPEN - {banner}")
        else:
            results[port] = {'status': 'closed'}
    
    print("-" * 50)
    print(f"Completed at {datetime.now()}")
    return results

def scan_port(host, port):
    """Check if port is open"""
    try:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(1)
        result = sock.connect_ex((host, port))
        sock.close()
        return result == 0
    except:
        return False

if __name__ == "__main__":
    if len(sys.argv) != 2:
        print(f"Usage: {sys.argv[0]} <host>")
        sys.exit(1)
    
    target = sys.argv[1]
    common_ports = [21, 22, 23, 25, 80, 443, 3306, 3389, 8080]
    scan_ports(target, common_ports)

Cryptography Examples​

from cryptography.fernet import Fernet
import hashlib

# Symmetric encryption
key = Fernet.generate_key()
cipher = Fernet(key)
encrypted = cipher.encrypt(b"Sensitive data")
decrypted = cipher.decrypt(encrypted)

# Hashing
def hash_password(password):
    return hashlib.sha256(password.encode()).hexdigest()

# Verify password
def verify_password(password, hash):
    return hash_password(password) == hash

SSH Automation with Paramiko​

import paramiko

def ssh_execute(host, username, password, command):
    """Execute command via SSH"""
    client = paramiko.SSHClient()
    client.set_missing_host_key_policy(paramiko.AutoAddPolicy())
    
    try:
        client.connect(host, username=username, password=password)
        stdin, stdout, stderr = client.exec_command(command)
        output = stdout.read().decode()
        error = stderr.read().decode()
        return output, error
    finally:
        client.close()
