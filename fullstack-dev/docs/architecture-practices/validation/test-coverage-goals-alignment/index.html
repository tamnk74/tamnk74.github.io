Test Coverage Goals & Stakeholder Alignment

This guide establishes automated test coverage goals for Next.js/NestJS applications, ensuring team and stakeholder alignment, visibility, and continuous monitoring through GitHub Actions and GCP.

1. Overview​

Test coverage measures the percentage of code exercised by automated tests. This document defines coverage targets, alignment strategies, and automated enforcement mechanisms to ensure all stakeholders understand and commit to quality goals.

2. Coverage Goals by Layer​

Recommended Coverage Targets​

LayerMinimum TargetOptimal TargetRationale
Business Logic (Services)85%90%+Critical paths must be thoroughly tested
API Endpoints (Controllers)75%85%Request/response validation essential
Database Layer (Repositories)70%80%Data integrity verification
Utility Functions90%95%Reusable code must be bulletproof
UI Components (React)70%80%User-facing functionality
Integration Tests60%75%End-to-end flow coverage

3.2 Stakeholder Communication Plan​

// src/reporting/stakeholder-report.service.ts
import { Injectable } from '@nestjs/common';
import { Storage } from '@google-cloud/storage';

export interface CoverageReport {
  timestamp: Date;
  overall: number;
  byLayer: Record<string, number>;
  trend: 'improving' | 'stable' | 'declining';
  blockers: string[];
  nextMilestone: string;
}

@Injectable()
export class StakeholderReportService {
  private readonly storage = new Storage();

  async generateWeeklyReport(): Promise<CoverageReport> {
    const coverage = await this.fetchLatestCoverage();
    const previous = await this.fetchPreviousWeekCoverage();

    return {
      timestamp: new Date(),
      overall: coverage.overall,
      byLayer: coverage.byLayer,
      trend: this.calculateTrend(coverage.overall, previous.overall),
      blockers: await this.identifyBlockers(),
      nextMilestone: this.getNextMilestone(coverage.overall),
    };
  }

  private calculateTrend(current: number, previous: number): 'improving' | 'stable' | 'declining' {
    const diff = current - previous;
    if (diff > 1) return 'improving';
    if (diff < -1) return 'declining';
    return 'stable';
  }

  private async identifyBlockers(): Promise<string[]> {
    // Analyze coverage data to find problem areas
    return ['Legacy payment module below 60%', 'New feature branches not meeting threshold'];
  }

  private getNextMilestone(current: number): string {
    if (current < 70) return 'Reach 70% baseline';
    if (current < 80) return 'Achieve 80% team goal';
    return 'Maintain 80%+ excellence';
  }

  async uploadReportToGCS(report: CoverageReport): Promise<string> {
    const bucket = this.storage.bucket('test-coverage-reports');
    const fileName = `weekly-report-${report.timestamp.toISOString()}.json`;

    await bucket.file(fileName).save(JSON.stringify(report, null, 2), {
      contentType: 'application/json',
      metadata: {
        type: 'stakeholder-report',
        timestamp: report.timestamp.toISOString(),
      },
    });

    return `gs://test-coverage-reports/${fileName}`;
  }
}

4.2 Coverage Monitoring Service​

// src/monitoring/coverage-monitor.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { Cron, CronExpression } from '@nestjs/schedule';
import { readFileSync } from 'fs';
import { join } from 'path';

interface CoverageSummary {
  total: {
    lines: { pct: number };
    statements: { pct: number };
    functions: { pct: number };
    branches: { pct: number };
  };
}

@Injectable()
export class CoverageMonitorService {
  private readonly logger = new Logger(CoverageMonitorService.name);

  @Cron(CronExpression.EVERY_DAY_AT_MIDNIGHT)
  async checkCoverageGoals() {
    const summary = this.loadCoverageSummary();
    const goals = this.loadCoverageGoals();

    const violations = this.detectViolations(summary, goals);

    if (violations.length > 0) {
      await this.notifyStakeholders(violations);
    } else {
      this.logger.log('✅ All coverage goals met');
    }
  }

  private loadCoverageSummary(): CoverageSummary {
    const path = join(process.cwd(), 'coverage', 'coverage-summary.json');
    return JSON.parse(readFileSync(path, 'utf-8'));
  }

  private loadCoverageGoals() {
    return {
      lines: 80,
      statements: 80,
      functions: 80,
      branches: 75,
    };
  }

  private detectViolations(summary: CoverageSummary, goals: any): string[] {
    const violations: string[] = [];
    const total = summary.total;

    if (total.lines.pct < goals.lines) {
      violations.push(`Lines coverage ${total.lines.pct}% below goal ${goals.lines}%`);
    }
    if (total.functions.pct < goals.functions) {
      violations.push(`Functions coverage ${total.functions.pct}% below goal ${goals.functions}%`);
    }
    if (total.branches.pct < goals.branches) {
      violations.push(`Branches coverage ${total.branches.pct}% below goal ${goals.branches}%`);
    }

    return violations;
  }

  private async notifyStakeholders(violations: string[]) {
    this.logger.warn(`⚠️ Coverage violations detected:`);
    violations.forEach((v) => this.logger.warn(`  - ${v}`));

    // Send to Slack, email, etc.
  }
}

5.2 Coverage Dashboard Component​

// app/admin/coverage-dashboard/page.tsx
'use client';
import { useState, useEffect } from 'react';
import useSWR from 'swr';

const fetcher = (url: string) => fetch(url).then((res) => res.json());

interface CoverageData {
  timestamp: string;
  overall: number;
  byLayer: {
    services: number;
    controllers: number;
    components: number;
    utils: number;
  };
  goals: {
    services: number;
    controllers: number;
    components: number;
    utils: number;
  };
  trend: 'improving' | 'stable' | 'declining';
}

export default function CoverageDashboard() {
  const { data, error, isLoading } = useSWR<CoverageData>('/api/coverage/status', fetcher, {
    refreshInterval: 300000, // 5 minutes
  });

  if (isLoading) return <LoadingState />;
  if (error) return <ErrorState error={error} />;

  return (
    <div className="p-6 space-y-6">
      <header>
        <h1 className="text-3xl font-bold">Test Coverage Dashboard</h1>
        <p className="text-gray-600">Real-time view of automated test coverage goals and progress</p>
      </header>

      {/* Overall Coverage */}
      <section className="bg-white rounded-lg shadow p-6">
        <h2 className="text-xl font-semibold mb-4">Overall Coverage</h2>
        <div className="flex items-center gap-4">
          <div className="flex-1">
            <CoverageProgress value={data.overall} target={80} />
          </div>
          <TrendIndicator trend={data.trend} />
        </div>
      </section>

      {/* Layer Breakdown */}
      <section className="bg-white rounded-lg shadow p-6">
        <h2 className="text-xl font-semibold mb-4">Coverage by Layer</h2>
        <div className="grid gap-4 md:grid-cols-2">
          {Object.entries(data.byLayer).map(([layer, coverage]) => (
            <LayerCard
              key={layer}
              layer={layer}
              coverage={coverage}
              goal={data.goals[layer as keyof typeof data.goals]}
            />
          ))}
        </div>
      </section>

      {/* Stakeholder View */}
      <section className="bg-white rounded-lg shadow p-6">
        <h2 className="text-xl font-semibold mb-4">Stakeholder Summary</h2>
        <StakeholderSummary data={data} />
      </section>
    </div>
  );
}

function CoverageProgress({ value, target }: { value: number; target: number }) {
  const isAboveTarget = value >= target;
  const barColor = isAboveTarget ? 'bg-green-500' : value >= target - 5 ? 'bg-yellow-500' : 'bg-red-500';

  return (
    <div className="space-y-2">
      <div className="flex justify-between text-sm">
        <span className="font-medium">Current: {value.toFixed(1)}%</span>
        <span className="text-gray-500">Goal: {target}%</span>
      </div>
      <div className="w-full bg-gray-200 rounded-full h-4">
        <div
          className={`h-4 rounded-full ${barColor} transition-all duration-500`}
          style={{ width: `${Math.min(value, 100)}%` }}
        />
      </div>
    </div>
  );
}

function TrendIndicator({ trend }: { trend: 'improving' | 'stable' | 'declining' }) {
  const configs = {
    improving: { icon: '📈', color: 'text-green-600', label: 'Improving' },
    stable: { icon: '➡️', color: 'text-blue-600', label: 'Stable' },
    declining: { icon: '📉', color: 'text-red-600', label: 'Declining' },
  };

  const config = configs[trend];

  return (
    <div className={`flex items-center gap-2 ${config.color}`}>
      <span className="text-2xl">{config.icon}</span>
      <span className="font-semibold">{config.label}</span>
    </div>
  );
}

function LayerCard({ layer, coverage, goal }: { layer: string; coverage: number; goal: number }) {
  const meetsGoal = coverage >= goal;
  const statusColor = meetsGoal ? 'text-green-600' : 'text-orange-600';

  return (
    <div className="border rounded-lg p-4">
      <h3 className="text-lg font-medium capitalize mb-2">{layer}</h3>
      <div className="space-y-2">
        <CoverageProgress value={coverage} target={goal} />
        <p className={`text-sm font-semibold ${statusColor}`}>
          {meetsGoal ? '✅ Goal Met' : `⚠️ ${(goal - coverage).toFixed(1)}% to goal`}
        </p>
      </div>
    </div>
  );
}

function StakeholderSummary({ data }: { data: CoverageData }) {
  const layersAboveGoal = Object.entries(data.byLayer).filter(
    ([layer, coverage]) => coverage >= data.goals[layer as keyof typeof data.goals],
  ).length;

  const totalLayers = Object.keys(data.byLayer).length;

  return (
    <div className="space-y-3">
      <div className="flex justify-between items-center">
        <span className="text-gray-700">Layers Meeting Goals:</span>
        <span className="font-bold text-lg">
          {layersAboveGoal} / {totalLayers}
        </span>
      </div>
      <div className="flex justify-between items-center">
        <span className="text-gray-700">Overall Health:</span>
        <span
          className={`px-3 py-1 rounded-full text-sm font-semibold ${
            data.overall >= 80
              ? 'bg-green-100 text-green-700'
              : data.overall >= 70
              ? 'bg-yellow-100 text-yellow-700'
              : 'bg-red-100 text-red-700'
          }`}
        >
          {data.overall >= 80 ? 'Excellent' : data.overall >= 70 ? 'Good' : 'Needs Improvement'}
        </span>
      </div>
      <div className="text-sm text-gray-600">Last updated: {new Date(data.timestamp).toLocaleString()}</div>
    </div>
  );
}

function LoadingState() {
  return <div className="p-6">Loading coverage data...</div>;
}

function ErrorState({ error }: { error: any }) {
  return <div className="p-6 text-red-500">Failed to load coverage data: {error.message}</div>;
}

7.1 Cloud Function for Report Generation​

// functions/coverage-report/index.ts
import { Storage } from '@google-cloud/storage';
import { CloudEvent } from '@google-cloud/functions-framework';

const storage = new Storage();

export async function generateCoverageReport(cloudEvent: CloudEvent) {
  const bucket = storage.bucket('coverage-reports');

  // Fetch latest coverage files
  const [backendFiles] = await bucket.getFiles({ prefix: 'backend/' });
  const [frontendFiles] = await bucket.getFiles({ prefix: 'frontend/' });

  const latestBackend = await getLatestFile(backendFiles);
  const latestFrontend = await getLatestFile(frontendFiles);

  const report = {
    timestamp: new Date().toISOString(),
    backend: latestBackend,
    frontend: latestFrontend,
    overall: calculateOverall(latestBackend, latestFrontend),
  };

  // Store aggregated report
  await bucket.file('reports/latest.json').save(JSON.stringify(report, null, 2));

  console.log('Coverage report generated successfully');
}

async function getLatestFile(files: any[]) {
  const sorted = files.sort(
    (a, b) => new Date(b.metadata.timeCreated).getTime() - new Date(a.metadata.timeCreated).getTime(),
  );

  const [content] = await sorted[0].download();
  return JSON.parse(content.toString());
}

function calculateOverall(backend: any, frontend: any) {
  return {
    lines: (backend.total.lines.pct + frontend.total.lines.pct) / 2,
    statements: (backend.total.statements.pct + frontend.total.statements.pct) / 2,
    functions: (backend.total.functions.pct + frontend.total.functions.pct) / 2,
    branches: (backend.total.branches.pct + frontend.total.branches.pct) / 2,
  };
}

7.2 BigQuery Analytics​

-- BigQuery: Coverage trend analysis
CREATE OR REPLACE TABLE `project.analytics.coverage_history` AS
SELECT
  timestamp,
  backend_lines_pct,
  frontend_lines_pct,
  (backend_lines_pct + frontend_lines_pct) / 2 AS overall_lines_pct,
  CASE
    WHEN (backend_lines_pct + frontend_lines_pct) / 2 >= 80 THEN 'Excellent'
    WHEN (backend_lines_pct + frontend_lines_pct) / 2 >= 70 THEN 'Good'
    ELSE 'Needs Improvement'
  END AS health_status
FROM `project.analytics.coverage_raw`
WHERE timestamp >= TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 90 DAY)
ORDER BY timestamp DESC;

-- Weekly improvement tracking
SELECT
  EXTRACT(WEEK FROM timestamp) AS week_number,
  AVG(overall_lines_pct) AS avg_coverage,
  COUNT(*) AS measurements,
  MAX(overall_lines_pct) - MIN(overall_lines_pct) AS weekly_variance
FROM `project.analytics.coverage_history`
GROUP BY week_number
ORDER BY week_number DESC
LIMIT 12;

8.2 Slack Bot Notifications​

// src/notifications/coverage-bot.service.ts
import { Injectable } from '@nestjs/common';
import { WebClient } from '@slack/web-api';

@Injectable()
export class CoverageBotService {
  private slack: WebClient;

  constructor() {
    this.slack = new WebClient(process.env.SLACK_BOT_TOKEN);
  }

  async notifyDailyCoverage(coverage: any) {
    const message = this.buildDailyMessage(coverage);
    await this.slack.chat.postMessage({
      channel: '#engineering',
      text: 'Daily Coverage Update',
      blocks: message,
    });
  }

  async notifyGoalAchieved(layer: string, coverage: number) {
    await this.slack.chat.postMessage({
      channel: '#engineering',
      text: `🎉 ${layer} coverage goal achieved: ${coverage}%!`,
    });
  }

  async notifyCoverageDecline(details: any) {
    await this.slack.chat.postMessage({
      channel: '#engineering-alerts',
      text: `⚠️ Coverage decline detected`,
      blocks: [
        {
          type: 'section',
          text: {
            type: 'mrkdwn',
            text: `*Coverage Alert*\n${details.layer}: ${details.previous}% → ${details.current}%`,
          },
        },
      ],
    });
  }

  private buildDailyMessage(coverage: any) {
    return [
      {
        type: 'section',
        text: {
          type: 'mrkdwn',
          text: `*📊 Daily Coverage Update*\nOverall: *${coverage.overall}%* (Goal: 80%)`,
        },
      },
      {
        type: 'divider',
      },
      {
        type: 'section',
        fields: [
          { type: 'mrkdwn', text: `*Backend:* ${coverage.backend}%` },
          { type: 'mrkdwn', text: `*Frontend:* ${coverage.frontend}%` },
        ],
      },
    ];
  }
}

9. Continuous Improvement Plan​

Quarterly Review Checklist​

  • Review and update coverage goals based on project maturity
  • Analyze coverage trends and identify systemic issues
  • Conduct team retrospective on testing practices
  • Update stakeholder on progress and challenges
  • Adjust thresholds if consistently meeting/missing goals
  • Recognize teams/individuals with high coverage contributions
  • Plan training sessions for areas with low coverage
