Automated Database Migration Deployment

This guide establishes automated database change management for NestJS applications on GCP, ensuring database scripts are automatically deployed through CI/CD pipelines using tools like Prisma Migrate, TypeORM, or Liquibase equivalents.

1. Overview​

Automated database migrations eliminate manual schema changes, reduce deployment errors, and maintain version control for database evolution. This document covers migration strategies, tooling, and complete automation through GitHub Actions and GCP.

2. Migration Strategy Selection​

2.1 Tool Comparison​

Recommendation: Use Prisma Migrate or TypeORM for NestJS projects due to tight integration and TypeScript support.

// src/database/migration.module.ts
import { Module } from '@nestjs/common';
import { MigrationService } from './migration.service';
import { MigrationController } from './migration.controller';

@Module({
  providers: [MigrationService],
  controllers: [MigrationController],
  exports: [MigrationService],
})
export class MigrationModule {}

// src/database/migration.service.ts
import { Injectable, Logger, OnModuleInit } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { execSync } from 'child_process';
import * as fs from 'fs';
import * as path from 'path';

export interface MigrationStatus {
  applied: string[];
  pending: string[];
  lastMigration: string | null;
  databaseUrl: string;
}

@Injectable()
export class MigrationService implements OnModuleInit {
  private readonly logger = new Logger(MigrationService.name);

  constructor(private readonly config: ConfigService) {}

  async onModuleInit() {
    // Auto-run migrations on startup in non-production
    if (this.config.get('AUTO_MIGRATE') === 'true') {
      await this.runMigrations();
    }
  }

  async runMigrations(): Promise<void> {
    try {
      this.logger.log('🔄 Running database migrations...');

      const result = execSync('npx prisma migrate deploy', {
        encoding: 'utf-8',
        env: {
          ...process.env,
          DATABASE_URL: this.config.get('DATABASE_URL'),
        },
      });

      this.logger.log(result);
      this.logger.log('✅ Migrations completed successfully');
    } catch (error) {
      this.logger.error('❌ Migration failed:', error.message);
      throw error;
    }
  }

  async getMigrationStatus(): Promise<MigrationStatus> {
    try {
      const result = execSync('npx prisma migrate status --json', {
        encoding: 'utf-8',
        env: {
          ...process.env,
          DATABASE_URL: this.config.get('DATABASE_URL'),
        },
      });

      const status = JSON.parse(result);

      return {
        applied: status.appliedMigrations || [],
        pending: status.pendingMigrations || [],
        lastMigration: status.appliedMigrations?.[status.appliedMigrations.length - 1] || null,
        databaseUrl: this.maskDatabaseUrl(this.config.get('DATABASE_URL')),
      };
    } catch (error) {
      this.logger.error('Failed to get migration status:', error.message);
      throw error;
    }
  }

  async createMigration(name: string): Promise<string> {
    try {
      this.logger.log(`Creating migration: ${name}`);

      const result = execSync(`npx prisma migrate dev --name ${name} --create-only`, {
        encoding: 'utf-8',
      });

      this.logger.log(result);
      return `Migration created: ${name}`;
    } catch (error) {
      this.logger.error('Failed to create migration:', error.message);
      throw error;
    }
  }

  async rollbackMigration(): Promise<void> {
    // Prisma doesn't have built-in rollback
    // Implement custom rollback logic
    this.logger.warn('⚠️ Manual rollback required for Prisma migrations');
    throw new Error('Rollback not supported - create reverse migration');
  }

  async validateMigrations(): Promise<boolean> {
    try {
      execSync('npx prisma validate', { encoding: 'utf-8' });
      return true;
    } catch (error) {
      this.logger.error('Schema validation failed:', error.message);
      return false;
    }
  }

  private maskDatabaseUrl(url: string): string {
    return url.replace(/:[^:@]+@/, ':***@');
  }
}

// src/database/migration.controller.ts
import { Controller, Get, Post, Body, UseGuards } from '@nestjs/common';
import { ApiTags, ApiOperation, ApiResponse, ApiBearerAuth } from '@nestjs/swagger';
import { MigrationService } from './migration.service';

@ApiTags('Database Migrations')
@Controller('api/migrations')
@ApiBearerAuth()
// @UseGuards(AdminGuard) // Restrict to admins only
export class MigrationController {
  constructor(private readonly migrationService: MigrationService) {}

  @Get('status')
  @ApiOperation({ summary: 'Get current migration status' })
  @ApiResponse({ status: 200, description: 'Migration status retrieved' })
  async getStatus() {
    return this.migrationService.getMigrationStatus();
  }

  @Post('run')
  @ApiOperation({ summary: 'Run pending migrations' })
  @ApiResponse({ status: 200, description: 'Migrations executed successfully' })
  async runMigrations() {
    await this.migrationService.runMigrations();
    return { message: 'Migrations executed successfully' };
  }

  @Get('validate')
  @ApiOperation({ summary: 'Validate schema and migrations' })
  async validate() {
    const isValid = await this.migrationService.validateMigrations();
    return { valid: isValid };
  }
}

4.2 Migration Service​

// src/database/typeorm-migration.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { DataSource } from 'typeorm';

@Injectable()
export class TypeOrmMigrationService {
  private readonly logger = new Logger(TypeOrmMigrationService.name);

  constructor(private readonly dataSource: DataSource) {}

  async runMigrations(): Promise<void> {
    try {
      this.logger.log('🔄 Running TypeORM migrations...');

      const migrations = await this.dataSource.runMigrations();

      if (migrations.length === 0) {
        this.logger.log('✅ No pending migrations');
      } else {
        migrations.forEach((migration) => {
          this.logger.log(`✅ Applied: ${migration.name}`);
        });
      }
    } catch (error) {
      this.logger.error('❌ Migration failed:', error.message);
      throw error;
    }
  }

  async getMigrationStatus() {
    const pending = await this.dataSource.showMigrations();
    const executed = await this.dataSource.query('SELECT * FROM migrations ORDER BY timestamp DESC');

    return {
      pending,
      executed: executed.map((m: any) => m.name),
      total: executed.length,
    };
  }

  async rollbackMigration(): Promise<void> {
    try {
      this.logger.log('⏪ Rolling back last migration...');
      await this.dataSource.undoLastMigration();
      this.logger.log('✅ Rollback successful');
    } catch (error) {
      this.logger.error('❌ Rollback failed:', error.message);
      throw error;
    }
  }
}

5.2 Migration Validation Script​

// scripts/validate-migrations.js
const fs = require('fs');
const path = require('path');

const migrationsDir = path.join(__dirname, '../prisma/migrations');

console.log('🔍 Validating migrations...\n');

// Check migrations directory exists
if (!fs.existsSync(migrationsDir)) {
  console.error('❌ Migrations directory not found');
  process.exit(1);
}

// Get all migration folders
const migrations = fs
  .readdirSync(migrationsDir)
  .filter((file) => fs.statSync(path.join(migrationsDir, file)).isDirectory())
  .sort();

if (migrations.length === 0) {
  console.log('ℹ️  No migrations found');
  process.exit(0);
}

let hasErrors = false;

// Validate each migration
migrations.forEach((migration, index) => {
  const migrationPath = path.join(migrationsDir, migration);
  const sqlFile = path.join(migrationPath, 'migration.sql');

  // Check SQL file exists
  if (!fs.existsSync(sqlFile)) {
    console.error(`❌ ${migration}: migration.sql not found`);
    hasErrors = true;
    return;
  }

  // Check SQL content
  const sql = fs.readFileSync(sqlFile, 'utf8');
  if (sql.trim().length === 0) {
    console.error(`❌ ${migration}: Empty SQL file`);
    hasErrors = true;
    return;
  }

  // Check for dangerous operations in production
  const dangerousOps = ['DROP TABLE', 'DROP COLUMN', 'TRUNCATE'];
  dangerousOps.forEach((op) => {
    if (sql.includes(op)) {
      console.warn(`⚠️  ${migration}: Contains ${op} - review carefully`);
    }
  });

  console.log(`✅ ${migration}: Valid`);
});

if (hasErrors) {
  console.error('\n❌ Migration validation failed');
  process.exit(1);
}

console.log('\n✅ All migrations validated successfully');

7.2 Zero-Downtime Migration Strategies​

// Example: Adding a column with zero downtime

// Step 1: Add nullable column
// Migration 1: 1703764800000-AddEmailVerifiedColumn.ts
export class AddEmailVerifiedColumn1703764800000 implements MigrationInterface {
  public async up(queryRunner: QueryRunner): Promise<void> {
    await queryRunner.addColumn(
      'users',
      new TableColumn({
        name: 'email_verified',
        type: 'boolean',
        isNullable: true, // Start as nullable
      }),
    );
  }

  public async down(queryRunner: QueryRunner): Promise<void> {
    await queryRunner.dropColumn('users', 'email_verified');
  }
}

// Step 2: Backfill data (run after deployment)
// Migration 2: 1703764900000-BackfillEmailVerified.ts
export class BackfillEmailVerified1703764900000 implements MigrationInterface {
  public async up(queryRunner: QueryRunner): Promise<void> {
    await queryRunner.query(`
      UPDATE users 
      SET email_verified = false 
      WHERE email_verified IS NULL
    `);
  }

  public async down(queryRunner: QueryRunner): Promise<void> {
    // Optional: revert to null
  }
}

// Step 3: Make column non-nullable (after all data backfilled)
// Migration 3: 1703765000000-MakeEmailVerifiedNonNullable.ts
export class MakeEmailVerifiedNonNullable1703765000000 implements MigrationInterface {
  public async up(queryRunner: QueryRunner): Promise<void> {
    await queryRunner.changeColumn(
      'users',
      'email_verified',
      new TableColumn({
        name: 'email_verified',
        type: 'boolean',
        isNullable: false,
        default: false,
      }),
    );
  }

  public async down(queryRunner: QueryRunner): Promise<void> {
    await queryRunner.changeColumn(
      'users',
      'email_verified',
      new TableColumn({
        name: 'email_verified',
        type: 'boolean',
        isNullable: true,
      }),
    );
  }
}

8. Monitoring & Alerting​

8.1 Migration Metrics​

// src/monitoring/migration-metrics.service.ts
import { Injectable } from '@nestjs/common';
import { Monitoring } from '@google-cloud/monitoring';

@Injectable()
export class MigrationMetricsService {
  private monitoring = new Monitoring.MetricServiceClient();

  async recordMigrationSuccess(duration: number, migrationsCount: number) {
    const projectId = process.env.GCP_PROJECT_ID;

    await this.monitoring.createTimeSeries({
      name: this.monitoring.projectPath(projectId),
      timeSeries: [
        {
          metric: {
            type: 'custom.googleapis.com/database/migration_duration',
            labels: {
              status: 'success',
            },
          },
          points: [
            {
              interval: {
                endTime: {
                  seconds: Date.now() / 1000,
                },
              },
              value: {
                doubleValue: duration,
              },
            },
          ],
        },
      ],
    });
  }

  async recordMigrationFailure(error: string) {
    // Log error and alert
    console.error('Migration failed:', error);
    // Trigger alert
  }
}

9. Troubleshooting Guide​

Common Issues and Solutions​

IssueCauseSolution
Migration timeoutLong-running migrationSplit into smaller migrations, use batching
Lock conflictsTable locked by other processRun during maintenance window, kill competing connections
Rollback failedMissing down migrationCreate compensating migration, restore from backup
Schema driftManual changes to DBGenerate migration from difference, enforce process
Connection refusedCloud SQL Auth Proxy not runningVerify proxy, check IAM permissions
Permission deniedInsufficient database privilegesGrant necessary permissions to migration user

By implementing automated database migration deployment through CI/CD pipelines, teams eliminate manual schema changes, reduce errors, and maintain version-controlled database evolution that's traceable, testable, and reversible.
