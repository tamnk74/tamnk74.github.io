Static Code Analysis Goals & Team Communication

This guide establishes static code analysis metrics, goals, and communication strategies for Next.js/NestJS applications, ensuring every team member understands quality targets and can access real-time feedback through GitHub Actions and GCP monitoring.

1. Overview​

Static code analysis examines code without executing it, identifying issues related to complexity, duplication, maintainability, security, and code smells. This document defines measurable goals for key metrics and ensures team-wide awareness and accountability.

2. Static Analysis Metrics & Goals​

4.1 Quality Metrics Service​

// src/quality/quality-metrics.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { HttpService } from '@nestjs/axios';
import { ConfigService } from '@nestjs/config';
import { firstValueFrom } from 'rxjs';

export interface QualityMetrics {
  complexity: {
    cyclomatic: number;
    cognitive: number;
  };
  duplication: {
    percentage: number;
    blocks: number;
  };
  maintainability: {
    index: number;
    rating: string;
  };
  reliability: {
    bugs: number;
    rating: string;
  };
  security: {
    vulnerabilities: number;
    rating: string;
  };
  codeSmells: {
    count: number;
    debtRatio: number;
  };
  coverage: {
    percentage: number;
    lines: number;
  };
}

export interface MetricGoal {
  metric: string;
  current: number;
  target: number;
  status: 'meeting' | 'warning' | 'failing';
}

@Injectable()
export class QualityMetricsService {
  private readonly logger = new Logger(QualityMetricsService.name);

  constructor(private readonly http: HttpService, private readonly config: ConfigService) {}

  async fetchSonarQubeMetrics(projectKey: string): Promise<QualityMetrics> {
    const sonarUrl = this.config.get('SONARQUBE_URL');
    const token = this.config.get('SONARQUBE_TOKEN');

    const response = await firstValueFrom(
      this.http.get(`${sonarUrl}/api/measures/component`, {
        params: {
          component: projectKey,
          metricKeys: [
            'complexity',
            'cognitive_complexity',
            'duplicated_lines_density',
            'sqale_index',
            'reliability_rating',
            'security_rating',
            'code_smells',
            'sqale_debt_ratio',
            'coverage',
          ].join(','),
        },
        auth: {
          username: token,
          password: '',
        },
      }),
    );

    return this.transformSonarMetrics(response.data);
  }

  private transformSonarMetrics(data: any): QualityMetrics {
    const measures = data.component.measures.reduce((acc: any, m: any) => {
      acc[m.metric] = parseFloat(m.value);
      return acc;
    }, {});

    return {
      complexity: {
        cyclomatic: measures.complexity || 0,
        cognitive: measures.cognitive_complexity || 0,
      },
      duplication: {
        percentage: measures.duplicated_lines_density || 0,
        blocks: measures.duplicated_blocks || 0,
      },
      maintainability: {
        index: 100 - (measures.sqale_index || 0),
        rating: this.getRatingLetter(measures.sqale_rating || 1),
      },
      reliability: {
        bugs: measures.bugs || 0,
        rating: this.getRatingLetter(measures.reliability_rating || 1),
      },
      security: {
        vulnerabilities: measures.vulnerabilities || 0,
        rating: this.getRatingLetter(measures.security_rating || 1),
      },
      codeSmells: {
        count: measures.code_smells || 0,
        debtRatio: measures.sqale_debt_ratio || 0,
      },
      coverage: {
        percentage: measures.coverage || 0,
        lines: measures.lines_to_cover || 0,
      },
    };
  }

  async compareWithGoals(metrics: QualityMetrics): Promise<MetricGoal[]> {
    const goals = this.loadGoals();
    const comparisons: MetricGoal[] = [];

    // Complexity check
    comparisons.push({
      metric: 'Cyclomatic Complexity',
      current: metrics.complexity.cyclomatic,
      target: goals.complexity.cyclomatic.per_file,
      status: this.getStatus(metrics.complexity.cyclomatic, goals.complexity.cyclomatic.per_file, 'lower'),
    });

    // Duplication check
    comparisons.push({
      metric: 'Code Duplication',
      current: metrics.duplication.percentage,
      target: goals.duplication.target,
      status: this.getStatus(metrics.duplication.percentage, goals.duplication.target, 'lower'),
    });

    // Maintainability check
    comparisons.push({
      metric: 'Maintainability Index',
      current: metrics.maintainability.index,
      target: goals.maintainability.index_minimum,
      status: this.getStatus(metrics.maintainability.index, goals.maintainability.index_minimum, 'higher'),
    });

    // Code smells check
    comparisons.push({
      metric: 'Code Smells',
      current: metrics.codeSmells.count,
      target: goals.code_smells.maximum,
      status: this.getStatus(metrics.codeSmells.count, goals.code_smells.maximum, 'lower'),
    });

    return comparisons;
  }

  private getStatus(current: number, target: number, direction: 'lower' | 'higher'): 'meeting' | 'warning' | 'failing' {
    const threshold = target * 0.1; // 10% threshold

    if (direction === 'lower') {
      if (current <= target) return 'meeting';
      if (current <= target + threshold) return 'warning';
      return 'failing';
    } else {
      if (current >= target) return 'meeting';
      if (current >= target - threshold) return 'warning';
      return 'failing';
    }
  }

  private getRatingLetter(rating: number): string {
    const map: Record<number, string> = { 1: 'A', 2: 'B', 3: 'C', 4: 'D', 5: 'E' };
    return map[rating] || 'Unknown';
  }

  private loadGoals() {
    // Load from configuration or database
    return {
      complexity: {
        cyclomatic: { per_function: 10, per_file: 50 },
        cognitive: { per_function: 15, per_file: 100 },
      },
      duplication: { target: 3, threshold: 5 },
      maintainability: { index_minimum: 70 },
      code_smells: { maximum: 50, debt_ratio_max: 5 },
    };
  }
}

4.2 Quality Gate Controller​

// src/quality/quality-gate.controller.ts
import { Controller, Get, Param, Query } from '@nestjs/common';
import { ApiTags, ApiOperation, ApiResponse } from '@nestjs/swagger';
import { QualityMetricsService } from './quality-metrics.service';

@ApiTags('Quality Gate')
@Controller('api/quality')
export class QualityGateController {
  constructor(private readonly metricsService: QualityMetricsService) {}

  @Get('metrics/:project')
  @ApiOperation({ summary: 'Get current quality metrics for a project' })
  @ApiResponse({ status: 200, description: 'Quality metrics retrieved' })
  async getMetrics(@Param('project') project: string) {
    return this.metricsService.fetchSonarQubeMetrics(project);
  }

  @Get('goals/:project')
  @ApiOperation({ summary: 'Compare current metrics with goals' })
  async compareWithGoals(@Param('project') project: string) {
    const metrics = await this.metricsService.fetchSonarQubeMetrics(project);
    const comparison = await this.metricsService.compareWithGoals(metrics);

    return {
      project,
      timestamp: new Date().toISOString(),
      metrics,
      goals: comparison,
      overallStatus: this.determineOverallStatus(comparison),
    };
  }

  @Get('dashboard')
  @ApiOperation({ summary: 'Get quality dashboard data' })
  async getDashboard() {
    const backendMetrics = await this.metricsService.fetchSonarQubeMetrics('backend');
    const frontendMetrics = await this.metricsService.fetchSonarQubeMetrics('frontend');

    return {
      backend: await this.metricsService.compareWithGoals(backendMetrics),
      frontend: await this.metricsService.compareWithGoals(frontendMetrics),
      timestamp: new Date().toISOString(),
    };
  }

  private determineOverallStatus(goals: any[]): string {
    const failing = goals.filter((g) => g.status === 'failing').length;
    const warning = goals.filter((g) => g.status === 'warning').length;

    if (failing > 0) return 'failing';
    if (warning > 0) return 'warning';
    return 'passing';
  }
}

'use client';
import { useState } from 'react';

function MetricsGrid({ goals }: { goals: MetricGoal[] }) {
  return (
    <div className="grid gap-4 md:grid-cols-2 lg:grid-cols-3">
      {goals.map((goal) => (
        <MetricCard key={goal.metric} goal={goal} />
      ))}
    </div>
  );
}

function MetricCard({ goal }: { goal: MetricGoal }) {
  const statusConfig = {
    meeting: { bg: 'bg-green-50', border: 'border-green-200', text: 'text-green-700', icon: '✅' },
    warning: { bg: 'bg-yellow-50', border: 'border-yellow-200', text: 'text-yellow-700', icon: '⚠️' },
    failing: { bg: 'bg-red-50', border: 'border-red-200', text: 'text-red-700', icon: '❌' },
  };

  const config = statusConfig[goal.status];

  return (
    <div className={`rounded-lg border-2 ${config.border} ${config.bg} p-4 shadow-sm`}>
      <div className="flex items-start justify-between">
        <h3 className="text-sm font-medium text-gray-700">{goal.metric}</h3>
        <span className="text-xl">{config.icon}</span>
      </div>
      <div className="mt-3 space-y-2">
        <div className="flex justify-between text-sm">
          <span className="text-gray-600">Current:</span>
          <span className={`font-semibold ${config.text}`}>{goal.current.toFixed(1)}</span>
        </div>
        <div className="flex justify-between text-sm">
          <span className="text-gray-600">Goal:</span>
          <span className="font-semibold text-gray-900">{goal.target.toFixed(1)}</span>
        </div>
        <ProgressBar
          current={goal.current}
          target={goal.target}
          inverse={goal.metric.includes('Duplication') || goal.metric.includes('Smells')}
        />
      </div>
    </div>
  );
}

function ProgressBar({ current, target, inverse = false }: { current: number; target: number; inverse?: boolean }) {
  let percentage: number;
  let color: string;

  if (inverse) {
    // For metrics where lower is better
    percentage = Math.max(0, Math.min(100, (target / current) * 100));
    color = current <= target ? 'bg-green-500' : 'bg-red-500';
  } else {
    // For metrics where higher is better
    percentage = Math.max(0, Math.min(100, (current / target) * 100));
    color = current >= target ? 'bg-green-500' : 'bg-yellow-500';
  }

  return (
    <div className="mt-2 h-2 w-full rounded-full bg-gray-200">
      <div className={`h-2 rounded-full ${color} transition-all duration-500`} style={{ width: `${percentage}%` }} />
    </div>
  );
}

function QualityGateBadge({ status }: { status: string }) {
  const config = {
    meeting: { label: 'Passing', color: 'bg-green-100 text-green-800' },
    warning: { label: 'Warning', color: 'bg-yellow-100 text-yellow-800' },
    failing: { label: 'Failing', color: 'bg-red-100 text-red-800' },
  };

  const badge = config[status as keyof typeof config] || config.warning;

  return (
    <span className={`rounded-full px-3 py-1 text-xs font-semibold ${badge.color}`}>Quality Gate: {badge.label}</span>
  );
}

function TeamGoalsSummary({ backend, frontend }: { backend: MetricGoal[]; frontend: MetricGoal[] }) {
  const allGoals = [...backend, ...frontend];
  const meeting = allGoals.filter((g) => g.status === 'meeting').length;
  const warning = allGoals.filter((g) => g.status === 'warning').length;
  const failing = allGoals.filter((g) => g.status === 'failing').length;

  return (
    <div className="space-y-4">
      <div className="grid gap-4 md:grid-cols-3">
        <StatCard label="Meeting Goals" value={meeting} total={allGoals.length} color="green" />
        <StatCard label="Warnings" value={warning} total={allGoals.length} color="yellow" />
        <StatCard label="Failing" value={failing} total={allGoals.length} color="red" />
      </div>
      <div className="mt-6 rounded-lg bg-blue-50 p-4">
        <h3 className="font-semibold text-blue-900">📚 Quick Reference</h3>
        <ul className="mt-2 space-y-1 text-sm text-blue-800">
          <li>• Complexity: Keep functions under 10 cyclomatic complexity</li>
          <li>• Duplication: Target &lt;3% code duplication</li>
          <li>• Maintainability: Maintain index above 70</li>
          <li>• Code Smells: Keep under 50 total issues</li>
        </ul>
      </div>
    </div>
  );
}

function StatCard({ label, value, total, color }: { label: string; value: number; total: number; color: string }) {
  const colors = {
    green: 'bg-green-50 text-green-700 border-green-200',
    yellow: 'bg-yellow-50 text-yellow-700 border-yellow-200',
    red: 'bg-red-50 text-red-700 border-red-200',
  };

  return (
    <div className={`rounded-lg border-2 p-4 ${colors[color as keyof typeof colors]}`}>
      <p className="text-sm font-medium">{label}</p>
      <p className="mt-2 text-3xl font-bold">
        {value}
        <span className="text-lg font-normal">/{total}</span>
      </p>
    </div>
  );
}

function LoadingState() {
  return <div className="p-6">Loading quality metrics...</div>;
}

function ErrorState({ error }: { error: any }) {
  return <div className="p-6 text-red-500">Failed to load metrics: {error.message}</div>;
}

6.1 Static Analysis Workflow​

# .github/workflows/static-analysis.yml
name: Static Code Analysis

on:
  pull_request:
    branches: [main, develop]
  push:
    branches: [main]
  schedule:
    - cron: '0 2 * * *' # Daily at 2 AM

jobs:
  sonarqube-scan:
    name: SonarQube Scan
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0 # Full history for better analysis

      - uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'pnpm'

      - name: Install dependencies
        run: pnpm install

      - name: Run tests with coverage
        run: pnpm test:cov

      - name: SonarQube Scan
        uses: SonarSource/sonarqube-scan-action@master
        env:
          SONAR_TOKEN: ${{ secrets.SONAR_TOKEN }}
          SONAR_HOST_URL: ${{ secrets.SONAR_HOST_URL }}

      - name: SonarQube Quality Gate
        uses: SonarSource/sonarqube-quality-gate-action@master
        timeout-minutes: 5
        env:
          SONAR_TOKEN: ${{ secrets.SONAR_TOKEN }}

  eslint-check:
    name: ESLint Analysis
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'pnpm'

      - name: Install dependencies
        run: pnpm install

      - name: Run ESLint with complexity checks
        run: pnpm lint --format json --output-file eslint-report.json
        continue-on-error: true

      - name: Analyze ESLint results
        run: |
          node scripts/analyze-eslint.js eslint-report.json

      - name: Upload ESLint report
        uses: actions/upload-artifact@v4
        with:
          name: eslint-report
          path: eslint-report.json

  code-duplication:
    name: Code Duplication Check
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: '20'

      - name: Install jscpd
        run: npm install -g jscpd

      - name: Check for duplications
        run: |
          jscpd src --min-lines 10 --min-tokens 50 --format json --output ./jscpd-report.json

      - name: Validate duplication threshold
        run: |
          node scripts/check-duplication-threshold.js

      - name: Upload duplication report
        uses: actions/upload-artifact@v4
        with:
          name: duplication-report
          path: jscpd-report.json

  comment-pr:
    name: Comment PR with Results
    needs: [sonarqube-scan, eslint-check, code-duplication]
    runs-on: ubuntu-latest
    if: github.event_name == 'pull_request'

    steps:
      - uses: actions/checkout@v4

      - name: Download artifacts
        uses: actions/download-artifact@v4

      - name: Generate PR comment
        uses: actions/github-script@v7
        with:
          script: |
            const fs = require('fs');

            // Read reports
            const eslintReport = JSON.parse(fs.readFileSync('eslint-report/eslint-report.json', 'utf8'));
            const duplicationReport = JSON.parse(fs.readFileSync('duplication-report/jscpd-report.json', 'utf8'));

            // Build comment
            const comment = `
            ## 🔍 Static Analysis Results

            ### ESLint Issues
            - Errors: ${eslintReport.errorCount}
            - Warnings: ${eslintReport.warningCount}
            - Complexity violations: ${eslintReport.complexityViolations || 0}

            ### Code Duplication

6.2 Duplication Threshold Check Script​

// scripts/check-duplication-threshold.js
const fs = require('fs');

const report = JSON.parse(fs.readFileSync('jscpd-report.json', 'utf8'));
const duplicationRate = report.statistics.total.percentage;

const THRESHOLD = 5.0;
const TARGET = 3.0;

console.log(`\n📊 Code Duplication Analysis:`);
console.log(`  Current: ${duplicationRate.toFixed(2)}%`);
console.log(`  Target: ${TARGET}%`);
console.log(`  Threshold: ${THRESHOLD}%\n`);

if (duplicationRate > THRESHOLD) {
  console.error(`❌ Duplication rate exceeds threshold!`);
  process.exit(1);
} else if (duplicationRate > TARGET) {
  console.warn(`⚠️  Duplication rate above target but within threshold`);
  process.exit(0);
} else {
  console.log(`✅ Duplication rate meets target`);
  process.exit(0);
}

7.2 Weekly Team Communication​

// src/notifications/quality-alert.service.ts
import { Injectable } from '@nestjs/common';
import { Cron, CronExpression } from '@nestjs/schedule';
import { WebClient } from '@slack/web-api';
import { QualityMetricsService } from '../quality/quality-metrics.service';

@Injectable()
export class QualityAlertService {
  private slack: WebClient;

  constructor(private readonly metricsService: QualityMetricsService) {
    this.slack = new WebClient(process.env.SLACK_BOT_TOKEN);
  }

  @Cron(CronExpression.EVERY_WEEK)
  async sendWeeklyReport() {
    const metrics = await this.metricsService.fetchSonarQubeMetrics('backend');
    const goals = await this.metricsService.compareWithGoals(metrics);

    await this.slack.chat.postMessage({
      channel: '#engineering',
      text: 'Weekly Quality Report',
      blocks: this.buildWeeklyReportBlocks(metrics, goals),
    });
  }

  @Cron(CronExpression.EVERY_DAY_AT_9AM)
  async sendDailyReminder() {
    await this.slack.chat.postMessage({
      channel: '#engineering',
      text: '🎯 Daily Quality Reminder',
      blocks: [
        {
          type: 'section',
          text: {
            type: 'mrkdwn',
            text: `*Quality Goals Reminder*\n\n• Keep functions under 10 complexity\n• Maintain <3% duplication\n• Write meaningful comments\n• Run static analysis before PR\n\nDashboard: <https://quality.example.com|View Metrics>`,
          },
        },
      ],
    });
  }

  async notifyGoalViolation(metric: string, current: number, target: number) {
    await this.slack.chat.postMessage({
      channel: '#engineering-alerts',
      text: `⚠️ Quality Goal Violation: ${metric}`,
      blocks: [
        {
          type: 'section',
          text: {
            type: 'mrkdwn',
            text: `*Quality Alert*\n\n${metric} has exceeded the goal:\nCurrent: ${current}\nTarget: ${target}\n\nAction required to maintain code quality standards.`,
          },
        },
      ],
    });
  }

  private buildWeeklyReportBlocks(metrics: any, goals: any[]) {
    return [
      {
        type: 'header',
        text: {
          type: 'plain_text',
          text: '📊 Weekly Quality Report',
        },
      },
      {
        type: 'section',
        fields: goals.slice(0, 4).map((goal) => ({
          type: 'mrkdwn',
          text: `*${goal.metric}*\n${goal.current.toFixed(1)} / ${goal.target} ${this.getStatusEmoji(goal.status)}`,
        })),
      },
      {
        type: 'divider',
      },
      {
        type: 'section',
        text: {
          type: 'mrkdwn',
          text: `View full dashboard: https://quality.example.com (Quality Metrics)`,
        },
      },
    ];
  }

  private getStatusEmoji(status: string): string {
    const map: Record<string, string> = {
      meeting: '✅',
      warning: '⚠️',
      failing: '❌',
    };
    return map[status] || '❓';
  }
}

8.1 Cloud Function for Metrics Aggregation​

// functions/quality-aggregator/index.ts
import { Storage } from '@google-cloud/storage';
import { BigQuery } from '@google-cloud/bigquery';

const storage = new Storage();
const bigquery = new BigQuery();

export async function aggregateQualityMetrics(event: any) {
  const bucket = storage.bucket('quality-metrics');

  // Fetch latest reports
  const [eslintFiles] = await bucket.getFiles({ prefix: 'eslint/' });
  const [duplicationFiles] = await bucket.getFiles({ prefix: 'duplication/' });

  const latestEslint = await getLatestReport(eslintFiles);
  const latestDuplication = await getLatestReport(duplicationFiles);

  // Insert into BigQuery
  await insertIntoBigQuery({
    timestamp: new Date().toISOString(),
    eslint_errors: latestEslint.errorCount,
    eslint_warnings: latestEslint.warningCount,
    duplication_rate: latestDuplication.statistics.total.percentage,
  });
}

async function getLatestReport(files: any[]) {
  const sorted = files.sort(
    (a, b) => new Date(b.metadata.timeCreated).getTime() - new Date(a.metadata.timeCreated).getTime(),
  );

  const [content] = await sorted[0].download();
  return JSON.parse(content.toString());
}

async function insertIntoBigQuery(data: any) {
  const datasetId = 'quality_metrics';
  const tableId = 'daily_metrics';

  await bigquery.dataset(datasetId).table(tableId).insert([data]);
}
