Critical Performance Areas Testing Guide

This guide provides a comprehensive approach to ensuring all critical performance areas are covered by performance testing. Performance testing is a non-functional software testing technique that determines how the stability, speed, scalability, and responsiveness of an application hold up under a given workload.

Overview​

Covering critical performance areas means that the performance of the main business flows in the application is being tested. This includes user registration, login, checkout, search, and other core business processes that directly impact user experience and business revenue.

Why Critical Performance Areas Matter​

Business Impact​

  • User Experience: Slow performance leads to user frustration and abandonment
  • Conversion Rates: Every 100ms delay can reduce conversion by 1%
  • SEO Rankings: Page speed is a ranking factor for search engines
  • Revenue Impact: Amazon calculated that 1 second delay costs $1.6 billion annually

Technical Impact​

  • Capacity Planning: Understanding limits helps plan infrastructure
  • Bottleneck Identification: Reveals performance issues before production
  • Regression Detection: Ensures new features don't degrade performance
  • SLA Compliance: Validates ability to meet service level agreements

// tests/performance/critical-flows/user-registration.test.ts
import http from 'k6/http';
import { check, sleep, group } from 'k6';
import { Rate, Trend, Counter } from 'k6/metrics';
import { randomString, randomIntBetween } from 'https://jslib.k6.io/k6-utils/1.4.0/index.js';

// Custom metrics for registration flow
const registrationTime = new Trend('registration_time');
const registrationErrorRate = new Rate('registration_error_rate');
const registrationSuccess = new Counter('registration_success');
const registrationFailure = new Counter('registration_failure');

// Performance targets
const REGISTRATION_TIME_TARGET = 2000; // 2 seconds

export const options = {
  scenarios: {
    registration_load: {
      executor: 'ramping-vus',
      startVUs: 0,
      stages: [
        { duration: '1m', target: 10 }, // Warm-up
        { duration: '3m', target: 50 }, // Normal load
        { duration: '2m', target: 100 }, // Peak load
        { duration: '3m', target: 100 }, // Sustained peak
        { duration: '1m', target: 0 }, // Ramp down
      ],
    },
    registration_spike: {
      executor: 'ramping-vus',
      startVUs: 0,
      stages: [
        { duration: '30s', target: 5 },
        { duration: '10s', target: 200 }, // Spike
        { duration: '1m', target: 200 },
        { duration: '10s', target: 5 },
      ],
      startTime: '10m', // Start after main test
    },
  },
  thresholds: {
    registration_time: ['p(95)<2000', 'p(99)<3000'],
    registration_error_rate: ['rate<0.01'],
    'http_req_duration{name:registration}': ['p(95)<2000'],
    'http_req_failed{name:registration}': ['rate<0.01'],
  },
};

const BASE_URL = __ENV.BASE_URL || 'http://localhost:3001';

// Generate unique user data
function generateUserData() {
  const uniqueId = randomString(8);
  return {
    email: `perf_test_${uniqueId}@example.com`,
    password: 'TestPassword123!',
    firstName: `TestUser_${uniqueId}`,
    lastName: 'PerformanceTest',
    phone: `+1${randomIntBetween(1000000000, 9999999999)}`,
  };
}

export default function () {
  group('User Registration Flow', () => {
    const userData = generateUserData();
    const startTime = Date.now();

    // Step 1: Check email availability (optional pre-check)
    const emailCheckResponse = http.post(
      `${BASE_URL}/api/auth/check-email`,
      JSON.stringify({ email: userData.email }),
      {
        headers: { 'Content-Type': 'application/json' },
        tags: { name: 'email_check' },
      },
    );

    check(emailCheckResponse, {
      'email check returns 200': (r) => r.status === 200,
      'email is available': (r) => r.json('available') === true,
    });

    // Step 2: Submit registration
    const registrationResponse = http.post(`${BASE_URL}/api/auth/register`, JSON.stringify(userData), {
      headers: { 'Content-Type': 'application/json' },
      tags: { name: 'registration' },
    });

    const registrationDuration = Date.now() - startTime;
    registrationTime.add(registrationDuration);

    const registrationChecks = check(registrationResponse, {
      'registration status is 201': (r) => r.status === 201,
      'registration response has user id': (r) => r.json('id') !== undefined,
      'registration response has email': (r) => r.json('email') === userData.email,
      'registration completes within target': () => registrationDuration < REGISTRATION_TIME_TARGET,
    });

    if (registrationChecks) {
      registrationSuccess.add(1);
    } else {
      registrationFailure.add(1);
      registrationErrorRate.add(1);
      console.log(`Registration failed: ${registrationResponse.status} - ${registrationResponse.body}`);
    }

    // Step 3: Verify email (simulated - in real tests may need to handle email verification)
    if (registrationResponse.status === 201) {
      const verifyResponse = http.post(
        `${BASE_URL}/api/auth/verify-email`,
        JSON.stringify({
          email: userData.email,
          token: 'test-verification-token', // In real scenario, extract from email
        }),
        {
          headers: { 'Content-Type': 'application/json' },
          tags: { name: 'email_verification' },
        },
      );

      check(verifyResponse, {
        'email verification completes': (r) => r.status === 200 || r.status === 400, // 400 if token invalid in test
      });
    }

    sleep(randomIntBetween(1, 3));
  });
}

export function handleSummary(data) {
  return {
    'reports/registration-performance.json': JSON.stringify(data, null, 2),
    stdout: textSummary(data, { indent: ' ', enableColors: true }),
  };
}

// tests/performance/critical-flows/user-login.test.ts
import http from 'k6/http';
import { check, sleep, group } from 'k6';
import { Rate, Trend, Counter } from 'k6/metrics';
import { SharedArray } from 'k6/data';

// Custom metrics for login flow
const loginTime = new Trend('login_time');
const loginErrorRate = new Rate('login_error_rate');
const loginSuccess = new Counter('login_success');
const tokenRefreshTime = new Trend('token_refresh_time');

// Performance targets
const LOGIN_TIME_TARGET = 1000; // 1 second
const TOKEN_REFRESH_TARGET = 500; // 500ms

// Test users (pre-created for performance testing)
const testUsers = new SharedArray('users', function () {
  return JSON.parse(open('./test-users.json'));
});

export const options = {
  scenarios: {
    // Normal login load
    login_normal: {
      executor: 'constant-arrival-rate',
      rate: 50, // 50 logins per second
      timeUnit: '1s',
      duration: '5m',
      preAllocatedVUs: 100,
      maxVUs: 200,
    },
    // Peak hour simulation
    login_peak: {
      executor: 'ramping-arrival-rate',
      startRate: 10,
      timeUnit: '1s',
      stages: [
        { duration: '1m', target: 100 },
        { duration: '3m', target: 100 },
        { duration: '1m', target: 10 },
      ],
      preAllocatedVUs: 200,
      maxVUs: 500,
      startTime: '6m',
    },
  },
  thresholds: {
    login_time: ['p(95)<1000', 'p(99)<1500'],
    login_error_rate: ['rate<0.005'], // Less than 0.5% error rate
    'http_req_duration{name:login}': ['p(95)<1000'],
    token_refresh_time: ['p(95)<500'],
  },
};

const BASE_URL = __ENV.BASE_URL || 'http://localhost:3001';

export default function () {
  const user = testUsers[Math.floor(Math.random() * testUsers.length)];

  group('User Login Flow', () => {
    // Step 1: Login request
    const loginStartTime = Date.now();

    const loginResponse = http.post(
      `${BASE_URL}/api/auth/login`,
      JSON.stringify({
        email: user.email,
        password: user.password,
      }),
      {
        headers: { 'Content-Type': 'application/json' },
        tags: { name: 'login' },
      },
    );

    const loginDuration = Date.now() - loginStartTime;
    loginTime.add(loginDuration);

    const loginChecks = check(loginResponse, {
      'login status is 200': (r) => r.status === 200,
      'login response has access token': (r) => r.json('accessToken') !== undefined,
      'login response has refresh token': (r) => r.json('refreshToken') !== undefined,
      'login completes within target': () => loginDuration < LOGIN_TIME_TARGET,
    });

    if (loginChecks) {
      loginSuccess.add(1);
      const accessToken = loginResponse.json('accessToken');
      const refreshToken = loginResponse.json('refreshToken');

      // Step 2: Validate token by accessing protected resource
      const profileResponse = http.get(`${BASE_URL}/api/users/profile`, {
        headers: {
          Authorization: `Bearer ${accessToken}`,
          'Content-Type': 'application/json',
        },
        tags: { name: 'profile_access' },
      });

      check(profileResponse, {
        'profile access with token succeeds': (r) => r.status === 200,
        'profile returns user data': (r) => r.json('email') === user.email,
      });

      // Step 3: Test token refresh (every 10th iteration)
      if (__ITER % 10 === 0) {
        sleep(1);
        const refreshStartTime = Date.now();

        const refreshResponse = http.post(`${BASE_URL}/api/auth/refresh`, JSON.stringify({ refreshToken }), {
          headers: { 'Content-Type': 'application/json' },
          tags: { name: 'token_refresh' },
        });

        const refreshDuration = Date.now() - refreshStartTime;
        tokenRefreshTime.add(refreshDuration);

        check(refreshResponse, {
          'token refresh succeeds': (r) => r.status === 200,
          'refresh returns new access token': (r) => r.json('accessToken') !== undefined,
          'token refresh within target': () => refreshDuration < TOKEN_REFRESH_TARGET,
        });
      }
    } else {
      loginErrorRate.add(1);
      console.log(`Login failed for ${user.email}: ${loginResponse.status}`);
    }

    sleep(Math.random() * 2 + 0.5);
  });
}

// Setup: Create test users if needed
export function setup() {
  console.log('Login performance test setup...');

  // Verify service is healthy
  const healthCheck = http.get(`${BASE_URL}/health`);
  if (healthCheck.status !== 200) {
    throw new Error('Service is not healthy');
  }

  return { startTime: Date.now() };
}

export function teardown(data) {
  console.log(`Test duration: ${(Date.now() - data.startTime) / 1000}s`);
}

// tests/performance/critical-flows/frontend-performance.ts
import { chromium, Browser, Page } from 'playwright';
import lighthouse from 'lighthouse';
import * as chromeLauncher from 'chrome-launcher';

interface PagePerformanceMetrics {
  url: string;
  loadTime: number;
  timeToFirstByte: number;
  firstContentfulPaint: number;
  largestContentfulPaint: number;
  timeToInteractive: number;
  cumulativeLayoutShift: number;
  totalBlockingTime: number;
  lighthouseScore: number;
}

interface CriticalFlowMetrics {
  flowName: string;
  totalDuration: number;
  steps: Array<{
    name: string;
    duration: number;
    status: 'pass' | 'fail';
  }>;
}

const PERFORMANCE_TARGETS = {
  registration: {
    pageLoad: 2000,
    formSubmission: 2000,
    lighthouseScore: 70,
  },
  login: {
    pageLoad: 1500,
    formSubmission: 1000,
    lighthouseScore: 80,
  },
  productSearch: {
    pageLoad: 1500,
    searchResponse: 500,
    lighthouseScore: 75,
  },
  checkout: {
    pageLoad: 2000,
    paymentProcess: 3000,
    lighthouseScore: 70,
  },
};

class CriticalFlowPerformanceTester {
  private browser: Browser | null = null;
  private baseUrl: string;

  constructor(baseUrl: string = 'http://localhost:3000') {
    this.baseUrl = baseUrl;
  }

  async initialize(): Promise<void> {
    this.browser = await chromium.launch({
      headless: true,
      args: ['--disable-gpu', '--no-sandbox', '--disable-dev-shm-usage'],
    });
  }

  async cleanup(): Promise<void> {
    if (this.browser) {
      await this.browser.close();
    }
  }

  // Registration page performance test
  async testRegistrationFlow(): Promise<CriticalFlowMetrics> {
    const page = await this.browser!.newPage();
    const steps: CriticalFlowMetrics['steps'] = [];
    const startTime = Date.now();

    try {
      // Step 1: Navigate to registration page
      const navStart = Date.now();
      await page.goto(`${this.baseUrl}/auth/register`, { waitUntil: 'networkidle' });
      const navDuration = Date.now() - navStart;

      steps.push({
        name: 'Page Load',
        duration: navDuration,
        status: navDuration < PERFORMANCE_TARGETS.registration.pageLoad ? 'pass' : 'fail',
      });

      // Step 2: Fill registration form
      const fillStart = Date.now();
      await page.fill('[data-testid="email-input"]', `test_${Date.now()}@example.com`);
      await page.fill('[data-testid="password-input"]', 'TestPassword123!');
      await page.fill('[data-testid="confirm-password-input"]', 'TestPassword123!');
      await page.fill('[data-testid="first-name-input"]', 'Test');
      await page.fill('[data-testid="last-name-input"]', 'User');
      const fillDuration = Date.now() - fillStart;

      steps.push({
        name: 'Form Fill',
        duration: fillDuration,
        status: fillDuration < 1000 ? 'pass' : 'fail',
      });

      // Step 3: Submit form and wait for response
      const submitStart = Date.now();
      await Promise.all([
        page.waitForResponse((response) => response.url().includes('/api/auth/register') && response.status() === 201),
        page.click('[data-testid="submit-button"]'),
      ]);
      const submitDuration = Date.now() - submitStart;

      steps.push({
        name: 'Form Submission',
        duration: submitDuration,
        status: submitDuration < PERFORMANCE_TARGETS.registration.formSubmission ? 'pass' : 'fail',
      });

      // Step 4: Verify redirect to success/verification page
      await page.waitForURL('**/auth/verify-email**', { timeout: 5000 });

      steps.push({
        name: 'Redirect Complete',
        duration: Date.now() - submitStart,
        status: 'pass',
      });
    } catch (error) {
      steps.push({
        name: 'Error',
        duration: 0,
        status: 'fail',
      });
      console.error('Registration flow error:', error);
    } finally {
      await page.close();
    }

    return {
      flowName: 'User Registration',
      totalDuration: Date.now() - startTime,
      steps,
    };
  }

  // Lighthouse audit for critical pages
  async runLighthouseAudit(url: string): Promise<PagePerformanceMetrics> {
    const chrome = await chromeLauncher.launch({
      chromeFlags: ['--headless', '--disable-gpu', '--no-sandbox'],
    });

    try {
      const options = {
        logLevel: 'error' as const,
        output: 'json' as const,
        port: chrome.port,
        onlyCategories: ['performance'],
      };

      const result = await lighthouse(url, options);
      const lhr = result!.lhr;

      return {
        url,
        loadTime: lhr.audits['interactive']?.numericValue || 0,
        timeToFirstByte: lhr.audits['server-response-time']?.numericValue || 0,
        firstContentfulPaint: lhr.audits['first-contentful-paint']?.numericValue || 0,
        largestContentfulPaint: lhr.audits['largest-contentful-paint']?.numericValue || 0,
        timeToInteractive: lhr.audits['interactive']?.numericValue || 0,
        cumulativeLayoutShift: lhr.audits['cumulative-layout-shift']?.numericValue || 0,
        totalBlockingTime: lhr.audits['total-blocking-time']?.numericValue || 0,
        lighthouseScore: Math.round((lhr.categories.performance?.score || 0) * 100),
      };
    } finally {
      await chrome.kill();
    }
  }
}

// Run all critical flow tests
async function runCriticalPerformanceTests() {
  const tester = new CriticalFlowPerformanceTester(process.env.BASE_URL || 'http://localhost:3000');

  await tester.initialize();

  const results = {
    timestamp: new Date().toISOString(),
    flows: [] as CriticalFlowMetrics[],
    lighthouseAudits: [] as PagePerformanceMetrics[],
  };

  // Run flow tests
  results.flows.push(await tester.testRegistrationFlow());
  results.flows.push(
    await tester.testLoginFlow({
      email: 'test@example.com',

NestJS Performance Tracking Service​

// services/performance-tracking.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { BigQuery } from '@google-cloud/bigquery';

interface PerformanceBaseline {
  flowName: string;
  metric: string;
  p50: number;
  p95: number;
  p99: number;
  errorRate: number;
  timestamp: Date;
}

interface PerformanceComparison {
  flowName: string;
  currentP95: number;
  baselineP95: number;
  percentageChange: number;
  status: 'improved' | 'degraded' | 'stable';
}

@Injectable()
export class PerformanceTrackingService {
  private readonly logger = new Logger(PerformanceTrackingService.name);
  private readonly bigquery: BigQuery;

  constructor() {
    this.bigquery = new BigQuery();
  }

  // Store performance baseline
  async storeBaseline(baseline: PerformanceBaseline): Promise<void> {
    const datasetId = 'performance_data';
    const tableId = 'baselines';

    await this.bigquery
      .dataset(datasetId)
      .table(tableId)
      .insert([
        {
          flow_name: baseline.flowName,
          metric: baseline.metric,
          p50: baseline.p50,
          p95: baseline.p95,
          p99: baseline.p99,
          error_rate: baseline.errorRate,
          timestamp: baseline.timestamp.toISOString(),
        },
      ]);

    this.logger.log(`Stored baseline for ${baseline.flowName}: P95=${baseline.p95}ms`);
  }

  // Compare current performance against baseline
  async compareWithBaseline(currentMetrics: Map<string, number>): Promise<PerformanceComparison[]> {
    const query = `
      SELECT 
        flow_name,
        AVG(p95) as baseline_p95
      FROM \`${process.env.GCP_PROJECT}.performance_data.baselines\`
      WHERE timestamp >= TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 7 DAY)
      GROUP BY flow_name
    `;

    const [rows] = await this.bigquery.query({ query });

    const comparisons: PerformanceComparison[] = [];

    for (const row of rows) {
      const currentP95 = currentMetrics.get(row.flow_name);
      if (currentP95 !== undefined) {
        const percentageChange = ((currentP95 - row.baseline_p95) / row.baseline_p95) * 100;

        let status: 'improved' | 'degraded' | 'stable';
        if (percentageChange > 10) {
          status = 'degraded';
        } else if (percentageChange < -10) {
          status = 'improved';
        } else {
          status = 'stable';
        }

        comparisons.push({
          flowName: row.flow_name,
          currentP95,
          baselineP95: row.baseline_p95,
          percentageChange,
          status,
        });
      }
    }

    return comparisons;
  }

  // Generate performance trend report
  async generateTrendReport(flowName: string, days: number = 30): Promise<any> {
    const query = `
      SELECT 
        DATE(timestamp) as date,
        AVG(p50) as avg_p50,
        AVG(p95) as avg_p95,
        AVG(p99) as avg_p99,
        AVG(error_rate) as avg_error_rate
      FROM \`${process.env.GCP_PROJECT}.performance_data.baselines\`
      WHERE flow_name = @flowName
        AND timestamp >= TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL @days DAY)
      GROUP BY date
      ORDER BY date
    `;

    const options = {
      query,
      params: { flowName, days },
    };

    const [rows] = await this.bigquery.query(options);

    return {
      flowName,
      period: `${days} days`,
      trends: rows,
      summary: this.calculateTrendSummary(rows),
    };
  }

  private calculateTrendSummary(rows: any[]): any {
    if (rows.length < 2) return { trend: 'insufficient_data' };

    const firstWeek = rows.slice(0, 7);
    const lastWeek = rows.slice(-7);

    const avgFirstWeekP95 = firstWeek.reduce((sum, r) => sum + r.avg_p95, 0) / firstWeek.length;
    const avgLastWeekP95 = lastWeek.reduce((sum, r) => sum + r.avg_p95, 0) / lastWeek.length;

    const percentageChange = ((avgLastWeekP95 - avgFirstWeekP95) / avgFirstWeekP95) * 100;

    return {
      trend: percentageChange > 5 ? 'degrading' : percentageChange < -5 ? 'improving' : 'stable',
      percentageChange: percentageChange.toFixed(2),
      avgP95FirstWeek: avgFirstWeekP95.toFixed(2),
      avgP95LastWeek: avgLastWeekP95.toFixed(2),
    };
  }
}
