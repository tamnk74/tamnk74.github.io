RabbitMQ Setup with Kubernetes

Overview​

This guide covers deploying RabbitMQ on Kubernetes with various architectures including standalone, clustered, and production-ready configurations. Learn how to set up RabbitMQ for high availability, scalability, and fault tolerance using StatefulSets, persistent storage, clustering, and monitoring.

Architecture Patterns​

1. RabbitMQ Cluster Architecture​

2. RabbitMQ with Quorum Queues​

3. RabbitMQ Operator Architecture​

Deployment Examples​

1. Single Node RabbitMQ (Development)​

# rabbitmq-single-node.yaml
apiVersion: v1
kind: Namespace
metadata:
  name: rabbitmq

---
apiVersion: v1

3. RabbitMQ Cluster Operator​

The recommended production approach using Kubernetes operators.

Install RabbitMQ Cluster Operator:

# Install the operator
kubectl apply -f https://github.com/rabbitmq/cluster-operator/releases/latest/download/cluster-operator.yml

# Verify installation
kubectl get pods -n rabbitmq-system
kubectl get customresourcedefinitions | grep rabbitmq

RabbitMQ Cluster Custom Resource:

# rabbitmq-cluster-operator.yaml
apiVersion: v1
kind: Namespace
metadata:
  name: rabbitmq

---
apiVersion: rabbitmq.com/v1beta1
kind: RabbitmqCluster
metadata:
  name: production-rabbitmq
  namespace: rabbitmq
spec:
  replicas: 3
  image: rabbitmq:3.12-management-alpine

  # Service configuration
  service:
    type: ClusterIP
    annotations:
      service.beta.kubernetes.io/aws-load-balancer-type: nlb

  # Persistence
  persistence:
    storageClassName: standard-rwo
    storage: 20Gi

  # Resources
  resources:
    requests:
      cpu: 1000m
      memory: 2Gi
    limits:
      cpu: 2000m
      memory: 4Gi

  # Affinity rules
  affinity:
    podAntiAffinity:
      requiredDuringSchedulingIgnoredDuringExecution:
        - labelSelector:
            matchLabels:
              app.kubernetes.io/name: production-rabbitmq
          topologyKey: kubernetes.io/hostname

  # RabbitMQ configuration
  rabbitmq:
    additionalPlugins:
      - rabbitmq_federation
      - rabbitmq_shovel
      - rabbitmq_prometheus
    additionalConfig: |
      # Memory and disk
      vm_memory_high_watermark.relative = 0.6
      vm_memory_high_watermark_paging_ratio = 0.75
      disk_free_limit.absolute = 2GB

      # Cluster partition handling
      cluster_partition_handling = autoheal

      # Queue master locator
      queue_master_locator = min-masters

      # Logging
      log.console = true
      log.console.level = info

      # Prometheus
      prometheus.return_per_object_metrics = true

    advancedConfig: |
      [
        {rabbit, [
          {collect_statistics_interval, 10000}
        ]},
        {rabbitmq_management, [
          {rates_mode, detailed}
        ]}
      ].

  # TLS configuration (optional)
  tls:
    secretName: rabbitmq-tls-secret
    caSecretName: rabbitmq-ca-secret
    disableNonTLSListeners: false

  # Override settings
  override:
    statefulSet:
      spec:
        template:
          spec:
            containers:
              - name: rabbitmq
                env:
                  - name: RABBITMQ_SERVER_ADDITIONAL_ERL_ARGS
                    value: '+sbwt none +sbwtdcpu none +sbwtdio none'
            securityContext:
              fsGroup: 999
              runAsUser: 999
              runAsNonRoot: true

---
# Default user secret
apiVersion: v1
kind: Secret
metadata:
  name: production-rabbitmq-default-user
  namespace: rabbitmq
type: Opaque
stringData:
  username: admin
  password: CHANGE-THIS-SECURE-PASSWORD
  default_user.conf: |
    default_user = admin
    default_pass = CHANGE-THIS-SECURE-PASSWORD

rabbitmq.conf: |
  # TLS listeners
  listeners.ssl.default = 5671
  ssl_options.cacertfile = /etc/rabbitmq/certs/ca.crt
  ssl_options.certfile = /etc/rabbitmq/certs/tls.crt
  ssl_options.keyfile = /etc/rabbitmq/certs/tls.key
  ssl_options.verify = verify_peer
  ssl_options.fail_if_no_peer_cert = false
  ssl_options.versions.1 = tlsv1.3
  ssl_options.versions.2 = tlsv1.2

  # Management UI with TLS
  management.ssl.port = 15671
  management.ssl.cacertfile = /etc/rabbitmq/certs/ca.crt
  management.ssl.certfile = /etc/rabbitmq/certs/tls.crt
  management.ssl.keyfile = /etc/rabbitmq/certs/tls.key

2. Network Policies​

# rabbitmq-network-policy.yaml
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: rabbitmq-network-policy
  namespace: rabbitmq
spec:
  podSelector:
    matchLabels:
      app: rabbitmq
  policyTypes:
    - Ingress
    - Egress
  ingress:
    # Allow from application pods
    - from:
        - namespaceSelector:
            matchLabels:
              name: default
        - podSelector:
            matchLabels:
              app: backend
      ports:
        - protocol: TCP
          port: 5672
        - protocol: TCP
          port: 5671
    # Allow management access
    - from:
        - namespaceSelector:
            matchLabels:
              name: monitoring
      ports:
        - protocol: TCP
          port: 15672
        - protocol: TCP
          port: 15692
    # Allow inter-cluster communication
    - from:
        - podSelector:
            matchLabels:
              app: rabbitmq
      ports:
        - protocol: TCP
          port: 4369 # EPMD
        - protocol: TCP
          port: 25672 # Distribution
  egress:
    # Allow DNS
    - to:
        - namespaceSelector:
            matchLabels:
              name: kube-system
      ports:
        - protocol: UDP
          port: 53
    # Allow inter-cluster
    - to:
        - podSelector:
            matchLabels:
              app: rabbitmq
      ports:
        - protocol: TCP
          port: 4369
        - protocol: TCP
          port: 25672
    # Allow Kubernetes API
    - to:
        - namespaceSelector: {}
          podSelector:
            matchLabels:
              component: apiserver
      ports:
        - protocol: TCP
          port: 443

Producer Service:

// rabbitmq-producer.service.ts
import { Injectable, Inject, OnModuleInit } from '@nestjs/common';
import { ClientProxy } from '@nestjs/microservices';
import { lastValueFrom } from 'rxjs';

interface MessageOptions {
  priority?: number;
  expiration?: string;
  headers?: Record<string, any>;
  correlationId?: string;
}

@Injectable()
export class RabbitmqProducerService implements OnModuleInit {
  constructor(
    @Inject('RABBITMQ_SERVICE')
    private readonly client: ClientProxy,
  ) {}

  async onModuleInit() {
    await this.client.connect();
  }

  async sendMessage(pattern: string, data: any, options?: MessageOptions) {
    return this.client.emit(pattern, {
      data,
      options,
      timestamp: new Date().toISOString(),
    });
  }

  async sendMessageAndWaitForResponse(pattern: string, data: any, options?: MessageOptions) {
    return lastValueFrom(
      this.client.send(pattern, {
        data,
        options,
        timestamp: new Date().toISOString(),
      }),
    );
  }

  async publishToExchange(exchange: string, routingKey: string, message: any, options?: MessageOptions) {
    return this.client.emit(`${exchange}.${routingKey}`, {
      message,
      options,
      timestamp: new Date().toISOString(),
    });
  }
}

@Controller()
export class RabbitmqConsumerController {
  private readonly logger = new Logger(RabbitmqConsumerController.name);

  @EventPattern('order.created')
  async handleOrderCreated(@Payload() data: any, @Ctx() context: RmqContext) {
    const channel = context.getChannelRef();
    const originalMessage = context.getMessage();

    this.logger.log('Received order.created event', {
      data,
      properties: originalMessage.properties,
      fields: originalMessage.fields,
    });

    try {
      // Process the message
      await this.processOrder(data);

      // Manually acknowledge
      channel.ack(originalMessage);
    } catch (error) {
      this.logger.error('Error processing order', error);

      // Reject and requeue (or send to DLQ)
      channel.nack(originalMessage, false, false);
    }
  }

  @MessagePattern('order.status')
  async getOrderStatus(@Payload() data: any, @Ctx() context: RmqContext) {
    const channel = context.getChannelRef();
    const originalMessage = context.getMessage();

    try {
      const status = await this.checkOrderStatus(data.orderId);

      channel.ack(originalMessage);

      return {
        orderId: data.orderId,
        status,
        timestamp: new Date().toISOString(),
      };
    } catch (error) {
      this.logger.error('Error checking order status', error);
      channel.nack(originalMessage, false, false);
      throw error;
    }
  }

    this.logger.log('Processing order:', data);
  }

Advanced Consumer with Retry Logic:

// rabbitmq-advanced-consumer.controller.ts
import { Controller, Logger } from '@nestjs/common';
import { EventPattern, Payload, Ctx, RmqContext } from '@nestjs/microservices';

interface RetryMetadata {
  retryCount: number;
  originalTimestamp: string;
  errors: string[];
}

@Controller()
export class RabbitmqAdvancedConsumerController {
  private readonly logger = new Logger(RabbitmqAdvancedConsumerController.name);
  private readonly maxRetries = 3;
  private readonly retryDelays = [1000, 5000, 10000]; // Exponential backoff

  @EventPattern('payment.process')
  async handlePayment(@Payload() data: any, @Ctx() context: RmqContext) {
    const channel = context.getChannelRef();
    const originalMessage = context.getMessage();
    const headers = originalMessage.properties.headers || {};

    const retryMetadata: RetryMetadata = {
      retryCount: headers['x-retry-count'] || 0,
      originalTimestamp: headers['x-original-timestamp'] || new Date().toISOString(),
      errors: headers['x-errors'] ? JSON.parse(headers['x-errors']) : [],
    };

    try {
      // Process payment
      await this.processPayment(data);

      // Success - acknowledge
      channel.ack(originalMessage);

      this.logger.log('Payment processed successfully', {
        data,
        retryCount: retryMetadata.retryCount,
      });
    } catch (error) {
      this.logger.error('Payment processing failed', {
        error: error.message,
        retryCount: retryMetadata.retryCount,
      });

      retryMetadata.errors.push(error.message);

      if (retryMetadata.retryCount < this.maxRetries) {
        // Retry with delay
        await this.retryWithDelay(channel, originalMessage, data, retryMetadata);
      } else {
        // Max retries exceeded - send to DLQ
        await this.sendToDeadLetterQueue(channel, originalMessage, data, retryMetadata);
      }
    }
  }

  private async retryWithDelay(channel: any, originalMessage: any, data: any, retryMetadata: RetryMetadata) {
    const delay = this.retryDelays[retryMetadata.retryCount] || 10000;

    this.logger.warn(`Retrying message after ${delay}ms`, {
      retryCount: retryMetadata.retryCount + 1,
    });

    // Publish to retry queue with delay
    channel.publish('', 'payment.process.retry', Buffer.from(JSON.stringify(data)), {
      persistent: true,
      headers: {
        'x-retry-count': retryMetadata.retryCount + 1,
        'x-original-timestamp': retryMetadata.originalTimestamp,
        'x-errors': JSON.stringify(retryMetadata.errors),
        'x-delay': delay,
      },
    });

    // Acknowledge original message
    channel.ack(originalMessage);
  }

  private async sendToDeadLetterQueue(channel: any, originalMessage: any, data: any, retryMetadata: RetryMetadata) {
    this.logger.error('Max retries exceeded - sending to DLQ', {
      data,
      retryMetadata,
    });

    // Publish to DLQ
    channel.publish('', 'payment.process.dlq', Buffer.from(JSON.stringify(data)), {
      persistent: true,
      headers: {
        'x-retry-count': retryMetadata.retryCount,
        'x-original-timestamp': retryMetadata.originalTimestamp,
        'x-errors': JSON.stringify(retryMetadata.errors),
        'x-death-reason': 'max-retries-exceeded',
      },
    });

    // Acknowledge original message
    channel.ack(originalMessage);
  }

  private async processPayment(data: any): Promise<void> {
    // Implementation
    this.logger.log('Processing payment:', data);
  }
}

2. Data Backup with Snapshots​

#!/bin/bash
# backup-rabbitmq-data.sh

NAMESPACE="rabbitmq"
BACKUP_DIR="/backup/rabbitmq/data"
TIMESTAMP=$(date +%Y%m%d-%H%M%S)

mkdir -p ${BACKUP_DIR}/${TIMESTAMP}

# Backup each RabbitMQ node's data
for i in 0 1 2; do
  POD_NAME="rabbitmq-${i}"

  echo "Backing up ${POD_NAME}..."

  # Create tar archive
  kubectl exec -n ${NAMESPACE} ${POD_NAME} -- \
    tar czf /tmp/rabbitmq-backup.tar.gz /var/lib/rabbitmq

  # Copy to local
  kubectl cp ${NAMESPACE}/${POD_NAME}:/tmp/rabbitmq-backup.tar.gz \
    ${BACKUP_DIR}/${TIMESTAMP}/${POD_NAME}.tar.gz

  # Upload to GCS
  gcloud storage cp ${BACKUP_DIR}/${TIMESTAMP}/${POD_NAME}.tar.gz \
    gs://your-backup-bucket/rabbitmq/data/${TIMESTAMP}/

  # Cleanup
  kubectl exec -n ${NAMESPACE} ${POD_NAME} -- rm /tmp/rabbitmq-backup.tar.gz
done

echo "Data backup completed: ${TIMESTAMP}"

3. Restore Procedure​

#!/bin/bash
# restore-rabbitmq.sh

NAMESPACE="rabbitmq"
BACKUP_TIMESTAMP="20240101-120000"

# Scale down to prevent conflicts
kubectl scale statefulset rabbitmq -n ${NAMESPACE} --replicas=0

# Wait for pods to terminate
kubectl wait --for=delete pod -l app=rabbitmq -n ${NAMESPACE} --timeout=120s

# Restore data for each node
for i in 0 1 2; do
  POD_NAME="rabbitmq-${i}"
  PVC_NAME="data-rabbitmq-${i}"

  # Download backup
  gcloud storage cp \
    gs://your-backup-bucket/rabbitmq/data/${BACKUP_TIMESTAMP}/${POD_NAME}.tar.gz \
    /tmp/

  # Create temporary pod to restore data
  kubectl run restore-${i} -n ${NAMESPACE} \
    --image=busybox \
    --restart=Never \
    --overrides="{
      \"spec\": {
        \"volumes\": [{
          \"name\": \"data\",
          \"persistentVolumeClaim\": {
            \"claimName\": \"${PVC_NAME}\"
          }
        }],
        \"containers\": [{
          \"name\": \"restore\",
          \"image\": \"busybox\",
          \"command\": [\"sleep\", \"3600\"],
          \"volumeMounts\": [{
            \"name\": \"data\",
            \"mountPath\": \"/data\"
          }]
        }]
      }
    }"

  # Wait for pod
  kubectl wait --for=condition=ready pod restore-${i} -n ${NAMESPACE} --timeout=60s

  # Copy backup to pod
  kubectl cp /tmp/${POD_NAME}.tar.gz ${NAMESPACE}/restore-${i}:/tmp/

  # Extract backup
  kubectl exec -n ${NAMESPACE} restore-${i} -- \
    tar xzf /tmp/${POD_NAME}.tar.gz -C /data

  # Cleanup
  kubectl delete pod restore-${i} -n ${NAMESPACE}
done

# Restore definitions
gcloud storage cp \
  gs://your-backup-bucket/rabbitmq/definitions/${BACKUP_TIMESTAMP}/definitions.json \
  /tmp/

# Scale up
kubectl scale statefulset rabbitmq -n ${NAMESPACE} --replicas=3

# Wait for cluster to be ready
kubectl wait --for=condition=ready pod -l app=rabbitmq -n ${NAMESPACE} --timeout=300s

# Import definitions
kubectl exec -n ${NAMESPACE} rabbitmq-0 -- \
  curl -u admin:password \
  -X POST \
  -H "Content-Type: application/json" \
  -d @/tmp/definitions.json \
  http://localhost:15672/api/definitions

echo "Restore completed"

Troubleshooting​

Common Issues​

1. Cluster formation issues:

# Check cluster status
kubectl exec -it rabbitmq-0 -n rabbitmq -- rabbitmqctl cluster_status

# Check node connectivity
kubectl exec -it rabbitmq-0 -n rabbitmq -- rabbitmqctl ping

# Reset node and rejoin cluster
kubectl exec -it rabbitmq-1 -n rabbitmq -- rabbitmqctl stop_app
kubectl exec -it rabbitmq-1 -n rabbitmq -- rabbitmqctl reset
kubectl exec -it rabbitmq-1 -n rabbitmq -- rabbitmqctl join_cluster rabbit@rabbitmq-0.rabbitmq-headless
kubectl exec -it rabbitmq-1 -n rabbitmq -- rabbitmqctl start_app

2. Memory alarms:

# Check memory status
kubectl exec -it rabbitmq-0 -n rabbitmq -- rabbitmqctl status | grep memory

# Clear memory alarm (after investigating)
kubectl exec -it rabbitmq-0 -n rabbitmq -- rabbitmqctl set_vm_memory_high_watermark 0.7

# Force garbage collection
kubectl exec -it rabbitmq-0 -n rabbitmq -- rabbitmqctl eval 'garbage_collect().'

3. Quorum queue issues:

# Check quorum queue status
kubectl exec -it rabbitmq-0 -n rabbitmq -- rabbitmqctl list_queues name type messages_ready messages_unacknowledged

# Grow quorum queue cluster
kubectl exec -it rabbitmq-0 -n rabbitmq -- rabbitmqctl grow_quorum_queue queue-name

# Shrink quorum queue cluster
kubectl exec -it rabbitmq-0 -n rabbitmq -- rabbitmqctl shrink_quorum_queue queue-name rabbit@rabbitmq-2.rabbitmq-headless

4. Unacknowledged messages:

# List consumer details
kubectl exec -it rabbitmq-0 -n rabbitmq -- rabbitmqctl list_consumers

# Purge queue (use with caution)
kubectl exec -it rabbitmq-0 -n rabbitmq -- rabbitmqctl purge_queue queue-name

5. Network partition resolution:

# Check for partitions
kubectl exec -it rabbitmq-0 -n rabbitmq -- rabbitmqctl cluster_status

# Restart nodes one by one to resolve partition
kubectl delete pod rabbitmq-1 -n rabbitmq
kubectl wait --for=condition=ready pod rabbitmq-1 -n rabbitmq

# Or use manual partition resolution
kubectl exec -it rabbitmq-0 -n rabbitmq -- rabbitmqctl forget_cluster_node rabbit@rabbitmq-2.rabbitmq-headless
