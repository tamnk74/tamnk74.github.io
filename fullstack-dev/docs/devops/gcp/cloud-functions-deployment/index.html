Deploy Cloud Functions with Node.js and GitHub Actions

Overview​

This guide covers deploying Google Cloud Functions using Node.js and automating deployments with GitHub Actions. Cloud Functions is a serverless execution environment for building and connecting cloud services, allowing you to run code without managing servers.

Cloud Functions Fundamentals​

What are Cloud Functions?​

Cloud Functions are lightweight, event-driven, serverless compute solutions that allow you to create small, single-purpose functions that respond to cloud events.

Key Features​

  • Serverless: No server management required
  • Auto-scaling: Automatically scales from zero to planet-scale
  • Event-driven: Triggered by HTTP requests, Cloud Storage, Pub/Sub, etc.
  • Pay-per-use: Only pay for execution time
  • Multiple runtimes: Node.js 16, 18, 20

functions/http-function/index.js

const functions = require('@google-cloud/functions-framework');

/**
 * HTTP Cloud Function - Hello World
 * @param {Object} req - Express request object
 * @param {Object} res - Express response object
 */
functions.http('helloWorld', (req, res) => {
  const name = req.query.name || req.body.name || 'World';

  res.status(200).json({
    message: `Hello, ${name}!`,
    timestamp: new Date().toISOString(),
    method: req.method,
    headers: req.headers,
  });
});

/**
 * HTTP Cloud Function - REST API Example
 */
functions.http('apiFunction', async (req, res) => {
  // CORS headers
  res.set('Access-Control-Allow-Origin', '*');
  res.set('Access-Control-Allow-Methods', 'GET, POST, PUT, DELETE');
  res.set('Access-Control-Allow-Headers', 'Content-Type, Authorization');

  // Handle preflight
  if (req.method === 'OPTIONS') {
    res.status(204).send('');
    return;
  }

  try {
    const { method, path } = req;

    switch (method) {
      case 'GET':
        res.status(200).json({
          data: [
            { id: 1, name: 'Item 1' },
            { id: 2, name: 'Item 2' },
          ],
        });
        break;

      case 'POST':
        const body = req.body;
        res.status(201).json({
          message: 'Created successfully',
          data: { id: Date.now(), ...body },
        });
        break;

      case 'PUT':
        res.status(200).json({
          message: 'Updated successfully',
        });
        break;

      case 'DELETE':
        res.status(200).json({
          message: 'Deleted successfully',
        });
        break;

      default:
        res.status(405).json({ error: 'Method not allowed' });
    }
  } catch (error) {
    console.error('Error:', error);
    res.status(500).json({
      error: 'Internal server error',
      message: error.message,
    });
  }
});

Function Types and Examples​

1. HTTP Function (Gen2)​

functions/http-function/index.js

const functions = require('@google-cloud/functions-framework');
const { SecretManagerServiceClient } = require('@google-cloud/secret-manager');

// Initialize Secret Manager
const secretManager = new SecretManagerServiceClient();

/**
 * Advanced HTTP Function with authentication and secrets
 */
functions.http('secureApi', async (req, res) => {
  // CORS
  res.set('Access-Control-Allow-Origin', '*');

  if (req.method === 'OPTIONS') {
    res.set('Access-Control-Allow-Methods', 'GET, POST');
    res.set('Access-Control-Allow-Headers', 'Authorization, Content-Type');
    res.status(204).send('');
    return;
  }

  try {
    // Verify authorization
    const authHeader = req.headers.authorization;
    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      return res.status(401).json({ error: 'Unauthorized' });
    }

    const token = authHeader.split('Bearer ')[1];

    // Get API key from Secret Manager
    const projectId = process.env.GCP_PROJECT || process.env.GOOGLE_CLOUD_PROJECT;
    const [version] = await secretManager.accessSecretVersion({
      name: `projects/${projectId}/secrets/api-key/versions/latest`,
    });

    const apiKey = version.payload.data.toString('utf8');

    if (token !== apiKey) {
      return res.status(403).json({ error: 'Invalid token' });
    }

    // Process request
    const result = await processRequest(req.body);

    res.status(200).json({
      success: true,
      data: result,
    });
  } catch (error) {
    console.error('Error:', error);
    res.status(500).json({
      error: 'Internal server error',
      message: process.env.NODE_ENV === 'development' ? error.message : undefined,
    });
  }
});

async function processRequest(data) {
  // Your business logic here
  return {
    processed: true,
    timestamp: new Date().toISOString(),
    input: data,
  };
}

2. Pub/Sub Function​

functions/pubsub-function/index.js

const functions = require('@google-cloud/functions-framework');
const { BigQuery } = require('@google-cloud/bigquery');
const { PubSub } = require('@google-cloud/pubsub');

const bigquery = new BigQuery();
const pubsub = new PubSub();

/**
 * Pub/Sub triggered function
 * Processes messages from a Pub/Sub topic
 */
functions.cloudEvent('processPubSubMessage', async (cloudEvent) => {
  try {
    // Decode Pub/Sub message
    const base64Data = cloudEvent.data.message.data;
    const messageData = JSON.parse(Buffer.from(base64Data, 'base64').toString());

    console.log('Received message:', messageData);

    // Process the message
    const result = await processMessage(messageData);

    // Insert into BigQuery
    const datasetId = 'analytics';
    const tableId = 'events';

    await bigquery.dataset(datasetId).table(tableId).insert([result]);

    console.log('Message processed successfully');
  } catch (error) {
    console.error('Error processing message:', error);
    throw error; // Will retry based on Pub/Sub retry policy
  }
});

async function processMessage(data) {
  return {
    event_id: data.id || generateId(),
    event_type: data.type,
    user_id: data.userId,
    timestamp: new Date().toISOString(),
    properties: JSON.stringify(data.properties),
    processed_at: new Date().toISOString(),
  };
}

function generateId() {
  return `${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
}

/**
 * Publish message example function
 */
functions.http('publishMessage', async (req, res) => {
  try {
    const topicName = 'events-topic';
    const messageData = req.body;

    const dataBuffer = Buffer.from(JSON.stringify(messageData));

    const messageId = await pubsub.topic(topicName).publishMessage({ data: dataBuffer });

    res.status(200).json({
      success: true,
      messageId: messageId,
    });
  } catch (error) {
    console.error('Error publishing message:', error);
    res.status(500).json({ error: error.message });
  }
});

3. Cloud Storage Function​

functions/storage-function/index.js

const functions = require('@google-cloud/functions-framework');
const { Storage } = require('@google-cloud/storage');
const sharp = require('sharp');
const path = require('path');

const storage = new Storage();

/**
 * Cloud Storage triggered function
 * Processes uploaded images - creates thumbnails
 */
functions.cloudEvent('processImage', async (cloudEvent) => {
  const file = cloudEvent.data;

  console.log(`Processing file: ${file.name}`);
  console.log(`Bucket: ${file.bucket}`);

  // Only process images
  if (!file.contentType || !file.contentType.startsWith('image/')) {
    console.log('Not an image file, skipping');
    return;
  }

  // Avoid processing thumbnails
  if (file.name.includes('_thumb')) {
    console.log('Already a thumbnail, skipping');
    return;
  }

  try {
    const bucket = storage.bucket(file.bucket);
    const fileObject = bucket.file(file.name);

    // Download file
    const [imageBuffer] = await fileObject.download();

    // Create thumbnail
    const thumbnailBuffer = await sharp(imageBuffer)
      .resize(200, 200, {
        fit: 'inside',
        withoutEnlargement: true,
      })
      .toBuffer();

    // Upload thumbnail
    const thumbFileName = getThumbFileName(file.name);
    const thumbFile = bucket.file(thumbFileName);

    await thumbFile.save(thumbnailBuffer, {
      metadata: {
        contentType: file.contentType,
        metadata: {
          originalFile: file.name,
          createdBy: 'cloud-function',
        },
      },
    });

    console.log(`Thumbnail created: ${thumbFileName}`);
  } catch (error) {
    console.error('Error processing image:', error);
    throw error;
  }
});

function getThumbFileName(fileName) {
  const ext = path.extname(fileName);
  const basename = path.basename(fileName, ext);
  const dirname = path.dirname(fileName);

  return path.join(dirname, `${basename}_thumb${ext}`);
}

4. Scheduled Function​

functions/scheduled-function/index.js

const functions = require('@google-cloud/functions-framework');
const { BigQuery } = require('@google-cloud/bigquery');
const { Storage } = require('@google-cloud/storage');

const bigquery = new BigQuery();
const storage = new Storage();

/**
 * Scheduled function triggered by Cloud Scheduler
 * Runs daily data export job
 */
functions.http('dailyExport', async (req, res) => {
  try {
    console.log('Starting daily export job');

    const projectId = process.env.GCP_PROJECT;
    const datasetId = 'analytics';
    const tableId = 'daily_summary';
    const bucketName = 'export-bucket';

    // Query data
    const query = `
      SELECT
        DATE(timestamp) as date,
        COUNT(*) as total_events,
        COUNT(DISTINCT user_id) as unique_users
      FROM \`${projectId}.${datasetId}.events\`
      WHERE DATE(timestamp) = CURRENT_DATE() - 1
      GROUP BY date
    `;

    const [rows] = await bigquery.query({ query });

    // Export to Cloud Storage
    const fileName = `exports/daily-${new Date().toISOString().split('T')[0]}.json`;
    const file = storage.bucket(bucketName).file(fileName);

    await file.save(JSON.stringify(rows, null, 2), {
      contentType: 'application/json',
      metadata: {
        exportDate: new Date().toISOString(),
      },
    });

    console.log(`Export completed: ${fileName}`);

    res.status(200).json({
      success: true,
      fileName: fileName,
      rowCount: rows.length,
    });
  } catch (error) {
    console.error('Export failed:', error);
    res.status(500).json({
      success: false,
      error: error.message,
    });
  }
});

GitHub Secrets Configuration​

Add these secrets to your GitHub repository (Settings → Secrets and variables → Actions):

  • GCP_PROJECT_ID: Your GCP project ID
  • GCP_SA_KEY: Content of key-base64.txt
  • GCP_REGION: Deployment region (e.g., us-central1)

Complete GitHub Actions Workflow​

.github/workflows/deploy-cloud-functions.yml

name: Deploy Cloud Functions

on:
  push:
    branches:
      - main
      - develop
    paths:
      - 'functions/**'
      - '.github/workflows/deploy-cloud-functions.yml'
  pull_request:
    branches:
      - main
    paths:
      - 'functions/**'
  workflow_dispatch:
    inputs:
      environment:
        description: 'Environment to deploy to'
        required: true
        default: 'dev'
        type: choice
        options:
          - dev
          - staging
          - prod

env:
  NODE_VERSION: '20'
  GCP_REGION: ${{ secrets.GCP_REGION || 'us-central1' }}

jobs:
  detect-changes:
    runs-on: ubuntu-latest
    outputs:
      http-function: ${{ steps.filter.outputs.http-function }}
      pubsub-function: ${{ steps.filter.outputs.pubsub-function }}
      storage-function: ${{ steps.filter.outputs.storage-function }}
    steps:
      - uses: actions/checkout@v4

      - uses: dorny/paths-filter@v2
        id: filter
        with:
          filters: |
            http-function:
              - 'functions/http-function/**'
            pubsub-function:
              - 'functions/pubsub-function/**'
            storage-function:
              - 'functions/storage-function/**'

  test:
    runs-on: ubuntu-latest
    needs: detect-changes
    if: |
      needs.detect-changes.outputs.http-function == 'true' ||
      needs.detect-changes.outputs.pubsub-function == 'true' ||
      needs.detect-changes.outputs.storage-function == 'true'

    strategy:
      matrix:
        function:
          - name: http-function
            changed: ${{ needs.detect-changes.outputs.http-function }}
          - name: pubsub-function
            changed: ${{ needs.detect-changes.outputs.pubsub-function }}
          - name: storage-function
            changed: ${{ needs.detect-changes.outputs.storage-function }}

    steps:
      - name: Checkout code
        if: matrix.function.changed == 'true'
        uses: actions/checkout@v4

      - name: Setup Node.js
        if: matrix.function.changed == 'true'
        uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'npm'
          cache-dependency-path: functions/${{ matrix.function.name }}/package-lock.json

      - name: Install dependencies
        if: matrix.function.changed == 'true'
        working-directory: functions/${{ matrix.function.name }}
        run: npm ci

      - name: Run linter
        if: matrix.function.changed == 'true'
        working-directory: functions/${{ matrix.function.name }}
        run: npm run lint || echo "No lint script found"

      - name: Run tests
        if: matrix.function.changed == 'true'
        working-directory: functions/${{ matrix.function.name }}
        run: npm test || echo "No tests found"

      - name: Build
        if: matrix.function.changed == 'true'
        working-directory: functions/${{ matrix.function.name }}
        run: npm run build || echo "No build script found"

  deploy-dev:
    runs-on: ubuntu-latest
    needs: [detect-changes, test]
    if: |
      github.ref == 'refs/heads/develop' &&
      github.event_name == 'push'
    environment: development
    strategy:
      matrix:
        function:
          - name: http-function
            changed: ${{ needs.detect-changes.outputs.http-function }}
            entry-point: helloWorld
            trigger: --trigger-http --allow-unauthenticated
          - name: pubsub-function
            changed: ${{ needs.detect-changes.outputs.pubsub-function }}
            entry-point: processPubSubMessage
            trigger: --trigger-topic=events-topic
          - name: storage-function
            changed: ${{ needs.detect-changes.outputs.storage-function }}
            entry-point: processImage
            trigger: --trigger-bucket=uploads-dev

    steps:
      - name: Checkout code
        if: matrix.function.changed == 'true'
        uses: actions/checkout@v4

      - name: Authenticate to Google Cloud
        if: matrix.function.changed == 'true'
        uses: google-github-actions/auth@v2
        with:
          credentials_json: ${{ secrets.GCP_SA_KEY }}

      - name: Set up Cloud SDK
        if: matrix.function.changed == 'true'
        uses: google-github-actions/setup-gcloud@v2

      - name: Deploy function
        if: matrix.function.changed == 'true'
        working-directory: functions/${{ matrix.function.name }}
        run: |
          gcloud functions deploy ${{ matrix.function.name }}-dev \
            --gen2 \
            --runtime=nodejs20 \
            --region=${{ env.GCP_REGION }} \
            --source=. \
            --entry-point=${{ matrix.function.entry-point }} \
            ${{ matrix.function.trigger }} \
            --set-env-vars=ENVIRONMENT=dev,NODE_ENV=development \
            --memory=256Mi \
            --timeout=60s \
            --max-instances=10 \
            --service-account=${{ secrets.GCP_SA_EMAIL }}

      - name: Test deployed function
        if: matrix.function.changed == 'true' && contains(matrix.function.trigger, 'http')
        run: |
          FUNCTION_URL=$(gcloud functions describe ${{ matrix.function.name }}-dev \
            --gen2 \
            --region=${{ env.GCP_REGION }} \
            --format='value(serviceConfig.uri)')

          echo "Testing function at: $FUNCTION_URL"
          curl -f $FUNCTION_URL || exit 1

  deploy-staging:
    runs-on: ubuntu-latest
    needs: [detect-changes, test]
    if: |
      github.ref == 'refs/heads/main' &&
      github.event_name == 'push'
    environment: staging
    strategy:
      matrix:
        function:
          - name: http-function
            changed: ${{ needs.detect-changes.outputs.http-function }}
            entry-point: helloWorld
            trigger: --trigger-http --allow-unauthenticated
          - name: pubsub-function
            changed: ${{ needs.detect-changes.outputs.pubsub-function }}
            entry-point: processPubSubMessage
            trigger: --trigger-topic=events-topic-staging
          - name: storage-function
            changed: ${{ needs.detect-changes.outputs.storage-function }}
            entry-point: processImage
            trigger: --trigger-bucket=uploads-staging

    steps:
      - name: Checkout code
        if: matrix.function.changed == 'true'
        uses: actions/checkout@v4

      - name: Authenticate to Google Cloud
        if: matrix.function.changed == 'true'
        uses: google-github-actions/auth@v2
        with:
          credentials_json: ${{ secrets.GCP_SA_KEY }}

      - name: Set up Cloud SDK
        if: matrix.function.changed == 'true'
        uses: google-github-actions/setup-gcloud@v2

      - name: Deploy function
        if: matrix.function.changed == 'true'
        working-directory: functions/${{ matrix.function.name }}
        run: |
          gcloud functions deploy ${{ matrix.function.name }}-staging \
            --gen2 \
            --runtime=nodejs20 \
            --region=${{ env.GCP_REGION }} \
            --source=. \
            --entry-point=${{ matrix.function.entry-point }} \
            ${{ matrix.function.trigger }} \
            --set-env-vars=ENVIRONMENT=staging,NODE_ENV=production \
            --memory=512Mi \
            --timeout=120s \
            --max-instances=50 \
            --min-instances=1 \
            --service-account=${{ secrets.GCP_SA_EMAIL }}

  deploy-production:
    runs-on: ubuntu-latest
    needs: [detect-changes, deploy-staging]
    if: github.ref == 'refs/heads/main'
    environment: production

    strategy:
      matrix:
        function:
          - name: http-function
            changed: ${{ needs.detect-changes.outputs.http-function }}
            entry-point: helloWorld
            trigger: --trigger-http
          - name: pubsub-function
            changed: ${{ needs.detect-changes.outputs.pubsub-function }}
            entry-point: processPubSubMessage
            trigger: --trigger-topic=events-topic-prod
          - name: storage-function
            changed: ${{ needs.detect-changes.outputs.storage-function }}
            entry-point: processImage
            trigger: --trigger-bucket=uploads-prod

    steps:
      - name: Checkout code
        if: matrix.function.changed == 'true'
        uses: actions/checkout@v4

      - name: Authenticate to Google Cloud
        if: matrix.function.changed == 'true'
        uses: google-github-actions/auth@v2
        with:
          credentials_json: ${{ secrets.GCP_SA_KEY }}

      - name: Set up Cloud SDK
        if: matrix.function.changed == 'true'
        uses: google-github-actions/setup-gcloud@v2

      - name: Deploy function with gradual rollout
        if: matrix.function.changed == 'true'
        working-directory: functions/${{ matrix.function.name }}
        run: |
          # Deploy new revision
          gcloud functions deploy ${{ matrix.function.name }}-prod \
            --gen2 \
            --runtime=nodejs20 \
            --region=${{ env.GCP_REGION }} \
            --source=. \
            --entry-point=${{ matrix.function.entry-point }} \
            ${{ matrix.function.trigger }} \
            --set-env-vars=ENVIRONMENT=production,NODE_ENV=production \
            --memory=1Gi \
            --timeout=300s \
            --max-instances=100 \
            --min-instances=2 \
            --cpu=1 \
            --concurrency=80 \
            --service-account=${{ secrets.GCP_SA_EMAIL }} \
            --no-traffic

      - name: Gradually shift traffic
        if: matrix.function.changed == 'true'
        run: |
          # Get the new revision
          NEW_REVISION=$(gcloud functions describe ${{ matrix.function.name }}-prod \
            --gen2 \
            --region=${{ env.GCP_REGION }} \
            --format='value(serviceConfig.revision)')

          echo "New revision: $NEW_REVISION"

          # Gradual rollout: 10% -> 50% -> 100%
          echo "Shifting 10% traffic to new revision"
          gcloud run services update-traffic ${{ matrix.function.name }}-prod \
            --region=${{ env.GCP_REGION }} \
            --to-revisions=$NEW_REVISION=10

          sleep 300  # Wait 5 minutes

          echo "Shifting 50% traffic to new revision"
          gcloud run services update-traffic ${{ matrix.function.name }}-prod \
            --region=${{ env.GCP_REGION }} \
            --to-revisions=$NEW_REVISION=50

          sleep 300  # Wait 5 minutes

          echo "Shifting 100% traffic to new revision"
          gcloud run services update-traffic ${{ matrix.function.name }}-prod \
            --region=${{ env.GCP_REGION }} \
            --to-latest

      - name: Run smoke tests
        if: matrix.function.changed == 'true' && contains(matrix.function.trigger, 'http')
        run: |
          FUNCTION_URL=$(gcloud functions describe ${{ matrix.function.name }}-prod \
            --gen2 \
            --region=${{ env.GCP_REGION }} \
            --format='value(serviceConfig.uri)')

          echo "Running smoke tests on: $FUNCTION_URL"

          # Test basic functionality
          response=$(curl -s -o /dev/null -w "%{http_code}" $FUNCTION_URL)
          if [ $response -ne 200 ]; then
            echo "Smoke test failed with status code: $response"
            exit 1
          fi

          echo "Smoke tests passed!"

  notify:
    runs-on: ubuntu-latest
    needs: [deploy-dev, deploy-staging, deploy-production]
    if: always()
    steps:
      - name: Send notification
        run: |
          echo "Deployment completed!"
          # Add Slack/Discord/Email notification here

const { SecretManagerServiceClient } = require('@google-cloud/secret-manager');
const functions = require('@google-cloud/functions-framework');

async function getSecret(secretName) {
  const projectId = process.env.GCP_PROJECT;
  const name = `projects/${projectId}/secrets/${secretName}/versions/latest`;

  const [version] = await client.accessSecretVersion({ name });
  return version.payload.data.toString('utf8');
}

Cloud Monitoring​

Custom metrics

const { MetricServiceClient } = require('@google-cloud/monitoring');
const functions = require('@google-cloud/functions-framework');

const client = new MetricServiceClient();

async function writeCustomMetric(metricValue) {
  const projectId = process.env.GCP_PROJECT;

  const dataPoint = {
    interval: {
      endTime: {
        seconds: Date.now() / 1000,
      },
    },
    value: {
      doubleValue: metricValue,
    },
  };

  const timeSeriesData = {
    metric: {
      type: 'custom.googleapis.com/function/processing_time',
      labels: {
        function_name: process.env.FUNCTION_NAME,
      },
    },
    resource: {
      type: 'cloud_function',
      labels: {
        function_name: process.env.FUNCTION_NAME,
        region: process.env.FUNCTION_REGION,
      },
    },
    points: [dataPoint],
  };

  const request = {
    name: client.projectPath(projectId),
    timeSeries: [timeSeriesData],
  };

  await client.createTimeSeries(request);
}

functions.http('trackedFunction', async (req, res) => {
  const startTime = Date.now();

  try {
    // Your function logic
    await processRequest(req.body);

    const duration = Date.now() - startTime;
    await writeCustomMetric(duration);

    res.send('Success!');
  } catch (error) {
    console.error('Error:', error);
    res.status(500).send('Error');
  }
});

Performance Monitoring​

const functions = require('@google-cloud/functions-framework');
const { performance } = require('perf_hooks');

functions.http('performanceTracking', async (req, res) => {
  const marks = {};

  // Mark start
  marks.start = performance.now();

  // Database query
  marks.dbStart = performance.now();
  await queryDatabase();
  marks.dbEnd = performance.now();

  // External API call
  marks.apiStart = performance.now();
  await callExternalAPI();
  marks.apiEnd = performance.now();

  // Calculate durations
  const metrics = {
    totalDuration: performance.now() - marks.start,
    dbDuration: marks.dbEnd - marks.dbStart,

      severity: 'INFO',

    }),

});

Best Practices​

1. Function Design​

✅ Do:

  • Keep functions small and focused (single responsibility)
  • Use async/await for asynchronous operations
  • Implement proper error handling
  • Set appropriate timeouts
  • Use environment variables for configuration

❌ Don't:

  • Store state in global variables
  • Perform long-running operations
  • Make unnecessary cold starts
  • Ignore errors
  • Hard-code configuration values

2. Performance Optimization​

Minimize cold starts

// ✅ Good: Initialize outside handler
const { BigQuery } = require('@google-cloud/bigquery');
const bigquery = new BigQuery();

functions.http('optimized', async (req, res) => {
  // Use pre-initialized client
  const [rows] = await bigquery.query('SELECT 1');
  res.json(rows);
});

// ❌ Bad: Initialize inside handler
functions.http('slow', async (req, res) => {
  const { BigQuery } = require('@google-cloud/bigquery');
  const bigquery = new BigQuery(); // Creates new client every time
  const [rows] = await bigquery.query('SELECT 1');
  res.json(rows);
});

Connection pooling

const { Pool } = require('pg');

// Create pool outside handler
const pool = new Pool({
  host: process.env.DB_HOST,
  database: process.env.DB_NAME,
  user: process.env.DB_USER,
  password: process.env.DB_PASSWORD,
  max: 1, // Cloud Functions: use 1 connection per instance
});

functions.http('dbQuery', async (req, res) => {
  const client = await pool.connect();
  try {
    const result = await client.query('SELECT * FROM users LIMIT 10');
    res.json(result.rows);
  } finally {
    client.release();
  }
});

3. Security Best Practices​

Validate input

const Joi = require('joi');

const schema = Joi.object({
  email: Joi.string().email().required(),
  name: Joi.string().min(3).max(50).required(),
  age: Joi.number().integer().min(0).max(150),
});

functions.http('secureEndpoint', async (req, res) => {
  // Validate input
  const { error, value } = schema.validate(req.body);

  if (error) {
    return res.status(400).json({
      error: 'Validation failed',
      details: error.details,
    });
  }

  // Process validated data
  await processData(value);
  res.json({ success: true });
});

Rate limiting

const { RateLimiterMemory } = require('rate-limiter-flexible');

const rateLimiter = new RateLimiterMemory({
  points: 10, // 10 requests
  duration: 60, // per 60 seconds
});

functions.http('rateLimited', async (req, res) => {
  const key = req.ip || 'anonymous';

  try {
    await rateLimiter.consume(key);

    // Process request
    res.json({ message: 'Success' });
  } catch (error) {
    res.status(429).json({
      error: 'Too many requests',
      retryAfter: error.msBeforeNext / 1000,
    });
  }
});

  --memory=256Mi  # Don't over-provision

gcloud functions deploy my-function \
