Google Cloud Professional DevOps Engineer - Complete Learning Guide

Overview​

The Google Cloud Professional DevOps Engineer certification validates your ability to build and maintain software delivery pipelines, deploy and monitor services, and manage incidents using Google Cloud technologies.

Exam Details:

  • Duration: 2 hours
  • Format: Multiple choice and multiple select questions
  • Cost: $200 USD
  • Validity: 2 years
  • Languages: English, Japanese, Spanish, Portuguese, Indonesian
  • Delivery: Online or at a test center
  • Prerequisites: Recommended 3+ years of industry experience with 1+ year designing and managing solutions on GCP

  - name: 'node:18'
    entrypoint: npm

  # Run linting
  - name: 'node:18'
    entrypoint: npm
    args: ['run', 'lint']

  # Run unit tests
  - name: 'node:18'
    entrypoint: npm
    args: ['run', 'test:unit']
    env:
      - 'NODE_ENV=test'

  # Run security scan
  - name: 'aquasec/trivy'
    args:
      - 'fs'
      - '--severity'
      - 'HIGH,CRITICAL'
      - '--exit-code'
      - '1'
      - '.'

  # Build Docker image
  - name: 'gcr.io/cloud-builders/docker'
    args:
      - 'build'
      - '-t'
      - 'us-central1-docker.pkg.dev/${PROJECT_ID}/my-repo/my-app:${SHORT_SHA}'
      - '-t'
      - 'us-central1-docker.pkg.dev/${PROJECT_ID}/my-repo/my-app:latest'
      - '.'

  # Push to Artifact Registry
  - name: 'gcr.io/cloud-builders/docker'
    args:
      - 'push'
      - '--all-tags'
      - 'us-central1-docker.pkg.dev/${PROJECT_ID}/my-repo/my-app'

# Store build artifacts
artifacts:
  objects:
    location: 'gs://${PROJECT_ID}_cloudbuild/artifacts'
    paths: ['coverage/**', 'test-results/**']

  machineType: 'N1_HIGHCPU_8'
  logging: CLOUD_LOGGING_ONLY

# monitoring_setup.py - Create custom metrics and dashboards
from google.cloud import monitoring_v3
from google.api import metric_pb2 as ga_metric
from google.api import label_pb2 as ga_label

def create_custom_metric(project_id, metric_type):
    """Create a custom metric descriptor"""
    client = monitoring_v3.MetricServiceClient()
    project_name = f"projects/{project_id}"

    descriptor = ga_metric.MetricDescriptor()
    descriptor.type = f"custom.googleapis.com/{metric_type}"
    descriptor.metric_kind = ga_metric.MetricDescriptor.MetricKind.GAUGE
    descriptor.value_type = ga_metric.MetricDescriptor.ValueType.DOUBLE
    descriptor.description = "Custom application metric"

    # Add labels
    label = ga_label.LabelDescriptor()
    label.key = "environment"
    label.value_type = ga_label.LabelDescriptor.ValueType.STRING
    descriptor.labels.append(label)

    descriptor = client.create_metric_descriptor(
        name=project_name, metric_descriptor=descriptor
    )
    print(f"Created metric: {descriptor.name}")
    return descriptor

def write_time_series(project_id, metric_type, value):
    """Write a time series data point"""
    client = monitoring_v3.MetricServiceClient()
    project_name = f"projects/{project_id}"

    series = monitoring_v3.TimeSeries()
    series.metric.type = f"custom.googleapis.com/{metric_type}"
    series.metric.labels["environment"] = "production"

    series.resource.type = "gke_container"
    series.resource.labels["project_id"] = project_id
    series.resource.labels["location"] = "us-central1"
    series.resource.labels["cluster_name"] = "my-cluster"
    series.resource.labels["namespace_name"] = "default"
    series.resource.labels["pod_name"] = "my-pod"
    series.resource.labels["container_name"] = "my-container"

    now = time.time()
    seconds = int(now)
    nanos = int((now - seconds) * 10 ** 9)
    interval = monitoring_v3.TimeInterval(
        {"end_time": {"seconds": seconds, "nanos": nanos}}
    )
    point = monitoring_v3.Point({
        "interval": interval,
        "value": {"double_value": value}
    })
    series.points = [point]

    client.create_time_series(name=project_name, time_series=[series])
    print(f"Wrote time series data: {value}")

# Create alerting policy
def create_alert_policy(project_id):
    """Create an alerting policy for high error rate"""
    client = monitoring_v3.AlertPolicyServiceClient()
    project_name = f"projects/{project_id}"

    policy = monitoring_v3.AlertPolicy()
    policy.display_name = "High Error Rate Alert"
    policy.documentation.content = """
    ## High Error Rate Detected

    The error rate has exceeded the threshold of 5% over a 5-minute window.

    ### Immediate Actions:
    1. Check the application logs
    2. Review recent deployments
    3. Check external dependencies
    4. Escalate to on-call engineer if not resolved in 10 minutes
    """

    # Define condition
    condition = monitoring_v3.AlertPolicy.Condition()
    condition.display_name = "Error rate > 5%"
    condition.condition_threshold.filter = '''
        resource.type="k8s_pod"
        AND metric.type="logging.googleapis.com/user/error_count"
    '''
    condition.condition_threshold.comparison = (
        monitoring_v3.ComparisonType.COMPARISON_GT
    )
    condition.condition_threshold.threshold_value = 0.05
    condition.condition_threshold.duration.seconds = 300  # 5 minutes
    condition.condition_threshold.aggregations.append({
        "alignment_period": {"seconds": 60},
        "per_series_aligner": monitoring_v3.Aggregation.Aligner.ALIGN_RATE,
    })

    policy.conditions.append(condition)
    policy.combiner = monitoring_v3.AlertPolicy.ConditionCombinerType.AND

    # Notification channels (must be created separately)
    policy.notification_channels.extend([
        f"projects/{project_id}/notificationChannels/{channel_id}"
        for channel_id in ["channel-1", "channel-2"]
    ])

    policy = client.create_alert_policy(name=project_name, alert_policy=policy)
    print(f"Created alert policy: {policy.name}")
    return policy

# structured_logging.py - Implement structured logging
import logging
import json
from google.cloud import logging as cloud_logging
from google.cloud.logging.handlers import CloudLoggingHandler

# Initialize Cloud Logging
client = cloud_logging.Client()
handler = CloudLoggingHandler(client, name="my-app-log")

# Set up logging
logger = logging.getLogger("my-app")
logger.setLevel(logging.INFO)
logger.addHandler(handler)

class StructuredMessage:
    def __init__(self, message, **kwargs):
        self.message = message
        self.kwargs = kwargs

    def __str__(self):
        return json.dumps({
            'message': self.message,
            **self.kwargs
        })

# Usage in application
def process_request(request_id, user_id, action):
    logger.info(StructuredMessage(
        "Processing request",
        request_id=request_id,
        user_id=user_id,
        action=action,
        environment="production",
        version="1.2.3"
    ))

    try:
        # Process request
        result = perform_action(action)

        logger.info(StructuredMessage(
            "Request processed successfully",
            request_id=request_id,
            user_id=user_id,
            action=action,
            duration_ms=result.duration,
            status="success"
        ))
    except Exception as e:
        logger.error(StructuredMessage(
            "Request processing failed",
            request_id=request_id,
            user_id=user_id,
            action=action,
            error=str(e),
            error_type=type(e).__name__,
            status="error"
        ), exc_info=True)
        raise

Incident Response Runbook Template:

# Incident Response Runbook - High Error Rate

## Severity Classification

- **P0 (Critical)**: Service completely down, affecting all users
- **P1 (High)**: Major functionality impaired, affecting >50% of users
- **P2 (Medium)**: Partial functionality degraded, affecting <50% of users
- **P3 (Low)**: Minor issues, no immediate user impact

## Detection

- Alert: Error rate > 5% for 5 minutes
- SLO burn rate: Error budget consumed > 2x normal rate
- User reports: Multiple support tickets

## Initial Response (5 minutes)

1. Acknowledge the incident in PagerDuty/Opsgenie
2. Join incident Slack channel: #incident-YYYY-MM-DD-NNN
3. Assess severity and impact
4. Notify stakeholders if P0/P1

## Investigation (15 minutes)

1. **Check Recent Changes**

   ```bash
   # List recent deployments
   gcloud deploy releases list \
     --delivery-pipeline=my-app-pipeline \
     --region=us-central1 \
     --limit=5

   # Check recent configuration changes
   kubectl get events --sort-by='.lastTimestamp' | head -20
   ```

  1. Check Application Logs

    # Query error logs
    gcloud logging read "resource.type=k8s_pod
      AND resource.labels.namespace_name=production
      AND severity=ERROR" \
      --limit 100 \
      --format json \
      --freshness 30m

  2. Check Service Health

    # Check pod status
    kubectl get pods -n production

    # Check service endpoints
    kubectl get endpoints -n production

    # Check HPA status
    kubectl get hpa -n production

  3. Check Dependencies

    • Database connectivity
    • External API status
    • Network connectivity

Mitigation Actions​

If caused by recent deployment:​

# Rollback to previous version
gcloud deploy releases rollback release-XXX \
  --delivery-pipeline=my-app-pipeline \
  --region=us-central1 \
  --to-target=prod

If caused by traffic spike:​

# Scale up manually
kubectl scale deployment my-app \
  --replicas=20 \
  -n production

# Or adjust HPA
kubectl patch hpa my-app -n production \
  -p '{"spec":{"maxReplicas":30}}'

If caused by external dependency:​

  • Enable circuit breaker
  • Switch to backup service
  • Enable degraded mode

Communication​

  • Update status page: status.mycompany.com
  • Post updates in #incidents channel every 15 minutes
  • Email notification to affected customers if P0/P1

Resolution​

  1. Confirm metrics return to normal
  2. Verify error rate < 1% for 10 minutes
  3. Check SLO dashboard
  4. Close incident in PagerDuty
  5. Update status page

Post-Incident​

  1. Schedule postmortem within 48 hours
  2. Document timeline and actions taken
  3. Identify root cause
  4. Create action items to prevent recurrence
  5. Update runbook based on learnings

from google.cloud import bigquery
import pandas as pd

def analyze_gke_costs(project_id, dataset_id, days=30):
    """Analyze GKE costs from billing export"""
    client = bigquery.Client(project=project_id)

    query = f"""
    SELECT
      service.description AS service,
      sku.description AS sku,
      SUM(cost) AS total_cost,
      SUM(usage.amount) AS total_usage,
      usage.unit AS unit
    FROM
      `{project_id}.{dataset_id}.gcp_billing_export_v1_*`
    WHERE
      _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL {days} DAY))
      AND service.description = 'Kubernetes Engine'
    GROUP BY
      service, sku, unit
    ORDER BY
      total_cost DESC
    LIMIT 20
    """

    df = client.query(query).to_dataframe()
    return df

Exam Tips​

  1. Time Management

    • 2 hours for 50-60 questions
    • ~2 minutes per question
    • Flag difficult questions and return later
    • Don't spend too much time on any single question

  2. Question Strategies

    • Read carefully - look for keywords like "least", "most", "not"
    • Eliminate obviously wrong answers first
    • Consider the most cost-effective and scalable solution
    • Think about Google Cloud-native solutions first
    • Consider security and compliance requirements

  3. Common Traps

    • Overthinking questions - often the obvious answer is correct
    • Ignoring cost considerations
    • Choosing complex solutions when simple ones exist
    • Not considering multi-region or disaster recovery requirements

  4. Day Before Exam

    • Review key concepts and services
    • Review sample questions
    • Get good sleep
    • Prepare your workspace (if remote exam)
