Elasticsearch Stack Implementation Guide

Overview​

The Elasticsearch Stack (ELK Stack) combines Elasticsearch, Logstash, and Kibana to provide powerful search, analytics, and visualization capabilities. This guide demonstrates production-ready implementations for e-commerce platforms including product search, log aggregation, analytics, and monitoring.

Key Components​

  • Elasticsearch: Distributed search and analytics engine
  • Logstash: Data processing pipeline for ingesting and transforming data
  • Kibana: Visualization and exploration platform
  • Beats: Lightweight data shippers (Filebeat, Metricbeat, APM)

Prerequisites​

# Install Elasticsearch client
npm install @elastic/elasticsearch

# Install additional dependencies
npm install @nestjs/elasticsearch @nestjs/config

# For development
npm install @types/node --save-dev

import { Injectable, Logger, OnModuleInit } from '@nestjs/common';
import { ElasticsearchService } from '@nestjs/elasticsearch';

export interface ProductDocument {
  id: string;
  name: string;
  description: string;
  category: string;
  brand: string;
  price: number;
  salePrice?: number;
  stock: number;
  rating: number;
  reviewCount: number;
  tags: string[];
  attributes: Record<string, any>;
  images: string[];
  createdAt: Date;
  updatedAt: Date;
}

@Injectable()
export class SearchIndexService implements OnModuleInit {
  private readonly logger = new Logger(SearchIndexService.name);
  private readonly PRODUCT_INDEX = 'products';

  constructor(private readonly elasticsearchService: ElasticsearchService) {}

  async onModuleInit() {
    await this.createProductIndex();
  }

  async createProductIndex() {
    const indexExists = await this.elasticsearchService.indices.exists({
      index: this.PRODUCT_INDEX,
    });

    if (indexExists) {
      this.logger.log(`Index ${this.PRODUCT_INDEX} already exists`);
      return;
    }

    try {
      await this.elasticsearchService.indices.create({
        index: this.PRODUCT_INDEX,
        body: {
          settings: {
            number_of_shards: 3,
            number_of_replicas: 2,
            analysis: {
              analyzer: {
                // Custom analyzer for product names
                product_name_analyzer: {
                  type: 'custom',
                  tokenizer: 'standard',
                  filter: ['lowercase', 'asciifolding', 'product_synonym', 'edge_ngram_filter'],
                },
                // Autocomplete analyzer
                autocomplete_analyzer: {
                  type: 'custom',
                  tokenizer: 'standard',
                  filter: ['lowercase', 'autocomplete_filter'],
                },
                autocomplete_search_analyzer: {
                  type: 'custom',
                  tokenizer: 'standard',
                  filter: ['lowercase'],
                },
              },
              filter: {
                edge_ngram_filter: {
                  type: 'edge_ngram',
                  min_gram: 2,
                  max_gram: 20,
                },
                autocomplete_filter: {
                  type: 'edge_ngram',
                  min_gram: 1,
                  max_gram: 20,
                },
                product_synonym: {
                  type: 'synonym',
                  synonyms: ['phone, smartphone, mobile', 'laptop, notebook, computer', 'tv, television'],
                },
              },
            },
          },
          mappings: {
            properties: {
              id: { type: 'keyword' },
              name: {
                type: 'text',
                analyzer: 'product_name_analyzer',
                fields: {
                  keyword: { type: 'keyword' },
                  autocomplete: {
                    type: 'text',
                    analyzer: 'autocomplete_analyzer',
                    search_analyzer: 'autocomplete_search_analyzer',
                  },
                },
              },
              description: {
                type: 'text',
                analyzer: 'standard',
              },
              category: {
                type: 'keyword',
                fields: {
                  text: { type: 'text' },
                },
              },
              brand: {
                type: 'keyword',
                fields: {
                  text: { type: 'text' },
                },
              },
              price: { type: 'float' },
              salePrice: { type: 'float' },
              stock: { type: 'integer' },
              rating: { type: 'float' },
              reviewCount: { type: 'integer' },
              tags: { type: 'keyword' },
              attributes: { type: 'object', enabled: true },
              images: { type: 'keyword' },
              createdAt: { type: 'date' },
              updatedAt: { type: 'date' },
            },
          },
        },
      });

      this.logger.log(`Created index: ${this.PRODUCT_INDEX}`);
    } catch (error) {
      this.logger.error(`Failed to create index: ${error.message}`);
      throw error;
    }
  }

  async indexProduct(product: ProductDocument): Promise<void> {
    try {
      await this.elasticsearchService.index({
        index: this.PRODUCT_INDEX,
        id: product.id,
        document: product,
      });

      this.logger.log(`Indexed product: ${product.id}`);
    } catch (error) {
      this.logger.error(`Failed to index product ${product.id}: ${error.message}`);
      throw error;
    }
  }

  async bulkIndexProducts(products: ProductDocument[]): Promise<void> {
    if (products.length === 0) {
      return;
    }

    try {
      const operations = products.flatMap((product) => [
        { index: { _index: this.PRODUCT_INDEX, _id: product.id } },
        product,
      ]);

      const bulkResponse = await this.elasticsearchService.bulk({
        refresh: true,
        operations,
      });

      if (bulkResponse.errors) {
        const erroredDocuments = [];
        bulkResponse.items.forEach((action, i) => {
          const operation = Object.keys(action)[0];
          if (action[operation].error) {
            erroredDocuments.push({
              status: action[operation].status,
              error: action[operation].error,
              document: products[i],
            });
          }
        });

        this.logger.error(`Bulk indexing errors: ${JSON.stringify(erroredDocuments)}`);
      }

      this.logger.log(`Bulk indexed ${products.length} products`);
    } catch (error) {
      this.logger.error(`Bulk indexing failed: ${error.message}`);
      throw error;
    }
  }

  async deleteProduct(productId: string): Promise<void> {
    try {
      await this.elasticsearchService.delete({
        index: this.PRODUCT_INDEX,
        id: productId,
      });

      this.logger.log(`Deleted product: ${productId}`);
    } catch (error) {
      if (error.meta?.statusCode === 404) {
        this.logger.warn(`Product not found: ${productId}`);
        return;
      }
      this.logger.error(`Failed to delete product ${productId}: ${error.message}`);
      throw error;
    }
  }

  async updateProduct(productId: string, updates: Partial<ProductDocument>): Promise<void> {
    try {
      await this.elasticsearchService.update({
        index: this.PRODUCT_INDEX,
        id: productId,
        doc: updates,
      });

      this.logger.log(`Updated product: ${productId}`);
    } catch (error) {
      this.logger.error(`Failed to update product ${productId}: ${error.message}`);
      throw error;
    }
  }

  async reindexAll(products: ProductDocument[]): Promise<void> {
    this.logger.log('Starting full reindex...');

    try {
      // Delete existing index
      const indexExists = await this.elasticsearchService.indices.exists({
        index: this.PRODUCT_INDEX,
      });

      if (indexExists) {
        await this.elasticsearchService.indices.delete({
          index: this.PRODUCT_INDEX,
        });
      }

      // Create new index
      await this.createProductIndex();

      // Bulk index all products
      await this.bulkIndexProducts(products);

      this.logger.log('Full reindex completed');
    } catch (error) {
      this.logger.error(`Full reindex failed: ${error.message}`);
      throw error;
    }
  }
}
// product-search.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { ElasticsearchService } from '@nestjs/elasticsearch';

export interface SearchQuery {
  query?: string;
  category?: string;
  brand?: string;
  minPrice?: number;
  maxPrice?: number;
  minRating?: number;
  tags?: string[];
  inStock?: boolean;
  sortBy?: 'relevance' | 'price_asc' | 'price_desc' | 'rating' | 'newest';
  page?: number;
  limit?: number;
}

export interface SearchResult {
  products: any[];
  total: number;
  page: number;
  limit: number;
  aggregations?: {
    categories: Array<{ key: string; count: number }>;
    brands: Array<{ key: string; count: number }>;
    priceRanges: Array<{ key: string; count: number; from?: number; to?: number }>;
    tags: Array<{ key: string; count: number }>;
  };
}

@Injectable()
export class ProductSearchService {
  private readonly logger = new Logger(ProductSearchService.name);
  private readonly PRODUCT_INDEX = 'products';

  constructor(private readonly elasticsearchService: ElasticsearchService) {}

  async search(searchQuery: SearchQuery): Promise<SearchResult> {
    const {
      query,
      category,
      brand,
      minPrice,
      maxPrice,
      minRating,
      tags,
      inStock,
      sortBy = 'relevance',
      page = 1,
      limit = 20,
    } = searchQuery;

    const from = (page - 1) * limit;

    try {
      // Build query
      const must: any[] = [];
      const filter: any[] = [];

      // Text search
      if (query) {
        must.push({
          multi_match: {
            query,
            fields: ['name^3', 'description', 'brand^2', 'category^2', 'tags'],
            type: 'best_fields',
            fuzziness: 'AUTO',
          },
        });
      } else {
        must.push({ match_all: {} });
      }

      // Category filter
      if (category) {
        filter.push({ term: { category } });
      }

      // Brand filter
      if (brand) {
        filter.push({ term: { brand } });
      }

      // Price range filter
      if (minPrice !== undefined || maxPrice !== undefined) {
        const priceRange: any = {};
        if (minPrice !== undefined) priceRange.gte = minPrice;
        if (maxPrice !== undefined) priceRange.lte = maxPrice;
        filter.push({ range: { price: priceRange } });
      }

      // Rating filter
      if (minRating !== undefined) {
        filter.push({ range: { rating: { gte: minRating } } });
      }

      // Tags filter
      if (tags && tags.length > 0) {
        filter.push({ terms: { tags } });
      }

      // Stock filter
      if (inStock) {
        filter.push({ range: { stock: { gt: 0 } } });
      }

      // Build sort
      const sort: any[] = [];
      switch (sortBy) {
        case 'price_asc':
          sort.push({ price: 'asc' });
          break;
        case 'price_desc':
          sort.push({ price: 'desc' });
          break;
        case 'rating':
          sort.push({ rating: 'desc' }, { reviewCount: 'desc' });
          break;
        case 'newest':
          sort.push({ createdAt: 'desc' });
          break;
        default:
          sort.push('_score');
      }

      const response = await this.elasticsearchService.search({
        index: this.PRODUCT_INDEX,
        body: {
          query: {
            bool: {
              must,
              filter,
            },
          },
          sort,
          from,
          size: limit,
          aggs: {
            categories: {
              terms: { field: 'category', size: 20 },
            },
            brands: {
              terms: { field: 'brand', size: 20 },
            },
            priceRanges: {
              range: {
                field: 'price',
                ranges: [
                  { key: 'Under $25', to: 25 },
                  { key: '$25 - $50', from: 25, to: 50 },
                  { key: '$50 - $100', from: 50, to: 100 },
                  { key: '$100 - $200', from: 100, to: 200 },
                  { key: 'Over $200', from: 200 },
                ],
              },
            },
            tags: {
              terms: { field: 'tags', size: 30 },
            },
          },
        },
      });

      const hits = response.hits.hits;
      const total = typeof response.hits.total === 'number' ? response.hits.total : response.hits.total.value;

      const products = hits.map((hit) => ({
        id: hit._id,
        score: hit._score,
        ...hit._source,
      }));

      const aggregations = {
        categories: response.aggregations.categories.buckets.map((b: any) => ({
          key: b.key,
          count: b.doc_count,
        })),
        brands: response.aggregations.brands.buckets.map((b: any) => ({
          key: b.key,
          count: b.doc_count,
        })),
        priceRanges: response.aggregations.priceRanges.buckets.map((b: any) => ({
          key: b.key,
          count: b.doc_count,
          from: b.from,
          to: b.to,
        })),
        tags: response.aggregations.tags.buckets.map((b: any) => ({
          key: b.key,
          count: b.doc_count,
        })),
      };

      return {
        products,
        total,
        page,
        limit,
        aggregations,
      };
    } catch (error) {
      this.logger.error(`Search failed: ${error.message}`);
      throw error;
    }
  }

  async autocomplete(query: string, limit = 10): Promise<string[]> {
    if (!query || query.length < 2) {
      return [];
    }

    try {
      const response = await this.elasticsearchService.search({
        index: this.PRODUCT_INDEX,
        body: {
          query: {
            match: {
              'name.autocomplete': query,
            },
          },
          _source: ['name'],
          size: limit,
        },
      });

      return response.hits.hits.map((hit: any) => hit._source.name);
    } catch (error) {
      this.logger.error(`Autocomplete failed: ${error.message}`);
      return [];
    }
  }

  async getSimilarProducts(productId: string, limit = 6): Promise<any[]> {
    try {
      // Get the product
      const product = await this.elasticsearchService.get({
        index: this.PRODUCT_INDEX,
        id: productId,
      });

      const productData: any = product._source;

      // Find similar products
      const response = await this.elasticsearchService.search({
        index: this.PRODUCT_INDEX,
        body: {
          query: {
            bool: {
              must: [
                {
                  more_like_this: {
                    fields: ['name', 'description', 'tags'],
                    like: [
                      {
                        _index: this.PRODUCT_INDEX,
                        _id: productId,
                      },
                    ],
                    min_term_freq: 1,
                    max_query_terms: 12,
                  },
                },
              ],
              filter: [{ term: { category: productData.category } }],
              must_not: [{ term: { id: productId } }],
            },
          },
          size: limit,
        },
      });

      return response.hits.hits.map((hit: any) => ({
        id: hit._id,
        score: hit._score,
        ...hit._source,
      }));
    } catch (error) {
      this.logger.error(`Similar products search failed: ${error.message}`);
      return [];
    }
  }
}

// product-search.controller.ts
import { Controller, Get, Query, Param } from '@nestjs/common';
import { ProductSearchService, SearchQuery } from './product-search.service';

@Controller('api/search')
export class ProductSearchController {
  constructor(private readonly searchService: ProductSearchService) {}

  @Get('products')
  async searchProducts(
    @Query('q') query?: string,
    @Query('category') category?: string,
    @Query('brand') brand?: string,
    @Query('minPrice') minPrice?: string,
    @Query('maxPrice') maxPrice?: string,
    @Query('minRating') minRating?: string,
    @Query('tags') tags?: string,
    @Query('inStock') inStock?: string,
    @Query('sortBy') sortBy?: string,
    @Query('page') page?: string,
    @Query('limit') limit?: string,
  ) {
    const searchQuery: SearchQuery = {
      query,
      category,
      brand,
      minPrice: minPrice ? parseFloat(minPrice) : undefined,
      maxPrice: maxPrice ? parseFloat(maxPrice) : undefined,
      minRating: minRating ? parseFloat(minRating) : undefined,
      tags: tags ? tags.split(',') : undefined,
      inStock: inStock === 'true',
      sortBy: sortBy as any,
      page: page ? parseInt(page, 10) : 1,
      limit: limit ? parseInt(limit, 10) : 20,
    };

    return this.searchService.search(searchQuery);
  }

  @Get('autocomplete')
  async autocomplete(@Query('q') query: string) {
    return this.searchService.autocomplete(query);
  }

  @Get('products/:id/similar')
  async getSimilarProducts(@Param('id') productId: string) {
    return this.searchService.getSimilarProducts(productId);
  }
}

// winston-elasticsearch.config.ts
import { WinstonModule } from 'nest-winston';
import * as winston from 'winston';
import { ElasticsearchTransport } from 'winston-elasticsearch';

export const createLogger = () => {
  const esTransportOpts = {
    level: 'info',
    clientOpts: {
      node: process.env.ELASTICSEARCH_NODE || 'http://localhost:9200',
      auth: {
        username: process.env.ELASTICSEARCH_USERNAME || 'elastic',
        password: process.env.ELASTICSEARCH_PASSWORD || 'changeme',
      },
    },
    index: 'logs-ecommerce',
    transformer: (logData: any) => {
      return {
        '@timestamp': new Date().toISOString(),
        message: logData.message,
        level: logData.level,
        service: process.env.SERVICE_NAME || 'unknown',
        environment: process.env.NODE_ENV || 'development',
        ...logData.meta,
      };
    },
  };

  return WinstonModule.createLogger({
    transports: [
      // Console transport for development
      new winston.transports.Console({
        format: winston.format.combine(
          winston.format.timestamp(),
          winston.format.colorize(),
          winston.format.printf(({ timestamp, level, message, ...meta }) => {
            return `${timestamp} [${level}]: ${message} ${
              Object.keys(meta).length ? JSON.stringify(meta, null, 2) : ''
            }`;
          }),
        ),
      }),
      // Elasticsearch transport for centralized logging
      new ElasticsearchTransport(esTransportOpts),
    ],
  });
};

// logging.interceptor.ts
import { Injectable, NestInterceptor, ExecutionContext, CallHandler, Logger } from '@nestjs/common';
import { Observable } from 'rxjs';
import { tap, catchError } from 'rxjs/operators';

@Injectable()
export class LoggingInterceptor implements NestInterceptor {
  private readonly logger = new Logger(LoggingInterceptor.name);

  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const request = context.switchToHttp().getRequest();
    const { method, url, body, headers, ip } = request;
    const userAgent = headers['user-agent'] || '';
    const startTime = Date.now();

    // Log request
    this.logger.log({
      message: 'Incoming request',
      req: {
        method,
        url,
        body,
        userAgent,
        clientIp: ip,
      },
    });

    return next.handle().pipe(
      tap((data) => {
        const response = context.switchToHttp().getResponse();
        const { statusCode } = response;
        const duration = Date.now() - startTime;

        // Log successful response
        this.logger.log({
          message: 'Request completed',
          req: { method, url },
          res: { statusCode, duration },
        });
      }),
      catchError((error) => {
        const duration = Date.now() - startTime;

        // Log error
        this.logger.error({
          message: 'Request failed',
          req: { method, url, body },
          error: {
            message: error.message,
            stack: error.stack,
            statusCode: error.status,
          },
          duration,
        });

        throw error;
      }),
    );
  }
}

  constructor(private readonly elasticsearchService: ElasticsearchService) {}

  async trackEvent(event: AnalyticsEvent): Promise<void> {
    try {
      await this.elasticsearchService.index({
        index: this.ANALYTICS_INDEX,
        document: {
          ...event,
          '@timestamp': event.timestamp.toISOString(),
        },
      });

      this.logger.debug(`Tracked event: ${event.eventType}`);
    } catch (error) {
      this.logger.error(`Failed to track event: ${error.message}`);
    }
  }

  async getEventCounts(query: AnalyticsQuery): Promise<any> {
    const { startDate, endDate, eventType, interval = 'day' } = query;

    try {
      const must: any[] = [
        {
          range: {
            '@timestamp': {
              gte: startDate.toISOString(),
              lte: endDate.toISOString(),
            },
          },
        },
      ];

      if (eventType) {
        must.push({ term: { eventType } });
      }

      const response = await this.elasticsearchService.search({
        index: this.ANALYTICS_INDEX,
        body: {
          size: 0,
          query: { bool: { must } },
          aggs: {
            events_over_time: {
              date_histogram: {
                field: '@timestamp',
                calendar_interval: interval,
                min_doc_count: 0,
              },
              aggs: {
                by_type: {
                  terms: { field: 'eventType' },
                },
              },
            },
            total_by_type: {
              terms: { field: 'eventType' },
            },
          },
        },
      });

      return {
        timeline: response.aggregations.events_over_time.buckets.map((bucket: any) => ({
          timestamp: bucket.key_as_string,
          total: bucket.doc_count,
          byType: bucket.by_type.buckets.reduce((acc: any, b: any) => {
            acc[b.key] = b.doc_count;
            return acc;
          }, {}),
        })),
        totals: response.aggregations.total_by_type.buckets.reduce((acc: any, b: any) => {
          acc[b.key] = b.doc_count;
          return acc;
        }, {}),
      };
    } catch (error) {
      this.logger.error(`Failed to get event counts: ${error.message}`);
      throw error;
    }
  }

// apm.interceptor.ts
import { Injectable, NestInterceptor, ExecutionContext, CallHandler } from '@nestjs/common';
import { Observable } from 'rxjs';
import { tap, catchError } from 'rxjs/operators';
import apm from 'elastic-apm-node';

@Injectable()
export class ApmInterceptor implements NestInterceptor {
  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const request = context.switchToHttp().getRequest();
    const { method, url } = request;

    // Start a custom transaction
    const transaction = apm.startTransaction(`${method} ${url}`, 'request');

    // Add custom context
    if (transaction) {
      transaction.setCustomContext({
        userId: request.user?.id,
        sessionId: request.sessionID,
      });
    }

    return next.handle().pipe(
      tap(() => {
        // Mark transaction as successful
        if (transaction) {
          transaction.result = 'success';
          transaction.end();
        }
      }),
      catchError((error) => {
        // Capture error in APM
        if (transaction) {
          apm.captureError(error, {
            request,
            custom: {
              transactionId: transaction.id,
            },
          });
          transaction.result = 'error';
          transaction.end();
        }
        throw error;
      }),
    );
  }
}

1. Index Lifecycle Management​

// index-lifecycle.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { ElasticsearchService } from '@nestjs/elasticsearch';
import { Cron } from '@nestjs/schedule';

@Injectable()
export class IndexLifecycleService {
  private readonly logger = new Logger(IndexLifecycleService.name);

  constructor(private readonly elasticsearchService: ElasticsearchService) {}

  async setupILMPolicy() {
    const policyName = 'logs-lifecycle-policy';

    try {
      await this.elasticsearchService.ilm.putLifecycle({
        name: policyName,
        body: {
          policy: {
            phases: {
              hot: {
                actions: {
                  rollover: {
                    max_size: '50GB',
                    max_age: '7d',
                  },
                },
              },
              warm: {
                min_age: '7d',
                actions: {
                  shrink: {
                    number_of_shards: 1,
                  },
                  forcemerge: {
                    max_num_segments: 1,
                  },
                },
              },
              cold: {
                min_age: '30d',
                actions: {
                  freeze: {},
                },
              },
              delete: {
                min_age: '90d',
                actions: {
                  delete: {},
                },
              },
            },
          },
        },
      });

      this.logger.log(`ILM policy created: ${policyName}`);
    } catch (error) {
      this.logger.error(`Failed to create ILM policy: ${error.message}`);
      throw error;
    }
  }

  @Cron('0 0 * * *') // Run daily at midnight
  async cleanupOldIndices() {
    const cutoffDate = new Date();
    cutoffDate.setDate(cutoffDate.getDate() - 90); // Keep last 90 days

    try {
      const indices = await this.elasticsearchService.cat.indices({
        format: 'json',
        h: ['index', 'creation.date'],
      });

      for (const index of indices as any[]) {
        const creationDate = new Date(parseInt(index['creation.date'], 10));

        if (creationDate < cutoffDate && index.index.startsWith('logs-')) {
          await this.elasticsearchService.indices.delete({
            index: index.index,
          });
          this.logger.log(`Deleted old index: ${index.index}`);
        }
      }
    } catch (error) {
      this.logger.error(`Failed to cleanup old indices: ${error.message}`);
    }
  }
}

// optimized-search.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { ElasticsearchService } from '@nestjs/elasticsearch';

@Injectable()
export class OptimizedSearchService {
  private readonly logger = new Logger(OptimizedSearchService.name);

  constructor(private readonly elasticsearchService: ElasticsearchService) {}

  async searchWithCache(query: any, cacheKey: string) {
    try {
      const response = await this.elasticsearchService.search({
        index: 'products',
        body: query,
        // Use request cache for frequently accessed queries
        request_cache: true,
        // Limit the number of fields returned
        _source: ['id', 'name', 'price', 'image'],
        // Use search_after for pagination instead of from/size
        // This is more efficient for deep pagination
        track_total_hits: 10000, // Limit total hits tracking
      });

      return response;
    } catch (error) {
      this.logger.error(`Optimized search failed: ${error.message}`);
      throw error;
    }
  }

  async bulkSearch(queries: any[]) {
    // Use msearch for multiple searches in one request
    const searches = queries.flatMap((query) => [{ index: 'products' }, query]);

    try {
      const response = await this.elasticsearchService.msearch({
        body: searches,
      });

      return response.responses;
    } catch (error) {
      this.logger.error(`Bulk search failed: ${error.message}`);
      throw error;
    }
  }

// elasticsearch-health.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { ElasticsearchService } from '@nestjs/elasticsearch';
import { Cron } from '@nestjs/schedule';

@Injectable()
export class ElasticsearchHealthService {
  private readonly logger = new Logger(ElasticsearchHealthService.name);

  constructor(private readonly elasticsearchService: ElasticsearchService) {}

  @Cron('*/5 * * * *') // Every 5 minutes
  async checkClusterHealth() {
    try {
      const health = await this.elasticsearchService.cluster.health();

      if (health.status === 'red') {
        this.logger.error('Elasticsearch cluster is RED!');
        // Send alert
      } else if (health.status === 'yellow') {
        this.logger.warn('Elasticsearch cluster is YELLOW');
      } else {
        this.logger.debug('Elasticsearch cluster is GREEN');
      }

      return {
        status: health.status,
        numberOfNodes: health.number_of_nodes,
        numberOfDataNodes: health.number_of_data_nodes,
        activePrimaryShards: health.active_primary_shards,
        activeShards: health.active_shards,
        relocatingShards: health.relocating_shards,
        initializingShards: health.initializing_shards,
        unassignedShards: health.unassigned_shards,
      };
    } catch (error) {
      this.logger.error(`Health check failed: ${error.message}`);
      throw error;
    }
  }

  async getNodeStats() {
    try {
      const stats = await this.elasticsearchService.nodes.stats();
      return stats;
    } catch (error) {
      this.logger.error(`Failed to get node stats: ${error.message}`);

    }
  }
