Recommendation Service

Collaborative filtering + RAG-based personalized recommendations: Elasticsearch behavioral index, Vertex AI vector store for embeddings, and real-time re-ranking.

Responsibilities​

  • Serve personalized product recommendations on the homepage, PDP, and cart page
  • Combine collaborative filtering (users who bought X also bought Y) with RAG-based semantic similarity (embedding-based product retrieval)
  • Consume user-events and order-events from Kafka to keep the behavioral index current
  • Cache recommendation results in Redis (TTL configurable per placement)

Architecture​

Recommendation Strategies​

StrategyWhen UsedLatency
Collaborative filteringUser has purchase/click history (cold-start after ≥3 interactions)<50 ms (Elasticsearch)
Content-based (RAG)New user, or browsing a specific product<100 ms (vector search)
TrendingFallback — no history, no product context<10 ms (Redis sorted set)
"Frequently bought together"Cart page, PDP upsell<50 ms (Elasticsearch co-occurrence)

NestJS Recommendation Service​

// src/recommendation/recommendation.service.ts
import { Injectable } from '@nestjs/common';
import { ElasticsearchService } from '@nestjs/elasticsearch';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';
import { VertexAIClient } from './vertexai.client';

@Injectable()
export class RecommendationService {
  constructor(
    private readonly es: ElasticsearchService,
    @InjectRedis() private readonly redis: Redis,
    private readonly vertexai: VertexAIClient,
  ) {}

  /**
   * Main entry point — selects strategy based on available signals.
   */
  async getRecommendations(
    userId: string | null,
    context: RecommendationContext,
    limit = 12,
  ): Promise<ProductRec[]> {
    const cacheKey = `recs:${userId ?? 'anon'}:${context.placement}:${context.skuContext ?? ''}`;
    const cached = await this.redis.get(cacheKey);
    if (cached) return JSON.parse(cached) as ProductRec[];

    let results: ProductRec[];

    if (context.skuContext) {
      // Product page: similar items (RAG + co-occurrence blend)
      results = await this.similarItems(context.skuContext, userId, limit);
    } else if (userId) {
      // Personalized: collaborative filtering
      results = await this.personalizedForUser(userId, limit);
    } else {
      // Fallback: trending
      results = await this.getTrending(limit);
    }

    await this.redis.set(cacheKey, JSON.stringify(results), 'EX', 300);
    return results;
  }

  /** Collaborative filtering via Elasticsearch more-like-this on purchase history */
  private async personalizedForUser(userId: string, limit: number): Promise<ProductRec[]> {
    const result = await this.es.search({
      index: 'product-recommendations',
      body: {
        query: {
          more_like_this: {
            fields:        ['sku', 'category', 'tags'],
            like: [
              {
                _index: 'user-purchase-history',
                _id:    userId,
              },
            ],
            min_term_freq:   1,
            min_doc_freq:    2,
            max_query_terms: 25,
          },
        },
        size: limit * 2,    // fetch extra for re-ranking
        _source: ['sku', 'title', 'price', 'imageUrl', 'score'],
      },
    });

    const candidates = (result.hits.hits as any[]).map(h => ({
      sku:       h._source.sku,
      title:     h._source.title,
      price:     h._source.price,
      imageUrl:  h._source.imageUrl,
      score:     h._score,
      strategy:  'collaborative',
    }));

    // Re-rank top candidates
    return this.rerank(candidates, userId, limit);
  }

  /** Semantic similarity via Vertex AI embeddings */
  private async similarItems(
    sku: string,
    userId: string | null,
    limit: number,
  ): Promise<ProductRec[]> {
    // 1. Get product embedding
    const vector = await this.vertexai.getProductEmbedding(sku);

    // 2. Vector search for similar products
    const similar = await this.vertexai.findSimilar(vector, limit * 2);

    // 3. Blend with Elasticsearch co-occurrence (frequently bought together)
    const coOccurrence = await this.getCoOccurrence(sku, limit);

    // 4. Merge + dedupe
    const merged = this.mergeAndDedupe(similar, coOccurrence, sku);
    return this.rerank(merged, userId, limit);
  }

  private async getCoOccurrence(sku: string, limit: number): Promise<ProductRec[]> {
    const result = await this.es.search({
      index: 'order-cooccurrence',
      body: {
        query: {
          term: { base_sku: sku },
        },
        sort: [{ co_occurrence_count: 'desc' }],
        size: limit,
        _source: ['related_sku', 'co_occurrence_count'],
      },
    });
    return (result.hits.hits as any[]).map(h => ({
      sku:      h._source.related_sku,
      score:    h._source.co_occurrence_count,
      strategy: 'co_occurrence',
    }));
  }

  private async rerank(
    candidates: ProductRec[],
    userId: string | null,
    limit: number,
  ): Promise<ProductRec[]> {
    if (candidates.length <= limit) return candidates;

    // Use Vertex AI Ranking API for personalized re-ranking
    const ranked = await this.vertexai.rerank(
      candidates.map(c => ({ id: c.sku, content: JSON.stringify(c) })),
      userId ?? 'anonymous',
    );

    return ranked.slice(0, limit).map(r => candidates.find(c => c.sku === r.id)!);
  }

  /** Trending: Redis sorted set updated by analytics pipeline */
  private async getTrending(limit: number): Promise<ProductRec[]> {
    const skus = await this.redis.zrevrange('trending:products:24h', 0, limit - 1);
    return skus.map(sku => ({ sku, strategy: 'trending' }));
  }

  private mergeAndDedupe(a: ProductRec[], b: ProductRec[], excludeSku: string): ProductRec[] {
    const seen = new Set<string>([excludeSku]);
    const result: ProductRec[] = [];
    for (const item of [...a, ...b]) {
      if (!seen.has(item.sku)) {
        seen.add(item.sku);
        result.push(item);
      }
    }
    return result;
  }
}

export interface ProductRec {
  sku:       string;
  title?:    string;
  price?:    number;
  imageUrl?: string;
  score?:    number;
  strategy:  string;
}

interface RecommendationContext {
  placement:   'homepage' | 'pdp' | 'cart' | 'search';
  skuContext?: string;  // current product SKU for PDP/cart
}

Kafka Consumer — Behavioral Index Update​

// src/recommendation/behavior.consumer.ts
import { Controller } from '@nestjs/common';
import { EventPattern, Payload } from '@nestjs/microservices';
import { ElasticsearchService } from '@nestjs/elasticsearch';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';

@Controller()
export class BehaviorConsumer {
  constructor(
    private readonly es: ElasticsearchService,
    @InjectRedis() private readonly redis: Redis,
  ) {}

  @EventPattern('analytics-events')
  async onAnalyticsEvent(@Payload() event: any): Promise<void> {
    if (event.eventType === 'product_click' && event.userId) {
      await this.es.update({
        index: 'user-purchase-history',
        id:    event.userId,
        body: {
          script: {
            source: `
              if (ctx._source.viewed == null) ctx._source.viewed = [];
              ctx._source.viewed.add(params.sku);
              if (ctx._source.viewed.length > 100) ctx._source.viewed.remove(0);
            `,
            params: { sku: event.properties.sku },
          },
          upsert: { viewed: [event.properties.sku], purchased: [] },
        },
      });
    }

    if (event.eventType === 'purchase_completed') {
      // Update trending sorted set (24-hour window)
      for (const item of event.properties.items ?? []) {
        await this.redis.zincrby('trending:products:24h', 1, item.sku);
        await this.redis.expire('trending:products:24h', 86400);
      }
    }
  }

  @EventPattern('order-events')
  async onOrderEvent(@Payload() event: any): Promise<void> {
    if (event.type !== 'order.confirmed') return;

    const skus: string[] = event.items?.map((i: any) => i.sku) ?? [];
    if (skus.length < 2) return;

    // Update co-occurrence index: all pairs in this order
    const pairs: Array<[string, string]> = [];
    for (let i = 0; i < skus.length; i++) {
      for (let j = i + 1; j < skus.length; j++) {
        pairs.push([skus[i], skus[j]]);
        pairs.push([skus[j], skus[i]]);
      }
    }

    for (const [base, related] of pairs) {
      await this.es.update({
        index: 'order-cooccurrence',
        id:    `${base}:${related}`,
        body: {
          script: {
            source: 'ctx._source.co_occurrence_count += 1',
          },
          upsert: { base_sku: base, related_sku: related, co_occurrence_count: 1 },
        },
      });
    }
  }
}
