Inventory Service

Stock levels, reservations, warehouse management: PostgreSQL as source of truth, Redis for low-latency reads, Kafka events for cross-service stock updates.

Responsibilities​

  • Maintain authoritative stock levels per SKU per warehouse
  • Handle soft reservations (stock held during checkout, released on payment failure)
  • Emit inventory-events for downstream consumers (search sync, analytics, notification)
  • Provide stock-level reads with <30 ms p99 via Redis cache
  • Support bulk stock imports from warehouse management systems (WMS)

Data Model​

CREATE TABLE inventory (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  sku           TEXT NOT NULL,
  warehouse_id  UUID NOT NULL REFERENCES warehouses(id),
  available_qty INT NOT NULL DEFAULT 0 CHECK (available_qty >= 0),
  reserved_qty  INT NOT NULL DEFAULT 0 CHECK (reserved_qty >= 0),
  reorder_level INT NOT NULL DEFAULT 10,
  updated_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  UNIQUE (sku, warehouse_id)
);

-- Soft reservations (held during checkout)
CREATE TABLE inventory_reservations (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  sku           TEXT NOT NULL,
  warehouse_id  UUID NOT NULL,
  order_id      UUID NOT NULL UNIQUE,
  qty           INT NOT NULL CHECK (qty > 0),
  status        TEXT NOT NULL DEFAULT 'held'
                CHECK (status IN ('held', 'confirmed', 'released')),
  expires_at    TIMESTAMPTZ NOT NULL,  -- auto-release after 15 min
  created_at    TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_reservations_expires ON inventory_reservations (expires_at)
  WHERE status = 'held';

CREATE TABLE warehouses (
  id     UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name   TEXT NOT NULL,
  region TEXT NOT NULL,
  active BOOLEAN NOT NULL DEFAULT true
);

NestJS Service​

// src/inventory/inventory.service.ts
import { Injectable, ConflictException, NotFoundException } from '@nestjs/common';
import { InjectDataSource } from '@nestjs/typeorm';
import { DataSource } from 'typeorm';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';
import { EventEmitter2 } from '@nestjs/event-emitter';

@Injectable()
export class InventoryService {
  constructor(
    @InjectDataSource() private readonly ds: DataSource,
    @InjectRedis() private readonly redis: Redis,
    private readonly events: EventEmitter2,
  ) {}

  /** Reserve stock for an order (Saga step 1) */
  async reserve(sku: string, qty: number, orderId: string): Promise<void> {
    await this.ds.transaction(async (em) => {
      const row = await em.query(
        `UPDATE inventory
         SET available_qty = available_qty - $1,
             reserved_qty  = reserved_qty  + $1,
             updated_at    = NOW()
         WHERE sku = $2 AND available_qty >= $1
         RETURNING id`,
        [qty, sku]
      );
      if (row.length === 0) throw new ConflictException(`Insufficient stock for ${sku}`);

      await em.query(
        `INSERT INTO inventory_reservations (sku, warehouse_id, order_id, qty, expires_at)
         SELECT $1, warehouse_id, $2, $3, NOW() + INTERVAL '15 minutes'
         FROM inventory WHERE sku = $1 AND available_qty >= 0
         LIMIT 1`,
        [sku, orderId, qty]
      );
    });

    await this.invalidateCache(sku);
    this.events.emit('inventory.reserved', { sku, qty, orderId });
  }

  /** Confirm reservation → fulfillment (Saga step: payment captured) */
  async confirmReservation(orderId: string): Promise<void> {
    const res = await this.ds.query(
      `UPDATE inventory_reservations SET status = 'confirmed'
       WHERE order_id = $1 AND status = 'held'
       RETURNING sku, warehouse_id, qty`,
      [orderId]
    );
    if (res.length === 0) throw new NotFoundException(`Reservation not found for order ${orderId}`);

    await this.ds.query(
      `UPDATE inventory
       SET reserved_qty = reserved_qty - $1, updated_at = NOW()
       WHERE sku = $2 AND warehouse_id = $3`,
      [res[0].qty, res[0].sku, res[0].warehouse_id]
    );
    this.events.emit('inventory.confirmed', { orderId, sku: res[0].sku, qty: res[0].qty });
  }

  /** Release reservation — Saga compensation (payment failed / order cancelled) */
  async releaseReservation(orderId: string): Promise<void> {
    const res = await this.ds.query(
      `UPDATE inventory_reservations SET status = 'released'
       WHERE order_id = $1 AND status = 'held'
       RETURNING sku, warehouse_id, qty`,
      [orderId]
    );
    if (res.length === 0) return; // idempotent

    await this.ds.query(
      `UPDATE inventory
       SET available_qty = available_qty + $1,
           reserved_qty  = reserved_qty  - $1,
           updated_at    = NOW()
       WHERE sku = $2 AND warehouse_id = $3`,
      [res[0].qty, res[0].sku, res[0].warehouse_id]
    );
    await this.invalidateCache(res[0].sku);
    this.events.emit('inventory.released', { orderId, sku: res[0].sku });
  }

  async getStock(sku: string): Promise<number> {
    const cached = await this.redis.get(`stock:db:${sku}`);
    if (cached !== null) return parseInt(cached, 10);

    const rows = await this.ds.query(
      `SELECT COALESCE(SUM(available_qty), 0) AS total
       FROM inventory WHERE sku = $1`,
      [sku]
    );
    const total = parseInt(rows[0].total, 10);
    await this.redis.set(`stock:db:${sku}`, total, 'EX', 30);
    return total;
  }

  private async invalidateCache(sku: string): Promise<void> {
    await this.redis.del(`stock:db:${sku}`);
  }
}

Expired reservation cleanup (scheduled)​

// src/inventory/reservation-cleanup.service.ts
import { Injectable } from '@nestjs/common';
import { Cron, CronExpression } from '@nestjs/schedule';
import { InjectDataSource } from '@nestjs/typeorm';
import { DataSource } from 'typeorm';

@Injectable()
export class ReservationCleanupService {
  constructor(@InjectDataSource() private readonly ds: DataSource) {}

  @Cron(CronExpression.EVERY_MINUTE)
  async releaseExpired(): Promise<void> {
    // Atomically release expired holds, restore available_qty
    await this.ds.query(`
      WITH expired AS (
        UPDATE inventory_reservations
        SET status = 'released'
        WHERE status = 'held' AND expires_at < NOW()
        RETURNING sku, warehouse_id, qty
      )
      UPDATE inventory i
      SET available_qty = i.available_qty + e.qty,
          reserved_qty  = i.reserved_qty  - e.qty,
          updated_at    = NOW()
      FROM expired e
      WHERE i.sku = e.sku AND i.warehouse_id = e.warehouse_id
    `);
  }
}

FailureMitigation
Double-reservation on Kafka replayorder_id UNIQUE in inventory_reservations — second insert is a no-op
Negative stock due to raceCHECK (available_qty >= 0) + row-level lock in UPDATE
Stale Redis cache after DB updateinvalidateCache() called in same transaction completion handler
