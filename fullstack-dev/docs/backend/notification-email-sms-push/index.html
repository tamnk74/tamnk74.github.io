Email / SMS / Push Notification Delivery

Transactional notification service: multi-channel delivery (Email via SendGrid, SMS via Twilio, Push via FCM/APNS), PostgreSQL outbox pattern for at-least-once guarantees, Kafka consumer, deduplication, and user preference management.

Problem Statement​

The notification service must:

  • Deliver order confirmations, shipping updates, and payment receipts across Email, SMS, and Push channels
  • Guarantee delivery even when downstream providers (SendGrid, Twilio, FCM) are temporarily unavailable
  • Never send duplicate notifications for the same event (idempotency)
  • Respect user channel preferences and opt-outs
  • Handle 50,000+ notifications/hour at peak (flash sales, order confirmations)

Core challenge: A Kafka consumer processes order events, but if the notification is emitted before the DB transaction commits (dual-write problem), it is lost on failure. The outbox pattern solves this.

Architecture​

Key invariant: The notification row is written inside the same DB transaction as the business event (or within the consuming service's own outbox). The outbox worker polls for undelivered rows and dispatches them — provider failure only delays, never loses, a notification.

Database Schema​

-- notification_outbox: the reliable dispatch queue
CREATE TABLE notification_outbox (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  event_id      TEXT NOT NULL UNIQUE,        -- idempotency key (Kafka message key)
  user_id       UUID NOT NULL,
  channel       TEXT NOT NULL CHECK (channel IN ('email', 'sms', 'push')),
  template_id   TEXT NOT NULL,
  payload       JSONB NOT NULL,
  status        TEXT NOT NULL DEFAULT 'pending'
                CHECK (status IN ('pending', 'sent', 'failed', 'skipped')),
  attempts      INT NOT NULL DEFAULT 0,
  next_attempt  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  sent_at       TIMESTAMPTZ,
  error         TEXT,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_outbox_pending ON notification_outbox (status, next_attempt)
  WHERE status = 'pending';

-- user_notification_prefs: channel opt-in/out per category
CREATE TABLE user_notification_prefs (
  user_id       UUID NOT NULL,
  category      TEXT NOT NULL,   -- 'order', 'marketing', 'security', 'shipping'
  channel       TEXT NOT NULL,   -- 'email', 'sms', 'push'
  enabled       BOOLEAN NOT NULL DEFAULT true,
  updated_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  PRIMARY KEY (user_id, category, channel)
);

-- notification_log: audit trail (never deleted)
CREATE TABLE notification_log (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  outbox_id     UUID REFERENCES notification_outbox(id),
  user_id       UUID NOT NULL,
  channel       TEXT NOT NULL,
  template_id   TEXT NOT NULL,
  provider_msg_id TEXT,           -- SendGrid message-id, Twilio SID, FCM message_id
  delivered_at  TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

Kafka Consumer — write to outbox​

// src/consumers/order-events.consumer.ts
import { Controller, Logger } from '@nestjs/common';
import { EventPattern, Payload, Ctx, KafkaContext } from '@nestjs/microservices';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';
import { OutboxService } from '../outbox/outbox.service';
import { PreferencesService } from '../preferences/preferences.service';

@Controller()
export class OrderEventsConsumer {
  private readonly logger = new Logger(OrderEventsConsumer.name);

  constructor(
    private readonly outbox: OutboxService,
    private readonly prefs: PreferencesService,
    @InjectRedis() private readonly redis: Redis,
  ) {}

  @EventPattern('order-events')
  async handleOrderEvent(
    @Payload() event: OrderEvent,
    @Ctx() ctx: KafkaContext,
  ): Promise<void> {
    const { eventId, type, userId } = event;

    // Dedup: skip if already enqueued (Redis NX with 24h TTL)
    const dedupKey = `notif:dedup:${eventId}`;
    const isNew = await this.redis.set(dedupKey, '1', 'EX', 86400, 'NX');
    if (!isNew) {
      this.logger.debug(`Skipping duplicate event ${eventId}`);
      ctx.getMessage().offset; // still commit offset
      return;
    }

    const templateMap: Partial<Record<string, string>> = {
      'order.placed':      'order-confirmation',
      'order.shipped':     'shipping-update',
      'order.delivered':   'delivery-confirmed',
      'payment.failed':    'payment-failure',
      'payment.refunded':  'refund-issued',
    };
    const templateId = templateMap[type];
    if (!templateId) return;

    // Determine channels based on user preferences
    const channels = await this.prefs.getEnabledChannels(userId, 'order');

    await this.outbox.enqueue(
      channels.map(channel => ({
        eventId: `${eventId}:${channel}`,
        userId,
        channel,
        templateId,
        payload: event,
      }))
    );
  }
}

interface OrderEvent {
  eventId: string;
  type: string;
  userId: string;
  orderId: string;
  [key: string]: unknown;
}

Outbox Service​

// src/outbox/outbox.service.ts
import { Injectable } from '@nestjs/common';
import { InjectDataSource } from '@nestjs/typeorm';
import { DataSource } from 'typeorm';

@Injectable()
export class OutboxService {
  constructor(@InjectDataSource() private readonly ds: DataSource) {}

  async enqueue(rows: OutboxRow[]): Promise<void> {
    if (rows.length === 0) return;
    await this.ds.query(
      `INSERT INTO notification_outbox
         (event_id, user_id, channel, template_id, payload)
       SELECT * FROM UNNEST($1::text[], $2::uuid[], $3::text[], $4::text[], $5::jsonb[])
       ON CONFLICT (event_id) DO NOTHING`,
      [
        rows.map(r => r.eventId),
        rows.map(r => r.userId),
        rows.map(r => r.channel),
        rows.map(r => r.templateId),
        rows.map(r => JSON.stringify(r.payload)),
      ]
    );
  }

  async fetchPending(limit = 50): Promise<OutboxRecord[]> {
    return this.ds.query(
      `SELECT * FROM notification_outbox
       WHERE status = 'pending' AND next_attempt <= NOW()
       ORDER BY next_attempt
       LIMIT $1
       FOR UPDATE SKIP LOCKED`,   // prevent concurrent worker contention
      [limit]
    );
  }

  async markSent(id: string, providerMsgId: string): Promise<void> {
    await this.ds.query(
      `UPDATE notification_outbox
       SET status = 'sent', sent_at = NOW()
       WHERE id = $1`,
      [id]
    );
    await this.ds.query(
      `INSERT INTO notification_log (outbox_id, user_id, channel, template_id, provider_msg_id)
       SELECT id, user_id, channel, template_id, $2
       FROM notification_outbox WHERE id = $1`,
      [id, providerMsgId]
    );
  }

  async markFailed(id: string, error: string, attempts: number): Promise<void> {
    // Exponential backoff: 1 min, 5 min, 30 min, 2 h, give up at 5 attempts
    const backoffSeconds = [60, 300, 1800, 7200, 0];
    const delay = backoffSeconds[Math.min(attempts, 4)];
    const nextStatus = delay === 0 ? 'failed' : 'pending';

    await this.ds.query(
      `UPDATE notification_outbox
       SET status = $2, attempts = $3, error = $4,
           next_attempt = NOW() + ($5 || ' seconds')::interval
       WHERE id = $1`,
      [id, nextStatus, attempts + 1, error, delay]
    );
  }
}

interface OutboxRow {
  eventId: string;
  userId: string;
  channel: 'email' | 'sms' | 'push';
  templateId: string;
  payload: Record<string, unknown>;
}

interface OutboxRecord extends OutboxRow {
  id: string;
  attempts: number;
}

Outbox Worker (polling dispatcher)​

// src/outbox/outbox.worker.ts
import { Injectable, Logger, OnApplicationBootstrap } from '@nestjs/common';
import { Interval } from '@nestjs/schedule';
import { OutboxService } from './outbox.service';
import { EmailChannel } from '../channels/email.channel';
import { SmsChannel } from '../channels/sms.channel';
import { PushChannel } from '../channels/push.channel';
import { TemplateService } from '../templates/template.service';

@Injectable()
export class OutboxWorker implements OnApplicationBootstrap {
  private readonly logger = new Logger(OutboxWorker.name);

  private readonly channels = {
    email: this.email,
    sms:   this.sms,
    push:  this.push,
  };

  constructor(
    private readonly outbox: OutboxService,
    private readonly template: TemplateService,
    private readonly email: EmailChannel,
    private readonly sms: SmsChannel,
    private readonly push: PushChannel,
  ) {}

  onApplicationBootstrap() { /* worker starts via @Interval */ }

  @Interval(2000)  // poll every 2 s
  async dispatch(): Promise<void> {
    const rows = await this.outbox.fetchPending(50);
    if (rows.length === 0) return;

    await Promise.allSettled(
      rows.map(row => this.dispatchOne(row))
    );
  }

  private async dispatchOne(row: any): Promise<void> {
    const rendered = await this.template.render(row.templateId, row.payload);
    const channel = this.channels[row.channel as 'email' | 'sms' | 'push'];

    try {
      const providerMsgId = await channel.send(row.userId, rendered);
      await this.outbox.markSent(row.id, providerMsgId);
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : String(err);
      this.logger.warn(`Failed to deliver ${row.id} via ${row.channel}: ${msg}`);
      await this.outbox.markFailed(row.id, msg, row.attempts);
    }
  }
}

Channel Adapters​

Email — SendGrid​

// src/channels/email.channel.ts
import { Injectable, Logger } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import * as sgMail from '@sendgrid/mail';

@Injectable()
export class EmailChannel {
  private readonly logger = new Logger(EmailChannel.name);

  constructor(private readonly config: ConfigService) {
    sgMail.setApiKey(this.config.getOrThrow('SENDGRID_API_KEY'));
  }

  async send(userId: string, rendered: RenderedNotification): Promise<string> {
    const msg: sgMail.MailDataRequired = {
      to:          rendered.to,
      from:        { email: 'no-reply@ecom.io', name: 'E-Commerce' },
      subject:     rendered.subject,
      html:        rendered.html,
      text:        rendered.text,
      trackingSettings: {
        clickTracking:  { enable: true },
        openTracking:   { enable: true },
      },
    };

    const [response] = await sgMail.send(msg);
    const msgId = response.headers['x-message-id'] as string;
    this.logger.debug(`Email sent to ${rendered.to}, msgId=${msgId}`);
    return msgId;
  }
}

SMS — Twilio​

// src/channels/sms.channel.ts
import { Injectable, Logger } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import Twilio from 'twilio';

@Injectable()
export class SmsChannel {
  private readonly logger = new Logger(SmsChannel.name);
  private readonly client: Twilio.Twilio;

  constructor(private readonly config: ConfigService) {
    this.client = Twilio(
      this.config.getOrThrow('TWILIO_ACCOUNT_SID'),
      this.config.getOrThrow('TWILIO_AUTH_TOKEN'),
    );
  }

  async send(userId: string, rendered: RenderedNotification): Promise<string> {
    const message = await this.client.messages.create({
      body: rendered.text,
      from: this.config.getOrThrow('TWILIO_FROM_NUMBER'),
      to:   rendered.to,  // E.164 format: +84912345678
    });

    this.logger.debug(`SMS sent to ${rendered.to}, SID=${message.sid}`);
    return message.sid;
  }
}

Push — FCM (Android) + APNS (iOS)​

// src/channels/push.channel.ts
import { Injectable, Logger } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import * as admin from 'firebase-admin';

@Injectable()
export class PushChannel {
  private readonly logger = new Logger(PushChannel.name);
  private readonly messaging: admin.messaging.Messaging;

  constructor(private readonly config: ConfigService) {
    if (!admin.apps.length) {
      admin.initializeApp({
        credential: admin.credential.cert(
          JSON.parse(this.config.getOrThrow('FIREBASE_SERVICE_ACCOUNT_JSON'))
        ),
      });
    }
    this.messaging = admin.messaging();
  }

  async send(userId: string, rendered: RenderedNotification): Promise<string> {
    // rendered.to = FCM registration token (stored in user device registry)
    const message: admin.messaging.Message = {
      token: rendered.to,
      notification: {
        title: rendered.subject,
        body:  rendered.text,
      },
      data: rendered.data as Record<string, string>,
      android: {
        priority:     'high',
        notification: { sound: 'default', clickAction: 'FLUTTER_NOTIFICATION_CLICK' },
      },
      apns: {
        payload: {
          aps: { sound: 'default', badge: 1 },
        },
      },
    };

    const messageId = await this.messaging.send(message);
    this.logger.debug(`Push sent to ${userId}, FCM messageId=${messageId}`);
    return messageId;
  }
}

export interface RenderedNotification {
  to:       string;
  subject:  string;
  html?:    string;
  text:     string;
  data?:    Record<string, unknown>;
}

Template Service (Handlebars)​

// src/templates/template.service.ts
import { Injectable } from '@nestjs/common';
import * as Handlebars from 'handlebars';
import * as fs from 'fs/promises';
import * as path from 'path';
import { RenderedNotification } from '../channels/push.channel';

@Injectable()
export class TemplateService {
  private readonly cache = new Map<string, HandlebarsTemplateDelegate>();

  async render(
    templateId: string,
    payload: Record<string, unknown>,
  ): Promise<RenderedNotification> {
    const tmpl = await this.loadTemplate(templateId);
    return {
      to:      payload['recipientAddress'] as string,
      subject: tmpl.subject(payload),
      html:    tmpl.html?.(payload),
      text:    tmpl.text(payload),
      data:    payload,
    };
  }

  private async loadTemplate(
    id: string,
  ): Promise<{
    subject: HandlebarsTemplateDelegate;
    html?: HandlebarsTemplateDelegate;
    text: HandlebarsTemplateDelegate;
  }> {
    const dir = path.join(process.cwd(), 'templates', id);
    const [subjectSrc, htmlSrc, textSrc] = await Promise.all([
      fs.readFile(path.join(dir, 'subject.hbs'), 'utf8'),
      fs.readFile(path.join(dir, 'body.html.hbs'), 'utf8').catch(() => null),
      fs.readFile(path.join(dir, 'body.txt.hbs'), 'utf8'),
    ]);
    return {
      subject: Handlebars.compile(subjectSrc),
      html:    htmlSrc ? Handlebars.compile(htmlSrc) : undefined,
      text:    Handlebars.compile(textSrc),
    };
  }
}

User Preference Management​

// src/preferences/preferences.service.ts
import { Injectable } from '@nestjs/common';
import { InjectDataSource } from '@nestjs/typeorm';
import { DataSource } from 'typeorm';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';

type Channel = 'email' | 'sms' | 'push';

@Injectable()
export class PreferencesService {
  constructor(
    @InjectDataSource() private readonly ds: DataSource,
    @InjectRedis() private readonly redis: Redis,
  ) {}

  async getEnabledChannels(userId: string, category: string): Promise<Channel[]> {
    const cacheKey = `notif-prefs:${userId}:${category}`;
    const cached = await this.redis.get(cacheKey);
    if (cached) return JSON.parse(cached) as Channel[];

    const rows = await this.ds.query(
      `SELECT channel FROM user_notification_prefs
       WHERE user_id = $1 AND category = $2 AND enabled = true`,
      [userId, category]
    );
    const channels = rows.map((r: any) => r.channel as Channel);

    // Default to email if no preferences stored yet
    const result = channels.length > 0 ? channels : ['email' as Channel];
    await this.redis.set(cacheKey, JSON.stringify(result), 'EX', 300);
    return result;
  }

  async updatePreference(
    userId: string,
    category: string,
    channel: Channel,
    enabled: boolean,
  ): Promise<void> {
    await this.ds.query(
      `INSERT INTO user_notification_prefs (user_id, category, channel, enabled)
       VALUES ($1, $2, $3, $4)
       ON CONFLICT (user_id, category, channel)
       DO UPDATE SET enabled = $4, updated_at = NOW()`,
      [userId, category, channel, enabled]
    );
    // Invalidate cache
    await this.redis.del(`notif-prefs:${userId}:${category}`);
  }
}

Rate Limiting per Channel​

Prevent notification storms (e.g., sending 100 shipping updates to the same user):

// Per-user, per-channel rate limiting using Redis sliding window
async isRateLimited(
  userId: string,
  channel: Channel,
  limitPerHour = 10,
): Promise<boolean> {
  const key = `notif-rate:${userId}:${channel}`;
  const now = Date.now();
  const windowMs = 3600_000;

  const pipeline = this.redis.pipeline();
  pipeline.zremrangebyscore(key, 0, now - windowMs);
  pipeline.zadd(key, now, `${now}`);
  pipeline.zcard(key);
  pipeline.expire(key, 3600);
  const results = await pipeline.exec();

  const count = results?.[2]?.[1] as number;
  return count > limitPerHour;
}

Delivery Retry Strategy​

AttemptDelayStatus
1 (initial)Immediatepending
2+1 minutepending
3+5 minutespending
4+30 minutespending
5+2 hourspending
6+—failed (alert fires)

Failed rows are kept in notification_outbox for audit. A separate reconciliation job alerts on-call when status = 'failed' count exceeds threshold.
