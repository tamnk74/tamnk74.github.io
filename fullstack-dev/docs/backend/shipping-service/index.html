Shipping Service

Shipment creation, carrier selection, tracking webhooks: FedEx/DHL integration, shipment label generation, Redis-cached tracking status, and Kafka events for downstream consumers.

Responsibilities​

  • Select the optimal carrier (FedEx / DHL / local partner) based on destination, weight, and SLA
  • Create shipments and generate labels via carrier APIs
  • Receive real-time tracking webhooks and push shipment.status_updated events to Kafka
  • Cache tracking status in Redis (TTL 5 min) to serve high-frequency polling
  • Trigger notification and order status updates on key milestones (shipped, out for delivery, delivered)

Data Model​

CREATE TABLE shipments (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  order_id        UUID NOT NULL UNIQUE,
  carrier         TEXT NOT NULL CHECK (carrier IN ('fedex','dhl','ghn','internal')),
  tracking_number TEXT NOT NULL UNIQUE,
  label_url       TEXT,            -- GCS signed URL for label PDF
  status          TEXT NOT NULL DEFAULT 'label_created'
                  CHECK (status IN (
                    'label_created','picked_up','in_transit',
                    'out_for_delivery','delivered','failed_delivery','returned'
                  )),
  estimated_delivery DATE,
  shipped_at      TIMESTAMPTZ,
  delivered_at    TIMESTAMPTZ,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_shipments_status ON shipments (status)
  WHERE status NOT IN ('delivered','returned');

CREATE TABLE shipment_events (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  shipment_id     UUID NOT NULL REFERENCES shipments(id),
  carrier_event_id TEXT NOT NULL UNIQUE,  -- dedup carrier webhook
  event_code      TEXT NOT NULL,
  description     TEXT,
  location        TEXT,
  occurred_at     TIMESTAMPTZ NOT NULL,
  received_at     TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_shipment_events_shipment ON shipment_events (shipment_id, occurred_at DESC);

Carrier Selection Strategy​

// src/shipping/carrier-selector.service.ts
import { Injectable } from '@nestjs/common';

interface ShipmentRequirements {
  destinationCountry: string;
  weightKg:           number;
  deliverySlaHours:   number;
  orderValueUsd:      number;
}

@Injectable()
export class CarrierSelectorService {
  select(req: ShipmentRequirements): 'fedex' | 'dhl' | 'ghn' {
    // International express
    if (req.destinationCountry !== 'VN' && req.deliverySlaHours <= 48) {
      return req.weightKg > 30 ? 'dhl' : 'fedex';
    }
    // Domestic Vietnam
    if (req.destinationCountry === 'VN') return 'ghn';
    // International economy
    return 'dhl';
  }
}

NestJS Shipping Service​

// src/shipping/shipping.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { InjectDataSource } from '@nestjs/typeorm';
import { DataSource } from 'typeorm';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';
import { EventEmitter2 } from '@nestjs/event-emitter';
import { FedExClient } from './carriers/fedex.client';
import { DhlClient } from './carriers/dhl.client';
import { GhnClient } from './carriers/ghn.client';
import { CarrierSelectorService } from './carrier-selector.service';

@Injectable()
export class ShippingService {
  private readonly logger = new Logger(ShippingService.name);

  constructor(
    @InjectDataSource() private readonly ds: DataSource,
    @InjectRedis() private readonly redis: Redis,
    private readonly events: EventEmitter2,
    private readonly selector: CarrierSelectorService,
    private readonly fedex: FedExClient,
    private readonly dhl: DhlClient,
    private readonly ghn: GhnClient,
  ) {}

  async createShipment(dto: CreateShipmentDto): Promise<string> {
    const carrier = this.selector.select({
      destinationCountry: dto.address.country,
      weightKg:           dto.totalWeightKg,
      deliverySlaHours:   dto.slaHours ?? 72,
      orderValueUsd:      dto.orderValueUsd,
    });

    const client = this.getClient(carrier);
    const shipmentResult = await client.createShipment({
      orderId:    dto.orderId,
      address:    dto.address,
      weightKg:   dto.totalWeightKg,
      dimensions: dto.dimensions,
    });

    const [shipment] = await this.ds.query(
      `INSERT INTO shipments
         (order_id, carrier, tracking_number, label_url, estimated_delivery)
       VALUES ($1, $2, $3, $4, $5)
       RETURNING id`,
      [
        dto.orderId,
        carrier,
        shipmentResult.trackingNumber,
        shipmentResult.labelUrl,
        shipmentResult.estimatedDelivery,
      ]
    );

    await this.redis.set(
      `tracking:${shipmentResult.trackingNumber}`,
      JSON.stringify({ status: 'label_created', carrier }),
      'EX', 300
    );

    this.events.emit('shipment.created', {
      orderId:        dto.orderId,
      shipmentId:     shipment.id,
      carrier,
      trackingNumber: shipmentResult.trackingNumber,
    });

    return shipment.id;
  }

  async getTrackingStatus(trackingNumber: string): Promise<TrackingStatus> {
    const cached = await this.redis.get(`tracking:${trackingNumber}`);
    if (cached) return JSON.parse(cached) as TrackingStatus;

    const [shipment] = await this.ds.query(
      `SELECT s.status, s.carrier, s.estimated_delivery,
              json_agg(e.* ORDER BY e.occurred_at DESC) AS events
       FROM shipments s
       LEFT JOIN shipment_events e ON e.shipment_id = s.id
       WHERE s.tracking_number = $1
       GROUP BY s.id`,
      [trackingNumber]
    );
    if (!shipment) throw new Error(`Tracking number ${trackingNumber} not found`);

    const result: TrackingStatus = {
      status:            shipment.status,
      carrier:           shipment.carrier,
      estimatedDelivery: shipment.estimated_delivery,
      events:            shipment.events ?? [],
    };

    await this.redis.set(`tracking:${trackingNumber}`, JSON.stringify(result), 'EX', 300);
    return result;
  }

  /** Called by carrier webhook controller */
  async processCarrierEvent(
    trackingNumber: string,
    carrierEventId: string,
    eventCode: string,
    description: string,
    location: string,
    occurredAt: Date,
  ): Promise<void> {
    const [shipment] = await this.ds.query(
      `SELECT id, order_id FROM shipments WHERE tracking_number = $1`,
      [trackingNumber]
    );
    if (!shipment) {
      this.logger.warn(`Unknown tracking number: ${trackingNumber}`);
      return;
    }

    // Dedup
    await this.ds.query(
      `INSERT INTO shipment_events
         (shipment_id, carrier_event_id, event_code, description, location, occurred_at)
       VALUES ($1, $2, $3, $4, $5, $6)
       ON CONFLICT (carrier_event_id) DO NOTHING`,
      [shipment.id, carrierEventId, eventCode, description, location, occurredAt]
    );

    const statusMap: Record<string, string> = {
      'PU':   'picked_up',
      'IT':   'in_transit',
      'OD':   'out_for_delivery',
      'DL':   'delivered',
      'FD':   'failed_delivery',
    };
    const newStatus = statusMap[eventCode];
    if (newStatus) {
      await this.ds.query(
        `UPDATE shipments SET status = $1,
           ${newStatus === 'delivered' ? 'delivered_at = NOW(),' : ''}
           ${newStatus === 'picked_up' ? 'shipped_at = NOW(),' : ''}
           updated_at = NOW()
         WHERE id = $2`,
        [newStatus, shipment.id]
      );
      await this.redis.del(`tracking:${trackingNumber}`); // invalidate cache

      this.events.emit(`shipment.${newStatus}`, {
        orderId:       shipment.order_id,
        shipmentId:    shipment.id,
        trackingNumber,
        location,
        occurredAt,
      });
    }
  }

  private getClient(carrier: string) {
    return { fedex: this.fedex, dhl: this.dhl, ghn: this.ghn }[carrier]!;
  }
}

interface CreateShipmentDto {
  orderId:       string;
  address:       { country: string; [k: string]: unknown };
  totalWeightKg: number;
  dimensions?:   { l: number; w: number; h: number };
  slaHours?:     number;
  orderValueUsd: number;
}

interface TrackingStatus {
  status:            string;
  carrier:           string;
  estimatedDelivery: string | null;
  events:            unknown[];
}

Carrier Webhook Controller​

// src/shipping/carrier-webhook.controller.ts
import { Controller, Post, Body, Headers, BadRequestException } from '@nestjs/common';
import { createHmac } from 'crypto';
import { ConfigService } from '@nestjs/config';
import { ShippingService } from './shipping.service';

@Controller('webhooks/shipping')
export class CarrierWebhookController {
  constructor(
    private readonly config: ConfigService,
    private readonly shipping: ShippingService,
  ) {}

  @Post('fedex')
  async fedex(
    @Headers('x-fedex-signature') sig: string,
    @Body() body: any,
  ): Promise<void> {
    this.verifySignature(body, sig, 'FEDEX_WEBHOOK_SECRET');
    await this.shipping.processCarrierEvent(
      body.trackingNumber,
      body.eventId,
      body.eventCode,
      body.eventDescription,
      body.location?.city,
      new Date(body.eventDateTime),
    );
  }

  @Post('dhl')
  async dhl(@Headers('x-dhl-signature') sig: string, @Body() body: any): Promise<void> {
    this.verifySignature(body, sig, 'DHL_WEBHOOK_SECRET');
    const event = body.shipmentTrackingEvent;
    await this.shipping.processCarrierEvent(
      event.awbNumber,
      event.eventId,
      event.activityCode,
      event.description,
      event.location,
      new Date(event.timestamp),
    );
  }

  private verifySignature(body: unknown, sig: string, secretKey: string): void {
    const expected = createHmac('sha256', this.config.getOrThrow(secretKey))
      .update(JSON.stringify(body))
      .digest('hex');
    if (expected !== sig) throw new BadRequestException('Invalid webhook signature');
  }
}

Kafka Events Produced​

EventTrigger
shipment.createdLabel generated, carrier assigned
shipment.picked_upCarrier picks up package
shipment.in_transitIn-transit scan
shipment.out_for_deliveryOut for delivery scan
shipment.deliveredDelivered scan
shipment.failed_deliveryDelivery attempt failed

Order Service consumes shipment.created → SHIPPED. Notification Service consumes all milestone events to send SMS/push updates.

Failure Modes​

FailureMitigation
Carrier API timeout on label creationRetry with backoff (3 attempts); if all fail, put order in LABEL_PENDING queue
Duplicate carrier webhookcarrier_event_id UNIQUE — second insert skipped
Stale tracking cacheTTL 5 min; redis.del on any processCarrierEvent that changes status
Carrier changes tracking numberCarrier webhook includes old + new number; update shipments tracking_number
