Redis Comprehensive Guide

Overview​

Redis (Remote Dictionary Server) is an open-source, in-memory data structure store used as a database, cache, message broker, and streaming engine. It supports various data structures such as strings, hashes, lists, sets, sorted sets, bitmaps, and more.

Key Features:

  • ⚡ In-memory storage for sub-millisecond latency
  • 🔄 Persistence options (RDB snapshots, AOF logs)
  • 📊 Rich data structures
  • 🔁 Built-in replication
  • 🎯 Pub/Sub messaging
  • 🔒 Transactions and Lua scripting
  • 🌊 Redis Streams for event sourcing
  • ⚙️ Cluster mode for horizontal scaling

Architecture​

High-Level Architecture​

Architecture Components​

1. Master-Replica Architecture

  • Master nodes: Handle write operations
  • Replica nodes: Serve read operations and provide high availability
  • Automatic failover: Replicas promoted to master if master fails

2. Cluster Mode (Sharding)

  • 16,384 hash slots distributed across master nodes
  • Data automatically partitioned across nodes
  • Horizontal scaling by adding more nodes
  • Client-side routing or Redis Cluster Proxy

3. Persistence Options

  • RDB (Redis Database): Point-in-time snapshots
  • AOF (Append Only File): Log of every write operation
  • Hybrid: Combine both for durability

4. Eviction Policies

  • noeviction: Return errors when memory limit reached
  • allkeys-lru: Remove least recently used keys
  • volatile-lru: Remove LRU among keys with expiration
  • allkeys-random: Remove random keys
  • volatile-ttl: Remove keys with shortest TTL

Data Structures​

1. String​

Basic key-value storage. Max value size: 512MB.

// Use cases: Cache, counters, flags
await redis.set('user:1000', JSON.stringify({ name: 'John' }));
await redis.get('user:1000');
await redis.incr('page:views'); // Atomic increment

2. Hash​

Maps between string fields and values. Perfect for objects.

// Use cases: User profiles, shopping carts
await redis.hset('user:1000', 'name', 'John', 'email', 'john@example.com');
await redis.hgetall('user:1000');

3. List​

Linked lists of strings. Push/pop from both ends.

// Use cases: Message queues, activity feeds
await redis.lpush('queue:emails', 'email1', 'email2');
await redis.rpop('queue:emails');

4. Set​

Unordered collection of unique strings.

// Use cases: Tags, unique visitors
await redis.sadd('tags:post:1', 'redis', 'nodejs', 'cache');
await redis.sismember('tags:post:1', 'redis');

5. Sorted Set (ZSet)​

Sets ordered by score. Members are unique, scores can repeat.

// Use cases: Leaderboards, priority queues
await redis.zadd('leaderboard', 100, 'player1', 200, 'player2');
await redis.zrange('leaderboard', 0, 9, 'WITHSCORES');

6. Bitmap​

Bit-level operations on strings.

// Use cases: User analytics, feature flags
await redis.setbit('users:active:2024-01-25', 1000, 1);
await redis.getbit('users:active:2024-01-25', 1000);

7. HyperLogLog​

Probabilistic data structure for counting unique items.

// Use cases: Unique visitors, cardinality estimation
await redis.pfadd('unique:visitors', 'user1', 'user2');
await redis.pfcount('unique:visitors');

1. Session Management​

Store user sessions with automatic expiration.

// services/session.service.ts
import { Injectable } from '@nestjs/common';
import { InjectRedis } from '@nestjs/redis';
import Redis from 'ioredis';
import { randomUUID } from 'crypto';

interface SessionData {
  userId: string;
  email: string;
  roles: string[];
  createdAt: number;
}

@Injectable()
export class SessionService {
  constructor(@InjectRedis() private readonly redis: Redis) {}

  async createSession(userId: string, email: string, roles: string[]): Promise<string> {
    const sessionId = randomUUID();
    const sessionData: SessionData = {
      userId,
      email,
      roles,
      createdAt: Date.now(),
    };

    // Store session with 24 hour expiration
    await this.redis.setex(
      `session:${sessionId}`,
      86400, // 24 hours
      JSON.stringify(sessionData),
    );

    return sessionId;
  }

  async getSession(sessionId: string): Promise<SessionData | null> {
    const data = await this.redis.get(`session:${sessionId}`);
    return data ? JSON.parse(data) : null;
  }

  async refreshSession(sessionId: string): Promise<void> {
    await this.redis.expire(`session:${sessionId}`, 86400);
  }

  async deleteSession(sessionId: string): Promise<void> {
    await this.redis.del(`session:${sessionId}`);
  }

  async getUserSessions(userId: string): Promise<string[]> {
    const keys = await this.redis.keys(`session:*`);
    const sessions: string[] = [];

    for (const key of keys) {
      const data = await this.redis.get(key);
      if (data) {
        const session: SessionData = JSON.parse(data);
        if (session.userId === userId) {
          sessions.push(key.replace('session:', ''));
        }
      }
    }

    return sessions;
  }
}

2. Caching​

Cache frequently accessed data to reduce database load.

// services/cache.service.ts
import { Injectable } from '@nestjs/common';
import { InjectRedis } from '@nestjs/redis';
import Redis from 'ioredis';

@Injectable()
export class CacheService {
  constructor(@InjectRedis() private readonly redis: Redis) {}

  async get<T>(key: string): Promise<T | null> {
    const data = await this.redis.get(key);
    return data ? JSON.parse(data) : null;
  }

  async set(key: string, value: any, ttl?: number): Promise<void> {
    const serialized = JSON.stringify(value);
    if (ttl) {
      await this.redis.setex(key, ttl, serialized);
    } else {
      await this.redis.set(key, serialized);
    }
  }

  async del(key: string): Promise<void> {
    await this.redis.del(key);
  }

  async invalidatePattern(pattern: string): Promise<void> {
    const keys = await this.redis.keys(pattern);
    if (keys.length > 0) {
      await this.redis.del(...keys);
    }
  }
}

// services/product.service.ts
@Injectable()
export class ProductService {
  constructor(
    private readonly productRepository: ProductRepository,
    private readonly cacheService: CacheService,
  ) {}

  async getProduct(id: string): Promise<Product> {
    const cacheKey = `product:${id}`;
    
    // Try cache first
    const cached = await this.cacheService.get<Product>(cacheKey);
    if (cached) {
      return cached;
    }

    // Cache miss - fetch from database
    const product = await this.productRepository.findById(id);
    
    // Store in cache for 1 hour
    await this.cacheService.set(cacheKey, product, 3600);
    
    return product;
  }

  async updateProduct(id: string, updates: Partial<Product>): Promise<Product> {
    const product = await this.productRepository.update(id, updates);
    
    // Invalidate cache
    await this.cacheService.del(`product:${id}`);
    
    return product;
  }
}

3. Distributed Lock​

Prevent race conditions in distributed systems.

// services/distributed-lock.service.ts
import { Injectable } from '@nestjs/common';
import { InjectRedis } from '@nestjs/redis';
import Redis from 'ioredis';
import { randomUUID } from 'crypto';

@Injectable()
export class DistributedLockService {
  constructor(@InjectRedis() private readonly redis: Redis) {}

  async acquireLock(
    resource: string,
    ttl: number = 10000, // 10 seconds
  ): Promise<string | null> {
    const lockKey = `lock:${resource}`;
    const lockValue = randomUUID();

    // NX: Only set if not exists, PX: Set expiration in milliseconds
    const result = await this.redis.set(lockKey, lockValue, 'PX', ttl, 'NX');

    return result === 'OK' ? lockValue : null;
  }

  async releaseLock(resource: string, lockValue: string): Promise<boolean> {
    const lockKey = `lock:${resource}`;

    // Lua script to ensure atomicity
    const script = `
      if redis.call("get", KEYS[1]) == ARGV[1] then
        return redis.call("del", KEYS[1])
      else
        return 0
      end
    `;

    const result = await this.redis.eval(script, 1, lockKey, lockValue);
    return result === 1;
  }

  async withLock<T>(
    resource: string,
    callback: () => Promise<T>,
    ttl: number = 10000,
  ): Promise<T> {
    const lockValue = await this.acquireLock(resource, ttl);
    
    if (!lockValue) {
      throw new Error(`Failed to acquire lock for resource: ${resource}`);
    }

    try {
      return await callback();
    } finally {
      await this.releaseLock(resource, lockValue);
    }
  }
}

// Example usage
@Injectable()
export class OrderService {
  constructor(private readonly lockService: DistributedLockService) {}

  async processPayment(orderId: string): Promise<void> {
    await this.lockService.withLock(`order:${orderId}`, async () => {
      // Critical section - only one instance can execute this
      const order = await this.getOrder(orderId);
      
      if (order.status === 'paid') {
        throw new Error('Order already paid');
      }

      await this.chargePayment(order);
      await this.updateOrderStatus(orderId, 'paid');
    });
  }
}

4. Counters​

Atomic increment/decrement operations for metrics and analytics.

// services/counter.service.ts
import { Injectable } from '@nestjs/common';
import { InjectRedis } from '@nestjs/redis';
import Redis from 'ioredis';

@Injectable()
export class CounterService {
  constructor(@InjectRedis() private readonly redis: Redis) {}

  async increment(key: string, amount: number = 1): Promise<number> {
    return await this.redis.incrby(key, amount);
  }

  async decrement(key: string, amount: number = 1): Promise<number> {
    return await this.redis.decrby(key, amount);
  }

  async get(key: string): Promise<number> {
    const value = await this.redis.get(key);
    return value ? parseInt(value, 10) : 0;
  }

  async reset(key: string): Promise<void> {
    await this.redis.set(key, 0);
  }

  // Get multiple counters at once
  async getMany(keys: string[]): Promise<Record<string, number>> {
    const values = await this.redis.mget(...keys);
    const result: Record<string, number> = {};
    
    keys.forEach((key, index) => {
      result[key] = values[index] ? parseInt(values[index], 10) : 0;
    });
    
    return result;
  }
}

// Example: Page view counter
@Controller('analytics')
export class AnalyticsController {
  constructor(private readonly counterService: CounterService) {}

  @Post('page-view')
  async trackPageView(@Body() { page }: { page: string }) {
    const today = new Date().toISOString().split('T')[0];
    
    // Increment daily counter
    await this.counterService.increment(`views:${page}:${today}`);
    
    // Increment total counter
    await this.counterService.increment(`views:${page}:total`);
    
    return { success: true };
  }

  @Get('stats/:page')
  async getPageStats(@Param('page') page: string) {
    const today = new Date().toISOString().split('T')[0];
    
    const [todayViews, totalViews] = await Promise.all([
      this.counterService.get(`views:${page}:${today}`),
      this.counterService.get(`views:${page}:total`),
    ]);
    
    return { todayViews, totalViews };
  }
}

5. Rate Limiting​

Control request rates using sliding window or token bucket algorithms.

// services/rate-limiter.service.ts
import { Injectable } from '@nestjs/common';
import { InjectRedis } from '@nestjs/redis';
import Redis from 'ioredis';

@Injectable()
export class RateLimiterService {
  constructor(@InjectRedis() private readonly redis: Redis) {}

  // Fixed window rate limiter
  async checkRateLimit(
    identifier: string,
    maxRequests: number,
    windowSeconds: number,
  ): Promise<{ allowed: boolean; remaining: number; resetAt: number }> {
    const key = `rate_limit:${identifier}`;
    const now = Date.now();
    const windowMs = windowSeconds * 1000;
    const windowStart = Math.floor(now / windowMs) * windowMs;
    const resetAt = windowStart + windowMs;

    const current = await this.redis.incr(key);
    
    if (current === 1) {
      await this.redis.pexpire(key, windowMs);
    }

    return {
      allowed: current <= maxRequests,
      remaining: Math.max(0, maxRequests - current),
      resetAt,
    };
  }

  // Sliding window rate limiter (more accurate)
  async checkSlidingWindowRateLimit(
    identifier: string,
    maxRequests: number,
    windowSeconds: number,
  ): Promise<{ allowed: boolean; remaining: number }> {
    const key = `rate_limit:sliding:${identifier}`;
    const now = Date.now();
    const windowMs = windowSeconds * 1000;
    const windowStart = now - windowMs;

    // Remove old entries and add current request
    const pipeline = this.redis.pipeline();
    pipeline.zremrangebyscore(key, 0, windowStart);
    pipeline.zadd(key, now, `${now}-${Math.random()}`);
    pipeline.zcard(key);
    pipeline.expire(key, windowSeconds);
    
    const results = await pipeline.exec();
    const count = results[2][1] as number;

    return {
      allowed: count <= maxRequests,
      remaining: Math.max(0, maxRequests - count),
    };
  }

  // Token bucket rate limiter
  async checkTokenBucket(
    identifier: string,
    capacity: number,
    refillRate: number,
    refillIntervalMs: number,
  ): Promise<{ allowed: boolean; tokens: number }> {
    const key = `rate_limit:bucket:${identifier}`;
    const now = Date.now();

    const script = `
      local key = KEYS[1]
      local capacity = tonumber(ARGV[1])
      local refillRate = tonumber(ARGV[2])
      local refillInterval = tonumber(ARGV[3])
      local now = tonumber(ARGV[4])
      
      local bucket = redis.call('HMGET', key, 'tokens', 'lastRefill')
      local tokens = tonumber(bucket[1]) or capacity
      local lastRefill = tonumber(bucket[2]) or now
      
      -- Refill tokens
      local elapsed = now - lastRefill
      local refills = math.floor(elapsed / refillInterval)
      tokens = math.min(capacity, tokens + (refills * refillRate))
      
      -- Try to consume a token
      local allowed = tokens > 0
      if allowed then
        tokens = tokens - 1
      end
      
      redis.call('HMSET', key, 'tokens', tokens, 'lastRefill', now)
      redis.call('EXPIRE', key, 3600)
      
      return {allowed and 1 or 0, tokens}
    `;

    const result = await this.redis.eval(
      script,
      1,
      key,
      capacity,
      refillRate,
      refillIntervalMs,
      now,
    ) as [number, number];

    return {
      allowed: result[0] === 1,
      tokens: result[1],
    };
  }
}

// Rate limiting middleware/guard
@Injectable()
export class RateLimitGuard implements CanActivate {
  constructor(private readonly rateLimiter: RateLimiterService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const identifier = request.ip || request.user?.id || 'anonymous';

    const result = await this.rateLimiter.checkSlidingWindowRateLimit(
      identifier,
      100, // 100 requests
      60,  // per 60 seconds
    );

    if (!result.allowed) {
      throw new HttpException(
        'Rate limit exceeded',
        HttpStatus.TOO_MANY_REQUESTS,
      );
    }

    return true;
  }
}

6. Global ID Generator​

Generate unique, distributed IDs using Redis INCR.

// services/id-generator.service.ts
import { Injectable } from '@nestjs/common';
import { InjectRedis } from '@nestjs/redis';
import Redis from 'ioredis';

@Injectable()
export class IdGeneratorService {
  constructor(@InjectRedis() private readonly redis: Redis) {}

  // Simple sequential ID
  async generateId(prefix: string): Promise<string> {
    const id = await this.redis.incr(`id:${prefix}`);
    return `${prefix}-${id}`;
  }

  // Twitter Snowflake-like ID (timestamp + sequence)
  async generateSnowflakeId(workerId: number = 1): Promise<string> {
    const epoch = 1609459200000; // Jan 1, 2021
    const timestamp = Date.now() - epoch;
    
    const sequence = await this.redis.incr(`snowflake:${workerId}:${timestamp}`);
    await this.redis.expire(`snowflake:${workerId}:${timestamp}`, 1);

    // 41 bits timestamp, 10 bits worker ID, 12 bits sequence
    const id = (BigInt(timestamp) << BigInt(22)) |
                (BigInt(workerId) << BigInt(12)) |
                BigInt(sequence);

    return id.toString();
  }

  // UUID with Redis-backed sequence
  async generateOrderedUUID(prefix: string): Promise<string> {
    const timestamp = Date.now();
    const sequence = await this.redis.incr(`uuid:${prefix}:${timestamp}`);
    await this.redis.expire(`uuid:${prefix}:${timestamp}`, 1);

    return `${prefix}-${timestamp}-${sequence.toString().padStart(6, '0')}`;
  }
}

// Example usage
@Injectable()
export class OrderService {
  constructor(private readonly idGenerator: IdGeneratorService) {}

  async createOrder(userId: string, items: any[]): Promise<Order> {
    const orderId = await this.idGenerator.generateOrderedUUID('ORD');
    
    const order = await this.orderRepository.create({
      id: orderId,
      userId,
      items,
      createdAt: new Date(),
    });

    return order;
  }
}

7. Shopping Cart​

Store temporary shopping cart data with automatic expiration.

// services/cart.service.ts
import { Injectable } from '@nestjs/common';
import { InjectRedis } from '@nestjs/redis';
import Redis from 'ioredis';

interface CartItem {
  productId: string;
  quantity: number;
  price: number;
  name: string;
}

@Injectable()
export class CartService {
  constructor(@InjectRedis() private readonly redis: Redis) {}

  private getCartKey(userId: string): string {
    return `cart:${userId}`;
  }

  async addItem(userId: string, item: CartItem): Promise<void> {
    const key = this.getCartKey(userId);
    
    // Store as hash: field = productId, value = JSON
    await this.redis.hset(key, item.productId, JSON.stringify(item));
    
    // Set expiration to 7 days
    await this.redis.expire(key, 7 * 24 * 60 * 60);
  }

  async removeItem(userId: string, productId: string): Promise<void> {
    const key = this.getCartKey(userId);
    await this.redis.hdel(key, productId);
  }

  async updateQuantity(
    userId: string,
    productId: string,
    quantity: number,
  ): Promise<void> {
    const key = this.getCartKey(userId);
    const itemData = await this.redis.hget(key, productId);
    
    if (!itemData) {
      throw new Error('Item not found in cart');
    }

    const item: CartItem = JSON.parse(itemData);
    item.quantity = quantity;

    await this.redis.hset(key, productId, JSON.stringify(item));
    await this.redis.expire(key, 7 * 24 * 60 * 60);
  }

  async getCart(userId: string): Promise<CartItem[]> {
    const key = this.getCartKey(userId);
    const data = await this.redis.hgetall(key);

    return Object.values(data).map((item) => JSON.parse(item));
  }

  async getCartTotal(userId: string): Promise<number> {
    const items = await this.getCart(userId);
    return items.reduce((total, item) => total + item.price * item.quantity, 0);
  }

  async clearCart(userId: string): Promise<void> {
    const key = this.getCartKey(userId);
    await this.redis.del(key);
  }

  async mergeGuestCart(guestId: string, userId: string): Promise<void> {
    const guestKey = this.getCartKey(guestId);
    const userKey = this.getCartKey(userId);

    const guestCart = await this.redis.hgetall(guestKey);
    
    if (Object.keys(guestCart).length > 0) {
      // Merge guest cart into user cart
      for (const [productId, itemData] of Object.entries(guestCart)) {
        const existingItem = await this.redis.hget(userKey, productId);
        
        if (existingItem) {
          // Combine quantities
          const existing: CartItem = JSON.parse(existingItem);
          const guest: CartItem = JSON.parse(itemData);
          existing.quantity += guest.quantity;
          await this.redis.hset(userKey, productId, JSON.stringify(existing));
        } else {
          await this.redis.hset(userKey, productId, itemData);
        }
      }

      await this.redis.del(guestKey);
      await this.redis.expire(userKey, 7 * 24 * 60 * 60);
    }
  }
}

8. User Retention with Bitmaps​

Track user activity and calculate retention metrics efficiently.

// services/retention.service.ts
import { Injectable } from '@nestjs/common';
import { InjectRedis } from '@nestjs/redis';
import Redis from 'ioredis';

@Injectable()
export class RetentionService {
  constructor(@InjectRedis() private readonly redis: Redis) {}

  private getDateKey(date: Date): string {
    return date.toISOString().split('T')[0];
  }

  async markUserActive(userId: number, date: Date = new Date()): Promise<void> {
    const key = `active:${this.getDateKey(date)}`;
    await this.redis.setbit(key, userId, 1);
    
    // Keep for 90 days
    await this.redis.expire(key, 90 * 24 * 60 * 60);
  }

  async isUserActive(userId: number, date: Date): Promise<boolean> {
    const key = `active:${this.getDateKey(date)}`;
    const bit = await this.redis.getbit(key, userId);
    return bit === 1;
  }

  async getDailyActiveUsers(date: Date): Promise<number> {
    const key = `active:${this.getDateKey(date)}`;
    return await this.redis.bitcount(key);
  }

  async getMonthlyActiveUsers(year: number, month: number): Promise<number> {
    const daysInMonth = new Date(year, month, 0).getDate();
    const keys: string[] = [];

    for (let day = 1; day <= daysInMonth; day++) {
      const date = new Date(year, month - 1, day);
      keys.push(`active:${this.getDateKey(date)}`);
    }

    const destKey = `mau:${year}-${month}`;
    
    // OR operation across all days
    if (keys.length > 0) {
      await this.redis.bitop('OR', destKey, ...keys);
      const count = await this.redis.bitcount(destKey);
      await this.redis.del(destKey);
      return count;
    }

    return 0;
  }

  async getRetentionRate(cohortDate: Date, checkDate: Date): Promise<number> {
    const cohortKey = `active:${this.getDateKey(cohortDate)}`;
    const checkKey = `active:${this.getDateKey(checkDate)}`;
    const destKey = `retention:${cohortDate.getTime()}-${checkDate.getTime()}`;

    // AND operation to find users active on both days
    await this.redis.bitop('AND', destKey, cohortKey, checkKey);
    
    const retained = await this.redis.bitcount(destKey);
    const cohortSize = await this.redis.bitcount(cohortKey);
    
    await this.redis.del(destKey);

    return cohortSize > 0 ? (retained / cohortSize) * 100 : 0;
  }

  async getUsersActiveOnBothDays(date1: Date, date2: Date): Promise<number> {
    const key1 = `active:${this.getDateKey(date1)}`;
    const key2 = `active:${this.getDateKey(date2)}`;
    const destKey = `temp:${Date.now()}`;

    await this.redis.bitop('AND', destKey, key1, key2);
    const count = await this.redis.bitcount(destKey);
    await this.redis.del(destKey);

    return count;
  }
}

9. Message Queue with Lists​

Implement simple message queues using Redis Lists.

// services/queue.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { InjectRedis } from '@nestjs/redis';
import Redis from 'ioredis';

interface QueueMessage<T = any> {
  id: string;
  data: T;
  attempts: number;
  createdAt: number;
}

@Injectable()
export class QueueService {
  private readonly logger = new Logger(QueueService.name);

  constructor(@InjectRedis() private readonly redis: Redis) {}

  async enqueue<T>(queueName: string, data: T): Promise<void> {
    const message: QueueMessage<T> = {
      id: `${Date.now()}-${Math.random()}`,
      data,
      attempts: 0,
      createdAt: Date.now(),
    };

    await this.redis.lpush(queueName, JSON.stringify(message));
  }

  async dequeue<T>(queueName: string, timeout: number = 0): Promise<QueueMessage<T> | null> {
    const result = await this.redis.brpop(queueName, timeout);
    
    if (!result) {
      return null;
    }

    const [, messageStr] = result;
    return JSON.parse(messageStr);
  }

  async getQueueLength(queueName: string): Promise<number> {
    return await this.redis.llen(queueName);
  }

  async processQueue<T>(
    queueName: string,
    processor: (data: T) => Promise<void>,
    options: { maxRetries?: number; retryDelay?: number } = {},
  ): Promise<void> {
    const { maxRetries = 3, retryDelay = 1000 } = options;

    while (true) {
      try {
        const message = await this.dequeue<T>(queueName, 5);
        
        if (!message) {
          continue;
        }

        try {
          await processor(message.data);
          this.logger.log(`Processed message ${message.id}`);
        } catch (error) {
          this.logger.error(`Failed to process message ${message.id}:`, error);

          message.attempts++;

          if (message.attempts < maxRetries) {
            // Re-queue with delay
            setTimeout(() => {
              this.redis.lpush(queueName, JSON.stringify(message));
            }, retryDelay * message.attempts);
          } else {
            // Move to dead letter queue
            await this.redis.lpush(`${queueName}:failed`, JSON.stringify(message));
          }
        }
      } catch (error) {
        this.logger.error('Queue processing error:', error);
        await new Promise((resolve) => setTimeout(resolve, 1000));
      }
    }
  }
}

// Example: Email queue
@Injectable()
export class EmailQueueService {
  constructor(
    private readonly queueService: QueueService,
    private readonly emailService: EmailService,
  ) {
    this.startWorker();
  }

  async queueEmail(to: string, subject: string, body: string): Promise<void> {
    await this.queueService.enqueue('emails', { to, subject, body });
  }

  private async startWorker(): Promise<void> {
    await this.queueService.processQueue(
      'emails',
      async (data: { to: string; subject: string; body: string }) => {
        await this.emailService.send(data.to, data.subject, data.body);
      },
      { maxRetries: 3, retryDelay: 2000 },
    );
  }
}

10. Leaderboard with Sorted Sets​

Implement real-time leaderboards and rankings.

// services/leaderboard.service.ts
import { Injectable } from '@nestjs/common';
import { InjectRedis } from '@nestjs/redis';
import Redis from 'ioredis';

interface LeaderboardEntry {
  userId: string;
  score: number;
  rank: number;
}

@Injectable()
export class LeaderboardService {
  constructor(@InjectRedis() private readonly redis: Redis) {}

  async addScore(
    leaderboardName: string,
    userId: string,
    score: number,
  ): Promise<void> {
    await this.redis.zadd(leaderboardName, score, userId);
  }

  async incrementScore(
    leaderboardName: string,
    userId: string,
    increment: number,
  ): Promise<number> {
    return await this.redis.zincrby(leaderboardName, increment, userId);
  }

  async getTopPlayers(
    leaderboardName: string,
    limit: number = 10,
  ): Promise<LeaderboardEntry[]> {
    const results = await this.redis.zrevrange(
      leaderboardName,
      0,
      limit - 1,
      'WITHSCORES',
    );

    const entries: LeaderboardEntry[] = [];
    for (let i = 0; i < results.length; i += 2) {
      entries.push({
        userId: results[i],
        score: parseFloat(results[i + 1]),
        rank: Math.floor(i / 2) + 1,
      });
    }

    return entries;
  }

  async getPlayerRank(
    leaderboardName: string,
    userId: string,
  ): Promise<LeaderboardEntry | null> {
    const rank = await this.redis.zrevrank(leaderboardName, userId);
    
    if (rank === null) {
      return null;
    }

    const score = await this.redis.zscore(leaderboardName, userId);

    return {
      userId,
      score: parseFloat(score),
      rank: rank + 1, // Redis uses 0-based indexing
    };
  }

  async getPlayersInRange(
    leaderboardName: string,
    start: number,
    end: number,
  ): Promise<LeaderboardEntry[]> {
    const results = await this.redis.zrevrange(
      leaderboardName,
      start - 1, // Convert to 0-based
      end - 1,
      'WITHSCORES',
    );

    const entries: LeaderboardEntry[] = [];
    for (let i = 0; i < results.length; i += 2) {
      entries.push({
        userId: results[i],
        score: parseFloat(results[i + 1]),
        rank: start + Math.floor(i / 2),
      });
    }

    return entries;
  }

  async getPlayersSurrounding(
    leaderboardName: string,
    userId: string,
    range: number = 5,
  ): Promise<LeaderboardEntry[]> {
    const rank = await this.redis.zrevrank(leaderboardName, userId);
    
    if (rank === null) {
      return [];
    }

    const start = Math.max(0, rank - range);
    const end = rank + range;

    const results = await this.redis.zrevrange(
      leaderboardName,
      start,
      end,
      'WITHSCORES',
    );

    const entries: LeaderboardEntry[] = [];
    for (let i = 0; i < results.length; i += 2) {
      entries.push({
        userId: results[i],
        score: parseFloat(results[i + 1]),
        rank: start + Math.floor(i / 2) + 1,
      });
    }

    return entries;
  }

  async removePlayer(leaderboardName: string, userId: string): Promise<void> {
    await this.redis.zrem(leaderboardName, userId);
  }

  async getTotalPlayers(leaderboardName: string): Promise<number> {
    return await this.redis.zcard(leaderboardName);
  }
}

// Example: Game leaderboard controller
@Controller('leaderboard')
export class LeaderboardController {
  constructor(private readonly leaderboardService: LeaderboardService) {}

  @Post('score')
  async submitScore(
    @Body() { userId, score }: { userId: string; score: number },
  ) {
    await this.leaderboardService.addScore('game:global', userId, score);
    const rank = await this.leaderboardService.getPlayerRank('game:global', userId);
    return { rank };
  }

  @Get('top/:limit')
  async getTop(@Param('limit') limit: string) {
    return this.leaderboardService.getTopPlayers('game:global', parseInt(limit, 10));
  }

  @Get('player/:userId')
  async getPlayerStats(@Param('userId') userId: string) {
    const [rank, surrounding] = await Promise.all([
      this.leaderboardService.getPlayerRank('game:global', userId),
      this.leaderboardService.getPlayersSurrounding('game:global', userId, 3),
    ]);

    return { rank, surrounding };
  }
}

// ❌ Bad: Blocks Redis
await redis.keys('user:*');

Monitoring​

@Injectable()
export class RedisHealthService {
  constructor(@InjectRedis() private readonly redis: Redis) {}

  async checkHealth(): Promise<{
    status: string;
    latency: number;
    memoryUsage: string;
    connectedClients: number;
  }> {
    const start = Date.now();
    await this.redis.ping();
    const latency = Date.now() - start;

    const info = await this.redis.info();
    const lines = info.split('\r\n');

    const memoryLine = lines.find((l) => l.startsWith('used_memory_human:'));
    const clientsLine = lines.find((l) => l.startsWith('connected_clients:'));

    return {
      status: 'healthy',
      latency,
      memoryUsage: memoryLine?.split(':')[1] || 'unknown',
      connectedClients: parseInt(clientsLine?.split(':')[1] || '0', 10),
    };
  }
}
