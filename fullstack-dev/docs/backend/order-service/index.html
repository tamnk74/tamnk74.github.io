Order Service

Order lifecycle management: state machine from PENDING → DELIVERED, Saga orchestration via Kafka, outbox pattern for reliable event publishing, and order history queries.

Responsibilities​

  • Create orders from a cart snapshot and drive them through the lifecycle state machine
  • Coordinate the checkout Saga via domain events (inventory → payment → fulfillment)
  • Write order-events to Kafka via the outbox pattern (no dual-write risk)
  • Serve order history with Redis caching (status TTL 60 s)
  • Execute Saga compensation (cancel + release stock + void payment) on any step failure

Data Model​

CREATE TABLE orders (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id         UUID NOT NULL,
  status          TEXT NOT NULL DEFAULT 'PENDING'
                  CHECK (status IN (
                    'PENDING','CONFIRMED','PAYMENT_FAILED',
                    'PROCESSING','SHIPPED','DELIVERED',
                    'CANCELLATION_REQUESTED','CANCELLED','REFUNDED'
                  )),
  subtotal        NUMERIC(12,2) NOT NULL,
  discount_amount NUMERIC(12,2) NOT NULL DEFAULT 0,
  total           NUMERIC(12,2) NOT NULL,
  currency        CHAR(3) NOT NULL DEFAULT 'USD',
  shipping_address_id UUID NOT NULL,
  idempotency_key TEXT NOT NULL UNIQUE, -- client-generated, prevents duplicate orders
  created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at      TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_orders_user ON orders (user_id, created_at DESC);
CREATE INDEX idx_orders_status ON orders (status) WHERE status NOT IN ('DELIVERED','CANCELLED');

CREATE TABLE order_items (
  id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  order_id   UUID NOT NULL REFERENCES orders(id),
  sku        TEXT NOT NULL,
  product_name TEXT NOT NULL,
  qty        INT NOT NULL CHECK (qty > 0),
  unit_price NUMERIC(12,2) NOT NULL,
  total      NUMERIC(12,2) NOT NULL
);
CREATE INDEX idx_order_items_order ON order_items (order_id);

-- Outbox for reliable Kafka publishing
CREATE TABLE order_outbox (
  id             UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  aggregate_id   UUID NOT NULL,
  event_type     TEXT NOT NULL,
  payload        JSONB NOT NULL,
  published      BOOLEAN NOT NULL DEFAULT false,
  created_at     TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_order_outbox_unpublished ON order_outbox (created_at)
  WHERE published = false;

-- Status transitions audit trail
CREATE TABLE order_status_history (
  id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  order_id   UUID NOT NULL REFERENCES orders(id),
  from_status TEXT,
  to_status  TEXT NOT NULL,
  reason     TEXT,
  changed_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  changed_by TEXT NOT NULL  -- 'saga', 'admin', 'user'
);

NestJS Order Service​

// src/order/order.service.ts
import { Injectable, ConflictException, BadRequestException } from '@nestjs/common';
import { InjectDataSource } from '@nestjs/typeorm';
import { DataSource } from 'typeorm';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';

export interface CreateOrderDto {
  userId:            string;
  cartSnapshot:      CartSnapshot;
  shippingAddressId: string;
  idempotencyKey:    string;
}

@Injectable()
export class OrderService {
  constructor(
    @InjectDataSource() private readonly ds: DataSource,
    @InjectRedis() private readonly redis: Redis,
  ) {}

  async createOrder(dto: CreateOrderDto): Promise<string> {
    const { userId, cartSnapshot, shippingAddressId, idempotencyKey } = dto;

    // Idempotency: return existing order if already created
    const existing = await this.ds.query(
      `SELECT id FROM orders WHERE idempotency_key = $1`, [idempotencyKey]
    );
    if (existing.length > 0) return existing[0].id;

    return this.ds.transaction(async (em) => {
      // 1. Insert order
      const [order] = await em.query(
        `INSERT INTO orders
           (user_id, status, subtotal, discount_amount, total, shipping_address_id, idempotency_key)
         VALUES ($1, 'PENDING', $2, $3, $4, $5, $6)
         RETURNING id`,
        [
          userId,
          cartSnapshot.subtotal,
          cartSnapshot.coupon?.discountAmount ?? 0,
          cartSnapshot.total,
          shippingAddressId,
          idempotencyKey,
        ]
      );

      // 2. Insert line items
      for (const [sku, item] of Object.entries(cartSnapshot.items)) {
        await em.query(
          `INSERT INTO order_items (order_id, sku, product_name, qty, unit_price, total)
           VALUES ($1, $2, $3, $4, $5, $6)`,
          [order.id, sku, item.productName ?? sku, item.qty, item.price, item.price * item.qty]
        );
      }

      // 3. Write to outbox (same transaction — no dual-write risk)
      await em.query(
        `INSERT INTO order_outbox (aggregate_id, event_type, payload)
         VALUES ($1, 'order.created', $2)`,
        [order.id, JSON.stringify({ orderId: order.id, userId, total: cartSnapshot.total })]
      );

      return order.id;
    });
  }

  async transitionStatus(
    orderId: string,
    toStatus: string,
    changedBy: string,
    reason?: string,
  ): Promise<void> {
    await this.ds.transaction(async (em) => {
      const [current] = await em.query(
        `SELECT status FROM orders WHERE id = $1 FOR UPDATE`, [orderId]
      );
      if (!current) throw new BadRequestException(`Order ${orderId} not found`);

      await em.query(
        `UPDATE orders SET status = $1, updated_at = NOW() WHERE id = $2`,
        [toStatus, orderId]
      );
      await em.query(
        `INSERT INTO order_status_history (order_id, from_status, to_status, reason, changed_by)
         VALUES ($1, $2, $3, $4, $5)`,
        [orderId, current.status, toStatus, reason ?? null, changedBy]
      );
      await em.query(
        `INSERT INTO order_outbox (aggregate_id, event_type, payload)
         VALUES ($1, $2, $3)`,
        [orderId, `order.${toStatus.toLowerCase()}`, JSON.stringify({ orderId, status: toStatus })]
      );
    });

    // Invalidate status cache
    await this.redis.del(`order:status:${orderId}`);
  }

  async getOrder(orderId: string): Promise<OrderDetail> {
    const cacheKey = `order:status:${orderId}`;
    const cached = await this.redis.get(cacheKey);
    if (cached) return JSON.parse(cached) as OrderDetail;

    const [order] = await this.ds.query(
      `SELECT o.*, json_agg(i.*) AS items
       FROM orders o
       JOIN order_items i ON i.order_id = o.id
       WHERE o.id = $1
       GROUP BY o.id`,
      [orderId]
    );
    if (!order) throw new BadRequestException(`Order ${orderId} not found`);

    await this.redis.set(cacheKey, JSON.stringify(order), 'EX', 60);
    return order as OrderDetail;
  }
}

interface CartSnapshot {
  items:    Record<string, { qty: number; price: number; productName?: string }>;
  subtotal: number;
  coupon:   { code: string; discountAmount: number } | null;
  total:    number;
}

interface OrderDetail { id: string; status: string; items: unknown[] }

Saga Kafka Consumers​

The Order Service listens to cross-service events and drives state transitions:

// src/order/saga.consumer.ts
import { Controller } from '@nestjs/common';
import { EventPattern, Payload } from '@nestjs/microservices';
import { OrderService } from './order.service';

@Controller()
export class SagaConsumer {
  constructor(private readonly orders: OrderService) {}

  @EventPattern('inventory-events')
  async onInventory(@Payload() event: any): Promise<void> {
    if (event.type === 'inventory.reserved') {
      // Inventory held — payment will now be triggered by payment-service consuming
      // order.created; nothing to do here until payment result arrives.
    }
    if (event.type === 'inventory.reservation.failed') {
      await this.orders.transitionStatus(event.orderId, 'CANCELLED', 'saga',
        'Inventory reservation failed');
    }
  }

  @EventPattern('payment-events')
  async onPayment(@Payload() event: any): Promise<void> {
    if (event.type === 'payment.captured') {
      await this.orders.transitionStatus(event.orderId, 'CONFIRMED', 'saga');
    }
    if (event.type === 'payment.failed') {
      await this.orders.transitionStatus(event.orderId, 'PAYMENT_FAILED', 'saga',
        event.reason);
    }
  }

  @EventPattern('shipping-events')
  async onShipping(@Payload() event: any): Promise<void> {
    if (event.type === 'shipment.created') {
      await this.orders.transitionStatus(event.orderId, 'SHIPPED', 'saga');
    }
    if (event.type === 'delivery.confirmed') {
      await this.orders.transitionStatus(event.orderId, 'DELIVERED', 'saga');
    }
  }
}

Outbox Publisher​

// src/order/outbox.publisher.ts
import { Injectable } from '@nestjs/common';
import { Interval } from '@nestjs/schedule';
import { InjectDataSource } from '@nestjs/typeorm';
import { DataSource } from 'typeorm';
import { KafkaProducerService } from '../kafka/kafka-producer.service';

@Injectable()
export class OutboxPublisher {
  constructor(
    @InjectDataSource() private readonly ds: DataSource,
    private readonly kafka: KafkaProducerService,
  ) {}

  @Interval(500) // poll every 500 ms
  async publish(): Promise<void> {
    const rows = await this.ds.query(
      `SELECT * FROM order_outbox
       WHERE published = false
       ORDER BY created_at
       LIMIT 100
       FOR UPDATE SKIP LOCKED`
    );
    if (rows.length === 0) return;

    await this.kafka.sendBatch('order-events',
      rows.map((r: any) => ({ key: r.aggregate_id, value: JSON.stringify(r.payload) }))
    );

    await this.ds.query(
      `UPDATE order_outbox SET published = true
       WHERE id = ANY($1)`,
      [rows.map((r: any) => r.id)]
    );
  }
}
