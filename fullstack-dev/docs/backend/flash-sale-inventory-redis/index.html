Flash Sale Inventory — Redis Atomic Decrement

Handling 50,000+ concurrent stock reservation attempts without database pressure: Lua scripts, rate limiting, oversell prevention, and observability.

Problem Statement​

A flash sale creates a thundering-herd write problem:

  • 50,000 users click "Buy" simultaneously for a limited-stock item (e.g. 100 units)
  • Every request must atomically check stock and decrement — without race conditions
  • Database cannot handle 50,000 concurrent writes without overload
  • Must prevent overselling (selling more than available stock) under any concurrency

Why Redis? Sub-millisecond single-threaded operations. The Redis server processes commands sequentially — an EVAL (Lua) execution is atomic by design. No SELECT → check → UPDATE race window exists.

Architecture​

Key invariant: Redis is the authoritative stock counter during the flash sale window. PostgreSQL is written asynchronously via Kafka — it is never on the critical path for a reservation response.

Core Implementation: Lua Atomic Decrement​

Why Lua (not DECR alone)?​

DECR alone would decrement below zero (negative stock). The Lua script adds an atomic check-then-decrement in a single command execution — no other command can interleave between the check and the decrement.

// src/inventory/inventory.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';

@Injectable()
export class InventoryService {
  private readonly logger = new Logger(InventoryService.name);

  // Lua script: atomic check-and-decrement
  // Returns 1 on success, 0 if insufficient stock, -1 if key not found
  private static readonly RESERVE_SCRIPT = `
    local stock = tonumber(redis.call('GET', KEYS[1]))
    if stock == nil then return -1 end
    if stock < tonumber(ARGV[1]) then return 0 end
    redis.call('DECRBY', KEYS[1], ARGV[1])
    return 1
  `;

  constructor(@InjectRedis() private readonly redis: Redis) {}

  async reserveStock(sku: string, qty: number): Promise<ReserveResult> {
    const key = `stock:${sku}`;

    const result = await this.redis.eval(
      InventoryService.RESERVE_SCRIPT,
      1,       // number of keys
      key,     // KEYS[1]
      qty.toString()  // ARGV[1]
    ) as number;

    switch (result) {
      case 1:
        return { success: true };
      case 0:
        return { success: false, reason: 'INSUFFICIENT_STOCK' };
      case -1:
        return { success: false, reason: 'SKU_NOT_FOUND' };
      default:
        throw new Error(`Unexpected Lua result: ${result}`);
    }
  }

  async releaseStock(sku: string, qty: number): Promise<void> {
    // Saga compensation: release reservation on payment failure
    await this.redis.incrby(`stock:${sku}`, qty);
    this.logger.log(`Released ${qty} units of ${sku} back to stock`);
  }

  async getStock(sku: string): Promise<number | null> {
    const val = await this.redis.get(`stock:${sku}`);
    return val !== null ? parseInt(val, 10) : null;
  }
}

export interface ReserveResult {
  success: boolean;
  reason?: 'INSUFFICIENT_STOCK' | 'SKU_NOT_FOUND';
}

Stock Initialization (Pre-Sale Load)​

Stock counters must be loaded into Redis before the flash sale opens. Load them from PostgreSQL with a TTL that outlasts the sale window plus a buffer.

// src/inventory/flash-sale-loader.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';
import { InventoryItem } from './inventory-item.entity';

@Injectable()
export class FlashSaleLoaderService {
  private readonly logger = new Logger(FlashSaleLoaderService.name);

  constructor(
    @InjectRepository(InventoryItem)
    private readonly inventoryRepo: Repository<InventoryItem>,
    @InjectRedis() private readonly redis: Redis,
  ) {}

  /**
   * Load flash sale stock into Redis ~5 min before sale opens.
   * Called by CronJob or ArgoCD pre-sync hook.
   */
  async loadFlashSaleStock(eventId: string, skus: string[]): Promise<void> {
    const items = await this.inventoryRepo.findBy({ sku: skus as any });

    const pipeline = this.redis.pipeline();
    for (const item of items) {
      const key = `stock:${item.sku}`;
      pipeline.set(key, item.availableQty);
      pipeline.expire(key, 7200); // 2-hour TTL (sale window + buffer)
      this.logger.log(`Loaded stock:${item.sku} = ${item.availableQty}`);
    }
    await pipeline.exec();

    // Mark event as loaded (prevents double-load)
    await this.redis.set(`flash-sale:${eventId}:loaded`, '1', 'EX', 7200);
    this.logger.log(`Flash sale ${eventId}: loaded ${items.length} SKUs into Redis`);
  }

  /**
   * After sale ends, sync Redis counters back to PostgreSQL.
   * Reconciles the authoritative DB with what was actually sold.
   */
  async syncStockToDB(skus: string[]): Promise<void> {
    for (const sku of skus) {
      const remaining = await this.redis.get(`stock:${sku}`);
      if (remaining !== null) {
        await this.inventoryRepo.update(
          { sku },
          { availableQty: parseInt(remaining, 10) }
        );
      }
    }
    this.logger.log(`Post-sale DB sync completed for ${skus.length} SKUs`);
  }
}

Reservation with Idempotency​

Prevent double-reservations if a user retries (network timeout, browser refresh):

// src/cart/cart.service.ts
async reserveWithIdempotency(
  userId: string,
  sku: string,
  qty: number,
  idempotencyKey: string,
): Promise<ReserveResult> {
  const dedupKey = `reservation:${idempotencyKey}`;

  // Check if this request was already processed
  const existing = await this.redis.get(dedupKey);
  if (existing) {
    return JSON.parse(existing) as ReserveResult;
  }

  // Process the reservation
  const result = await this.inventoryService.reserveStock(sku, qty);

  // Store result for idempotency window (5 min)
  await this.redis.set(dedupKey, JSON.stringify(result), 'EX', 300);

  if (result.success) {
    // Publish event for order creation (async)
    await this.kafka.emit('order-events', {
      type: 'inventory.reserved',
      userId,
      sku,
      qty,
      idempotencyKey,
      reservedAt: new Date().toISOString(),
    });
  }

  return result;
}

Per-User Purchase Limit​

Prevent a single user from buying all stock by enforcing a per-user limit in Redis alongside the global counter:

// Lua script: atomic global decrement + per-user limit enforcement
private static readonly RESERVE_WITH_LIMIT_SCRIPT = `
  local stockKey  = KEYS[1]   -- stock:{sku}
  local userKey   = KEYS[2]   -- user-limit:{userId}:{sku}
  local qty       = tonumber(ARGV[1])
  local maxPerUser = tonumber(ARGV[2])

  -- Check per-user limit
  local userPurchased = tonumber(redis.call('GET', userKey)) or 0
  if userPurchased + qty > maxPerUser then
    return -2  -- per-user limit exceeded
  end

  -- Check global stock
  local stock = tonumber(redis.call('GET', stockKey))
  if stock == nil then return -1 end
  if stock < qty then return 0 end

  -- Atomic decrement both counters
  redis.call('DECRBY', stockKey, qty)
  redis.call('INCRBY', userKey, qty)
  redis.call('EXPIRE', userKey, 86400)  -- 24h per-user tracking TTL

  return 1
`;

async reserveWithUserLimit(
  userId: string,
  sku: string,
  qty: number,
  maxPerUser = 2,
): Promise<ReserveResult> {
  const result = await this.redis.eval(
    InventoryService.RESERVE_WITH_LIMIT_SCRIPT,
    2,
    `stock:${sku}`,
    `user-limit:${userId}:${sku}`,
    qty.toString(),
    maxPerUser.toString(),
  ) as number;

  const resultMap: Record<number, ReserveResult> = {
    1:  { success: true },
    0:  { success: false, reason: 'INSUFFICIENT_STOCK' },
    [-1]: { success: false, reason: 'SKU_NOT_FOUND' },
    [-2]: { success: false, reason: 'USER_LIMIT_EXCEEDED' as any },
  };
  return resultMap[result] ?? { success: false, reason: 'UNKNOWN' as any };
}

Rate Limiting at the API Layer​

The Lua script is atomic and fast, but it still consumes a Redis operation per request. Add Traefik rate limiting to cap simultaneous checkout attempts before they hit the service:

# traefik/middlewares/flash-sale-rate-limit.yaml
apiVersion: traefik.io/v1alpha1
kind: Middleware
metadata:
  name: flash-sale-rate-limit
  namespace: production
spec:
  rateLimit:
    average: 100      # 100 requests/s per IP
    burst: 200        # allow burst of 200 before throttling
    period: 1s
    sourceCriterion:
      ipStrategy:
        depth: 1      # respect X-Forwarded-For (Cloudflare proxied)

Apply to checkout route during flash sale:

# traefik/ingress-route.yaml (checkout route)
- match: Host(`api.ecom.io`) && PathPrefix(`/cart/reserve`)
  kind: Rule
  services: [{ name: cart-service, port: 3002 }]
  middlewares:
    - name: flash-sale-rate-limit
    - name: jwt-auth

# k8s/production/cart-service-flashsale-scaledobject.yaml
apiVersion: keda.sh/v1alpha1
kind: ScaledObject
metadata:
  name: cart-service-flashsale
  namespace: production
spec:
  scaleTargetRef:
    name: cart-service
  minReplicaCount: 5
  maxReplicaCount: 50
  triggers:
    - type: cron
      metadata:
        timezone: UTC
        start: "55 11 * * *"     # 5 min before noon flash sale
        end:   "30 12 * * *"     # 30 min after noon
        desiredReplicas: "50"
    - type: redis
      metadata:
        address: "${REDIS_HOST}:6379"
        listName: "checkout-queue"
        listLength: "100"         # scale on queue depth too

Post-Sale Reconciliation​

After the flash sale window closes, verify Redis counters match what was actually ordered:

// src/inventory/reconciliation.service.ts
async reconcilePostSale(eventId: string, skus: string[]): Promise<ReconciliationReport> {
  const report: ReconciliationReport = { skus: [], discrepancies: [] };

  for (const sku of skus) {
    const redisStock = parseInt((await this.redis.get(`stock:${sku}`)) ?? '0', 10);
    const dbStock = await this.inventoryRepo.findOneBy({ sku });
    const orderedQty = await this.orderRepo
      .createQueryBuilder('o')
      .select('SUM(o.qty)', 'total')
      .where('o.sku = :sku AND o.eventId = :eventId', { sku, eventId })
      .getRawOne()
      .then(r => parseInt(r?.total ?? '0', 10));

    const originalStock = (dbStock?.availableQty ?? 0) + orderedQty;
    const expectedRedis = originalStock - orderedQty;
    const delta = redisStock - expectedRedis;

    report.skus.push({ sku, redisStock, orderedQty, expectedRedis, delta });

    if (Math.abs(delta) > 0) {
      report.discrepancies.push({ sku, delta });
      this.logger.warn(`Stock discrepancy for ${sku}: delta=${delta}`);
    }
  }

  // Sync corrected values back to DB
  await this.flashSaleLoader.syncStockToDB(skus);

  return report;
}
