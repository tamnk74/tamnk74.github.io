Analytics Service — Clickstream, Revenue & Funnel

Real-time and batch analytics pipeline for e-commerce: Kafka-sourced clickstream events → Elasticsearch (real-time dashboards) + BigQuery (historical reporting, funnel analysis, revenue intelligence).

Problem Statement​

The analytics service must:

  • Capture every clickstream event (page views, product impressions, add-to-cart, checkout steps) without impacting the critical path
  • Compute real-time dashboards (current active users, live revenue, stock depletion rate during flash sales)
  • Power funnel analysis (where users drop off in the purchase journey)
  • Feed revenue reports (daily/weekly/monthly GMV, AOV, conversion rate)
  • Handle 50 million events/day (≈600 events/s avg, 5,000/s during flash sale peaks) without data loss

Architecture principle: analytics is a consumer, never a producer — services emit domain events to Kafka, and the analytics pipeline consumes them. No analytics code pollutes the order/cart/payment critical path.

Architecture​

Two storage tiers:

TierToolUse CaseRetention
Hot / real-timeElasticsearchLive dashboards, last-hour funnels, flash sale monitoring7 days
Cold / historicalBigQueryRevenue reports, cohort analysis, ML feature storeUnlimited (partitioned)

Clickstream Event Schema​

All frontend events share a common envelope:

// shared/analytics-event.schema.ts
export interface AnalyticsEvent {
  eventId:    string;          // UUID v4 (client-generated for dedup)
  eventType:  AnalyticsEventType;
  sessionId:  string;          // persisted in localStorage, rotates on login
  userId?:    string;          // nullable until authenticated
  deviceId:   string;          // fingerprint (hashed user-agent + screen)
  timestamp:  string;          // ISO 8601 UTC
  url:        string;
  referrer?:  string;
  properties: Record<string, unknown>;
}

export type AnalyticsEventType =
  | 'page_view'
  | 'product_impression'
  | 'product_click'
  | 'add_to_cart'
  | 'remove_from_cart'
  | 'checkout_started'
  | 'checkout_step_completed'
  | 'purchase_completed'
  | 'search_performed'
  | 'promo_applied';

Ingest Service (NestJS)​

// src/ingest/ingest.controller.ts
import {
  Controller, Post, Body, Ip, Headers, HttpCode, BadRequestException
} from '@nestjs/common';
import { KafkaProducerService } from '../kafka/kafka-producer.service';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';
import { AnalyticsEvent } from '../schemas/analytics-event.schema';

@Controller('v1/track')
export class IngestController {
  constructor(
    private readonly kafka: KafkaProducerService,
    @InjectRedis() private readonly redis: Redis,
  ) {}

  @Post()
  @HttpCode(202)
  async track(
    @Body() events: AnalyticsEvent[],
    @Ip() ip: string,
    @Headers('x-session-id') sessionId: string,
  ): Promise<{ accepted: number }> {
    if (!Array.isArray(events) || events.length > 100) {
      throw new BadRequestException('Expected array of 1–100 events');
    }

    // Dedup: skip events already seen (client retry on network failure)
    const pipeline = this.redis.pipeline();
    for (const e of events) {
      pipeline.set(`analytics:dedup:${e.eventId}`, '1', 'EX', 86400, 'NX');
    }
    const results = await pipeline.exec();

    const newEvents = events.filter((_, i) => results?.[i]?.[1] === 1);

    if (newEvents.length > 0) {
      await this.kafka.sendBatch(
        'analytics-events',
        newEvents.map(e => ({
          key:   e.sessionId,         // partition by session for ordered replay
          value: JSON.stringify({ ...e, ip }),
        }))
      );
    }

    return { accepted: newEvents.length };
  }
}

Kafka Producer (batched)​

// src/kafka/kafka-producer.service.ts
import { Injectable, OnModuleInit } from '@nestjs/common';
import { Kafka, Producer, CompressionTypes } from 'kafkajs';

@Injectable()
export class KafkaProducerService implements OnModuleInit {
  private producer: Producer;

  constructor() {
    const kafka = new Kafka({ brokers: process.env.KAFKA_BROKERS!.split(',') });
    this.producer = kafka.producer({
      idempotent: true,          // exactly-once produce semantics
      transactionTimeout: 30_000,
    });
  }

  async onModuleInit() { await this.producer.connect(); }

  async sendBatch(
    topic: string,
    messages: Array<{ key: string; value: string }>,
  ): Promise<void> {
    await this.producer.send({
      topic,
      compression: CompressionTypes.GZIP,
      messages,
    });
  }
}

Stream Processing (Dataflow / Flink)​

The Flink job consumes analytics-events plus domain events from order-events and payment-events, applying windowed aggregations before writing to Elasticsearch and BigQuery.

Windowed aggregations​

# analytics-pipeline/src/pipeline.py (Apache Beam / Dataflow)
import apache_beam as beam
from apache_beam.transforms.window import FixedWindows, SlidingWindows
from apache_beam.options.pipeline_options import PipelineOptions

def run():
    options = PipelineOptions(
        runner='DataflowRunner',
        project='ecom-prod',
        region='us-central1',
        streaming=True,
        job_name='analytics-streaming',
    )

    with beam.Pipeline(options=options) as p:
        events = (
            p
            | 'ReadKafka' >> beam.io.ReadFromKafka(
                consumer_config={'bootstrap.servers': '${KAFKA_BROKERS}'},
                topics=['analytics-events', 'order-events', 'payment-events'],
            )
            | 'ParseJSON' >> beam.Map(lambda m: json.loads(m[1]))
        )

        # 1-minute fixed windows → Elasticsearch (real-time dashboards)
        (
            events
            | 'Window1m' >> beam.WindowInto(FixedWindows(60))
            | 'CountByType' >> beam.combiners.Count.PerKey(
                beam.Map(lambda e: e['eventType'])
              )
            | 'WriteES' >> beam.io.WriteToElasticsearch(
                host='${ES_HOST}',
                index='analytics-realtime',
            )
        )

        # 60-second micro-batch → BigQuery (append-only partitioned table)
        (
            events
            | 'Window60s' >> beam.WindowInto(FixedWindows(60))
            | 'WriteBQ' >> beam.io.WriteToBigQuery(
                table='ecom-prod:analytics.events',
                schema='AUTO_DETECT',
                write_disposition=beam.io.BigQueryDisposition.WRITE_APPEND,
                create_disposition=beam.io.BigQueryDisposition.CREATE_IF_NEEDED,
            )
        )

BigQuery Schema & Partitioning​

-- analytics.events (partitioned by event_date, clustered by event_type)
CREATE TABLE `ecom-prod.analytics.events`
(
  event_id      STRING NOT NULL,
  event_type    STRING NOT NULL,
  session_id    STRING,
  user_id       STRING,
  device_id     STRING,
  event_date    DATE NOT NULL,
  ts            TIMESTAMP NOT NULL,
  url           STRING,
  referrer      STRING,
  properties    JSON
)
PARTITION BY event_date
CLUSTER BY event_type, user_id
OPTIONS (
  partition_expiration_days = 730,  -- 2-year retention
  require_partition_filter = TRUE   -- prevent full-table scans
);

-- analytics.orders (materialized from payment-events)
CREATE TABLE `ecom-prod.analytics.orders`
(
  order_id      STRING NOT NULL,
  user_id       STRING NOT NULL,
  order_date    DATE NOT NULL,
  revenue       NUMERIC NOT NULL,
  item_count    INT64,
  coupon_code   STRING,
  payment_method STRING,
  status        STRING
)
PARTITION BY order_date
CLUSTER BY user_id, status;

Key analytics queries​

-- Daily GMV (last 30 days)
SELECT
  order_date,
  SUM(revenue)    AS gmv,
  COUNT(*)        AS order_count,
  SUM(revenue) / COUNT(*) AS aov
FROM `ecom-prod.analytics.orders`
WHERE order_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 30 DAY)
  AND status = 'COMPLETED'
GROUP BY order_date
ORDER BY order_date DESC;

-- Checkout funnel (last 7 days)
SELECT
  step,
  COUNT(DISTINCT session_id) AS sessions,
  SAFE_DIVIDE(
    COUNT(DISTINCT session_id),
    FIRST_VALUE(COUNT(DISTINCT session_id)) OVER (ORDER BY step_order)
  ) AS conversion_rate
FROM (
  SELECT
    session_id,
    CASE event_type
      WHEN 'product_click'           THEN 'step_1_product_view'
      WHEN 'add_to_cart'             THEN 'step_2_add_to_cart'
      WHEN 'checkout_started'        THEN 'step_3_checkout'
      WHEN 'checkout_step_completed' THEN 'step_4_payment'
      WHEN 'purchase_completed'      THEN 'step_5_purchase'
    END AS step,
    CASE event_type
      WHEN 'product_click'           THEN 1
      WHEN 'add_to_cart'             THEN 2
      WHEN 'checkout_started'        THEN 3
      WHEN 'checkout_step_completed' THEN 4
      WHEN 'purchase_completed'      THEN 5
    END AS step_order
  FROM `ecom-prod.analytics.events`
  WHERE event_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY)
    AND event_type IN (
      'product_click','add_to_cart','checkout_started',
      'checkout_step_completed','purchase_completed'
    )
)
WHERE step IS NOT NULL
GROUP BY step, step_order
ORDER BY step_order;

-- Top products by revenue (last 7 days)
SELECT
  JSON_VALUE(properties, '$.sku')   AS sku,
  SUM(JSON_VALUE(properties, '$.price')::FLOAT64 * JSON_VALUE(properties, '$.qty')::INT64) AS revenue,
  COUNT(*)                          AS add_to_cart_events
FROM `ecom-prod.analytics.events`
WHERE event_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY)
  AND event_type = 'add_to_cart'
GROUP BY sku
ORDER BY revenue DESC
LIMIT 20;

Elasticsearch — Real-Time Dashboards​

Elasticsearch stores the last 7 days of raw and aggregated events for live dashboard queries (sub-second response).

// src/analytics/analytics-query.service.ts
import { Injectable } from '@nestjs/common';
import { ElasticsearchService } from '@nestjs/elasticsearch';

@Injectable()
export class AnalyticsQueryService {
  constructor(private readonly es: ElasticsearchService) {}

  /** Active users in last 5 minutes (flash sale monitor) */
  async getActiveUsers(): Promise<number> {
    const result = await this.es.search({
      index: 'analytics-realtime',
      body: {
        query: {
          range: { ts: { gte: 'now-5m' } },
        },
        aggs: {
          active_users: { cardinality: { field: 'session_id' } },
        },
        size: 0,
      },
    });
    return (result.aggregations?.active_users as any)?.value ?? 0;
  }

  /** Revenue in the last hour (rolling) */
  async getLiveRevenue(): Promise<number> {
    const result = await this.es.search({
      index: 'analytics-realtime',
      body: {
        query: {
          bool: {
            filter: [
              { term:  { eventType: 'purchase_completed' } },
              { range: { ts: { gte: 'now-1h' } } },
            ],
          },
        },
        aggs: {
          total_revenue: {
            sum: { field: 'properties.revenue' },
          },
        },
        size: 0,
      },
    });
    return (result.aggregations?.total_revenue as any)?.value ?? 0;
  }

  /** Real-time funnel drop-off (last 30 min) */
  async getFunnelStats(): Promise<FunnelStep[]> {
    const result = await this.es.search({
      index: 'analytics-realtime',
      body: {
        query: { range: { ts: { gte: 'now-30m' } } },
        aggs: {
          steps: {
            terms: {
              field: 'eventType',
              include: [
                'product_click', 'add_to_cart',
                'checkout_started', 'purchase_completed',
              ],
            },
            aggs: {
              unique_sessions: { cardinality: { field: 'session_id' } },
            },
          },
        },
        size: 0,
      },
    });
    return (result.aggregations?.steps as any)?.buckets ?? [];
  }
}

interface FunnelStep {
  key: string;
  doc_count: number;
  unique_sessions: { value: number };
}

Redis Live Counters​

For extremely low-latency reads (flash sale scoreboard, admin "live" view):

// src/analytics/live-counter.service.ts
import { Injectable } from '@nestjs/common';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';

@Injectable()
export class LiveCounterService {
  constructor(@InjectRedis() private readonly redis: Redis) {}

  /** Increment on every page_view */
  async recordPageView(sessionId: string): Promise<void> {
    const minute = Math.floor(Date.now() / 60000);
    await this.redis.pfadd(`pv:hll:${minute}`, sessionId);  // HyperLogLog
    await this.redis.expire(`pv:hll:${minute}`, 600);        // 10-min TTL
  }

  /** Active unique sessions in last 5 minutes */
  async getActiveSessionCount(): Promise<number> {
    const now = Math.floor(Date.now() / 60000);
    const keys = Array.from({ length: 5 }, (_, i) => `pv:hll:${now - i}`);
    const result = await this.redis.pfcount(...keys);
    return result;
  }

  /** Revenue counter (incremented by payment-events consumer) */
  async addRevenue(amountCents: number): Promise<void> {
    const key = `revenue:${new Date().toISOString().slice(0, 13)}`; // hourly bucket
    await this.redis.incrby(key, amountCents);
    await this.redis.expire(key, 86400 * 2); // 48 h
  }

  async getHourlyRevenue(hour: string): Promise<number> {
    const val = await this.redis.get(`revenue:${hour}`);
    return val ? parseInt(val, 10) / 100 : 0;
  }
}

Frontend Analytics SDK​

Thin batching SDK that collects events client-side and flushes them on idle or before unload:

// packages/analytics-sdk/src/index.ts
class AnalyticsSDK {
  private queue: AnalyticsEvent[] = [];
  private flushTimer?: ReturnType<typeof setTimeout>;

  track(eventType: string, properties: Record<string, unknown>): void {
    this.queue.push({
      eventId:   crypto.randomUUID(),
      eventType: eventType as any,
      sessionId: this.getSessionId(),
      userId:    this.getUserId(),
      deviceId:  this.getDeviceId(),
      timestamp: new Date().toISOString(),
      url:       window.location.href,
      referrer:  document.referrer || undefined,
      properties,
    });

    // Flush immediately at 20 events or after 5 s idle
    if (this.queue.length >= 20) {
      this.flush();
    } else {
      this.scheduleFlush(5000);
    }
  }

  private scheduleFlush(ms: number): void {
    clearTimeout(this.flushTimer);
    this.flushTimer = setTimeout(() => this.flush(), ms);
  }

  private async flush(): Promise<void> {
    if (this.queue.length === 0) return;
    const batch = this.queue.splice(0, 100);

    // Use sendBeacon for unload — guaranteed delivery even on page close
    if (!navigator.sendBeacon(
      '/api/v1/track',
      new Blob([JSON.stringify(batch)], { type: 'application/json' })
    )) {
      // Fallback to fetch for non-beacon environments
      await fetch('/api/v1/track', {
        method: 'POST', headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(batch),
        keepalive: true,
      });
    }
  }

  private getSessionId(): string {
    let id = sessionStorage.getItem('_aid');
    if (!id) { id = crypto.randomUUID(); sessionStorage.setItem('_aid', id); }
    return id;
  }

  private getUserId(): string | undefined {
    return localStorage.getItem('_uid') ?? undefined;
  }

  private getDeviceId(): string {
    let id = localStorage.getItem('_did');
    if (!id) { id = crypto.randomUUID(); localStorage.setItem('_did', id); }
    return id;
  }
}

export const analytics = new AnalyticsSDK();

// Auto-flush before page unload
window.addEventListener('visibilitychange', () => {
  if (document.visibilityState === 'hidden') analytics['flush']();
});
