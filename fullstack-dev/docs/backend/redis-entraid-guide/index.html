Redis with Microsoft Entra ID Authentication

Overview​

Microsoft Entra ID (formerly Azure Active Directory) integration with Azure Cache for Redis provides enterprise-grade authentication and authorization using managed identities and service principals. This eliminates the need for password-based authentication and provides centralized access management.

Key Benefits​

  • Password-less Authentication: Use managed identities instead of connection strings
  • Centralized Access Control: Manage permissions through Azure RBAC
  • Audit Trail: Track all Redis access through Entra ID logs
  • Token-based Security: Automatic token refresh and rotation
  • Compliance: Meet enterprise security requirements

Prerequisites​

# Install required packages
npm install @azure/identity redis

# For development/testing
npm install @azure/identity-cache-persistence --save-dev

Use Cases​

1. NestJS Service with Managed Identity​

Use Azure Managed Identity for production deployments on Azure App Service, AKS, or Azure Functions.

// redis-entraid.service.ts
import { Injectable, OnModuleInit, OnModuleDestroy, Logger } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { createClient, RedisClientType } from 'redis';
import {
  DefaultAzureCredential,
  ManagedIdentityCredential,
  AzureCliCredential,
  ChainedTokenCredential,
} from '@azure/identity';

@Injectable()
export class RedisEntraIdService implements OnModuleInit, OnModuleDestroy {
  private readonly logger = new Logger(RedisEntraIdService.name);
  private client: RedisClientType;
  private tokenRefreshInterval: NodeJS.Timeout;
  private currentToken: string;

  constructor(private configService: ConfigService) {}

  async onModuleInit() {
    await this.connect();
  }

  async onModuleDestroy() {
    if (this.tokenRefreshInterval) {
      clearInterval(this.tokenRefreshInterval);
    }
    await this.disconnect();
  }

  private async getCredential() {
    const credentialType = this.configService.get('redis-entraid.entraId.credentialType');

    if (credentialType === 'managed-identity') {
      // Production: Use Managed Identity
      this.logger.log('Using Managed Identity credential');
      return new ManagedIdentityCredential();
    }

    if (credentialType === 'azure-cli') {
      // Development: Use Azure CLI
      this.logger.log('Using Azure CLI credential');
      return new AzureCliCredential();
    }

    // Default: Try multiple credential types
    this.logger.log('Using DefaultAzureCredential');
    return new DefaultAzureCredential();
  }

  private async getAccessToken(): Promise<string> {
    const credential = await this.getCredential();
    const scope = this.configService.get('redis-entraid.entraId.scope');

    try {
      const tokenResponse = await credential.getToken(scope);

      if (!tokenResponse?.token) {
        throw new Error('Failed to get access token from Entra ID');
      }

      this.logger.log('Successfully retrieved Entra ID access token');
      return tokenResponse.token;
    } catch (error) {
      this.logger.error(`Failed to get Entra ID token: ${error.message}`);
      throw error;
    }
  }

  private async refreshToken() {
    try {
      this.currentToken = await this.getAccessToken();

      // Update the client's password (token)
      if (this.client?.isOpen) {
        // Re-authenticate with new token
        await this.client.auth({
          username: 'default',
          password: this.currentToken,
        });
        this.logger.log('Token refreshed successfully');
      }
    } catch (error) {
      this.logger.error(`Token refresh failed: ${error.message}`);
      // Attempt to reconnect
      await this.reconnect();
    }
  }

  private async connect() {
    const host = this.configService.get('redis-entraid.host');
    const port = this.configService.get('redis-entraid.port');
    const useTls = this.configService.get('redis-entraid.tls');
    const entraEnabled = this.configService.get('redis-entraid.entraId.enabled');

    try {
      if (!entraEnabled) {
        throw new Error('Entra ID authentication is not enabled');
      }

      // Get initial access token
      this.currentToken = await this.getAccessToken();

      // Create Redis client
      this.client = createClient({
        socket: {
          host,
          port,
          tls: useTls,
        },
        // Use token as password
        username: 'default',
        password: this.currentToken,
      });

      // Error handling
      this.client.on('error', (err) => {
        this.logger.error(`Redis client error: ${err.message}`);
      });

      this.client.on('ready', () => {
        this.logger.log('Redis client is ready');
      });

      this.client.on('reconnecting', () => {
        this.logger.warn('Redis client is reconnecting...');
      });

      await this.client.connect();
      this.logger.log(`Connected to Redis at ${host}:${port} using Entra ID`);

      // Set up token refresh (refresh every 30 minutes)
      this.tokenRefreshInterval = setInterval(() => this.refreshToken(), 30 * 60 * 1000);
    } catch (error) {
      this.logger.error(`Failed to connect to Redis: ${error.message}`);
      throw error;
    }
  }

  private async reconnect() {
    this.logger.warn('Attempting to reconnect to Redis...');

    try {
      await this.disconnect();
      await this.connect();
      this.logger.log('Reconnection successful');
    } catch (error) {
      this.logger.error(`Reconnection failed: ${error.message}`);
      throw error;
    }
  }

  private async disconnect() {
    if (this.client?.isOpen) {
      await this.client.quit();
      this.logger.log('Disconnected from Redis');
    }
  }

  // Public methods for Redis operations
  getClient(): RedisClientType {
    if (!this.client?.isOpen) {
      throw new Error('Redis client is not connected');
    }
    return this.client;
  }

  async get(key: string): Promise<string | null> {
    return this.getClient().get(key);
  }

  async set(key: string, value: string, ttl?: number): Promise<void> {
    const client = this.getClient();
    if (ttl) {
      await client.setEx(key, ttl, value);
    } else {
      await client.set(key, value);
    }
  }

  async del(key: string): Promise<void> {
    await this.getClient().del(key);
  }

  async exists(key: string): Promise<boolean> {
    const result = await this.getClient().exists(key);
    return result === 1;
  }

  async hGetAll(key: string): Promise<Record<string, string>> {
    return this.getClient().hGetAll(key);
  }

  async hSet(key: string, field: string, value: string): Promise<void> {
    await this.getClient().hSet(key, field, value);
  }
}

2. Multi-Environment Configuration with Service Principals​

Handle different authentication methods for local development, staging, and production.

// redis-auth.factory.ts
import {
  TokenCredential,
  DefaultAzureCredential,
  ClientSecretCredential,
  ManagedIdentityCredential,
  AzureCliCredential,
} from '@azure/identity';
import { Logger } from '@nestjs/common';

export enum CredentialType {
  DEFAULT = 'default',
  MANAGED_IDENTITY = 'managed-identity',
  SERVICE_PRINCIPAL = 'service-principal',
  AZURE_CLI = 'azure-cli',
}

export interface EntraIdConfig {
  credentialType: CredentialType;
  tenantId?: string;
  clientId?: string;
  clientSecret?: string;
  managedIdentityClientId?: string;
}

export class RedisAuthFactory {
  private static readonly logger = new Logger(RedisAuthFactory.name);

  static createCredential(config: EntraIdConfig): TokenCredential {
    switch (config.credentialType) {
      case CredentialType.MANAGED_IDENTITY:
        return this.createManagedIdentityCredential(config);

      case CredentialType.SERVICE_PRINCIPAL:
        return this.createServicePrincipalCredential(config);

      case CredentialType.AZURE_CLI:
        return this.createAzureCliCredential();

      case CredentialType.DEFAULT:
      default:
        return this.createDefaultCredential();
    }
  }

  private static createManagedIdentityCredential(config: EntraIdConfig): ManagedIdentityCredential {
    this.logger.log('Creating Managed Identity credential');

    if (config.managedIdentityClientId) {
      return new ManagedIdentityCredential({
        clientId: config.managedIdentityClientId,
      });
    }

    return new ManagedIdentityCredential();
  }

  private static createServicePrincipalCredential(config: EntraIdConfig): ClientSecretCredential {
    this.logger.log('Creating Service Principal credential');

    const { tenantId, clientId, clientSecret } = config;

    if (!tenantId || !clientId || !clientSecret) {
      throw new Error('Service Principal requires tenantId, clientId, and clientSecret');
    }

    return new ClientSecretCredential(tenantId, clientId, clientSecret);
  }

  private static createAzureCliCredential(): AzureCliCredential {
    this.logger.log('Creating Azure CLI credential');
    return new AzureCliCredential();
  }

  private static createDefaultCredential(): DefaultAzureCredential {
    this.logger.log('Creating DefaultAzureCredential (tries multiple auth methods)');
    return new DefaultAzureCredential();
  }
}

// enhanced-redis-entraid.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { createClient, RedisClientType } from 'redis';
import { RedisAuthFactory, CredentialType, EntraIdConfig } from './redis-auth.factory';

@Injectable()
export class EnhancedRedisEntraIdService {
  private readonly logger = new Logger(EnhancedRedisEntraIdService.name);
  private client: RedisClientType;
  private tokenRefreshInterval: NodeJS.Timeout;

  constructor(private configService: ConfigService) {}

  async connect() {
    const host = this.configService.get<string>('REDIS_HOST');
    const port = this.configService.get<number>('REDIS_PORT');
    const useTls = this.configService.get<boolean>('REDIS_TLS');
    const entraEnabled = this.configService.get<boolean>('REDIS_ENTRA_ENABLED');

    if (!entraEnabled) {
      throw new Error('Entra ID authentication is required');
    }

    // Build credential configuration
    const credentialConfig: EntraIdConfig = {
      credentialType: this.configService.get<CredentialType>('REDIS_ENTRA_CREDENTIAL_TYPE', CredentialType.DEFAULT),
      tenantId: this.configService.get<string>('REDIS_ENTRA_TENANT_ID'),
      clientId: this.configService.get<string>('REDIS_ENTRA_CLIENT_ID'),
      clientSecret: this.configService.get<string>('REDIS_ENTRA_CLIENT_SECRET'),
      managedIdentityClientId: this.configService.get<string>('REDIS_ENTRA_MANAGED_IDENTITY_CLIENT_ID'),
    };

    // Create credential using factory
    const credential = RedisAuthFactory.createCredential(credentialConfig);

    // Get access token
    const scope = 'https://redis.azure.com/.default';
    const tokenResponse = await credential.getToken(scope);

    if (!tokenResponse?.token) {
      throw new Error('Failed to obtain Entra ID token');
    }

    // Create Redis client
    this.client = createClient({
      socket: {
        host,
        port,
        tls: useTls,
        // Add connection timeout
        connectTimeout: 10000,
      },
      username: 'default',
      password: tokenResponse.token,
      // Add command timeout
      commandsQueueMaxLength: 1000,
    });

    await this.client.connect();
    this.logger.log(`Connected to Redis using ${credentialConfig.credentialType} authentication`);

    // Setup token refresh
    this.setupTokenRefresh(credential, scope);
  }

  private setupTokenRefresh(credential: any, scope: string) {
    // Refresh token every 30 minutes (tokens typically valid for 1 hour)
    this.tokenRefreshInterval = setInterval(async () => {
      try {
        const tokenResponse = await credential.getToken(scope);

        if (tokenResponse?.token && this.client?.isOpen) {
          await this.client.auth({
            username: 'default',
            password: tokenResponse.token,
          });
          this.logger.log('Token refreshed successfully');
        }
      } catch (error) {
        this.logger.error(`Token refresh failed: ${error.message}`);
      }
    }, 30 * 60 * 1000);
  }

  getClient(): RedisClientType {
    return this.client;
  }
}

3. Session Store with Entra ID Authentication​

Secure session management using Redis with Entra ID.

// session-store.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { RedisEntraIdService } from './redis-entraid.service';

export interface Session {
  userId: string;
  email: string;
  roles: string[];
  createdAt: number;
  lastActivity: number;
  metadata?: Record<string, any>;
}

@Injectable()
export class SessionStoreService {
  private readonly logger = new Logger(SessionStoreService.name);
  private readonly SESSION_PREFIX = 'session:';
  private readonly USER_SESSIONS_PREFIX = 'user:sessions:';
  private readonly SESSION_TTL = 3600; // 1 hour

  constructor(private redisEntraId: RedisEntraIdService) {}

  private getSessionKey(sessionId: string): string {
    return `${this.SESSION_PREFIX}${sessionId}`;
  }

  private getUserSessionsKey(userId: string): string {
    return `${this.USER_SESSIONS_PREFIX}${userId}`;
  }

  async createSession(sessionId: string, session: Session): Promise<void> {
    const client = this.redisEntraId.getClient();
    const sessionKey = this.getSessionKey(sessionId);
    const userSessionsKey = this.getUserSessionsKey(session.userId);

    try {
      // Use pipeline for atomic operations
      const pipeline = client.multi();

      // Store session data
      pipeline.hSet(sessionKey, {
        userId: session.userId,
        email: session.email,
        roles: JSON.stringify(session.roles),
        createdAt: session.createdAt.toString(),
        lastActivity: session.lastActivity.toString(),
        metadata: session.metadata ? JSON.stringify(session.metadata) : '{}',
      });

      // Set TTL on session
      pipeline.expire(sessionKey, this.SESSION_TTL);

      // Add to user's session set
      pipeline.sAdd(userSessionsKey, sessionId);
      pipeline.expire(userSessionsKey, this.SESSION_TTL);

      await pipeline.exec();

      this.logger.log(`Session created for user ${session.userId}`);
    } catch (error) {
      this.logger.error(`Failed to create session: ${error.message}`);
      throw error;
    }
  }

  async getSession(sessionId: string): Promise<Session | null> {
    try {
      const sessionKey = this.getSessionKey(sessionId);
      const data = await this.redisEntraId.hGetAll(sessionKey);

      if (!data || Object.keys(data).length === 0) {
        return null;
      }

      // Update last activity
      await this.updateSessionActivity(sessionId);

      return {
        userId: data.userId,
        email: data.email,
        roles: JSON.parse(data.roles),
        createdAt: parseInt(data.createdAt, 10),
        lastActivity: parseInt(data.lastActivity, 10),
        metadata: data.metadata ? JSON.parse(data.metadata) : {},
      };
    } catch (error) {
      this.logger.error(`Failed to get session: ${error.message}`);
      return null;
    }
  }

  async updateSessionActivity(sessionId: string): Promise<void> {
    const client = this.redisEntraId.getClient();
    const sessionKey = this.getSessionKey(sessionId);
    const now = Date.now();

    try {
      await client.hSet(sessionKey, 'lastActivity', now.toString());
      await client.expire(sessionKey, this.SESSION_TTL);
    } catch (error) {
      this.logger.error(`Failed to update session activity: ${error.message}`);
    }
  }

  async deleteSession(sessionId: string): Promise<void> {
    const client = this.redisEntraId.getClient();

    try {
      // Get session to find userId
      const session = await this.getSession(sessionId);

      if (session) {
        const sessionKey = this.getSessionKey(sessionId);
        const userSessionsKey = this.getUserSessionsKey(session.userId);

        const pipeline = client.multi();
        pipeline.del(sessionKey);
        pipeline.sRem(userSessionsKey, sessionId);
        await pipeline.exec();

        this.logger.log(`Session ${sessionId} deleted`);
      }
    } catch (error) {
      this.logger.error(`Failed to delete session: ${error.message}`);
      throw error;
    }
  }

  async getUserSessions(userId: string): Promise<string[]> {
    const client = this.redisEntraId.getClient();
    const userSessionsKey = this.getUserSessionsKey(userId);

    try {
      return await client.sMembers(userSessionsKey);
    } catch (error) {
      this.logger.error(`Failed to get user sessions: ${error.message}`);
      return [];
    }
  }

  async deleteAllUserSessions(userId: string): Promise<void> {
    const client = this.redisEntraId.getClient();

    try {
      const sessionIds = await this.getUserSessions(userId);

      if (sessionIds.length === 0) {
        return;
      }

      const pipeline = client.multi();

      // Delete all session keys
      sessionIds.forEach((sessionId) => {
        pipeline.del(this.getSessionKey(sessionId));
      });

      // Delete user sessions set
      pipeline.del(this.getUserSessionsKey(userId));

      await pipeline.exec();

      this.logger.log(`Deleted ${sessionIds.length} sessions for user ${userId}`);
    } catch (error) {
      this.logger.error(`Failed to delete user sessions: ${error.message}`);
      throw error;
    }
  }
}

Express middleware integration:

// session.middleware.ts
import { Injectable, NestMiddleware, UnauthorizedException } from '@nestjs/common';
import { Request, Response, NextFunction } from 'express';
import { SessionStoreService } from './session-store.service';

@Injectable()
export class SessionMiddleware implements NestMiddleware {
  constructor(private sessionStore: SessionStoreService) {}

  async use(req: Request, res: Response, next: NextFunction) {
    const sessionId = req.headers['x-session-id'] as string;

    if (!sessionId) {
      throw new UnauthorizedException('Session ID required');
    }

    const session = await this.sessionStore.getSession(sessionId);

    if (!session) {
      throw new UnauthorizedException('Invalid or expired session');
    }

    // Attach session to request
    req['session'] = session;

    next();
  }
}

4. Distributed Cache with RBAC​

Implement role-based access control for cached resources using Entra ID.

// rbac-cache.service.ts
import { Injectable, ForbiddenException, Logger } from '@nestjs/common';
import { RedisEntraIdService } from './redis-entraid.service';

export enum CachePermission {
  READ = 'read',
  WRITE = 'write',
  DELETE = 'delete',
  ADMIN = 'admin',
}

export interface CacheEntry {
  data: any;
  permissions: {
    [key: string]: CachePermission[]; // userId -> permissions
  };
  owner: string;
  createdAt: number;
  updatedAt: number;
  metadata?: Record<string, any>;
}

@Injectable()
export class RBACCacheService {
  private readonly logger = new Logger(RBACCacheService.name);
  private readonly CACHE_PREFIX = 'rbac:cache:';

  constructor(private redisEntraId: RedisEntraIdService) {}

  private getCacheKey(key: string): string {
    return `${this.CACHE_PREFIX}${key}`;
  }

  private hasPermission(entry: CacheEntry, userId: string, requiredPermission: CachePermission): boolean {
    // Owner has all permissions
    if (entry.owner === userId) {
      return true;
    }

    // Check if user has the required permission
    const userPermissions = entry.permissions[userId] || [];

    // Admin permission grants all access
    if (userPermissions.includes(CachePermission.ADMIN)) {
      return true;
    }

    return userPermissions.includes(requiredPermission);
  }

  async set(key: string, data: any, userId: string, ttl?: number): Promise<void> {
    const cacheKey = this.getCacheKey(key);
    const now = Date.now();

    const entry: CacheEntry = {
      data,
      permissions: {
        [userId]: [CachePermission.ADMIN], // Owner gets admin permission
      },
      owner: userId,
      createdAt: now,
      updatedAt: now,
    };

    try {
      const client = this.redisEntraId.getClient();
      await client.set(cacheKey, JSON.stringify(entry));

      if (ttl) {
        await client.expire(cacheKey, ttl);
      }

      this.logger.log(`Cache entry created: ${key} by ${userId}`);
    } catch (error) {
      this.logger.error(`Failed to set cache: ${error.message}`);
      throw error;
    }
  }

  async get(key: string, userId: string): Promise<any> {
    const cacheKey = this.getCacheKey(key);

    try {
      const raw = await this.redisEntraId.get(cacheKey);

      if (!raw) {
        return null;
      }

      const entry: CacheEntry = JSON.parse(raw);

      // Check read permission
      if (!this.hasPermission(entry, userId, CachePermission.READ)) {
        throw new ForbiddenException(`User ${userId} does not have read permission for ${key}`);
      }

      return entry.data;
    } catch (error) {
      if (error instanceof ForbiddenException) {
        throw error;
      }
      this.logger.error(`Failed to get cache: ${error.message}`);
      return null;
    }
  }

  async update(key: string, data: any, userId: string): Promise<void> {
    const cacheKey = this.getCacheKey(key);

    try {
      const raw = await this.redisEntraId.get(cacheKey);

      if (!raw) {
        throw new Error(`Cache entry not found: ${key}`);
      }

      const entry: CacheEntry = JSON.parse(raw);

      // Check write permission
      if (!this.hasPermission(entry, userId, CachePermission.WRITE)) {
        throw new ForbiddenException(`User ${userId} does not have write permission for ${key}`);
      }

      // Update entry
      entry.data = data;
      entry.updatedAt = Date.now();

      await this.redisEntraId.set(cacheKey, JSON.stringify(entry));

      this.logger.log(`Cache entry updated: ${key} by ${userId}`);
    } catch (error) {
      this.logger.error(`Failed to update cache: ${error.message}`);
      throw error;
    }
  }

  async delete(key: string, userId: string): Promise<void> {
    const cacheKey = this.getCacheKey(key);

    try {
      const raw = await this.redisEntraId.get(cacheKey);

      if (!raw) {
        return; // Already deleted
      }

      const entry: CacheEntry = JSON.parse(raw);

      // Check delete permission
      if (!this.hasPermission(entry, userId, CachePermission.DELETE)) {
        throw new ForbiddenException(`User ${userId} does not have delete permission for ${key}`);
      }

      await this.redisEntraId.del(cacheKey);

      this.logger.log(`Cache entry deleted: ${key} by ${userId}`);
    } catch (error) {
      this.logger.error(`Failed to delete cache: ${error.message}`);
      throw error;
    }
  }

  async grantPermission(
    key: string,
    ownerId: string,
    targetUserId: string,
    permissions: CachePermission[],
  ): Promise<void> {
    const cacheKey = this.getCacheKey(key);

    try {
      const raw = await this.redisEntraId.get(cacheKey);

      if (!raw) {
        throw new Error(`Cache entry not found: ${key}`);
      }

      const entry: CacheEntry = JSON.parse(raw);

      // Only owner or admin can grant permissions
      if (!this.hasPermission(entry, ownerId, CachePermission.ADMIN)) {
        throw new ForbiddenException(`User ${ownerId} cannot grant permissions for ${key}`);
      }

      // Grant permissions
      entry.permissions[targetUserId] = permissions;
      entry.updatedAt = Date.now();

      await this.redisEntraId.set(cacheKey, JSON.stringify(entry));

      this.logger.log(`Permissions granted to ${targetUserId} for ${key}: ${permissions.join(', ')}`);
    } catch (error) {
      this.logger.error(`Failed to grant permissions: ${error.message}`);
      throw error;
    }
  }

  async revokePermission(key: string, ownerId: string, targetUserId: string): Promise<void> {
    const cacheKey = this.getCacheKey(key);

    try {
      const raw = await this.redisEntraId.get(cacheKey);

      if (!raw) {
        throw new Error(`Cache entry not found: ${key}`);
      }

      const entry: CacheEntry = JSON.parse(raw);

      // Only owner or admin can revoke permissions
      if (!this.hasPermission(entry, ownerId, CachePermission.ADMIN)) {
        throw new ForbiddenException(`User ${ownerId} cannot revoke permissions for ${key}`);
      }

      // Cannot revoke owner's permissions
      if (targetUserId === entry.owner) {
        throw new ForbiddenException('Cannot revoke owner permissions');
      }

      // Revoke permissions
      delete entry.permissions[targetUserId];
      entry.updatedAt = Date.now();

      await this.redisEntraId.set(cacheKey, JSON.stringify(entry));

      this.logger.log(`Permissions revoked from ${targetUserId} for ${key}`);
    } catch (error) {
      this.logger.error(`Failed to revoke permissions: ${error.message}`);
      throw error;
    }
  }
}

Best Practices​

1. Token Management​

// token-manager.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { TokenCredential, AccessToken } from '@azure/identity';

@Injectable()
export class TokenManager {
  private readonly logger = new Logger(TokenManager.name);
  private tokenCache: Map<string, { token: string; expiresOn: number }> = new Map();

  async getToken(credential: TokenCredential, scope: string, forceRefresh = false): Promise<string> {
    const cacheKey = `${scope}`;

    // Check cache
    if (!forceRefresh && this.tokenCache.has(cacheKey)) {
      const cached = this.tokenCache.get(cacheKey);
      const now = Date.now();

      // Return cached token if valid for at least 5 more minutes
      if (cached.expiresOn - now > 5 * 60 * 1000) {
        this.logger.debug('Using cached token');
        return cached.token;
      }
    }

    // Get new token
    this.logger.debug('Fetching new token from Entra ID');
    const tokenResponse: AccessToken = await credential.getToken(scope);

    if (!tokenResponse?.token) {
      throw new Error('Failed to obtain access token');
    }

    // Cache token
    this.tokenCache.set(cacheKey, {
      token: tokenResponse.token,
      expiresOn: tokenResponse.expiresOnTimestamp,
    });

    return tokenResponse.token;
  }

  clearCache(): void {
    this.tokenCache.clear();
    this.logger.log('Token cache cleared');
  }
}

// redis-health.indicator.ts
import { Injectable } from '@nestjs/common';
import { HealthIndicator, HealthIndicatorResult, HealthCheckError } from '@nestjs/terminus';
import { RedisEntraIdService } from './redis-entraid.service';

@Injectable()
export class RedisHealthIndicator extends HealthIndicator {
  constructor(private redisEntraId: RedisEntraIdService) {
    super();
  }

  async isHealthy(key: string): Promise<HealthIndicatorResult> {
    try {
      const client = this.redisEntraId.getClient();

      // Test Redis connection with PING
      const startTime = Date.now();
      await client.ping();
      const responseTime = Date.now() - startTime;

      // Test authentication by setting a test key
      const testKey = `health:check:${Date.now()}`;
      await client.setEx(testKey, 10, 'ok');
      await client.del(testKey);

      return this.getStatus(key, true, {
        responseTime: `${responseTime}ms`,
        authentication: 'valid',
      });
    } catch (error) {
      throw new HealthCheckError(
        'Redis health check failed',
        this.getStatus(key, false, {
          error: error.message,
        }),
      );
    }
  }
}

3. Monitoring and Logging​

// redis-metrics.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { RedisEntraIdService } from './redis-entraid.service';

export interface RedisMetrics {
  totalOperations: number;
  successfulOperations: number;
  failedOperations: number;
  averageLatency: number;
  tokenRefreshCount: number;
  lastTokenRefresh: number;
}

@Injectable()
export class RedisMetricsService {
  private readonly logger = new Logger(RedisMetricsService.name);
  private metrics: RedisMetrics = {
    totalOperations: 0,
    successfulOperations: 0,
    failedOperations: 0,
    averageLatency: 0,
    tokenRefreshCount: 0,
    lastTokenRefresh: 0,
  };

  constructor(private redisEntraId: RedisEntraIdService) {}

  async executeWithMetrics<T>(operation: () => Promise<T>, operationName: string): Promise<T> {
    const startTime = Date.now();
    this.metrics.totalOperations++;

    try {
      const result = await operation();
      const latency = Date.now() - startTime;

      this.metrics.successfulOperations++;
      this.updateAverageLatency(latency);

      this.logger.debug(`${operationName} completed in ${latency}ms`);

      return result;
    } catch (error) {
      this.metrics.failedOperations++;
      this.logger.error(`${operationName} failed: ${error.message}`);
      throw error;
    }
  }

  private updateAverageLatency(latency: number): void {
    const total = this.metrics.averageLatency * (this.metrics.successfulOperations - 1);
    this.metrics.averageLatency = (total + latency) / this.metrics.successfulOperations;
  }

  recordTokenRefresh(): void {
    this.metrics.tokenRefreshCount++;
    this.metrics.lastTokenRefresh = Date.now();
  }

  getMetrics(): RedisMetrics {
    return { ...this.metrics };
  }

  getSuccessRate(): number {
    if (this.metrics.totalOperations === 0) {
      return 100;
    }
    return (this.metrics.successfulOperations / this.metrics.totalOperations) * 100;
  }

  resetMetrics(): void {
    this.metrics = {
      totalOperations: 0,
      successfulOperations: 0,
      failedOperations: 0,
      averageLatency: 0,
      tokenRefreshCount: 0,
      lastTokenRefresh: 0,
    };
  }
}

2. Token Refresh Failures

// Implement exponential backoff for token refresh
private async refreshTokenWithRetry(maxRetries = 3): Promise<void> {
  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      await this.refreshToken();
      return;
    } catch (error) {
      const delay = Math.pow(2, attempt) * 1000; // Exponential backoff
      this.logger.warn(`Token refresh attempt ${attempt} failed. Retrying in ${delay}ms...`);

      if (attempt === maxRetries) {
        throw error;
      }

      await new Promise(resolve => setTimeout(resolve, delay));
    }
  }
}

Security Considerations​

  1. Never log tokens: Always redact tokens from logs
  2. Use TLS: Always enable TLS for Redis connections in production
  3. Rotate managed identities: Regularly review and rotate credentials
  4. Implement rate limiting: Prevent token exhaustion attacks
  5. Monitor authentication failures: Alert on suspicious patterns
  6. Use least privilege: Grant minimum required permissions
