WebRTC Implementation Guide

Overview​

WebRTC (Web Real-Time Communication) is an open-source project that enables real-time communication capabilities directly in web browsers and mobile applications. This guide covers implementing WebRTC for video calls, audio streaming, and data channels in a NestJS microservices architecture.

2. WebRTC Gateway Implementation​

// services/webrtc-service/src/webrtc/webrtc.gateway.ts
import {
  WebSocketGateway,
  WebSocketServer,
  SubscribeMessage,
  OnGatewayConnection,
  OnGatewayDisconnect,
  ConnectedSocket,
  MessageBody,
} from '@nestjs/websockets';
import { Server, Socket } from 'socket.io';
import { Logger, UseGuards } from '@nestjs/common';
import { WebRtcService } from './webrtc.service';
import { RoomService } from './room.service';
import { WsJwtGuard } from '../auth/ws-jwt.guard';

interface SignalData {
  roomId: string;
  userId: string;
  signalData: any;
  type: 'offer' | 'answer' | 'ice-candidate';
}

@WebSocketGateway({
  cors: {
    origin: process.env.CORS_ORIGIN || '*',
    credentials: true,
  },
  namespace: '/webrtc',
})
export class WebRtcGateway implements OnGatewayConnection, OnGatewayDisconnect {
  @WebSocketServer()
  server: Server;

  private readonly logger = new Logger(WebRtcGateway.name);

  constructor(private readonly webrtcService: WebRtcService, private readonly roomService: RoomService) {}

  async handleConnection(client: Socket) {
    try {
      const userId = this.extractUserId(client);
      this.logger.log(`Client connected: ${client.id}, User: ${userId}`);

      // Store client connection
      await this.webrtcService.addConnection(userId, client.id);
    } catch (error) {
      this.logger.error(`Connection error: ${error.message}`);
      client.disconnect();
    }
  }

  async handleDisconnect(client: Socket) {
    try {
      const userId = this.extractUserId(client);
      this.logger.log(`Client disconnected: ${client.id}, User: ${userId}`);

      // Clean up connection and notify room participants
      const rooms = await this.webrtcService.getUserRooms(userId);

      for (const room of rooms) {
        await this.handleLeaveRoom(client, { roomId: room.id });
      }

      await this.webrtcService.removeConnection(userId, client.id);
    } catch (error) {
      this.logger.error(`Disconnect error: ${error.message}`);
    }
  }

  @UseGuards(WsJwtGuard)
  @SubscribeMessage('join-room')
  async handleJoinRoom(
    @ConnectedSocket() client: Socket,
    @MessageBody() data: { roomId: string; userId: string; userName: string },
  ) {
    try {
      const { roomId, userId, userName } = data;

      // Join socket room
      await client.join(roomId);

      // Add participant to room
      const room = await this.roomService.addParticipant(roomId, userId, userName);

      // Get existing participants
      const participants = await this.roomService.getParticipants(roomId);

      // Notify existing participants about new user
      client.to(roomId).emit('user-joined', {
        userId,
        userName,
        participants: participants.filter((p) => p.userId !== userId),
      });

      // Send current participants to new user
      client.emit('room-joined', {
        roomId,
        participants: participants.filter((p) => p.userId !== userId),
      });

      this.logger.log(`User ${userId} joined room ${roomId}`);

      return { success: true, room };
    } catch (error) {
      this.logger.error(`Join room error: ${error.message}`);
      return { success: false, error: error.message };
    }
  }

  @UseGuards(WsJwtGuard)
  @SubscribeMessage('leave-room')
  async handleLeaveRoom(@ConnectedSocket() client: Socket, @MessageBody() data: { roomId: string }) {
    try {
      const userId = this.extractUserId(client);
      const { roomId } = data;

      // Leave socket room
      await client.leave(roomId);

      // Remove participant from room
      await this.roomService.removeParticipant(roomId, userId);

      // Notify other participants
      client.to(roomId).emit('user-left', { userId });

      this.logger.log(`User ${userId} left room ${roomId}`);

      return { success: true };
    } catch (error) {
      this.logger.error(`Leave room error: ${error.message}`);
      return { success: false, error: error.message };
    }
  }

  @UseGuards(WsJwtGuard)
  @SubscribeMessage('signal')
  async handleSignal(@ConnectedSocket() client: Socket, @MessageBody() data: SignalData) {
    try {
      const { roomId, userId, signalData, type } = data;

      this.logger.debug(`Signal received: ${type} from ${userId} in room ${roomId}`);

      // Forward signal to specific user or broadcast to room
      if (signalData.targetUserId) {
        // Send to specific user
        const targetSocketId = await this.webrtcService.getConnectionId(signalData.targetUserId);
        if (targetSocketId) {
          this.server.to(targetSocketId).emit('signal', {
            type,
            signalData,
            fromUserId: userId,
          });
        }
      } else {
        // Broadcast to room (excluding sender)
        client.to(roomId).emit('signal', {
          type,
          signalData,
          fromUserId: userId,
        });
      }

      return { success: true };
    } catch (error) {
      this.logger.error(`Signal error: ${error.message}`);
      return { success: false, error: error.message };
    }
  }

  @UseGuards(WsJwtGuard)
  @SubscribeMessage('offer')
  async handleOffer(
    @ConnectedSocket() client: Socket,
    @MessageBody() data: { roomId: string; targetUserId: string; offer: RTCSessionDescriptionInit },
  ) {
    try {
      const userId = this.extractUserId(client);
      const { roomId, targetUserId, offer } = data;

      this.logger.debug(`Offer from ${userId} to ${targetUserId} in room ${roomId}`);

      const targetSocketId = await this.webrtcService.getConnectionId(targetUserId);
      if (targetSocketId) {
        this.server.to(targetSocketId).emit('offer', {
          fromUserId: userId,
          offer,
        });
      }

      return { success: true };
    } catch (error) {
      this.logger.error(`Offer error: ${error.message}`);
      return { success: false, error: error.message };
    }
  }

  @UseGuards(WsJwtGuard)
  @SubscribeMessage('answer')
  async handleAnswer(
    @ConnectedSocket() client: Socket,
    @MessageBody() data: { roomId: string; targetUserId: string; answer: RTCSessionDescriptionInit },
  ) {
    try {
      const userId = this.extractUserId(client);
      const { roomId, targetUserId, answer } = data;

      this.logger.debug(`Answer from ${userId} to ${targetUserId} in room ${roomId}`);

      const targetSocketId = await this.webrtcService.getConnectionId(targetUserId);
      if (targetSocketId) {
        this.server.to(targetSocketId).emit('answer', {
          fromUserId: userId,
          answer,
        });
      }

      return { success: true };
    } catch (error) {
      this.logger.error(`Answer error: ${error.message}`);
      return { success: false, error: error.message };
    }
  }

  @UseGuards(WsJwtGuard)
  @SubscribeMessage('ice-candidate')
  async handleIceCandidate(
    @ConnectedSocket() client: Socket,
    @MessageBody() data: { roomId: string; targetUserId: string; candidate: RTCIceCandidateInit },
  ) {
    try {
      const userId = this.extractUserId(client);
      const { roomId, targetUserId, candidate } = data;

      const targetSocketId = await this.webrtcService.getConnectionId(targetUserId);
      if (targetSocketId) {
        this.server.to(targetSocketId).emit('ice-candidate', {
          fromUserId: userId,
          candidate,
        });
      }

      return { success: true };
    } catch (error) {
      this.logger.error(`ICE candidate error: ${error.message}`);
      return { success: false, error: error.message };
    }
  }

  @UseGuards(WsJwtGuard)
  @SubscribeMessage('media-state')
  async handleMediaState(
    @ConnectedSocket() client: Socket,
    @MessageBody() data: { roomId: string; audio: boolean; video: boolean },
  ) {
    try {
      const userId = this.extractUserId(client);
      const { roomId, audio, video } = data;

      // Update participant media state
      await this.roomService.updateParticipantMedia(roomId, userId, { audio, video });

      // Notify other participants
      client.to(roomId).emit('media-state-changed', {
        userId,
        audio,
        video,
      });

      return { success: true };
    } catch (error) {
      this.logger.error(`Media state error: ${error.message}`);
      return { success: false, error: error.message };
    }
  }

  @UseGuards(WsJwtGuard)
  @SubscribeMessage('screen-share')
  async handleScreenShare(
    @ConnectedSocket() client: Socket,
    @MessageBody() data: { roomId: string; isSharing: boolean },
  ) {
    try {
      const userId = this.extractUserId(client);
      const { roomId, isSharing } = data;

      // Notify other participants
      client.to(roomId).emit('screen-share-changed', {
        userId,
        isSharing,
      });

      return { success: true };
    } catch (error) {
      this.logger.error(`Screen share error: ${error.message}`);
      return { success: false, error: error.message };
    }
  }

  private extractUserId(client: Socket): string {
    // Extract user ID from JWT token or socket handshake
    return client.handshake.auth?.userId || (client.handshake.query?.userId as string);
  }
}

3. WebRTC Service​

// services/webrtc-service/src/webrtc/webrtc.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { InjectRedis } from '@nestjs-modules/ioredis';
import { Redis } from 'ioredis';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { Call } from './entities/call.entity';

@Injectable()
export class WebRtcService {
  private readonly logger = new Logger(WebRtcService.name);
  private readonly CONNECTION_PREFIX = 'webrtc:connection:';
  private readonly ROOM_PREFIX = 'webrtc:room:';

  constructor(
    @InjectRedis() private readonly redis: Redis,
    @InjectRepository(Call)
    private readonly callRepository: Repository<Call>,
  ) {}

  async addConnection(userId: string, socketId: string): Promise<void> {
    const key = `${this.CONNECTION_PREFIX}${userId}`;
    await this.redis.set(key, socketId, 'EX', 3600); // 1 hour expiry
  }

  async removeConnection(userId: string, socketId: string): Promise<void> {
    const key = `${this.CONNECTION_PREFIX}${userId}`;
    await this.redis.del(key);
  }

  async getConnectionId(userId: string): Promise<string | null> {
    const key = `${this.CONNECTION_PREFIX}${userId}`;
    return await this.redis.get(key);
  }

  async getUserRooms(userId: string): Promise<any[]> {
    const pattern = `${this.ROOM_PREFIX}*`;
    const keys = await this.redis.keys(pattern);
    const rooms = [];

    for (const key of keys) {
      const members = await this.redis.smembers(key);
      if (members.includes(userId)) {
        const roomId = key.replace(this.ROOM_PREFIX, '');
        rooms.push({ id: roomId, members });
      }
    }

    return rooms;
  }

  async createCall(callData: {
    roomId: string;
    callerId: string;
    calleeId: string;
    type: 'audio' | 'video';
  }): Promise<Call> {
    const call = this.callRepository.create({
      ...callData,
      status: 'initiated',
      startedAt: new Date(),
    });

    return await this.callRepository.save(call);
  }

  async updateCallStatus(
    callId: string,
    status: 'initiated' | 'ringing' | 'connected' | 'ended' | 'failed',
  ): Promise<void> {
    await this.callRepository.update(callId, { status });

    if (status === 'ended') {
      await this.callRepository.update(callId, { endedAt: new Date() });
    }
  }

  async getCallHistory(userId: string, limit: number = 50): Promise<Call[]> {
    return await this.callRepository.find({
      where: [{ callerId: userId }, { calleeId: userId }],
      order: { startedAt: 'DESC' },
      take: limit,
    });
  }
}

4. Room Service​

// services/webrtc-service/src/webrtc/room.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { InjectRedis } from '@nestjs-modules/ioredis';
import { Redis } from 'ioredis';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { Room } from './entities/room.entity';
import { Participant } from './entities/participant.entity';

@Injectable()
export class RoomService {
  private readonly logger = new Logger(RoomService.name);
  private readonly ROOM_PREFIX = 'webrtc:room:';
  private readonly ROOM_PARTICIPANTS_PREFIX = 'webrtc:room:participants:';

  constructor(
    @InjectRedis() private readonly redis: Redis,
    @InjectRepository(Room)
    private readonly roomRepository: Repository<Room>,
    @InjectRepository(Participant)
    private readonly participantRepository: Repository<Participant>,
  ) {}

  async createRoom(roomData: {
    name: string;
    createdBy: string;
    maxParticipants?: number;
    isPrivate?: boolean;
  }): Promise<Room> {
    const room = this.roomRepository.create({
      ...roomData,
      maxParticipants: roomData.maxParticipants || 10,
      isPrivate: roomData.isPrivate || false,
      isActive: true,
    });

    const savedRoom = await this.roomRepository.save(room);

    // Store in Redis for quick access
    await this.redis.set(
      `${this.ROOM_PREFIX}${savedRoom.id}`,
      JSON.stringify(savedRoom),
      'EX',
      86400, // 24 hours
    );

    return savedRoom;
  }

  async addParticipant(roomId: string, userId: string, userName: string): Promise<Room> {
    const room = await this.roomRepository.findOne({ where: { id: roomId } });
    if (!room) {
      throw new Error('Room not found');
    }

    const participantCount = await this.getParticipantCount(roomId);
    if (participantCount >= room.maxParticipants) {
      throw new Error('Room is full');
    }

    // Add to Redis set
    const participantKey = `${this.ROOM_PARTICIPANTS_PREFIX}${roomId}`;
    const participantData = JSON.stringify({
      userId,
      userName,
      joinedAt: new Date(),
      audio: true,
      video: true,
    });

    await this.redis.sadd(participantKey, participantData);
    await this.redis.expire(participantKey, 86400);

    // Save to database
    const participant = this.participantRepository.create({
      roomId,
      userId,
      userName,
      joinedAt: new Date(),
    });

    await this.participantRepository.save(participant);

    return room;
  }

  async removeParticipant(roomId: string, userId: string): Promise<void> {
    const participantKey = `${this.ROOM_PARTICIPANTS_PREFIX}${roomId}`;
    const participants = await this.redis.smembers(participantKey);

    for (const participant of participants) {
      const data = JSON.parse(participant);
      if (data.userId === userId) {
        await this.redis.srem(participantKey, participant);
        break;
      }
    }

    // Update database
    await this.participantRepository.update({ roomId, userId, leftAt: null }, { leftAt: new Date() });

    // Check if room is empty
    const count = await this.getParticipantCount(roomId);
    if (count === 0) {
      await this.deactivateRoom(roomId);
    }
  }

  async getParticipants(roomId: string): Promise<any[]> {
    const participantKey = `${this.ROOM_PARTICIPANTS_PREFIX}${roomId}`;
    const participants = await this.redis.smembers(participantKey);
    return participants.map((p) => JSON.parse(p));
  }

  async getParticipantCount(roomId: string): Promise<number> {
    const participantKey = `${this.ROOM_PARTICIPANTS_PREFIX}${roomId}`;
    return await this.redis.scard(participantKey);
  }

  async updateParticipantMedia(
    roomId: string,
    userId: string,
    media: { audio?: boolean; video?: boolean },
  ): Promise<void> {
    const participantKey = `${this.ROOM_PARTICIPANTS_PREFIX}${roomId}`;
    const participants = await this.redis.smembers(participantKey);

    for (const participant of participants) {
      const data = JSON.parse(participant);
      if (data.userId === userId) {
        const updated = { ...data, ...media };
        await this.redis.srem(participantKey, participant);
        await this.redis.sadd(participantKey, JSON.stringify(updated));
        break;
      }
    }
  }

  async deactivateRoom(roomId: string): Promise<void> {
    await this.roomRepository.update(roomId, { isActive: false });
    await this.redis.del(`${this.ROOM_PREFIX}${roomId}`);
    await this.redis.del(`${this.ROOM_PARTICIPANTS_PREFIX}${roomId}`);
  }

  async getActiveRooms(): Promise<Room[]> {
    return await this.roomRepository.find({
      where: { isActive: true },
      order: { createdAt: 'DESC' },
    });
  }
}

3. ICE Server Configuration Service​

// services/webrtc-service/src/webrtc/ice-server.service.ts
import { Injectable } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';

export interface IceServer {
  urls: string | string[];
  username?: string;
  credential?: string;
  credentialType?: 'password' | 'oauth';
}

@Injectable()
export class IceServerService {
  constructor(private readonly configService: ConfigService) {}

  getIceServers(): IceServer[] {
    const stunServers = this.getStunServers();
    const turnServers = this.getTurnServers();

    return [...stunServers, ...turnServers];
  }

  private getStunServers(): IceServer[] {
    return [
      {
        urls: 'stun:stun.l.google.com:19302',
      },
      {
        urls: 'stun:stun1.l.google.com:19302',
      },
      {
        urls: this.configService.get<string>('STUN_SERVER_URL') || 'stun:your-stun-server.com:3478',
      },
    ];
  }

  private getTurnServers(): IceServer[] {
    const turnUrl = this.configService.get<string>('TURN_SERVER_URL');
    const turnUsername = this.configService.get<string>('TURN_USERNAME');
    const turnPassword = this.configService.get<string>('TURN_PASSWORD');

    if (!turnUrl || !turnUsername || !turnPassword) {
      return [];
    }

    return [
      {
        urls: [`turn:${turnUrl}:3478`, `turn:${turnUrl}:3478?transport=tcp`],
        username: turnUsername,
        credential: turnPassword,
        credentialType: 'password',
      },
      {
        urls: [`turns:${turnUrl}:5349`],
        username: turnUsername,
        credential: turnPassword,
        credentialType: 'password',
      },
    ];
  }

  // Generate time-limited TURN credentials using HMAC
  generateTurnCredentials(username: string, ttl: number = 86400): { username: string; credential: string } {
    const timestamp = Math.floor(Date.now() / 1000) + ttl;
    const tempUsername = `${timestamp}:${username}`;

    const crypto = require('crypto');
    const secret = this.configService.get<string>('TURN_SECRET');
    const hmac = crypto.createHmac('sha1', secret);
    hmac.update(tempUsername);
    const credential = hmac.digest('base64');

    return {
      username: tempUsername,
      credential,
    };
  }
}

// frontend/src/hooks/useWebRTC.ts
import { useEffect, useRef, useState, useCallback } from 'react';
import { io, Socket } from 'socket.io-client';

export interface UseWebRTCOptions {
  roomId: string;
  userId: string;
  userName: string;
  token: string;
  serverUrl: string;
  iceServers?: RTCIceServer[];
}

export interface Participant {
  userId: string;
  userName: string;
  stream?: MediaStream;
  audio: boolean;
  video: boolean;
}

export const useWebRTC = (options: UseWebRTCOptions) => {
  const [socket, setSocket] = useState<Socket | null>(null);
  const [localStream, setLocalStream] = useState<MediaStream | null>(null);
  const [participants, setParticipants] = useState<Map<string, Participant>>(new Map());
  const [isConnected, setIsConnected] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const peerConnections = useRef<Map<string, RTCPeerConnection>>(new Map());
  const { roomId, userId, userName, token, serverUrl, iceServers } = options;

  // Initialize socket connection
  useEffect(() => {
    const newSocket = io(`${serverUrl}/webrtc`, {
      auth: { token, userId },
      transports: ['websocket'],
    });

    newSocket.on('connect', () => {
      console.log('Connected to signaling server');
      setIsConnected(true);
    });

    newSocket.on('disconnect', () => {
      console.log('Disconnected from signaling server');
      setIsConnected(false);
    });

    newSocket.on('connect_error', (err) => {
      console.error('Connection error:', err);
      setError(err.message);
    });

    setSocket(newSocket);

    return () => {
      newSocket.close();
    };
  }, [serverUrl, token, userId]);

  // Get local media stream
  const getLocalStream = useCallback(async (audio: boolean = true, video: boolean = true) => {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        audio,
        video: video ? { width: 1280, height: 720 } : false,
      });
      setLocalStream(stream);
      return stream;
    } catch (err) {
      console.error('Error accessing media devices:', err);
      setError('Failed to access camera/microphone');
      throw err;
    }
  }, []);

  // Create peer connection
  const createPeerConnection = useCallback(
    (targetUserId: string): RTCPeerConnection => {
      const config: RTCConfiguration = {
        iceServers: iceServers || [{ urls: 'stun:stun.l.google.com:19302' }],
      };

      const pc = new RTCPeerConnection(config);

      // Add local stream tracks
      if (localStream) {
        localStream.getTracks().forEach((track) => {
          pc.addTrack(track, localStream);
        });
      }

      // Handle ICE candidates
      pc.onicecandidate = (event) => {
        if (event.candidate && socket) {
          socket.emit('ice-candidate', {
            roomId,
            targetUserId,
            candidate: event.candidate,
          });
        }
      };

      // Handle remote stream
      pc.ontrack = (event) => {
        console.log('Received remote track:', event.track.kind);
        setParticipants((prev) => {
          const updated = new Map(prev);
          const participant = updated.get(targetUserId);
          if (participant) {
            participant.stream = event.streams[0];
            updated.set(targetUserId, participant);
          }
          return updated;
        });
      };

      // Handle connection state changes
      pc.onconnectionstatechange = () => {
        console.log(`Connection state with ${targetUserId}:`, pc.connectionState);
        if (pc.connectionState === 'failed' || pc.connectionState === 'disconnected') {
          // Handle reconnection logic here
        }
      };

      peerConnections.current.set(targetUserId, pc);
      return pc;
    },
    [localStream, socket, roomId, iceServers],
  );

  // Join room
  const joinRoom = useCallback(async () => {
    if (!socket || !localStream) {
      console.error('Socket or local stream not ready');
      return;
    }

    socket.emit('join-room', { roomId, userId, userName });

    // Handle room joined
    socket.on('room-joined', async ({ participants: existingParticipants }) => {
      console.log('Room joined, existing participants:', existingParticipants);

      // Create peer connections and send offers to existing participants
      for (const participant of existingParticipants) {
        const pc = createPeerConnection(participant.userId);
        const offer = await pc.createOffer();
        await pc.setLocalDescription(offer);

        socket.emit('offer', {
          roomId,
          targetUserId: participant.userId,
          offer,
        });

        setParticipants((prev) =>
          new Map(prev).set(participant.userId, {
            ...participant,
            stream: undefined,
          }),
        );
      }
    });

    // Handle new user joined
    socket.on('user-joined', ({ userId: newUserId, userName: newUserName }) => {
      console.log('New user joined:', newUserId);
      setParticipants((prev) =>
        new Map(prev).set(newUserId, {
          userId: newUserId,
          userName: newUserName,
          audio: true,
          video: true,
        }),
      );
    });

    // Handle offer
    socket.on('offer', async ({ fromUserId, offer }) => {
      console.log('Received offer from:', fromUserId);
      const pc = createPeerConnection(fromUserId);
      await pc.setRemoteDescription(new RTCSessionDescription(offer));
      const answer = await pc.createAnswer();
      await pc.setLocalDescription(answer);

      socket.emit('answer', {
        roomId,
        targetUserId: fromUserId,
        answer,
      });
    });

    // Handle answer
    socket.on('answer', async ({ fromUserId, answer }) => {
      console.log('Received answer from:', fromUserId);
      const pc = peerConnections.current.get(fromUserId);
      if (pc) {
        await pc.setRemoteDescription(new RTCSessionDescription(answer));
      }
    });

    // Handle ICE candidate
    socket.on('ice-candidate', async ({ fromUserId, candidate }) => {
      const pc = peerConnections.current.get(fromUserId);
      if (pc) {
        await pc.addIceCandidate(new RTCIceCandidate(candidate));
      }
    });

    // Handle user left
    socket.on('user-left', ({ userId: leftUserId }) => {
      console.log('User left:', leftUserId);
      const pc = peerConnections.current.get(leftUserId);
      if (pc) {
        pc.close();
        peerConnections.current.delete(leftUserId);
      }
      setParticipants((prev) => {
        const updated = new Map(prev);
        updated.delete(leftUserId);
        return updated;
      });
    });

    // Handle media state changes
    socket.on('media-state-changed', ({ userId: changedUserId, audio, video }) => {
      setParticipants((prev) => {
        const updated = new Map(prev);
        const participant = updated.get(changedUserId);
        if (participant) {
          participant.audio = audio;
          participant.video = video;
          updated.set(changedUserId, participant);
        }
        return updated;
      });
    });
  }, [socket, localStream, roomId, userId, userName, createPeerConnection]);

  // Leave room
  const leaveRoom = useCallback(() => {
    if (socket) {
      socket.emit('leave-room', { roomId });
    }

    // Close all peer connections
    peerConnections.current.forEach((pc) => pc.close());
    peerConnections.current.clear();

    // Stop local stream
    if (localStream) {
      localStream.getTracks().forEach((track) => track.stop());
    }

    setParticipants(new Map());
  }, [socket, roomId, localStream]);

  // Toggle audio
  const toggleAudio = useCallback(() => {
    if (localStream) {
      const audioTrack = localStream.getAudioTracks()[0];
      if (audioTrack) {
        audioTrack.enabled = !audioTrack.enabled;
        socket?.emit('media-state', {
          roomId,
          audio: audioTrack.enabled,
          video: localStream.getVideoTracks()[0]?.enabled || false,
        });
      }
    }
  }, [localStream, socket, roomId]);

  // Toggle video
  const toggleVideo = useCallback(() => {
    if (localStream) {
      const videoTrack = localStream.getVideoTracks()[0];
      if (videoTrack) {
        videoTrack.enabled = !videoTrack.enabled;
        socket?.emit('media-state', {
          roomId,
          audio: localStream.getAudioTracks()[0]?.enabled || false,
          video: videoTrack.enabled,
        });
      }
    }
  }, [localStream, socket, roomId]);

  // Screen sharing
  const startScreenShare = useCallback(async () => {
    try {
      const screenStream = await navigator.mediaDevices.getDisplayMedia({
        video: true,
      });

      // Replace video track in all peer connections
      const screenTrack = screenStream.getVideoTracks()[0];
      peerConnections.current.forEach((pc) => {
        const sender = pc.getSenders().find((s) => s.track?.kind === 'video');
        if (sender) {
          sender.replaceTrack(screenTrack);
        }
      });

      socket?.emit('screen-share', { roomId, isSharing: true });

      // Handle screen share stopped
      screenTrack.onended = () => {
        stopScreenShare();
      };

      return screenStream;
    } catch (err) {
      console.error('Error starting screen share:', err);
      setError('Failed to start screen sharing');
      throw err;
    }
  }, [socket, roomId]);

  const stopScreenShare = useCallback(() => {
    if (localStream) {
      const videoTrack = localStream.getVideoTracks()[0];
      peerConnections.current.forEach((pc) => {
        const sender = pc.getSenders().find((s) => s.track?.kind === 'video');
        if (sender && videoTrack) {
          sender.replaceTrack(videoTrack);
        }
      });
    }

    socket?.emit('screen-share', { roomId, isSharing: false });
  }, [localStream, socket, roomId]);

  return {
    localStream,
    participants,
    isConnected,
    error,
    getLocalStream,
    joinRoom,
    leaveRoom,
    toggleAudio,
    toggleVideo,
    startScreenShare,
    stopScreenShare,
  };
};

          {isVideoEnabled ? '📹' : '📷'}

          🖥️

          className="px-6 py-4 rounded-full bg-red-600 text-white hover:bg-red-700 transition font-bold"
        >
          Leave Call
        </button>
      </div>

1. WebRTC Metrics Service​

// services/webrtc-service/src/monitoring/webrtc-metrics.service.ts
import { Injectable } from '@nestjs/common';
import { Counter, Histogram, Gauge, register } from 'prom-client';

@Injectable()
export class WebRtcMetricsService {
  private readonly activeConnections = new Gauge({
    name: 'webrtc_active_connections',
    help: 'Number of active WebRTC connections',
    labelNames: ['room'],
  });

  private readonly callsTotal = new Counter({
    name: 'webrtc_calls_total',
    help: 'Total number of WebRTC calls',
    labelNames: ['type', 'status'],
  });

  private readonly callDuration = new Histogram({
    name: 'webrtc_call_duration_seconds',
    help: 'Duration of WebRTC calls in seconds',
    labelNames: ['room'],
    buckets: [10, 30, 60, 300, 600, 1800, 3600],
  });

  private readonly signalLatency = new Histogram({
    name: 'webrtc_signal_latency_seconds',
    help: 'Latency of signaling messages',
    labelNames: ['type'],
    buckets: [0.001, 0.005, 0.01, 0.05, 0.1, 0.5, 1],
  });

  recordActiveConnection(roomId: string, count: number): void {
    this.activeConnections.set({ room: roomId }, count);
  }

  recordCall(type: 'audio' | 'video', status: 'initiated' | 'connected' | 'ended' | 'failed'): void {
    this.callsTotal.inc({ type, status });
  }

  recordCallDuration(roomId: string, duration: number): void {
    this.callDuration.observe({ room: roomId }, duration);
  }

  recordSignalLatency(type: string, latency: number): void {
    this.signalLatency.observe({ type }, latency);
  }

  getMetrics(): string {
    return register.metrics();
  }
}

Best Practices​

1. Connection Management​

  • Implement connection health checks
  • Handle reconnection logic for unstable networks
  • Use session affinity for load balancers
  • Implement proper cleanup on disconnect

2. Media Quality​

  • Implement adaptive bitrate based on network conditions
  • Use VP9 or H.264 codecs for better compression
  • Implement simulcast for multi-party calls
  • Monitor packet loss and adjust quality

3. Security​

  • Always use HTTPS for signaling
  • Implement proper authentication and authorization
  • Use DTLS-SRTP for media encryption
  • Validate all signaling messages
  • Implement rate limiting

4. Scalability​

  • Use Redis for session state management
  • Implement horizontal scaling for signaling servers
  • Use multiple TURN servers for load distribution
  • Consider using SFU for large conferences

5. Performance Optimization​

  • Enable gzip compression for signaling
  • Use WebSocket binary frames when possible
  • Implement connection pooling
  • Monitor and optimize bandwidth usage
  • Use CDN for STUN/TURN server distribution

This comprehensive guide covers the essential aspects of implementing WebRTC in a production environment. Adjust the configurations and code samples based on your specific requirements and infrastructure.
