Redis Pub/Sub Scaling Patterns

📖 Overview​

Redis Pub/Sub is a messaging pattern that enables publishers to send messages to channels without knowledge of subscribers, and subscribers to receive messages from channels without knowledge of publishers. This decoupling makes it perfect for building scalable, distributed real-time systems.

🎯 When to Use Redis Pub/Sub​

✅ Best Use Cases​

  • Multi-instance WebSocket broadcasting (Socket.IO, native WebSocket)
  • Real-time notifications across multiple app servers
  • Live updates (dashboards, analytics)
  • Cache invalidation in distributed systems
  • Event broadcasting in microservices
  • Chat applications with multiple servers
  • Live activity feeds
  • Real-time collaboration tools

❌ When NOT to Use​

ScenarioBetter Alternative
Message persistence requiredKafka, RabbitMQ, AWS SQS
Guaranteed delivery neededMessage queues with acknowledgments
Complex routing logicRabbitMQ with exchanges
Message ordering criticalKafka with partitions
Large message payloadsS3 + event notification

Example 1: Socket.IO with Redis Adapter (Production-Ready)​

Backend Implementation​

// redis-adapter.config.ts
import { createAdapter } from '@socket.io/redis-adapter';
import { createClient } from 'redis';
import { Logger } from '@nestjs/common';

const logger = new Logger('RedisAdapter');

export async function setupRedisAdapter(io: any) {
  const pubClient = createClient({
    url: process.env.REDIS_URL || 'redis://localhost:6379',
    password: process.env.REDIS_PASSWORD,
    socket: {
      reconnectStrategy: (retries) => {
        if (retries > 20) {
          logger.error('Redis connection failed after 20 retries');
          return new Error('Redis connection failed');
        }
        return Math.min(retries * 50, 2000);
      },
    },
  });

  const subClient = pubClient.duplicate();

  pubClient.on('error', (err) => logger.error('Redis Pub Client Error:', err));
  subClient.on('error', (err) => logger.error('Redis Sub Client Error:', err));

  pubClient.on('connect', () => logger.log('Redis Pub Client Connected'));
  subClient.on('connect', () => logger.log('Redis Sub Client Connected'));

  await Promise.all([pubClient.connect(), subClient.connect()]);

  io.adapter(createAdapter(pubClient, subClient));

  logger.log('Redis adapter configured successfully');

  return { pubClient, subClient };
}
// websocket.gateway.ts
import {
  WebSocketGateway,
  WebSocketServer,
  OnGatewayInit,
  OnGatewayConnection,
  OnGatewayDisconnect,
  SubscribeMessage,
  MessageBody,
  ConnectedSocket,
} from '@nestjs/websockets';
import { Server, Socket } from 'socket.io';
import { Logger } from '@nestjs/common';
import { setupRedisAdapter } from './redis-adapter.config';

@WebSocketGateway({
  cors: {
    origin: process.env.CORS_ORIGIN?.split(',') || ['http://localhost:3000'],
    credentials: true,
  },
})
export class EventsGateway implements OnGatewayInit, OnGatewayConnection, OnGatewayDisconnect {
  @WebSocketServer()
  server: Server;

  private logger = new Logger('EventsGateway');

  async afterInit(server: Server) {
    // Setup Redis adapter for multi-instance support
    await setupRedisAdapter(server);

    this.logger.log('WebSocket Gateway initialized with Redis adapter');
  }

  handleConnection(client: Socket) {
    const userId = client.handshake.query.userId as string;
    this.logger.log(`Client connected: ${client.id}, User: ${userId}`);

    // Join user to personal room
    client.join(`user:${userId}`);

    // Broadcast to other servers that user is online
    this.server.emit('user:online', { userId, socketId: client.id });
  }

  handleDisconnect(client: Socket) {
    const userId = client.handshake.query.userId as string;
    this.logger.log(`Client disconnected: ${client.id}, User: ${userId}`);

    // Broadcast to other servers that user is offline
    this.server.emit('user:offline', { userId, socketId: client.id });
  }

  /**
   * Broadcast message to all connected clients across all servers
   */
  @SubscribeMessage('broadcast')
  async handleBroadcast(@ConnectedSocket() client: Socket, @MessageBody() data: { message: string }) {
    // This will be broadcast to all servers via Redis
    this.server.emit('broadcast:message', {
      from: client.id,
      message: data.message,
      timestamp: new Date(),
    });
  }

  /**
   * Send message to specific user (may be on different server)
   */
  @SubscribeMessage('send:message')
  async handleSendMessage(@ConnectedSocket() client: Socket, @MessageBody() data: { userId: string; message: string }) {
    // Redis adapter will route this to the correct server
    this.server.to(`user:${data.userId}`).emit('receive:message', {
      from: client.handshake.query.userId,
      message: data.message,
      timestamp: new Date(),
    });
  }

  /**
   * Broadcast to specific room across all servers
   */
  broadcastToRoom(room: string, event: string, data: any) {
    this.server.to(room).emit(event, data);
  }
}

Example 2: Native Redis Pub/Sub for Cache Invalidation​

// redis-pubsub.service.ts
import { Injectable, Logger, OnModuleInit, OnModuleDestroy } from '@nestjs/common';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';

export interface CacheInvalidationMessage {
  key?: string;
  pattern?: string;
  namespace?: string;
  operation: 'delete' | 'clear' | 'invalidate';
  timestamp: Date;
  source: string; // server ID that triggered invalidation
}

@Injectable()
export class RedisPubSubService implements OnModuleInit, OnModuleDestroy {
  private readonly logger = new Logger('RedisPubSubService');
  private subscriber: Redis;
  private publisher: Redis;
  private readonly serverId = `server-${process.pid}-${Date.now()}`;

  // Channels
  private readonly CACHE_INVALIDATION_CHANNEL = 'cache:invalidation';
  private readonly USER_ACTIVITY_CHANNEL = 'user:activity';
  private readonly SYSTEM_EVENTS_CHANNEL = 'system:events';

  constructor(@InjectRedis() private readonly redis: Redis) {}

  async onModuleInit() {
    // Create separate connections for pub and sub
    this.publisher = this.redis.duplicate();
    this.subscriber = this.redis.duplicate();

    await Promise.all([this.publisher.connect(), this.subscriber.connect()]);

    // Subscribe to channels
    await this.subscriber.subscribe(
      this.CACHE_INVALIDATION_CHANNEL,
      this.USER_ACTIVITY_CHANNEL,
      this.SYSTEM_EVENTS_CHANNEL,
    );

    // Setup message handlers
    this.subscriber.on('message', this.handleMessage.bind(this));

    this.logger.log(`Redis Pub/Sub initialized for server: ${this.serverId}`);
  }

  async onModuleDestroy() {
    await this.subscriber.quit();
    await this.publisher.quit();
  }

  /**
   * Handle incoming messages from subscribed channels
   */
  private async handleMessage(channel: string, message: string) {
    try {
      const data = JSON.parse(message);

      // Ignore messages from self
      if (data.source === this.serverId) {
        return;
      }

      this.logger.log(`Received message on channel ${channel}:`, data);

      switch (channel) {
        case this.CACHE_INVALIDATION_CHANNEL:
          await this.handleCacheInvalidation(data);
          break;
        case this.USER_ACTIVITY_CHANNEL:
          await this.handleUserActivity(data);
          break;
        case this.SYSTEM_EVENTS_CHANNEL:
          await this.handleSystemEvent(data);
          break;
        default:
          this.logger.warn(`Unknown channel: ${channel}`);
      }
    } catch (error) {
      this.logger.error(`Error handling message from ${channel}:`, error);
    }
  }

  /**
   * Publish cache invalidation message
   */
  async publishCacheInvalidation(message: Omit<CacheInvalidationMessage, 'timestamp' | 'source'>) {
    const payload: CacheInvalidationMessage = {
      ...message,
      timestamp: new Date(),
      source: this.serverId,
    };

    await this.publisher.publish(this.CACHE_INVALIDATION_CHANNEL, JSON.stringify(payload));

    this.logger.log('Published cache invalidation:', payload);
  }

  /**
   * Handle cache invalidation from other servers
   */
  private async handleCacheInvalidation(message: CacheInvalidationMessage) {
    switch (message.operation) {
      case 'delete':
        if (message.key) {
          await this.redis.del(message.key);
          this.logger.log(`Cache key deleted: ${message.key}`);
        }
        break;

      case 'clear':
        if (message.namespace) {
          const keys = await this.redis.keys(`${message.namespace}:*`);
          if (keys.length > 0) {
            await this.redis.del(...keys);
            this.logger.log(`Cache namespace cleared: ${message.namespace}`);
          }
        }
        break;

      case 'invalidate':
        if (message.pattern) {
          const keys = await this.redis.keys(message.pattern);
          if (keys.length > 0) {
            await this.redis.del(...keys);
            this.logger.log(`Cache pattern invalidated: ${message.pattern}`);
          }
        }
        break;
    }
  }

  /**
   * Publish user activity event
   */
  async publishUserActivity(userId: string, action: string, data?: any) {
    const payload = {
      userId,
      action,
      data,
      timestamp: new Date(),
      source: this.serverId,
    };

    await this.publisher.publish(this.USER_ACTIVITY_CHANNEL, JSON.stringify(payload));
  }

  /**
   * Handle user activity from other servers
   */
  private async handleUserActivity(data: any) {
    this.logger.log(`User activity: ${data.userId} - ${data.action}`);
    // Update local cache, metrics, etc.
  }

  /**
   * Publish system event
   */
  async publishSystemEvent(event: string, data: any) {
    const payload = {
      event,
      data,
      timestamp: new Date(),
      source: this.serverId,
    };

    await this.publisher.publish(this.SYSTEM_EVENTS_CHANNEL, JSON.stringify(payload));
  }

  /**
   * Handle system events
   */
  private async handleSystemEvent(data: any) {
    this.logger.log(`System event: ${data.event}`, data.data);
    // Handle deployment, config changes, etc.
  }
}

Example 3: Distributed Cache Service with Pub/Sub​

// distributed-cache.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';
import { RedisPubSubService } from './redis-pubsub.service';

@Injectable()
export class DistributedCacheService {
  private readonly logger = new Logger('DistributedCacheService');
  private localCache = new Map<string, { value: any; expiry: number }>();

  constructor(@InjectRedis() private readonly redis: Redis, private readonly pubSubService: RedisPubSubService) {}

  /**
   * Get value with two-level caching (local + Redis)
   */
  async get<T>(key: string): Promise<T | null> {
    // Check local cache first
    const localValue = this.getFromLocalCache(key);
    if (localValue !== null) {
      this.logger.debug(`Cache hit (local): ${key}`);
      return localValue;
    }

    // Check Redis
    const redisValue = await this.redis.get(key);
    if (redisValue) {
      this.logger.debug(`Cache hit (Redis): ${key}`);
      const parsed = JSON.parse(redisValue);

      // Store in local cache
      this.setLocalCache(key, parsed, 60); // 60 seconds local TTL

      return parsed;
    }

    this.logger.debug(`Cache miss: ${key}`);
    return null;
  }

  /**
   * Set value in both caches and invalidate on other servers
   */
  async set(key: string, value: any, ttl = 3600): Promise<void> {
    // Set in Redis
    await this.redis.setex(key, ttl, JSON.stringify(value));

    // Set in local cache
    this.setLocalCache(key, value, Math.min(ttl, 60));

    this.logger.debug(`Cache set: ${key}`);
  }

  /**
   * Delete key and notify other servers
   */
  async delete(key: string): Promise<void> {
    // Delete from Redis
    await this.redis.del(key);

    // Delete from local cache
    this.localCache.delete(key);

    // Notify other servers to invalidate their local cache
    await this.pubSubService.publishCacheInvalidation({
      key,
      operation: 'delete',
    });

    this.logger.log(`Cache deleted: ${key}`);
  }

  /**
   * Clear entire namespace
   */
  async clearNamespace(namespace: string): Promise<void> {
    // Clear from Redis
    const keys = await this.redis.keys(`${namespace}:*`);
    if (keys.length > 0) {
      await this.redis.del(...keys);
    }

    // Clear from local cache
    for (const [key] of this.localCache) {
      if (key.startsWith(`${namespace}:`)) {
        this.localCache.delete(key);
      }
    }

    // Notify other servers
    await this.pubSubService.publishCacheInvalidation({
      namespace,
      operation: 'clear',
    });

    this.logger.log(`Cache namespace cleared: ${namespace}`);
  }

  /**
   * Get or set pattern (cache-aside)
   */
  async getOrSet<T>(key: string, factory: () => Promise<T>, ttl = 3600): Promise<T> {
    // Try to get from cache
    const cached = await this.get<T>(key);
    if (cached !== null) {
      return cached;
    }

    // Generate value
    const value = await factory();

    // Store in cache
    await this.set(key, value, ttl);

    return value;
  }

  /**
   * Local cache helpers
   */
  private getFromLocalCache(key: string): any | null {
    const entry = this.localCache.get(key);
    if (!entry) return null;

    // Check expiry
    if (Date.now() > entry.expiry) {
      this.localCache.delete(key);
      return null;
    }

    return entry.value;
  }

  private setLocalCache(key: string, value: any, ttlSeconds: number): void {
    this.localCache.set(key, {
      value,
      expiry: Date.now() + ttlSeconds * 1000,
    });
  }

  /**
   * Cleanup expired local cache entries
   */
  startCleanupInterval(intervalMs = 60000) {
    setInterval(() => {
      const now = Date.now();
      let cleaned = 0;

      for (const [key, entry] of this.localCache) {
        if (now > entry.expiry) {
          this.localCache.delete(key);
          cleaned++;
        }
      }

      if (cleaned > 0) {
        this.logger.debug(`Cleaned ${cleaned} expired local cache entries`);
      }
    }, intervalMs);
  }
}

Example 4: Real-Time Chat with Pub/Sub​

// chat.gateway.ts
import { WebSocketGateway, WebSocketServer, SubscribeMessage, MessageBody, ConnectedSocket } from '@nestjs/websockets';
import { Server, Socket } from 'socket.io';
import { Logger } from '@nestjs/common';
import { RedisPubSubService } from './redis-pubsub.service';

interface ChatMessage {
  chatId: string;
  userId: string;
  message: string;
  timestamp: Date;
}

@WebSocketGateway({ namespace: '/chat' })
export class ChatGateway {
  @WebSocketServer()
  server: Server;

  private readonly logger = new Logger('ChatGateway');
  private readonly CHAT_MESSAGE_CHANNEL = 'chat:messages';

  constructor(private readonly pubSubService: RedisPubSubService, private readonly chatService: ChatService) {
    // Subscribe to chat messages from other servers
    this.subscribeToMessages();
  }

  private async subscribeToMessages() {
    const subscriber = await this.pubSubService.createSubscriber();

    await subscriber.subscribe(this.CHAT_MESSAGE_CHANNEL);

    subscriber.on('message', (channel, message) => {
      if (channel === this.CHAT_MESSAGE_CHANNEL) {
        const chatMessage: ChatMessage = JSON.parse(message);

        // Emit to connected clients in this server
        this.server.to(`chat:${chatMessage.chatId}`).emit('message', chatMessage);

        this.logger.log(`Relayed message to chat ${chatMessage.chatId}`);
      }
    });
  }

  @SubscribeMessage('join:chat')
  async handleJoinChat(@ConnectedSocket() client: Socket, @MessageBody() data: { chatId: string; userId: string }) {
    await client.join(`chat:${data.chatId}`);
    this.logger.log(`User ${data.userId} joined chat ${data.chatId}`);
  }

  @SubscribeMessage('send:message')
  async handleSendMessage(
    @ConnectedSocket() client: Socket,
    @MessageBody() data: { chatId: string; userId: string; message: string },
  ) {
    const chatMessage: ChatMessage = {
      chatId: data.chatId,
      userId: data.userId,
      message: data.message,
      timestamp: new Date(),
    };

    // Persist message
    await this.chatService.saveMessage(chatMessage);

    // Publish to Redis for other servers
    await this.pubSubService.publish(this.CHAT_MESSAGE_CHANNEL, JSON.stringify(chatMessage));

    // Also emit locally (will be received by all servers including this one)
    this.logger.log(`Message published to Redis: ${chatMessage.chatId}`);
  }
}

Example 5: Real-Time Analytics Dashboard​

// analytics-pubsub.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';

interface AnalyticsEvent {
  type: 'pageview' | 'click' | 'conversion' | 'error';
  userId?: string;
  sessionId: string;
  page?: string;
  data: any;
  timestamp: Date;
}

interface AggregatedMetrics {
  pageviews: number;
  uniqueUsers: Set<string>;
  conversions: number;
  errors: number;
  lastUpdated: Date;
}

@Injectable()
export class AnalyticsPubSubService {
  private readonly logger = new Logger('AnalyticsPubSubService');
  private readonly ANALYTICS_CHANNEL = 'analytics:events';
  private subscriber: Redis;
  private publisher: Redis;

  // In-memory aggregation (per server instance)
  private metrics: AggregatedMetrics = {
    pageviews: 0,
    uniqueUsers: new Set(),
    conversions: 0,
    errors: 0,
    lastUpdated: new Date(),
  };

  constructor(@InjectRedis() private readonly redis: Redis) {
    this.initializePubSub();
    this.startMetricsFlush();
  }

  private async initializePubSub() {
    this.publisher = this.redis.duplicate();
    this.subscriber = this.redis.duplicate();

    await Promise.all([this.publisher.connect(), this.subscriber.connect()]);

    await this.subscriber.subscribe(this.ANALYTICS_CHANNEL);

    this.subscriber.on('message', (channel, message) => {
      if (channel === this.ANALYTICS_CHANNEL) {
        this.handleAnalyticsEvent(JSON.parse(message));
      }
    });

    this.logger.log('Analytics Pub/Sub initialized');
  }

  /**
   * Publish analytics event to all servers
   */
  async trackEvent(event: Omit<AnalyticsEvent, 'timestamp'>) {
    const analyticsEvent: AnalyticsEvent = {
      ...event,
      timestamp: new Date(),
    };

    // Publish to Redis for real-time aggregation across servers
    await this.publisher.publish(this.ANALYTICS_CHANNEL, JSON.stringify(analyticsEvent));

    // Also handle locally
    this.handleAnalyticsEvent(analyticsEvent);
  }

  /**
   * Handle incoming analytics events
   */
  private handleAnalyticsEvent(event: AnalyticsEvent) {
    switch (event.type) {
      case 'pageview':
        this.metrics.pageviews++;
        if (event.userId) {
          this.metrics.uniqueUsers.add(event.userId);
        }
        break;

      case 'conversion':
        this.metrics.conversions++;
        break;

      case 'error':
        this.metrics.errors++;
        break;
    }

    this.metrics.lastUpdated = new Date();
  }

  /**
   * Flush aggregated metrics to Redis periodically
   */
  private startMetricsFlush() {
    setInterval(async () => {
      const metricsSnapshot = {
        pageviews: this.metrics.pageviews,
        uniqueUsers: this.metrics.uniqueUsers.size,
        conversions: this.metrics.conversions,
        errors: this.metrics.errors,
        timestamp: new Date(),
        serverId: process.pid,
      };

      // Store in Redis sorted set for time-series data
      const key = `analytics:metrics:${Date.now()}`;
      await this.redis.setex(key, 3600, JSON.stringify(metricsSnapshot));

      this.logger.log('Metrics flushed to Redis:', metricsSnapshot);

      // Reset local counters
      this.metrics = {
        pageviews: 0,
        uniqueUsers: new Set(),
        conversions: 0,
        errors: 0,
        lastUpdated: new Date(),
      };
    }, 10000); // Flush every 10 seconds
  }

  /**
   * Get aggregated metrics from all servers
   */
  async getAggregatedMetrics(startTime: Date, endTime: Date): Promise<any> {
    const keys = await this.redis.keys('analytics:metrics:*');
    const metrics = await Promise.all(keys.map((key) => this.redis.get(key)));

    const aggregated = metrics
      .filter(Boolean)
      .map((m) => JSON.parse(m!))
      .filter((m) => {
        const timestamp = new Date(m.timestamp);
        return timestamp >= startTime && timestamp <= endTime;
      })
      .reduce(
        (acc, m) => ({
          pageviews: acc.pageviews + m.pageviews,
          uniqueUsers: acc.uniqueUsers + m.uniqueUsers,
          conversions: acc.conversions + m.conversions,
          errors: acc.errors + m.errors,
        }),
        { pageviews: 0, uniqueUsers: 0, conversions: 0, errors: 0 },
      );

    return aggregated;
  }
}

Example 6: Product Inventory Updates​

// inventory-pubsub.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';
import { EventEmitter2 } from '@nestjs/event-emitter';

interface InventoryUpdate {
  productId: string;
  warehouseId: string;
  quantity: number;
  operation: 'reserve' | 'release' | 'adjust';
  orderId?: string;
  timestamp: Date;
  source: string;
}

@Injectable()
export class InventoryPubSubService {
  private readonly logger = new Logger('InventoryPubSubService');
  private readonly INVENTORY_CHANNEL = 'inventory:updates';
  private subscriber: Redis;
  private publisher: Redis;
  private readonly serverId = `inventory-${process.pid}`;

  constructor(@InjectRedis() private readonly redis: Redis, private eventEmitter: EventEmitter2) {
    this.initializePubSub();
  }

  private async initializePubSub() {
    this.publisher = this.redis.duplicate();
    this.subscriber = this.redis.duplicate();

    await Promise.all([this.publisher.connect(), this.subscriber.connect()]);

    await this.subscriber.subscribe(this.INVENTORY_CHANNEL);

    this.subscriber.on('message', (channel, message) => {
      if (channel === this.INVENTORY_CHANNEL) {
        this.handleInventoryUpdate(JSON.parse(message));
      }
    });

    this.logger.log('Inventory Pub/Sub initialized');
  }

  /**
   * Publish inventory update
   */
  async publishInventoryUpdate(update: Omit<InventoryUpdate, 'timestamp' | 'source'>) {
    const inventoryUpdate: InventoryUpdate = {
      ...update,
      timestamp: new Date(),
      source: this.serverId,
    };

    // Update in Redis
    const key = `inventory:${update.productId}:${update.warehouseId}`;

    const multi = this.redis.multi();

    switch (update.operation) {
      case 'reserve':
        multi.decrby(key, update.quantity);
        break;
      case 'release':
        multi.incrby(key, update.quantity);
        break;
      case 'adjust':
        multi.set(key, update.quantity);
        break;
    }

    // Store update history
    multi.lpush(`inventory:history:${update.productId}`, JSON.stringify(inventoryUpdate));
    multi.ltrim(`inventory:history:${update.productId}`, 0, 99); // Keep last 100

    await multi.exec();

    // Publish to other servers
    await this.publisher.publish(this.INVENTORY_CHANNEL, JSON.stringify(inventoryUpdate));

    this.logger.log('Inventory update published:', inventoryUpdate);
  }

  /**
   * Handle inventory updates from other servers
   */
  private async handleInventoryUpdate(update: InventoryUpdate) {
    // Ignore updates from self
    if (update.source === this.serverId) {
      return;
    }

    this.logger.log('Received inventory update:', update);

    // Emit event for local subscribers (e.g., WebSocket to notify clients)
    this.eventEmitter.emit('inventory.updated', update);

    // Check if inventory is low
    const currentQty = await this.redis.get(`inventory:${update.productId}:${update.warehouseId}`);

    if (currentQty && parseInt(currentQty) < 10) {
      this.eventEmitter.emit('inventory.low', {
        productId: update.productId,
        warehouseId: update.warehouseId,
        quantity: parseInt(currentQty),
      });
    }
  }

  /**
   * Get current inventory
   */
  async getInventory(productId: string, warehouseId: string): Promise<number> {
    const key = `inventory:${productId}:${warehouseId}`;
    const qty = await this.redis.get(key);
    return qty ? parseInt(qty) : 0;
  }

  /**
   * Get inventory history
   */
  async getInventoryHistory(productId: string, limit = 20): Promise<InventoryUpdate[]> {
    const history = await this.redis.lrange(`inventory:history:${productId}`, 0, limit - 1);

    return history.map((h) => JSON.parse(h));
  }
}

1. Message Batching​

// batch-publisher.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';

@Injectable()
export class BatchPublisherService {
  private readonly logger = new Logger('BatchPublisherService');
  private messageQueue: Map<string, any[]> = new Map();
  private batchSize = 100;
  private batchInterval = 1000; // 1 second

  constructor(@InjectRedis() private readonly redis: Redis) {
    this.startBatchProcessing();
  }

  /**
   * Queue message for batched publishing
   */
  async queueMessage(channel: string, message: any) {
    if (!this.messageQueue.has(channel)) {
      this.messageQueue.set(channel, []);
    }

    const queue = this.messageQueue.get(channel)!;
    queue.push(message);

    // Flush if batch size reached
    if (queue.length >= this.batchSize) {
      await this.flushChannel(channel);
    }
  }

  /**
   * Process batches periodically
   */
  private startBatchProcessing() {
    setInterval(async () => {
      for (const channel of this.messageQueue.keys()) {
        await this.flushChannel(channel);
      }
    }, this.batchInterval);
  }

  /**
   * Flush messages for a channel
   */
  private async flushChannel(channel: string) {
    const queue = this.messageQueue.get(channel);
    if (!queue || queue.length === 0) return;

    const batch = {
      messages: queue.splice(0, this.batchSize),
      timestamp: new Date(),
      count: queue.length,
    };

    await this.redis.publish(channel, JSON.stringify(batch));

    this.logger.log(`Flushed ${batch.count} messages to ${channel}`);
  }
}

🚨 Error Handling & Resilience​

Retry Logic​

// resilient-publisher.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';

@Injectable()
export class ResilientPublisherService {
  private readonly logger = new Logger('ResilientPublisherService');
  private failedMessages: Map<string, any[]> = new Map();

  constructor(@InjectRedis() private readonly redis: Redis) {
    this.startRetryProcessor();
  }

  /**
   * Publish with retry logic
   */
  async publishWithRetry(channel: string, message: any, maxRetries = 3): Promise<boolean> {
    for (let attempt = 1; attempt <= maxRetries; attempt++) {
      try {
        await this.redis.publish(channel, JSON.stringify(message));
        return true;
      } catch (error) {
        this.logger.error(`Publish failed (attempt ${attempt}/${maxRetries}):`, error);

        if (attempt === maxRetries) {
          // Store for later retry
          this.storeFailedMessage(channel, message);
          return false;
        }

        // Exponential backoff
        await this.delay(Math.pow(2, attempt) * 100);
      }
    }

    return false;
  }

  /**
   * Store failed message for retry
   */
  private storeFailedMessage(channel: string, message: any) {
    if (!this.failedMessages.has(channel)) {
      this.failedMessages.set(channel, []);
    }

    this.failedMessages.get(channel)!.push({
      message,
      timestamp: new Date(),
      retries: 0,
    });

    this.logger.warn(`Failed message stored for channel: ${channel}`);
  }

  /**
   * Retry failed messages periodically
   */
  private startRetryProcessor() {
    setInterval(async () => {
      for (const [channel, messages] of this.failedMessages) {
        const remaining = [];

        for (const item of messages) {
          try {
            await this.redis.publish(channel, JSON.stringify(item.message));
            this.logger.log(`Retry successful for channel: ${channel}`);
          } catch (error) {
            item.retries++;

            // Keep trying for 24 hours
            if (item.retries < 288) {
              remaining.push(item);
            } else {
              this.logger.error(`Message abandoned after 24 hours: ${channel}`);
            }
          }
        }

        if (remaining.length > 0) {
          this.failedMessages.set(channel, remaining);
        } else {
          this.failedMessages.delete(channel);
        }
      }
    }, 300000); // Retry every 5 minutes
  }

  private delay(ms: number): Promise<void> {
    return new Promise((resolve) => setTimeout(resolve, ms));
  }
}

🎯 Summary​

Redis Pub/Sub is an excellent choice for:

✅ Real-time broadcasting across multiple server instances
✅ Cache invalidation in distributed systems
✅ Event-driven microservices communication
✅ Live notifications and updates
✅ Horizontal scaling of WebSocket servers

Key Considerations:

  • ⚠️ No message persistence - messages are lost if no subscribers
  • ⚠️ No delivery guarantees - fire-and-forget pattern
  • ⚠️ Memory bound - all messages held in memory
  • ✅ Very fast - microsecond latency
  • ✅ Simple - easy to implement and scale

For critical messages requiring guaranteed delivery, combine with message queues (Kafka, RabbitMQ) or use Redis Streams instead of Pub/Sub.
