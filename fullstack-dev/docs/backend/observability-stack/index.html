Correlation Strategy​

// shared/observability/src/correlation.service.ts
import { Injectable } from '@nestjs/common';
import { v4 as uuidv4 } from 'uuid';
import { AsyncLocalStorage } from 'async_hooks';

export interface CorrelationContext {
  traceId: string;
  spanId: string;
  userId?: string;
  requestId: string;
  timestamp: Date;
}

@Injectable()
export class CorrelationService {
  private readonly asyncLocalStorage = new AsyncLocalStorage<CorrelationContext>();

  createContext(traceId?: string): CorrelationContext {
    return {
      traceId: traceId || uuidv4(),
      spanId: uuidv4(),
      requestId: uuidv4(),
      timestamp: new Date(),
    };
  }

  run<T>(context: CorrelationContext, callback: () => T): T {
    return this.asyncLocalStorage.run(context, callback);
  }

  getContext(): CorrelationContext | undefined {
    return this.asyncLocalStorage.getStore();
  }

  getTraceId(): string | undefined {
    return this.getContext()?.traceId;
  }

  getRequestId(): string | undefined {
    return this.getContext()?.requestId;
  }

  enrichWithContext(data: any): any {
    const context = this.getContext();
    if (!context) return data;

    return {
      ...data,
      traceId: context.traceId,
      spanId: context.spanId,
      requestId: context.requestId,
    };
  }
}

// shared/metrics/src/metrics.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { InjectMetric } from '@willsoto/nestjs-prometheus';
import { Counter, Histogram, Gauge, Summary, Registry } from 'prom-client';

@Injectable()
export class MetricsService {
  private readonly logger = new Logger(MetricsService.name);

  constructor(
    @InjectMetric('http_requests_total')
    public httpRequestsCounter: Counter<string>,
    @InjectMetric('http_request_duration_seconds')
    public httpRequestDurationHistogram: Histogram<string>,
    @InjectMetric('active_connections')
    public activeConnectionsGauge: Gauge<string>,
    @InjectMetric('business_transactions_total')
    public businessTransactionsCounter: Counter<string>,
    @InjectMetric('database_query_duration_seconds')
    public dbQueryDurationHistogram: Histogram<string>,
  ) {}

  // HTTP Metrics
  recordHttpRequest(method: string, path: string, statusCode: number): void {
    this.httpRequestsCounter.inc({
      method,
      path,
      status_code: statusCode.toString(),
    });
  }

  recordHttpDuration(method: string, path: string, statusCode: number, durationMs: number): void {
    this.httpRequestDurationHistogram.observe(
      {
        method,
        path,
        status_code: statusCode.toString(),
      },
      durationMs / 1000,
    );
  }

  // Connection Metrics
  incrementActiveConnections(): void {
    this.activeConnectionsGauge.inc();
  }

  decrementActiveConnections(): void {
    this.activeConnectionsGauge.dec();
  }

  // Business Metrics
  recordBusinessTransaction(type: string, status: string): void {
    this.businessTransactionsCounter.inc({
      transaction_type: type,
      status,
    });
  }

  // Database Metrics
  recordDatabaseQuery(operation: string, table: string, durationMs: number): void {
    this.dbQueryDurationHistogram.observe(
      {
        operation,
        table,
      },
      durationMs / 1000,
    );
  }
}

// services/order-service/src/metrics/order-metrics.service.ts
import { Injectable } from '@nestjs/common';
import { Counter, Histogram, Gauge } from 'prom-client';

@Injectable()
export class OrderMetricsService {
  private ordersCreatedCounter = new Counter({
    name: 'orders_created_total',
    help: 'Total number of orders created',
    labelNames: ['status', 'payment_method'],
  });

  private orderValueHistogram = new Histogram({
    name: 'order_value_dollars',
    help: 'Distribution of order values',
    labelNames: ['payment_method'],
    buckets: [10, 25, 50, 100, 250, 500, 1000, 2500, 5000],
  });

  private activeOrdersGauge = new Gauge({
    name: 'orders_active_count',
    help: 'Number of currently active orders',
    labelNames: ['status'],
  });

  private orderProcessingDuration = new Histogram({
    name: 'order_processing_duration_seconds',
    help: 'Time taken to process an order',
    labelNames: ['status'],
    buckets: [0.1, 0.5, 1, 2, 5, 10, 30, 60],
  });

  private cartAbandonmentCounter = new Counter({
    name: 'cart_abandonment_total',
    help: 'Total number of abandoned carts',
    labelNames: ['reason'],
  });

  recordOrderCreated(status: string, paymentMethod: string): void {
    this.ordersCreatedCounter.inc({ status, payment_method: paymentMethod });
  }

  recordOrderValue(value: number, paymentMethod: string): void {
    this.orderValueHistogram.observe({ payment_method: paymentMethod }, value);
  }

  setActiveOrders(status: string, count: number): void {
    this.activeOrdersGauge.set({ status }, count);
  }

  recordOrderProcessing(status: string, durationSeconds: number): void {
    this.orderProcessingDuration.observe({ status }, durationSeconds);
  }

  recordCartAbandonment(reason: string): void {
    this.cartAbandonmentCounter.inc({ reason });
  }
}

// shared/logging/src/logger.module.ts
import { Module, DynamicModule, Global } from '@nestjs/common';
import { WinstonModule } from 'nest-winston';
import * as winston from 'winston';
import { LoggerService } from './logger.service';

@Global()
@Module({})
export class LoggerModule {
  static forRoot(): DynamicModule {
    return {
      module: LoggerModule,
      imports: [
        WinstonModule.forRoot({
          transports: [
            // Console transport for development
            new winston.transports.Console({
              format: winston.format.combine(
                winston.format.timestamp(),
                winston.format.ms(),
                winston.format.errors({ stack: true }),
                winston.format.colorize({ all: true }),
                winston.format.printf(({ timestamp, level, message, context, ...meta }) => {
                  return `${timestamp} [${context}] ${level}: ${message} ${
                    Object.keys(meta).length ? JSON.stringify(meta) : ''
                  }`;
                }),
              ),
            }),
            // JSON transport for production
            new winston.transports.File({
              filename: 'logs/app.log',
              format: winston.format.combine(
                winston.format.timestamp(),
                winston.format.errors({ stack: true }),
                winston.format.json(),
              ),
            }),
            // Error log file
            new winston.transports.File({
              filename: 'logs/error.log',
              level: 'error',
              format: winston.format.combine(
                winston.format.timestamp(),
                winston.format.errors({ stack: true }),
                winston.format.json(),
              ),
            }),
          ],
          exceptionHandlers: [new winston.transports.File({ filename: 'logs/exceptions.log' })],
          rejectionHandlers: [new winston.transports.File({ filename: 'logs/rejections.log' })],
        }),
      ],
      providers: [LoggerService],
      exports: [LoggerService],
    };
  }
}
// shared/logging/src/logger.service.ts
import { Injectable, LoggerService as NestLoggerService } from '@nestjs/common';
import { Logger } from 'winston';
import { WINSTON_MODULE_PROVIDER } from 'nest-winston';
import { Inject } from '@nestjs/common';
import { CorrelationService } from '@ecommerce/observability';

@Injectable()
export class LoggerService implements NestLoggerService {
  constructor(
    @Inject(WINSTON_MODULE_PROVIDER) private readonly logger: Logger,
    private readonly correlationService: CorrelationService,
  ) {}

  private enrichLogData(message: string, context?: string, meta?: any) {
    const correlationContext = this.correlationService.getContext();

    return {
      message,
      context,
      ...meta,
      traceId: correlationContext?.traceId,
      spanId: correlationContext?.spanId,
      requestId: correlationContext?.requestId,
      userId: correlationContext?.userId,
    };
  }

  log(message: string, context?: string, meta?: any) {
    this.logger.info(this.enrichLogData(message, context, meta));
  }

  error(message: string, trace?: string, context?: string, meta?: any) {
    this.logger.error(this.enrichLogData(message, context, { ...meta, trace }));
  }

  warn(message: string, context?: string, meta?: any) {
    this.logger.warn(this.enrichLogData(message, context, meta));
  }

  debug(message: string, context?: string, meta?: any) {
    this.logger.debug(this.enrichLogData(message, context, meta));
  }

  verbose(message: string, context?: string, meta?: any) {
    this.logger.verbose(this.enrichLogData(message, context, meta));
  }

  // Structured logging methods
  logHttpRequest(req: any, res: any, responseTime: number) {
    this.logger.info({
      type: 'http_request',
      method: req.method,
      url: req.url,
      statusCode: res.statusCode,
      responseTime,
      userAgent: req.headers['user-agent'],
      ip: req.ip,
      ...this.correlationService.enrichWithContext({}),
    });
  }

  logDatabaseQuery(query: string, duration: number, params?: any) {
    this.logger.debug({
      type: 'database_query',
      query,
      duration,
      params,
      ...this.correlationService.enrichWithContext({}),
    });
  }

  logBusinessEvent(event: string, data: any) {
    this.logger.info({
      type: 'business_event',
      event,
      data,
      ...this.correlationService.enrichWithContext({}),
    });
  }

  logSecurityEvent(event: string, severity: string, data: any) {
    this.logger.warn({
      type: 'security_event',
      event,
      severity,
      data,
      ...this.correlationService.enrichWithContext({}),
    });
  }
}

  static forRoot(options: { serviceName: string; serviceVersion?: string }): DynamicModule {

// shared/observability/src/observability.interceptor.ts
import { Injectable, NestInterceptor, ExecutionContext, CallHandler } from '@nestjs/common';
import { Observable } from 'rxjs';
import { tap } from 'rxjs/operators';
import { CorrelationService } from './correlation.service';
import { LoggerService } from '@ecommerce/logging';
import { MetricsService } from '@ecommerce/metrics';
import { TracingService } from '@ecommerce/tracing';

@Injectable()
export class ObservabilityInterceptor implements NestInterceptor {
  constructor(
    private correlationService: CorrelationService,
    private logger: LoggerService,
    private metrics: MetricsService,
    private tracing: TracingService,
  ) {}

  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const request = context.switchToHttp().getRequest();
    const response = context.switchToHttp().getResponse();
    const { method, url, headers } = request;

    // Create correlation context
    const traceId = headers['x-trace-id'] || this.tracing.getCurrentSpan()?.spanContext().traceId;
    const correlationContext = this.correlationService.createContext(traceId);

    return this.correlationService.run(correlationContext, () => {
      const startTime = Date.now();

      // Start trace span
      return this.tracing.startActiveSpanAsync(`${method} ${url}`, async (span) => {
        return next.handle().pipe(
          tap({
            next: (data) => {
              const duration = Date.now() - startTime;

              // Record metrics
              this.metrics.recordHttpRequest(method, url, response.statusCode);
              this.metrics.recordHttpDuration(method, url, response.statusCode, duration);

              // Log request
              this.logger.logHttpRequest(request, response, duration);

              // Add trace attributes
              span.setAttributes({
                'http.status_code': response.statusCode,
                'http.response_time_ms': duration,
              });
            },
            error: (error) => {
              const duration = Date.now() - startTime;

              // Record error metrics
              this.metrics.recordHttpRequest(method, url, response.statusCode || 500);

              // Log error
              this.logger.error(`Request failed: ${method} ${url}`, error.stack, 'ObservabilityInterceptor', {
                method,
                url,
                statusCode: response.statusCode || 500,
                duration,
              });

              // Record exception in trace
              this.tracing.recordException(error);
            },
          }),
        );
      });
    });
  }
}

// services/order-service/src/main.ts
import { NestFactory } from '@nestjs/core';
import { AppModule } from './app.module';
import { ObservabilityInterceptor } from '@ecommerce/observability';
import { LoggerService } from '@ecommerce/logging';

async function bootstrap() {
  const app = await NestFactory.create(AppModule, {
    bufferLogs: true,
  });

  // Set up global logger
  const logger = app.get(LoggerService);
  app.useLogger(logger);

  // Set up global observability interceptor
  const observabilityInterceptor = app.get(ObservabilityInterceptor);
  app.useGlobalInterceptors(observabilityInterceptor);

  await app.listen(3000);
  logger.log('Order Service started on port 3000', 'Bootstrap');
}

Sampling Strategies​

// Intelligent sampling based on request characteristics
export class IntelligentSampler {
  shouldSample(request: any): boolean {
    // Always sample errors
    if (request.statusCode >= 400) {
      return true;
    }

    // Always sample slow requests
    if (request.duration > 1000) {
      return true;
    }

    // Sample critical endpoints at higher rate
    const criticalEndpoints = ['/checkout', '/payment', '/order'];
    if (criticalEndpoints.some((path) => request.url.includes(path))) {
      return Math.random() < 0.5; // 50% sampling
    }

    // Sample regular traffic at lower rate
    return Math.random() < 0.1; // 10% sampling
  }
}

Best Practices​

1. Metrics Best Practices​

✅ Do:

  • Use consistent naming conventions (snake_case)
  • Include unit in metric names (_seconds, _bytes)
  • Use appropriate metric types (Counter, Gauge, Histogram)
  • Keep cardinality low (avoid high-cardinality labels)
  • Document metrics with help text

❌ Don't:

  • Create metrics with unbounded labels
  • Use Gauges for counters
  • Include PII in metric labels
  • Create too many metrics
  • Forget to add units

2. Logging Best Practices​

✅ Do:

  • Use structured logging (JSON)
  • Include correlation IDs
  • Log at appropriate levels
  • Sanitize sensitive data
  • Use consistent log formats

❌ Don't:

  • Log sensitive data (passwords, tokens)
  • Use string interpolation for structured data
  • Log at DEBUG level in production
  • Create circular references in log objects
  • Forget error stack traces

3. Tracing Best Practices​

✅ Do:

  • Propagate trace context across all services
  • Use semantic conventions for span names
  • Add meaningful attributes
  • Record exceptions
  • Sample intelligently

❌ Don't:

  • Create orphaned spans
  • Include large payloads in span attributes
  • Trace every single operation
  • Ignore sampling in high-traffic scenarios
  • Forget to end spans

4. Alerting Best Practices​

✅ Do:

  • Alert on symptoms, not causes
  • Set appropriate thresholds
  • Include runbooks in alerts
  • Route alerts to right teams
  • Test alert rules

❌ Don't:

  • Create alert fatigue with too many alerts
  • Alert on every anomaly
  • Ignore alert context
  • Forget to update oncall rotations
  • Skip alert testing

Summary​

A complete observability stack provides:

  • Metrics - What's happening (Prometheus + Grafana)
  • Logs - Why it's happening (Loki/ELK + Fluent Bit)
  • Traces - How requests flow (Jaeger/Tempo + OpenTelemetry)
  • Correlation - Connect metrics, logs, and traces
  • Alerting - Proactive problem detection
  • Dashboards - Visual insights and trends

Key implementation points:

  • Implement all three pillars together
  • Use correlation IDs across metrics, logs, and traces
  • Set up proper retention and sampling
  • Create actionable alerts
  • Monitor costs and optimize
  • Build comprehensive dashboards

Remember: Good observability is essential for maintaining reliable microservices!
