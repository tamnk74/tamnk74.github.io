Service Registry and Health Checks Implementation Guide

Overview​

Service registry and health checks are critical components of a microservices architecture. This guide covers implementing service discovery with Consul, health monitoring, and integration with NestJS microservices on Kubernetes.

NestJS Service Registration​

1. Consul Module​

// shared/consul/src/consul.module.ts
import { Module, DynamicModule, Global } from '@nestjs/common';
import { ConsulService } from './consul.service';
import { HealthCheckService } from './health-check.service';
import { CONSUL_OPTIONS } from './consul.constants';
import { ConsulModuleOptions } from './interfaces/consul-options.interface';

@Global()
@Module({})
export class ConsulModule {
  static forRoot(options: ConsulModuleOptions): DynamicModule {
    return {
      module: ConsulModule,
      providers: [
        {
          provide: CONSUL_OPTIONS,
          useValue: options,
        },
        ConsulService,
        HealthCheckService,
      ],
      exports: [ConsulService, HealthCheckService],
    };
  }

  static forRootAsync(options: {
    useFactory: (...args: any[]) => Promise<ConsulModuleOptions> | ConsulModuleOptions;
    inject?: any[];
  }): DynamicModule {
    return {
      module: ConsulModule,
      providers: [
        {
          provide: CONSUL_OPTIONS,
          useFactory: options.useFactory,
          inject: options.inject || [],
        },
        ConsulService,
        HealthCheckService,
      ],
      exports: [ConsulService, HealthCheckService],
    };
  }
}

2. Consul Service​

// shared/consul/src/consul.service.ts
import { Injectable, Inject, Logger, OnModuleInit, OnModuleDestroy } from '@nestjs/common';
import * as Consul from 'consul';
import { CONSUL_OPTIONS } from './consul.constants';
import { ConsulModuleOptions, ServiceRegistration } from './interfaces';

@Injectable()
export class ConsulService implements OnModuleInit, OnModuleDestroy {
  private readonly logger = new Logger(ConsulService.name);
  private consulClient: Consul.Consul;
  private registeredServices: Set<string> = new Set();

  constructor(@Inject(CONSUL_OPTIONS) private options: ConsulModuleOptions) {
    this.consulClient = new Consul({
      host: options.host || 'localhost',
      port: options.port || 8500,
      promisify: true,
    });
  }

  async onModuleInit() {
    if (this.options.autoRegister) {
      await this.registerService(this.options.service);
    }
  }

  async onModuleDestroy() {
    await this.deregisterAllServices();
  }

  async registerService(service: ServiceRegistration): Promise<void> {
    try {
      const registration: Consul.Agent.Service.RegisterOptions = {
        id: service.id,
        name: service.name,
        address: service.address,
        port: service.port,
        tags: service.tags || [],
        meta: service.meta || {},
        check: service.check
          ? {
              http: service.check.http,
              interval: service.check.interval || '10s',
              timeout: service.check.timeout || '5s',
              deregistercriticalserviceafter: service.check.deregisterAfter || '30s',
            }
          : undefined,
        checks: service.checks?.map((check) => ({
          http: check.http,
          tcp: check.tcp,
          grpc: check.grpc,
          interval: check.interval || '10s',
          timeout: check.timeout || '5s',
          deregistercriticalserviceafter: check.deregisterAfter || '30s',
          notes: check.notes,
        })),
      };

      await this.consulClient.agent.service.register(registration);
      this.registeredServices.add(service.id);

      this.logger.log(`Service registered: ${service.name} (${service.id}) at ${service.address}:${service.port}`);
    } catch (error) {
      this.logger.error(`Failed to register service: ${error.message}`, error.stack);
      throw error;
    }
  }

  async deregisterService(serviceId: string): Promise<void> {
    try {
      await this.consulClient.agent.service.deregister(serviceId);
      this.registeredServices.delete(serviceId);
      this.logger.log(`Service deregistered: ${serviceId}`);
    } catch (error) {
      this.logger.error(`Failed to deregister service: ${error.message}`, error.stack);
      throw error;
    }
  }

  async deregisterAllServices(): Promise<void> {
    const promises = Array.from(this.registeredServices).map((serviceId) => this.deregisterService(serviceId));
    await Promise.all(promises);
  }

  async getService(serviceName: string, options?: { passing?: boolean; tag?: string }): Promise<any[]> {
    try {
      const result = await this.consulClient.health.service({
        service: serviceName,
        passing: options?.passing ?? true,
        tag: options?.tag,
      });

      return result;
    } catch (error) {
      this.logger.error(`Failed to get service: ${error.message}`, error.stack);
      throw error;
    }
  }

  async getServiceEndpoint(serviceName: string): Promise<{ address: string; port: number } | null> {
    try {
      const services = await this.getService(serviceName, { passing: true });

      if (!services || services.length === 0) {
        this.logger.warn(`No healthy instances found for service: ${serviceName}`);
        return null;
      }

      // Simple round-robin (you can implement more sophisticated load balancing)
      const randomIndex = Math.floor(Math.random() * services.length);
      const service = services[randomIndex];

      return {
        address: service.Service.Address || service.Node.Address,
        port: service.Service.Port,
      };
    } catch (error) {
      this.logger.error(`Failed to get service endpoint: ${error.message}`, error.stack);
      return null;
    }
  }

  async getAllServices(): Promise<any> {
    try {
      return await this.consulClient.catalog.service.list();
    } catch (error) {
      this.logger.error(`Failed to get all services: ${error.message}`, error.stack);
      throw error;
    }
  }

  async getServiceHealth(serviceName: string): Promise<any[]> {
    try {
      return await this.consulClient.health.service(serviceName);
    } catch (error) {
      this.logger.error(`Failed to get service health: ${error.message}`, error.stack);
      throw error;
    }
  }

  async setKeyValue(key: string, value: string): Promise<void> {
    try {
      await this.consulClient.kv.set(key, value);
      this.logger.debug(`KV set: ${key}`);
    } catch (error) {
      this.logger.error(`Failed to set KV: ${error.message}`, error.stack);
      throw error;
    }
  }

  async getKeyValue(key: string): Promise<string | undefined> {
    try {
      const result = await this.consulClient.kv.get(key);
      return result?.Value;
    } catch (error) {
      this.logger.error(`Failed to get KV: ${error.message}`, error.stack);
      throw error;
    }
  }

  async deleteKeyValue(key: string): Promise<void> {
    try {
      await this.consulClient.kv.del(key);
      this.logger.debug(`KV deleted: ${key}`);
    } catch (error) {
      this.logger.error(`Failed to delete KV: ${error.message}`, error.stack);
      throw error;
    }
  }

  getClient(): Consul.Consul {
    return this.consulClient;
  }
}

Health Check Implementation​

1. Health Check Service​

// shared/consul/src/health-check.service.ts
import { Injectable, Logger } from '@nestjs/common';
import {
  HealthCheckService as TerminusHealthCheckService,
  HealthCheck,
  HttpHealthIndicator,
  TypeOrmHealthIndicator,
  MemoryHealthIndicator,
  DiskHealthIndicator,
} from '@nestjs/terminus';
import { InjectConnection } from '@nestjs/typeorm';
import { Connection } from 'typeorm';

export interface HealthCheckResult {
  status: 'ok' | 'error';
  info?: Record<string, any>;
  error?: Record<string, any>;
  details: Record<string, any>;
}

@Injectable()
export class HealthCheckService {
  private readonly logger = new Logger(HealthCheckService.name);

  constructor(
    private health: TerminusHealthCheckService,
    private http: HttpHealthIndicator,
    private db: TypeOrmHealthIndicator,
    private memory: MemoryHealthIndicator,
    private disk: DiskHealthIndicator,
    @InjectConnection() private connection: Connection,
  ) {}

  @HealthCheck()
  async check(): Promise<HealthCheckResult> {
    try {
      return await this.health.check([
        // Database check
        () => this.db.pingCheck('database'),

        // Memory check (heap should not use more than 300MB)
        () => this.memory.checkHeap('memory_heap', 300 * 1024 * 1024),

        // Memory check (RSS should not use more than 300MB)
        () => this.memory.checkRSS('memory_rss', 300 * 1024 * 1024),

        // Disk check (storage should not use more than 90% of available space)
        () =>
          this.disk.checkStorage('disk', {
            thresholdPercent: 0.9,
            path: '/',
          }),
      ]);
    } catch (error) {
      this.logger.error('Health check failed', error);
      throw error;
    }
  }

  @HealthCheck()
  async liveness(): Promise<HealthCheckResult> {
    // Simple liveness check - just verify the app is running
    return {
      status: 'ok',
      details: {
        uptime: process.uptime(),
        timestamp: new Date().toISOString(),
      },
    };
  }

  @HealthCheck()
  async readiness(): Promise<HealthCheckResult> {
    try {
      return await this.health.check([
        // Check if database is ready
        () => this.db.pingCheck('database'),

        // Check if we can allocate memory
        () => this.memory.checkHeap('memory_heap', 500 * 1024 * 1024),
      ]);
    } catch (error) {
      this.logger.error('Readiness check failed', error);
      throw error;
    }
  }

  async checkDatabase(): Promise<boolean> {
    try {
      await this.connection.query('SELECT 1');
      return true;
    } catch (error) {
      this.logger.error('Database health check failed', error);
      return false;
    }
  }

  async checkExternalService(url: string, timeout: number = 5000): Promise<boolean> {
    try {
      const result = await this.http.pingCheck('external_service', url, { timeout });
      return result.external_service.status === 'up';
    } catch (error) {
      this.logger.error(`External service health check failed for ${url}`, error);
      return false;
    }
  }

  getSystemInfo(): Record<string, any> {
    return {
      node: {
        version: process.version,
        platform: process.platform,
        arch: process.arch,
      },
      memory: {
        total: process.memoryUsage().heapTotal,
        used: process.memoryUsage().heapUsed,
        external: process.memoryUsage().external,
        rss: process.memoryUsage().rss,
      },
      uptime: process.uptime(),
      pid: process.pid,
      environment: process.env.NODE_ENV,
    };
  }
}

3. Service Bootstrap with Consul​

// services/user-service/src/main.ts
import { NestFactory } from '@nestjs/core';
import { AppModule } from './app.module';
import { ConsulService } from '@ecommerce/consul';
import { Logger } from '@nestjs/common';
import * as os from 'os';

async function bootstrap() {
  const logger = new Logger('Bootstrap');
  const app = await NestFactory.create(AppModule);

  const port = process.env.PORT || 3001;
  const serviceName = process.env.SERVICE_NAME || 'user-service';
  const serviceId = `${serviceName}-${process.env.HOSTNAME || os.hostname()}`;

  // Get the pod IP or local IP
  const getServiceAddress = (): string => {
    // In Kubernetes, use the pod IP
    if (process.env.POD_IP) {
      return process.env.POD_IP;
    }

    // In local development, use localhost
    if (process.env.NODE_ENV === 'development') {
      return 'localhost';
    }

    // Fallback to first non-internal IPv4 address
    const interfaces = os.networkInterfaces();
    for (const name of Object.keys(interfaces)) {
      for (const iface of interfaces[name]!) {
        if (iface.family === 'IPv4' && !iface.internal) {
          return iface.address;
        }
      }
    }

    return 'localhost';
  };

  await app.listen(port);

  const serviceAddress = getServiceAddress();
  logger.log(`Service started on ${serviceAddress}:${port}`);

  // Register with Consul
  const consulService = app.get(ConsulService);
  await consulService.registerService({
    id: serviceId,
    name: serviceName,
    address: serviceAddress,
    port: Number(port),
    tags: [
      'nestjs',
      'microservice',
      process.env.NODE_ENV || 'development',
      `version:${process.env.APP_VERSION || '1.0.0'}`,
    ],
    meta: {
      version: process.env.APP_VERSION || '1.0.0',
      environment: process.env.NODE_ENV || 'development',
      podName: process.env.HOSTNAME || os.hostname(),
      namespace: process.env.NAMESPACE || 'default',
    },
    check: {
      http: `http://${serviceAddress}:${port}/health`,
      interval: '10s',
      timeout: '5s',
      deregisterAfter: '30s',
    },
    checks: [
      {
        http: `http://${serviceAddress}:${port}/health/live`,
        interval: '10s',
        timeout: '3s',
        notes: 'Liveness probe',
      },
      {
        http: `http://${serviceAddress}:${port}/health/ready`,
        interval: '10s',
        timeout: '5s',
        notes: 'Readiness probe',
      },
    ],
  });

  // Graceful shutdown
  const signals = ['SIGTERM', 'SIGINT'];
  signals.forEach((signal) => {
    process.on(signal, async () => {
      logger.log(`Received ${signal}, starting graceful shutdown`);

      try {
        // Deregister from Consul
        await consulService.deregisterService(serviceId);
        logger.log('Service deregistered from Consul');

        // Close the application
        await app.close();
        logger.log('Application closed gracefully');

        process.exit(0);
      } catch (error) {
        logger.error('Error during graceful shutdown', error);
        process.exit(1);
      }
    });
  });

  logger.log(`${serviceName} is running on: ${await app.getUrl()}`);
}

1. Service Discovery Client​

// shared/consul/src/service-discovery.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { ConsulService } from './consul.service';
import { HttpService } from '@nestjs/axios';
import { firstValueFrom } from 'rxjs';

export interface ServiceInstance {
  id: string;
  name: string;
  address: string;
  port: number;
  tags: string[];
  meta: Record<string, string>;
  health: 'passing' | 'warning' | 'critical';
}

@Injectable()
export class ServiceDiscoveryService {
  private readonly logger = new Logger(ServiceDiscoveryService.name);
  private serviceCache: Map<string, { instances: ServiceInstance[]; timestamp: number }> = new Map();
  private readonly cacheTTL = 10000; // 10 seconds

  constructor(private consulService: ConsulService, private httpService: HttpService) {}

  async discoverService(serviceName: string, useCache: boolean = true): Promise<ServiceInstance[]> {
    // Check cache
    if (useCache) {
      const cached = this.serviceCache.get(serviceName);
      if (cached && Date.now() - cached.timestamp < this.cacheTTL) {
        return cached.instances;
      }
    }

    try {
      const services = await this.consulService.getService(serviceName, { passing: true });

      const instances: ServiceInstance[] = services.map((service: any) => ({
        id: service.Service.ID,
        name: service.Service.Service,
        address: service.Service.Address || service.Node.Address,
        port: service.Service.Port,
        tags: service.Service.Tags || [],
        meta: service.Service.Meta || {},
        health: this.mapHealthStatus(service.Checks),
      }));

      // Update cache
      this.serviceCache.set(serviceName, {
        instances,
        timestamp: Date.now(),
      });

      return instances;
    } catch (error) {
      this.logger.error(`Failed to discover service ${serviceName}:`, error);

      // Return cached data if available, even if expired
      const cached = this.serviceCache.get(serviceName);
      if (cached) {
        this.logger.warn(`Returning stale cache for ${serviceName}`);
        return cached.instances;
      }

      throw error;
    }
  }

  async getServiceUrl(serviceName: string, loadBalancing: 'random' | 'round-robin' = 'random'): Promise<string> {
    const instances = await this.discoverService(serviceName);

    if (instances.length === 0) {
      throw new Error(`No healthy instances found for service: ${serviceName}`);
    }

    const instance = this.selectInstance(instances, loadBalancing);
    return `http://${instance.address}:${instance.port}`;
  }

  async callService<T = any>(
    serviceName: string,
    path: string,
    options?: {
      method?: 'GET' | 'POST' | 'PUT' | 'DELETE' | 'PATCH';
      data?: any;
      headers?: Record<string, string>;
      timeout?: number;
      retries?: number;
    },
  ): Promise<T> {
    const method = options?.method || 'GET';
    const retries = options?.retries || 3;
    let lastError: Error;

    for (let attempt = 1; attempt <= retries; attempt++) {
      try {
        const baseUrl = await this.getServiceUrl(serviceName);
        const url = `${baseUrl}${path}`;

        this.logger.debug(`Calling ${serviceName} (attempt ${attempt}/${retries}): ${method} ${url}`);

        const response = await firstValueFrom(
          this.httpService.request({
            method,
            url,
            data: options?.data,
            headers: options?.headers,
            timeout: options?.timeout || 5000,
          }),
        );

        return response.data;
      } catch (error) {
        lastError = error;
        this.logger.warn(`Service call failed (attempt ${attempt}/${retries}):`, error.message);

        if (attempt < retries) {
          // Exponential backoff
          const delay = Math.min(1000 * Math.pow(2, attempt - 1), 5000);
          await new Promise((resolve) => setTimeout(resolve, delay));
        }
      }
    }

    this.logger.error(`All retries exhausted for ${serviceName}${path}`);
    throw lastError!;
  }

  private selectInstance(instances: ServiceInstance[], strategy: 'random' | 'round-robin'): ServiceInstance {
    if (strategy === 'random') {
      const randomIndex = Math.floor(Math.random() * instances.length);
      return instances[randomIndex];
    }

    // Simple round-robin (in production, use a more sophisticated implementation with state)
    const now = Date.now();
    const index = now % instances.length;
    return instances[index];
  }

  private mapHealthStatus(checks: any[]): 'passing' | 'warning' | 'critical' {
    if (!checks || checks.length === 0) return 'passing';

    const hasFailure = checks.some((check) => check.Status === 'critical');
    if (hasFailure) return 'critical';

    const hasWarning = checks.some((check) => check.Status === 'warning');
    if (hasWarning) return 'warning';

    return 'passing';
  }

  clearCache(serviceName?: string): void {
    if (serviceName) {
      this.serviceCache.delete(serviceName);
    } else {
      this.serviceCache.clear();
    }
  }
}

export class ConsulMetricsService {

    help: 'Total number of service discovery requests',
    labelNames: ['service', 'status'],
  });

  private readonly serviceDiscoveryDuration = new Histogram({
    name: 'consul_service_discovery_duration_seconds',
    help: 'Duration of service discovery requests',
    labelNames: ['service'],
    buckets: [0.001, 0.005, 0.01, 0.05, 0.1, 0.5, 1],
  });

  private readonly healthyServicesGauge = new Gauge({
    name: 'consul_healthy_services',
    help: 'Number of healthy service instances',
    labelNames: ['service'],
  });

  private readonly serviceCallCounter = new Counter({
    name: 'consul_service_call_total',
    help: 'Total number of service-to-service calls',
    labelNames: ['source', 'target', 'method', 'status'],
  });

  private readonly serviceCallDuration = new Histogram({
    name: 'consul_service_call_duration_seconds',
    help: 'Duration of service-to-service calls',
    labelNames: ['source', 'target', 'method'],
    buckets: [0.01, 0.05, 0.1, 0.5, 1, 5, 10],
  });

  recordServiceDiscovery(service: string, status: 'success' | 'error', duration: number): void {
    this.serviceDiscoveryCounter.inc({ service, status });
    this.serviceDiscoveryDuration.observe({ service }, duration);
  }

  updateHealthyServices(service: string, count: number): void {
    this.healthyServicesGauge.set({ service }, count);
  }

  recordServiceCall(source: string, target: string, method: string, status: number, duration: number): void {
    this.serviceCallCounter.inc({
      source,
      target,
      method,
      status: status >= 200 && status < 300 ? 'success' : 'error',
    });
    this.serviceCallDuration.observe({ source, target, method }, duration);
  }

  getMetrics(): string {
    return register.metrics();
  }
}

      `status:${statusCode}`,
    ]);

Best Practices​

1. Service Registration​

✅ Do:

  • Use unique service IDs (e.g., include hostname/pod name)
  • Include version tags for blue-green deployments
  • Set appropriate health check intervals (10-30 seconds)
  • Use deregistration timeout to prevent flapping
  • Include metadata for debugging and monitoring

❌ Don't:

  • Register services with duplicate IDs
  • Use very short health check intervals (< 5 seconds)
  • Skip health checks
  • Hard-code service addresses

2. Health Checks​

✅ Do:

  • Implement separate liveness and readiness probes
  • Check all critical dependencies (database, cache, etc.)
  • Return appropriate HTTP status codes (200, 503)
  • Include meaningful health check responses
  • Set reasonable timeouts (3-5 seconds)

❌ Don't:

  • Make health checks too expensive (avoid heavy queries)
  • Return 200 when the service can't handle requests
  • Check external services in liveness probes
  • Use the same endpoint for all probe types

3. Service Discovery​

✅ Do:

  • Cache service discovery results
  • Implement retry logic with exponential backoff
  • Handle service unavailability gracefully
  • Use circuit breakers for failing services
  • Monitor discovery latency and errors

❌ Don't:

  • Discover services on every request
  • Ignore service health status
  • Use blocking calls without timeouts
  • Skip error handling

4. Monitoring​

✅ Do:

  • Track service registration/deregistration events
  • Monitor health check success rates
  • Alert on service unavailability
  • Track service-to-service call metrics
  • Use distributed tracing

❌ Don't:

  • Ignore health check failures
  • Skip alerting configuration
  • Monitor only application metrics
  • Forget to track dependencies

5. Security​

✅ Do:

  • Enable Consul ACLs in production
  • Use TLS for Consul communication
  • Implement authentication for service-to-service calls
  • Rotate credentials regularly
  • Limit network access to Consul

❌ Don't:

  • Run Consul without ACLs
  • Use plain HTTP in production
  • Share service credentials
  • Expose Consul UI publicly without authentication
