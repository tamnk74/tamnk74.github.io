Pricing & Promo Service

Price rules engine, coupon/promotion validation, flash-sale pricing: PostgreSQL rules store, Redis computed-price cache (10 s TTL), and Kafka event invalidation.

Responsibilities​

  • Compute the effective price for any SKU at request time (base price + rules + active promotions)
  • Validate and apply coupon codes with usage limits, expiry, and per-user caps
  • Support flash-sale pricing (time-bounded price overrides loaded into Redis)
  • Broadcast pricing.updated events so caches are invalidated within 10 s

Data Model​

CREATE TABLE price_rules (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name          TEXT NOT NULL,
  rule_type     TEXT NOT NULL CHECK (rule_type IN (
                  'fixed_price','percentage_discount','flash_sale','bundle')),
  sku_pattern   TEXT,                  -- NULL = applies to all SKUs
  category_id   UUID,
  discount_pct  NUMERIC(5,2),
  fixed_price   NUMERIC(12,2),
  min_qty       INT NOT NULL DEFAULT 1,
  priority      INT NOT NULL DEFAULT 0, -- higher = evaluated first
  starts_at     TIMESTAMPTZ NOT NULL,
  ends_at       TIMESTAMPTZ,           -- NULL = no expiry
  active        BOOLEAN NOT NULL DEFAULT true
);
CREATE INDEX idx_price_rules_active ON price_rules (sku_pattern, starts_at, ends_at)
  WHERE active = true;

CREATE TABLE coupons (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  code          TEXT NOT NULL UNIQUE,
  discount_type TEXT NOT NULL CHECK (discount_type IN ('pct','fixed','free_shipping')),
  discount_value NUMERIC(12,2) NOT NULL,
  min_order_value NUMERIC(12,2) NOT NULL DEFAULT 0,
  max_uses      INT,                   -- NULL = unlimited
  used_count    INT NOT NULL DEFAULT 0,
  per_user_limit INT NOT NULL DEFAULT 1,
  starts_at     TIMESTAMPTZ NOT NULL,
  expires_at    TIMESTAMPTZ,
  active        BOOLEAN NOT NULL DEFAULT true
);

CREATE TABLE coupon_uses (
  coupon_id  UUID NOT NULL REFERENCES coupons(id),
  user_id    UUID NOT NULL,
  order_id   UUID NOT NULL,
  used_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  PRIMARY KEY (coupon_id, order_id)
);
CREATE INDEX idx_coupon_uses_user ON coupon_uses (coupon_id, user_id);

Price Computation​

// src/pricing/pricing.service.ts
import { Injectable } from '@nestjs/common';
import { InjectDataSource } from '@nestjs/typeorm';
import { DataSource } from 'typeorm';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import Redis from 'ioredis';

@Injectable()
export class PricingService {
  constructor(
    @InjectDataSource() private readonly ds: DataSource,
    @InjectRedis() private readonly redis: Redis,
  ) {}

  /**
   * Compute effective price for a SKU.
   * Returns cached value if available (TTL 10 s).
   */
  async getEffectivePrice(sku: string, qty = 1): Promise<PriceResult> {
    const cacheKey = `price:${sku}:${qty}`;
    const cached = await this.redis.get(cacheKey);
    if (cached) return JSON.parse(cached) as PriceResult;

    const result = await this.computePrice(sku, qty);
    await this.redis.set(cacheKey, JSON.stringify(result), 'EX', 10);
    return result;
  }

  private async computePrice(sku: string, qty: number): Promise<PriceResult> {
    // 1. Fetch base price
    const [product] = await this.ds.query(
      `SELECT base_price FROM products WHERE sku = $1`, [sku]
    );
    if (!product) throw new Error(`SKU not found: ${sku}`);
    const basePrice = parseFloat(product.base_price);

    // 2. Find highest-priority active rule for this SKU
    const rules = await this.ds.query(
      `SELECT * FROM price_rules
       WHERE active = true
         AND (sku_pattern IS NULL OR $1 LIKE sku_pattern)
         AND starts_at <= NOW()
         AND (ends_at IS NULL OR ends_at > NOW())
         AND min_qty <= $2
       ORDER BY priority DESC
       LIMIT 1`,
      [sku, qty]
    );

    if (rules.length === 0) {
      return { sku, originalPrice: basePrice, finalPrice: basePrice, discountApplied: null };
    }

    const rule = rules[0];
    let finalPrice = basePrice;

    if (rule.rule_type === 'fixed_price' || rule.rule_type === 'flash_sale') {
      finalPrice = parseFloat(rule.fixed_price);
    } else if (rule.rule_type === 'percentage_discount') {
      finalPrice = basePrice * (1 - parseFloat(rule.discount_pct) / 100);
    }

    return {
      sku,
      originalPrice: basePrice,
      finalPrice: Math.round(finalPrice * 100) / 100,
      discountApplied: { ruleId: rule.id, type: rule.rule_type },
    };
  }
}

export interface PriceResult {
  sku:            string;
  originalPrice:  number;
  finalPrice:     number;
  discountApplied: { ruleId: string; type: string } | null;
}

Coupon Validation​

// src/pricing/coupon.service.ts
import {
  Injectable, BadRequestException, ConflictException
} from '@nestjs/common';
import { InjectDataSource } from '@nestjs/typeorm';
import { DataSource } from 'typeorm';

@Injectable()
export class CouponService {
  constructor(@InjectDataSource() private readonly ds: DataSource) {}

  async validateAndApply(
    code: string,
    userId: string,
    orderId: string,
    orderTotal: number,
  ): Promise<CouponApplication> {
    return this.ds.transaction(async (em) => {
      // Lock the row to prevent concurrent over-use
      const [coupon] = await em.query(
        `SELECT * FROM coupons WHERE code = $1 AND active = true FOR UPDATE`,
        [code]
      );

      if (!coupon) throw new BadRequestException('Invalid or inactive coupon');
      if (coupon.expires_at && new Date(coupon.expires_at) < new Date()) {
        throw new BadRequestException('Coupon expired');
      }
      if (coupon.max_uses !== null && coupon.used_count >= coupon.max_uses) {
        throw new BadRequestException('Coupon usage limit reached');
      }
      if (orderTotal < parseFloat(coupon.min_order_value)) {
        throw new BadRequestException(
          `Minimum order value ${coupon.min_order_value} required`
        );
      }

      // Per-user limit check
      const [userUse] = await em.query(
        `SELECT COUNT(*) AS cnt FROM coupon_uses
         WHERE coupon_id = $1 AND user_id = $2`,
        [coupon.id, userId]
      );
      if (parseInt(userUse.cnt, 10) >= coupon.per_user_limit) {
        throw new ConflictException('Per-user coupon limit reached');
      }

      // Record usage and increment counter atomically
      await em.query(
        `INSERT INTO coupon_uses (coupon_id, user_id, order_id)
         VALUES ($1, $2, $3)`,
        [coupon.id, userId, orderId]
      );
      await em.query(
        `UPDATE coupons SET used_count = used_count + 1 WHERE id = $1`,
        [coupon.id]
      );

      const discount = coupon.discount_type === 'pct'
        ? orderTotal * (parseFloat(coupon.discount_value) / 100)
        : parseFloat(coupon.discount_value);

      return {
        couponId:      coupon.id,
        discountType:  coupon.discount_type,
        discountValue: parseFloat(coupon.discount_value),
        discountAmount: Math.min(discount, orderTotal),
      };
    });
  }
}

export interface CouponApplication {
  couponId:      string;
  discountType:  string;
  discountValue: number;
  discountAmount: number;
}

Flash Sale Price Loading​

Before a flash sale opens, pre-load overriding Redis keys to bypass DB lookups entirely during the spike (complementary to the inventory Redis counter):

// src/pricing/flash-sale-loader.service.ts
async loadFlashSalePrices(eventId: string): Promise<void> {
  const rules = await this.ds.query(
    `SELECT sku_pattern, fixed_price FROM price_rules
     WHERE rule_type = 'flash_sale' AND active = true
       AND starts_at <= NOW() AND ends_at > NOW()`
  );

  const pipeline = this.redis.pipeline();
  for (const rule of rules) {
    pipeline.set(`price:${rule.sku_pattern}:1`, JSON.stringify({
      sku: rule.sku_pattern,
      originalPrice: 0,   // populated at read time
      finalPrice: parseFloat(rule.fixed_price),
      discountApplied: { type: 'flash_sale' },
    }), 'EX', 7200);
  }
  await pipeline.exec();
}
